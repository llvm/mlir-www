42void ConvertACCToLLVMPass::runOnOperation() {
43 ModuleOp module = getOperation();
44
45 SymbolTable symbolTable(module);
46 LLVMTypeConverter converter(&getContext());
47 RewritePatternSet patterns(&getContext());
48 arith::populateArithToLLVMConversionPatterns(converter, patterns);
49 cf::populateControlFlowToLLVMConversionPatterns(converter, patterns);
50 populateFuncToLLVMConversionPatterns(converter, patterns);
51 populateFinalizeMemRefToLLVMConversionPatterns(converter, patterns);
52
53 acc::ACCRuntimeCallConfig runtimeConfig;
54
55 populateACCExecutableDirectivePatterns(
56 converter, patterns, module.getBodyRegion(), symbolTable, runtimeConfig);
57
58 acc::OpenACCSupport &accSupport = getAnalysis<acc::OpenACCSupport>();
59 populateACCDataDirectivePatterns(converter, patterns, accSupport,
60 module.getBodyRegion(), symbolTable,
61 runtimeConfig);
62 populateACCHostDataPatterns(converter, patterns, module.getBodyRegion(),
63 symbolTable, runtimeConfig);
64 populateACCKernelLaunchPatterns(converter, patterns, accSupport,
65 module.getBodyRegion(), symbolTable,
66 runtimeConfig);
67 populateACCAtomicPatterns(converter, patterns, accSupport);
68 populateACCDataClauseOpPatterns(converter, patterns, accSupport,
69 module.getBodyRegion(), symbolTable,
70 runtimeConfig);
71
72 LLVMConversionTarget target(getContext());
73 configureACCExecutableDirectiveConversionLegality(target);
74 configureACCDataDirectiveConversionLegality(target);
75 configureACCHostDataConversionLegality(target);
76 configureACCKernelLaunchConversionLegality(target);
77 configureACCAtomicConversionLegality(target);
78 if (failed(applyPartialConversion(module, target, std::move(patterns))))
79 signalPassFailure();
80}
