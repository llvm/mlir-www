35LLVM::LLVMFunctionType acc::getRuntimeFunctionType(MLIRContext *ctx,
36 RuntimeFunction fn) {
37 Type Void = LLVM::LLVMVoidType::get(ctx);
38 Type Ptr = LLVM::LLVMPointerType::get(ctx);
39 Type Int32 = IntegerType::get(ctx, 32);
40 Type Int64 = IntegerType::get(ctx, 64);

42 switch (fn) {
43#define ACC_RTL(Enum, Str, IsVarArg, ReturnType, ...) \
44 case RuntimeFunction::Enum: \
45 return LLVM::LLVMFunctionType::get(ReturnType, \
46 ArrayRef<Type>{__VA_ARGS__}, IsVarArg);
47#include "mlir/Dialect/OpenACC/OpenACCRuntimeFunctions.def"

136int64_t ACCRuntimeCallConfig::getMapFlagsRuntimeValue(MapFlags flags) const {
137 int64_t runtimeValue = 0;
138 for (unsigned bit = 0; bit != 32; ++bit) {
139 auto flag = static_cast<MapFlags>(1u << bit);
140 if (!bitEnumContainsAny(flags, flag))
141 continue;
142 auto it = mapFlagRuntimeValues.find(flag);
143 if (it == mapFlagRuntimeValues.end())
144 llvm::report_fatal_error(
145 llvm::Twine("missing OpenACC runtime map-flag mapping for ") +
146 stringifyMapFlags(flag));
147 runtimeValue |= it->second;
148 }
149 return runtimeValue;
150}

163std::string ACCRuntimeCallConfig::formatMapFlags(MapFlags flags) const {
164 int64_t runtimeValue = getMapFlagsRuntimeValue(flags);
165 std::string rendered = stringifyMapFlags(flags);
166 rendered += " (";
167 rendered += std::to_string(runtimeValue);

190void acc::populateDialectIdentityDeviceTypeMapping(
191 ACCRuntimeCallConfig &config) {
192 for (uint32_t value = 0; value <= getMaxEnumValForDeviceType(); ++value)
193 if (std::optional<DeviceType> type = symbolizeDeviceType(value))
194 config.setDeviceTypeRuntimeValue(*type, value);
195}

197void acc::populateDialectIdentityMapFlagsMapping(ACCRuntimeCallConfig &config) {
198 for (unsigned bit = 0; bit != 32; ++bit) {
199 uint32_t value = 1u << bit;
200 if (std::optional<MapFlags> flag = symbolizeMapFlags(value))
201 config.setMapFlagRuntimeValue(*flag, value);
202 }
203}

205FailureOr<LLVM::CallOp>
206acc::createRuntimeCall(Location loc, OpBuilder &builder,
207 Region &globalSymbolRegion, SymbolTable &symbolTable,
208 RuntimeFunction fn, const ACCRuntimeCallConfig &config,
209 ArrayRef<Value> arguments) {
210 MLIRContext *ctx = builder.getContext();
211 LLVM::LLVMFunctionType fnTy = getRuntimeFunctionType(ctx, fn);
212 StringRef symbolName = config.getName(fn);
213
214 auto func = symbolTable.lookup<LLVM::LLVMFuncOp>(symbolName);
215 if (func) {
216 // An existing declaration with a different signature cannot be called with
217 // the arguments expected by the runtime entry point.
218 if (func.getFunctionType() != fnTy)
219 return emitError(loc) << "OpenACC runtime function '" << symbolName
220 << "' is already declared with signature "
221 << func.getFunctionType() << ", expected " << fnTy;
222 } else {
223 OpBuilder moduleBuilder =
224 OpBuilder::atBlockEnd(&globalSymbolRegion.front());
225 func = LLVM::LLVMFuncOp::create(moduleBuilder, loc, symbolName, fnTy);
226 symbolTable.insert(func);
227 }
228
229 return LLVM::CallOp::create(builder, loc, func, arguments);
230}
