41static ParseResult parseCTAGroup(OpAsmParser &parser,
42 NVVM::CTAGroupKindAttr &groupAttr) {
43 StringRef keyword;
44 if (parser.parseKeyword(&keyword))
45 return failure();
46 std::optional<NVVM::CTAGroupKind> group =
47 NVVM::symbolizeCTAGroupKind(keyword);
48 if (!group)
49 return parser.emitError(parser.getNameLoc()) << " expected CTA group";
50 groupAttr = NVVM::CTAGroupKindAttr::get(parser.getContext(), *group);
51 return success();
52}

63// This verifier is shared among the following Ops:
64// CpAsyncBulkTensorSharedCTAToGlobalOp (TMA Store)
65// CpAsyncBulkTensorReduceOp (TMA Store-Reduce)
66static LogicalResult cpAsyncBulkTensorCommonVerifier(size_t tensorDims,
67 bool isIm2Col,
68 size_t numIm2ColOffsets,
69 Location loc) {
70 if (tensorDims < 1 || tensorDims > 5)
71 return emitError(loc, "expects coordinates between 1 to 5 dimension");
72
73 // For Im2Col mode, there are two constraints:
74 if (isIm2Col) {
75 // 1. Tensor must always be at least 3-d.
76 if (tensorDims < 3)
77 return emitError(
78 loc,
79 "to use im2col mode, the tensor has to be at least 3-dimensional");
80 // 2. When there are Im2ColOffsets, they must be (Dims - 2) in number.
81 if (numIm2ColOffsets && (tensorDims != (numIm2ColOffsets + 2)))
82 return emitError(
83 loc, "im2col offsets must be 2 less than number of coordinates");
84 }
85 return success();
86}

88LogicalResult CpAsyncBulkTensorOverrideAddrCommonVerifier(
89 OperandRange coordinates, OperandRange tensorSize, OperandRange lowerStride,
90 Value upperStride, bool isTile, Location loc) {
91 LogicalResult res = success();
92 if (!tensorSize.empty() && coordinates.size() != tensorSize.size()) {
93 res =
94 emitError(loc, "Expected coordinates size to be equal to tensor size");
95 }
96
97 if (!lowerStride.empty() && tensorSize.empty()) {
98 res = emitError(
99 loc,
100 "Expected tensor_size to be present when lower_stride is provided");
101 } else if (!lowerStride.empty() &&
102 lowerStride.size() != tensorSize.size() - 1) {
103 res = emitError(
104 loc,
105 "Expected lower_stride size to be equal to one less than tensor size");
106 }
107
108 if (!lowerStride.empty() != static_cast<bool>(upperStride)) {
109 res = emitError(loc,
110 "Expected lower_stride and upper_stride to be either both "
111 "present or both absent");
112 }
113
114 bool isDimStride = tensorSize.size() > 0;
115 if (!isTile && isDimStride) {
116 res = emitError(
117 loc, "Only tile mode supports override address with dim and stride");
118 }
119
120 return res;
121}

123LogicalResult CpAsyncBulkTensorSharedCTAToGlobalOp::verify() {
124 TMAStoreMode mode = getMode();
125 // We lower through inline-ptx when getPredicate() is true.
126 // a) Only TILE mode is supported
127 // b) Cache-hint is not supported
128 if (getPredicate()) {
129 if (mode != TMAStoreMode::TILE)
130 return emitError("Inline-ptx lowering supported only for Tile mode.");
131 if (getL2CacheHint())
132 return emitError("Inline-ptx lowering unsupported with L2 cache-hint.");
133 }
134
135 size_t dims = getCoordinates().size();
136 switch (mode) {
137 case TMAStoreMode::TILE:
138 return cpAsyncBulkTensorCommonVerifier(dims, false, 0, getLoc());
139 case TMAStoreMode::IM2COL:
140 case TMAStoreMode::IM2COL_W:
141 return cpAsyncBulkTensorCommonVerifier(dims, true, 0, getLoc());
142 case TMAStoreMode::TILE_SCATTER4:
143 if (dims != 5)
144 return emitError("Scatter4 mode expects 5 coordinates");
145 }
146 return success();
147}

149LogicalResult CpAsyncBulkTensorSharedCTAToGlobalOverrideAddrOp::verify() {
150 TMAStoreMode mode = getMode();
151 bool isIm2Col =
152 mode == TMAStoreMode::IM2COL || mode == TMAStoreMode::IM2COL_W;
153 bool isTile = mode == TMAStoreMode::TILE;
154
155 LogicalResult commonRes = cpAsyncBulkTensorCommonVerifier(
156 getCoordinates().size(), isIm2Col, 0, getLoc());
157
158 LogicalResult overrideAddrRes = CpAsyncBulkTensorOverrideAddrCommonVerifier(
159 getCoordinates(), getTensorSize(), getLowerStride(), getUpperStride(),
160 isTile, getLoc());
161
162 if (mode == TMAStoreMode::TILE_SCATTER4 && getCoordinates().size() != 5)
163 overrideAddrRes = emitError("Mode tile scatter4 expects 5 coordinates");
164
165 return failed(commonRes) || failed(overrideAddrRes) ? failure() : success();
166}

179// This verify params can be shared across TMA Load and Prefetch Ops.
180static LogicalResult verifyTMALoadParams(size_t tensorDims, size_t numIm2colOff,
181 TMALoadMode mode, Location loc) {
182 if (tensorDims < 1 || tensorDims > 5)
183 return emitError(loc, "expects coordinates between 1 to 5 dimension");
184
185 auto checkTMALoadParams = [&](TMALoadMode mode, bool isIm2col,
186 size_t expectedIm2colOff) -> LogicalResult {
187 if (isIm2col && (tensorDims < 3))
188 return emitError(loc)
189 << "to use " << mode
190 << " mode, the tensor has to be at least 3-dimensional";
191
192 if (numIm2colOff != expectedIm2colOff)
193 return emitError(loc) << " im2col offsets expected " << expectedIm2colOff
194 << " (provided " << numIm2colOff << ")";
195
196 return success();
197 };
198
199 switch (mode) {
200 case TMALoadMode::TILE:
201 return checkTMALoadParams(mode, false, 0);
202 case TMALoadMode::IM2COL:
203 return checkTMALoadParams(mode, true, tensorDims - 2);
204 case TMALoadMode::IM2COL_W:
205 case TMALoadMode::IM2COL_W_128:
206 return checkTMALoadParams(mode, true, 2);
207 case TMALoadMode::TILE_GATHER4:
208 return (tensorDims == 5)
209 ? checkTMALoadParams(mode, false, 0)
210 : emitError(loc, "Gather4 mode expects 5 coordinates");
211 }
212 return success();
213}

220LogicalResult CpAsyncBulkTensorGlobalToSharedClusterOp::verify() {
221 TMALoadMode mode = getMode();
222 bool isCTAOnly = getIsCTAOnly();
223 if (getPredicate()) { // Inline-asm based lowering
224 if (isCTAOnly)
225 return emitError("Predicate is supported only for shared::cluster mode.");
226 if (mode != TMALoadMode::TILE && mode != TMALoadMode::IM2COL)
227 return emitError(
228 "Predicate is supported only for Tile and Im2col modes.");
229 } else { // Intrinsics-based lowering
230 NVVMMemorySpace expectedAS =
231 isCTAOnly ? NVVMMemorySpace::Shared : NVVMMemorySpace::SharedCluster;
232 unsigned AS = llvm::cast<LLVM::LLVMPointerType>(getDstMem().getType())
233 .getAddressSpace();
234 if (AS != expectedAS)
235 return emitError()
236 << (isCTAOnly
237 ? "Shared::cta destination requires address-space 3."
238 : "Shared::cluster destination requires address-space 7.");
239 // Checks specific to shared::cta mode
240 if (isCTAOnly) {
241 if (getMulticastMask())
242 return emitError("Multicast is not supported with shared::cta mode.");
243 if (getGroup())
244 return emitError("CTAGroup is not supported with shared::cta mode.");
245 }
246 }
247
248 return verifyTMALoadParams(getCoordinates().size(), getIm2colOffsets().size(),
249 getMode(), getLoc());
250}

294LogicalResult NVVM::TensormapReplaceOp::verify() {
295 auto ord = getOrd();
296 Value newVal = getNewValue();
297 auto newValAttr = getNewValueAttr();
298 auto fieldName = stringifyEnum(getField());
299
300 if (ord && !llvm::is_contained({NVVM::TensormapField::BOX_DIM,
301 NVVM::TensormapField::GLOBAL_DIM,
302 NVVM::TensormapField::GLOBAL_STRIDE,
303 NVVM::TensormapField::ELEMENT_STRIDE},
304 getField()))
305 return emitOpError("ordinal is not supported for ")
306 << fieldName << " field";
307
308 auto invalidNewVal = [&](llvm::Twine type) -> std::string {
309 return llvm::Twine("new_value must be specified and must be an " + type +
310 " for " + llvm::Twine(fieldName) + " field")
311 .str();
312 };
313
314 auto invalidNewValAttr = [&]() -> std::string {
315 return (llvm::Twine(
316 "new_value_attr must be specified and must be a valid ") +
317 llvm::Twine(fieldName) + " attribute for " + fieldName + " field")
318 .str();
319 };
320
321 switch (getField()) {
322 case NVVM::TensormapField::GLOBAL_ADDRESS:
323 if (!(newVal && newVal.getType().isInteger(64)))
324 return emitOpError(invalidNewVal("i64"));
325 break;
326 case NVVM::TensormapField::RANK:
327 if (!(newVal && newVal.getType().isInteger(32)))
328 return emitOpError(invalidNewVal("i32"));
329 break;
330 case NVVM::TensormapField::GLOBAL_STRIDE:
331 if (!ord)
332 return emitOpError("ordinal is required for global_stride field");
333 if (!(newVal && newVal.getType().isInteger(64)))
334 return emitOpError(invalidNewVal("i64"));
335 break;
336 case NVVM::TensormapField::BOX_DIM:
337 case NVVM::TensormapField::GLOBAL_DIM:
338 case NVVM::TensormapField::ELEMENT_STRIDE:
339 if (!ord)
340 return emitOpError("ordinal is required for ")
341 << stringifyEnum(getField()) << " field";
342 if (!(newVal && newVal.getType().isInteger(32)))
343 return emitOpError(invalidNewVal("i32"));
344 break;
345 case NVVM::TensormapField::ELEMTYPE:
346 if (!(newValAttr && llvm::isa<TensormapElemtypeAttr>(*newValAttr)))
347 return emitOpError(invalidNewValAttr());
348 break;
349 case NVVM::TensormapField::INTERLEAVE_LAYOUT:
350 if (!(newValAttr && llvm::isa<TensormapInterleaveLayoutAttr>(*newValAttr)))
351 return emitOpError(invalidNewValAttr());
352 break;
353 case NVVM::TensormapField::SWIZZLE_MODE:
354 if (!(newValAttr && llvm::isa<TensormapSwizzleModeAttr>(*newValAttr)))
355 return emitOpError(invalidNewValAttr());
356 break;
357 case NVVM::TensormapField::SWIZZLE_ATOMICITY:
358 if (!(newValAttr && llvm::isa<TensormapSwizzleAtomicityAttr>(*newValAttr)))
359 return emitOpError(invalidNewValAttr());
360 break;
361 case NVVM::TensormapField::FILL_MODE:
362 if (!(newValAttr && llvm::isa<TensormapFillModeAttr>(*newValAttr)))
363 return emitOpError(invalidNewValAttr());
364 break;
365 }
366
367 return success();
368}

374#define CP_ASYNC_ID_IMPL(mod, size, suffix) \
375 llvm::Intrinsic::nvvm_cp_async_##mod##_shared_global_##size##suffix
376
377#define GET_CP_ASYNC_ID(mod, size, has_cpsize) \
378 has_cpsize ? CP_ASYNC_ID_IMPL(mod, size, _s) : CP_ASYNC_ID_IMPL(mod, size, )
379
380llvm::Intrinsic::ID
381CpAsyncOp::getIntrinsicIDAndArgs(Operation &op, LLVM::ModuleTranslation &mt,
382 llvm::SmallVector<llvm::Value *> &args) {
383 llvm::Intrinsic::ID id;
384
385 auto cpAsyncOp = cast<NVVM::CpAsyncOp>(op);
386 bool hasCpSize = static_cast<bool>(cpAsyncOp.getCpSize());
387 switch (cpAsyncOp.getSize()) {
388 case 4:
389 id = GET_CP_ASYNC_ID(ca, 4, hasCpSize);
390 break;
391 case 8:
392 id = GET_CP_ASYNC_ID(ca, 8, hasCpSize);
393 break;
394 case 16:
395 id = (cpAsyncOp.getModifier() == NVVM::LoadCacheModifierKind::CG)
396 ? GET_CP_ASYNC_ID(cg, 16, hasCpSize)
397 : GET_CP_ASYNC_ID(ca, 16, hasCpSize);
398 break;
399 default:
400 llvm_unreachable("Invalid copy size in CpAsyncOp.");
401 }
402
403 // Fill the Intrinsic Args
404 args.push_back(mt.lookupValue(cpAsyncOp.getDst()));
405 args.push_back(mt.lookupValue(cpAsyncOp.getSrc()));
406 if (hasCpSize)
407 args.push_back(mt.lookupValue(cpAsyncOp.getCpSize()));
408
409 return id;
410}

432mlir::NVVM::IDArgPair CpAsyncBulkGlobalToSharedClusterOp::getIntrinsicIDAndArgs(
433 Operation &op, LLVM::ModuleTranslation &mt, llvm::IRBuilderBase &builder) {
434 auto thisOp = cast<NVVM::CpAsyncBulkGlobalToSharedClusterOp>(op);
435 llvm::SmallVector<llvm::Value *> args;
436
437 // Fill the Intrinsic Args: dst, mbar, src, size.
438 args.push_back(mt.lookupValue(thisOp.getDstMem()));
439 args.push_back(mt.lookupValue(thisOp.getMbar()));
440 args.push_back(mt.lookupValue(thisOp.getSrcMem()));
441 args.push_back(mt.lookupValue(thisOp.getSize()));
442
443 // Multicast mask for shared::cluster only, if available.
444 mlir::Value multicastMask = thisOp.getMulticastMask();
445 const bool hasMulticastMask = static_cast<bool>(multicastMask);
446 const bool isSharedCTA = isPtrInSharedCTASpace(thisOp.getDstMem());
447 if (!isSharedCTA) {
448 llvm::Value *i16Unused = llvm::ConstantInt::get(builder.getInt16Ty(), 0);
449 args.push_back(hasMulticastMask ? mt.lookupValue(multicastMask)
450 : i16Unused);
451 } else {
452 args.push_back(builder.getInt32(0)); // ignore_bytes_left
453 args.push_back(builder.getInt32(0)); // ignore_bytes_right
454 }
455
456 // Cache hint, if available.
457 mlir::Value cacheHint = thisOp.getL2CacheHint();
458 const bool hasCacheHint = static_cast<bool>(cacheHint);
459 llvm::Value *i64Unused = llvm::ConstantInt::get(builder.getInt64Ty(), 0);
460 args.push_back(hasCacheHint ? mt.lookupValue(cacheHint) : i64Unused);
461
462 // Flag arguments for multicast/ignore_oob and cachehint.
463 if (isSharedCTA) {
464 args.push_back(builder.getInt1(hasCacheHint)); // flag_ch
465 args.push_back(builder.getInt1(false)); // flag_oob
466 } else {
467 args.push_back(builder.getInt1(hasMulticastMask)); // flag_mc
468 args.push_back(builder.getInt1(hasCacheHint)); // flag_ch
469 }
470
471 // flag_valid_pattern = disabled.
472 args.push_back(builder.getInt32(0));
473
474 llvm::Intrinsic::ID id =
475 isSharedCTA
476 ? llvm::Intrinsic::nvvm_cp_async_bulk_global_to_shared_cta
477 : llvm::Intrinsic::nvvm_cp_async_bulk_global_to_shared_cluster;
478
479 return {id, std::move(args)};
480}

485 llvm::SmallVector<llvm::Value *> args;

491 args.push_back(mt.lookupValue(thisOp.getSrcMem()));

501 // Choose the bytemask variant

506
507 return {id, std::move(args)};

523mlir::NVVM::IDArgPair
524CpAsyncBulkTensorGlobalToSharedClusterOp::getIntrinsicIDAndArgs(
525 Operation &op, LLVM::ModuleTranslation &mt, llvm::IRBuilderBase &builder) {
526 auto thisOp = cast<NVVM::CpAsyncBulkTensorGlobalToSharedClusterOp>(op);
527 const bool isCTAOnly = thisOp.getIsCTAOnly();
528 llvm::SmallVector<llvm::Value *> args;
529
530 // Fill the Intrinsic Args
531 args.push_back(mt.lookupValue(thisOp.getDstMem()));
532 args.push_back(mt.lookupValue(thisOp.getMbar()));
533 args.push_back(mt.lookupValue(thisOp.getTmaDescriptor()));
534
535 // Coordinates and im2col-offsets
536 for (mlir::Value v : thisOp.getCoordinates())
537 args.push_back(mt.lookupValue(v));
538 for (mlir::Value v : thisOp.getIm2colOffsets())
539 args.push_back(mt.lookupValue(v));
540
541 // MulticastMask, if available
542 mlir::Value mcMask = thisOp.getMulticastMask();
543 const bool hasMC = static_cast<bool>(mcMask);
544 llvm::Value *i16Zero =
545 llvm::ConstantInt::get(llvm::Type::getInt16Ty(mt.getLLVMContext()), 0);
546
547 // CacheHint, if available
548 mlir::Value cacheHint = thisOp.getL2CacheHint();
549 const bool hasCacheHint = static_cast<bool>(cacheHint);
550 llvm::Value *i64Zero =
551 llvm::ConstantInt::get(llvm::Type::getInt64Ty(mt.getLLVMContext()), 0);
552
553 // Flag argument CTAGroup
554 // CTA_1/2 is mapped to values 1 and 2 for the intrinsics.
555 // Hence, the +1 to getGroup().
556 const int32_t val =
557 thisOp.getGroup() ? (static_cast<int32_t>(*thisOp.getGroup()) + 1) : 0;
558 llvm::Value *cg =
559 llvm::ConstantInt::get(llvm::Type::getInt32Ty(mt.getLLVMContext()), val);
560
561 // flag_valid_pattern = disabled
562 llvm::Value *flagValidPattern = builder.getInt32(0);
563
564 if (!isCTAOnly) {
565 // For shared::cluster, all the arguments that we build are applicable.
566 args.push_back(hasMC ? mt.lookupValue(mcMask) : i16Zero);
567 args.push_back(hasCacheHint ? mt.lookupValue(cacheHint) : i64Zero);
568 args.push_back(builder.getInt1(hasMC));
569 args.push_back(builder.getInt1(hasCacheHint));
570 args.push_back(cg);
571 args.push_back(flagValidPattern);
572 } else {
573 // For shared::cta, only cache-hint is applicable.
574 args.push_back(hasCacheHint ? mt.lookupValue(cacheHint) : i64Zero);
575 args.push_back(builder.getInt1(hasCacheHint));
576 args.push_back(flagValidPattern);
577 }

581 using rowTy = std::array<llvm::Intrinsic::ID, numDims + 1>;
582 using TableTy = std::array<rowTy, numModes>;
583 static constexpr TableTy IDTable{
