58LogicalResult NVVMDialect::verifyOperationAttribute(Operation *op,
59 NamedAttribute attr) {
60 StringAttr attrName = attr.getName();
61 // Kernel function attribute should be attached to functions.
62 if (attrName == NVVMDialect::getKernelFuncAttrName()) {
63 if (!isa<LLVM::LLVMFuncOp>(op)) {
64 return op->emitError() << "'" << NVVMDialect::getKernelFuncAttrName()
65 << "' attribute attached to unexpected op";
66 }
67 }
68 // If maxntid / reqntid / cluster_dim exist, it must be an array with max 3
69 // dim
70 if (attrName == NVVMDialect::getMaxntidAttrName() ||
71 attrName == NVVMDialect::getReqntidAttrName() ||
72 attrName == NVVMDialect::getClusterDimAttrName()) {
73 auto values = llvm::dyn_cast<DenseI32ArrayAttr>(attr.getValue());
74 if (!values || values.empty() || values.size() > 3) {
75 return op->emitError()
76 << "'" << attrName
77 << "' attribute must be integer array with maximum 3 index";
78 }
79 }
80 // If minctasm / maxnreg / cluster_max_blocks exist, it must be an integer
81 // attribute
82 if (attrName == NVVMDialect::getMinctasmAttrName() ||
83 attrName == NVVMDialect::getMaxnregAttrName() ||
84 attrName == NVVMDialect::getClusterMaxBlocksAttrName()) {
85 if (!llvm::dyn_cast<IntegerAttr>(attr.getValue())) {
86 return op->emitError()
87 << "'" << attrName << "' attribute must be integer constant";
88 }
89 }
90 // blocksareclusters must be used along with reqntid and cluster_dim
91 if (attrName == NVVMDialect::getBlocksAreClustersAttrName()) {
92 if (!op->hasDiscardableAttr(NVVMDialect::getReqntidAttrName()) ||
93 !op->hasDiscardableAttr(NVVMDialect::getClusterDimAttrName())) {
94 return op->emitError()
95 << "'" << attrName << "' attribute must be used along with " << "'"
96 << NVVMDialect::getReqntidAttrName() << "' and " << "'"
97 << NVVMDialect::getClusterDimAttrName() << "'";
98 }
99 }
100
101 return success();
102}

104LogicalResult NVVMDialect::verifyRegionArgAttribute(Operation *op,
105 unsigned regionIndex,
106 unsigned argIndex,
107 NamedAttribute argAttr) {
108 auto funcOp = dyn_cast<FunctionOpInterface>(op);
109 if (!funcOp)
110 return success();
111
112 bool isKernel = op->hasDiscardableAttr(NVVMDialect::getKernelFuncAttrName());
113 StringAttr attrName = argAttr.getName();
114 if (attrName == NVVM::NVVMDialect::getGridConstantAttrName()) {
115 if (!isKernel) {
116 return op->emitError()
117 << "'" << attrName
118 << "' attribute must be present only on kernel arguments";
119 }
120 if (!isa<UnitAttr>(argAttr.getValue()))
121 return op->emitError() << "'" << attrName << "' must be a unit attribute";
122 if (!funcOp.getArgAttr(argIndex, LLVM::LLVMDialect::getByValAttrName())) {
123 return op->emitError()
124 << "'" << attrName
125 << "' attribute requires the argument to also have attribute '"
126 << LLVM::LLVMDialect::getByValAttrName() << "'";
127 }
128 }
129
130 return success();
131}
