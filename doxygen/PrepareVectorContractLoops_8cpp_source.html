23static vector::ContractionOp getCandidate(scf::ForOp loop) {
24 if (loop.getNumRegionIterArgs() != 1) {
25 LDBG() << "reject loop at " << loop.getLoc() << ": expected one iter_arg";
26 return {};
27 }
28
29 auto yield = cast<scf::YieldOp>(loop.getBody()->getTerminator());
30 for (Operation &op : loop.getBody()->without_terminator()) {
31 auto contract = dyn_cast<vector::ContractionOp>(&op);
32 if (!contract)
33 continue;
34 if (contract.getKind() != vector::CombiningKind::ADD ||
35 contract.getAcc() != loop.getRegionIterArgs().front() ||
36 !contract.getAcc().hasOneUse() || !contract.getResult().hasOneUse() ||
37 yield->getNumOperands() != 1 ||
38 yield->getOperand(0) != contract.getResult()) {
39 LDBG() << "reject contraction at " << contract.getLoc()
40 << ": not a single loop-carried accumulation";
41 continue;
42 }
43
44 auto iterators = contract.getIteratorTypesArray();
45 auto maps = contract.getIndexingMapsArray();
46 MLIRContext *ctx = contract.getContext();
47 auto m = getAffineDimExpr(0, ctx);
48 auto n = getAffineDimExpr(1, ctx);
49 auto k = getAffineDimExpr(2, ctx);
50 auto par = vector::IteratorType::parallel;
51 auto red = vector::IteratorType::reduction;
52 if (llvm::equal(iterators, ArrayRef{par, par, red}) &&
53 maps[0] == AffineMap::get(3, 0, {m, k}, ctx) &&
54 maps[1] == AffineMap::get(3, 0, {k, n}, ctx) &&
55 maps[2] == AffineMap::get(3, 0, {m, n}, ctx))
56 return contract;
57
58 auto v = getAffineDimExpr(3, ctx);
59 if (x86::isInVnniLayout(contract, maps) &&
60 llvm::equal(iterators, ArrayRef{par, par, red, red}) &&
61 maps[0] == AffineMap::get(4, 0, {m, k, v}, ctx) &&
62 maps[1] == AffineMap::get(4, 0, {k, n, v}, ctx) &&
63 maps[2] == AffineMap::get(4, 0, {m, n}, ctx))
64 return contract;
65 LDBG() << "reject contraction at " << contract.getLoc()
66 << ": noncanonical matmul layout or iterators";
67 }
68 return {};
69}

71static bool isSimpleReadOutsideLoop(vector::TransferReadOp read,
72 scf::ForOp loop) {
73 return read && read->getParentOp() == loop &&
74 isa<MemRefType>(read.getBase().getType()) &&
75 loop.isDefinedOutsideOfLoop(read.getBase()) &&
76 read.getPermutationMap().isMinorIdentity() && !read.getMask() &&
77 !read.hasOutOfBoundsDim();
78}

82 vector::TransferReadOp read;
83 MemRefType type;
84 SmallVector<ReassociationIndices> reassociation;
85 int64_t lastIndex;
86 // Static view of a dynamic last dimension, or null.
87 MemRefType view;
88};

91static std::optional<ExpandedRead>
92getExpandableRead(vector::TransferReadOp read, unsigned factor,
93 scf::ForOp loop) {
94 if (!isSimpleReadOutsideLoop(read, loop))
95 return std::nullopt;
96 auto type = cast<MemRefType>(read.getBase().getType());
97 if (type.getElementType() != read.getVectorType().getElementType() ||
98 type.getRank() < 2)
99 return std::nullopt;
100 auto lastIndex = getConstantIntValue(read.getIndices().back());
101 if (!lastIndex || *lastIndex < 0 || *lastIndex % factor != 0)
102 return std::nullopt;
103
104 MemRefType view;
105 if (type.isDynamicDim(type.getRank() - 1)) {
106 SmallVector<int64_t> strides;
107 int64_t offset;
108 if (read.getVectorType().isScalable() ||
109 failed(type.getStridesAndOffset(strides, offset)))
110 return std::nullopt;
111 // The full dimension has a dynamic size, but this in-bounds read ends at
112 // a known index. Create a view whose last dimension ends there and give
113 // the divisibility check below a static size to work with.
114 SmallVector<int64_t> shape(type.getShape());
115 shape.back() = *lastIndex + read.getVectorType().getShape().back();
116 view = type = MemRefType::get(
117 shape, type.getElementType(),
118 StridedLayoutAttr::get(type.getContext(), offset, strides),
119 type.getMemorySpace());
120 }
121 if (type.getShape().back() % factor != 0)
122 return std::nullopt;
123
124 SmallVector<int64_t> shape(type.getShape());
125 shape.back() /= factor;
126 shape.push_back(factor);
127 SmallVector<ReassociationIndices> reassociation;
128 for (int64_t i = 0; i < type.getRank(); ++i)
129 reassociation.push_back(i == type.getRank() - 1
130 ? ReassociationIndices{i, i + 1}
131 : ReassociationIndices{i});
132 FailureOr<MemRefType> expanded =
133 memref::ExpandShapeOp::computeExpandedType(type, shape, reassociation);
134 if (failed(expanded))
135 return std::nullopt;
136 return ExpandedRead{read, *expanded, reassociation, *lastIndex / factor,
137 view};
138}

140static Value buildExpandedRead(PatternRewriter &rewriter, Location loc,
141 ExpandedRead &info, VectorType resultType) {
142 OpBuilder::InsertionGuard guard(rewriter);
143 rewriter.setInsertionPoint(info.read->getParentOp());
144 Value base = info.read.getBase();
145 if (info.view) {
146 // Unlike memref.subview, memref.reinterpret_cast is valid even when
147 // the loop runs zero times (which makes the view out of bounds).
148 auto metadata =
149 memref::ExtractStridedMetadataOp::create(rewriter, loc, base);
150 SmallVector<OpFoldResult> sizes = metadata.getConstifiedMixedSizes();
151 sizes.back() = rewriter.getIndexAttr(info.view.getShape().back());
152 base = memref::ReinterpretCastOp::create(
153 rewriter, loc, info.view, base, metadata.getConstifiedMixedOffset(),
154 sizes, metadata.getConstifiedMixedStrides());
155 }
156 Value expanded = memref::ExpandShapeOp::create(rewriter, loc, info.type, base,
157 info.reassociation);
158 Value lastIndex =
159 arith::ConstantIndexOp::create(rewriter, loc, info.lastIndex);
160 Value zero = arith::ConstantIndexOp::create(rewriter, loc, 0);
161 rewriter.setInsertionPoint(info.read);
162 SmallVector<Value> indices(info.read.getIndices());
163 indices.back() = lastIndex;
164 indices.push_back(zero);
165 auto packed = vector::TransferReadOp::create(
166 rewriter, loc, resultType, expanded, indices, info.read.getPadding());
167 packed.setInBoundsAttr(
168 rewriter.getBoolArrayAttr(SmallVector<bool>(resultType.getRank(), true)));
169 return packed;
170}

173// Fold RHS unpacking into a VNNI contraction:
174// - Match a packed RHS read transposed and flattened for a flat matmul.
175// - Split the LHS memref's K dimension and reread it in packed lanes.
176// A dynamic K dimension is first narrowed to the read's static extent.
177// - Contract the packed reads directly over K and the VNNI lane dimension.
178// For bf16, 32x32 LHS and RHS become 32x16x2 LHS and 16x32x2 RHS.
179struct FoldVNNIUnpacking : OpRewritePattern<vector::ContractionOp> {
180 using OpRewritePattern::OpRewritePattern;
181
182 LogicalResult matchAndRewrite(vector::ContractionOp contract,
183 PatternRewriter &rewriter) const override {
184 auto loop = dyn_cast<scf::ForOp>(contract->getParentOp());
185 if (!loop)
186 return rewriter.notifyMatchFailure(contract,
187 "not directly in a for loop");
188 if (getCandidate(loop) != contract)
189 return rewriter.notifyMatchFailure(
190 contract, "not the loop's canonical contraction");
191
192 auto lhsType = contract.getLhsType();
193 auto rhsType = contract.getRhsType();
194 auto accType = dyn_cast<VectorType>(contract.getAccType());
195 Type element = lhsType.getElementType();
196 if (!accType || lhsType.getRank() != 2 || rhsType.getRank() != 2 ||
197 accType.getRank() != 2)
198 return rewriter.notifyMatchFailure(
199 contract, "expected 2D operands and accumulator");
200 if (lhsType.isScalable() || rhsType.isScalable())
201 return rewriter.notifyMatchFailure(contract, "scalable operand vectors");
202 if (element != rhsType.getElementType())
203 return rewriter.notifyMatchFailure(contract,
204 "operand element types differ");
205 if (!(element.isBF16() || element.isInteger(8)))
206 return rewriter.notifyMatchFailure(contract,
207 "expected bf16 or i8 operands");
208 unsigned factor = 32 / element.getIntOrFloatBitWidth();
209 if (lhsType.getDimSize(1) != rhsType.getDimSize(0))
210 return rewriter.notifyMatchFailure(contract, "K dimensions differ");
211 if (lhsType.getDimSize(1) % factor != 0)
212 return rewriter.notifyMatchFailure(
213 contract, "K dimension is not divisible by VNNI factor");
214
215 auto shapeCast = contract.getRhs().getDefiningOp<vector::ShapeCastOp>();
216 auto transpose =
217 shapeCast ? shapeCast.getSource().getDefiningOp<vector::TransposeOp>()
218 : vector::TransposeOp();
219 if (!transpose || transpose->getParentOp() != loop)
220 return rewriter.notifyMatchFailure(
221 contract, "RHS is not an in-loop transpose followed by shape_cast");
222 VectorType rhsPackedType =
223 VectorType::get({lhsType.getDimSize(1) / factor, rhsType.getDimSize(1),
224 int64_t(factor)},
225 element);
226 auto sourceType = cast<VectorType>(transpose.getVector().getType());
227 if (sourceType.getRank() < rhsPackedType.getRank())
228 return rewriter.notifyMatchFailure(contract,
229 "RHS source rank is too small");
230 unsigned leading = sourceType.getRank() - rhsPackedType.getRank();
231 if (!llvm::all_of(sourceType.getShape().take_front(leading),
232 [](int64_t dim) { return dim == 1; }) ||
233 !llvm::equal(sourceType.getShape().drop_front(leading),
234 rhsPackedType.getShape()))
235 return rewriter.notifyMatchFailure(contract,
236 "RHS source is not VNNI packed");
237 for (unsigned i = 0; i < leading + 1; ++i)
238 if (transpose.getPermutation()[i] != i)
239 return rewriter.notifyMatchFailure(
240 contract, "transpose reorders leading or K dimensions");
241 if (transpose.getPermutation()[leading + 1] != leading + 2 ||
242 transpose.getPermutation()[leading + 2] != leading + 1)
243 return rewriter.notifyMatchFailure(
244 contract, "transpose does not swap N and VNNI lanes");
245
246 auto lhsRead = contract.getLhs().getDefiningOp<vector::TransferReadOp>();
247 auto lhs = getExpandableRead(lhsRead, factor, loop);
248 auto broadcast = transpose.getVector().getDefiningOp<vector::BroadcastOp>();
249 Value packedSource =
250 broadcast ? broadcast.getSource() : transpose.getVector();
251 if (broadcast && packedSource.getType() != rhsPackedType)
252 return rewriter.notifyMatchFailure(contract,
253 "broadcast source is not VNNI packed");
254 auto packedCast = packedSource.getDefiningOp<vector::ShapeCastOp>();
255 auto rhsRead =
256 packedCast
257 ? packedCast.getSource().getDefiningOp<vector::TransferReadOp>()
258 : packedSource.getDefiningOp<vector::TransferReadOp>();
259 if (!lhs)
260 return rewriter.notifyMatchFailure(
261 contract, "LHS read cannot be expanded into VNNI layout");
262 if (!isSimpleReadOutsideLoop(rhsRead, loop))
263 return rewriter.notifyMatchFailure(contract,
264 "RHS is not a simple transfer read");
265 std::optional<ExpandedRead> splitRhs;
266 if (packedCast) {
267 auto flatType = rhsRead.getVectorType();
268 if (flatType.getRank() != 2 ||
269 flatType.getDimSize(0) != rhsPackedType.getDimSize(0) ||
270 flatType.getDimSize(1) % factor != 0 ||
271 flatType.getDimSize(1) / factor != rhsPackedType.getDimSize(1))
272 return rewriter.notifyMatchFailure(
273 contract, "flat RHS shape is not VNNI compatible");
274 if (!(splitRhs = getExpandableRead(rhsRead, factor, loop)))
275 return rewriter.notifyMatchFailure(
276 contract, "RHS read cannot be expanded into VNNI layout");
277 } else {
278 if (rhsRead.getVectorType() != rhsPackedType)
279 return rewriter.notifyMatchFailure(contract,
280 "RHS read is not VNNI packed");
281 if (getConstantIntValue(rhsRead.getIndices().back()) != 0)
282 return rewriter.notifyMatchFailure(
283 contract, "packed RHS has nonzero inner offset");
284 }
285
286 Location loc = contract.getLoc();
287 VectorType lhsPackedType =
288 VectorType::get({lhsType.getDimSize(0), lhsType.getDimSize(1) / factor,
289 int64_t(factor)},
290 element);
291 Value packedLhs = buildExpandedRead(rewriter, loc, *lhs, lhsPackedType);
292 Value packedRhs =
293 splitRhs ? buildExpandedRead(rewriter, loc, *splitRhs, rhsPackedType)
294 : rhsRead.getResult();
295 rewriter.setInsertionPoint(contract);
296 auto ctx = rewriter.getContext();
297 auto m = rewriter.getAffineDimExpr(0);
298 auto n = rewriter.getAffineDimExpr(1);
299 auto k = rewriter.getAffineDimExpr(2);
300 auto v = rewriter.getAffineDimExpr(3);
301 auto vnniMaps =
302 rewriter.getAffineMapArrayAttr({AffineMap::get(4, 0, {m, k, v}, ctx),
303 AffineMap::get(4, 0, {k, n, v}, ctx),
304 AffineMap::get(4, 0, {m, n}, ctx)});
305 auto iterators = rewriter.getArrayAttr(
306 {vector::IteratorTypeAttr::get(ctx, vector::IteratorType::parallel),
307 vector::IteratorTypeAttr::get(ctx, vector::IteratorType::parallel),
308 vector::IteratorTypeAttr::get(ctx, vector::IteratorType::reduction),
309 vector::IteratorTypeAttr::get(ctx, vector::IteratorType::reduction)});
310 rewriter.replaceOpWithNewOp<vector::ContractionOp>(
311 contract, packedLhs, packedRhs, contract.getAcc(), vnniMaps, iterators,
312 contract.getKind(), contract.getFastmath());
313 return success();
314 }
315};
316} // namespace
