21/// Verifies that all elements of `array` are instances of `Attr`.
22template <class AttrT>
23static LogicalResult isArrayOf(Operation *op, ArrayAttr array) {
24 for (Attribute iter : array)
25 if (!isa<AttrT>(iter))
26 return op->emitOpError("expected op to return array of ")
27 << AttrT::getMnemonic() << " attributes";
28 return success();
29}

36 auto iface = cast<AccessGroupOpInterface>(op);

45// AliasAnalysisOpInterface
46//===----------------------------------------------------------------------===//
47
48LogicalResult
49mlir::LLVM::detail::verifyAliasAnalysisOpInterface(Operation *op) {
50 auto iface = cast<AliasAnalysisOpInterface>(op);
51
52 if (auto aliasScopes = iface.getAliasScopesOrNull())
53 if (failed(isArrayOf<AliasScopeAttr>(op, aliasScopes)))
54 return failure();
55
56 if (auto noAliasScopes = iface.getNoAliasScopesOrNull())
57 if (failed(isArrayOf<AliasScopeAttr>(op, noAliasScopes)))
58 return failure();
59
60 ArrayAttr tags = iface.getTBAATagsOrNull();
61 if (!tags)
62 return success();
63
64 return isArrayOf<TBAATagAttr>(op, tags);
65}

85// RangeOpInterface
86//===----------------------------------------------------------------------===//
87
88LogicalResult mlir::LLVM::detail::verifyRangeOpInterface(Operation *op) {
89 auto iface = cast<RangeOpInterface>(op);
90
91 ConstantRangeAttr range = iface.getRangeOrNull();
92 if (!range)
93 return success();
94 if (op->getNumResults() != 1)
95 return op->emitOpError("expected op to have a single result");
96 auto intType =
97 dyn_cast<IntegerType>(getElementTypeOrSelf(op->getResult(0).getType()));
98 if (!intType || intType.getWidth() != range.getLower().getBitWidth())
99 return op->emitOpError(
100 "expected range bitwidth to match the integer (element) result type");
101 if (range.getLower() == range.getUpper())
102 return op->emitOpError("expected range to be neither empty nor full");
103 return success();
104}

158SmallVector<Value> mlir::LLVM::CallOp::getAccessedOperands() {
159 // Note: This must not use `getArgOperands`, which excludes the variadic
160 // arguments of a call to a variadic callee. Those are passed to the callee
161 // and may well be accessed by it.
162 Operation::operand_range operands = getCalleeOperands();
163 // In an indirect call, the first callee operand is the callee itself.
164 if (!getCallee().has_value() && !operands.empty())
165 operands = operands.drop_front();
166 return llvm::filter_to_vector(
167 operands, [](Value arg) { return isa<LLVMPointerType>(arg.getType()); });
168}
