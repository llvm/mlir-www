139/// The clauses of a construct can be given once per device type. Of the values
140/// that reach a given device type, the ones naming it are the most specific,
141/// then the ones naming every device type, then the ones given before any
142/// device_type clause.
143SmallVector<DeviceType, 3> getDeviceTypesByPrecedence(DeviceType deviceType);

145namespace detail {
146/// The constructs carrying a `device_type` clause hold their async and wait
147/// clauses per device type. The others, such as `acc.enter_data` or
148/// `acc.kernel_environment`, hold a single value for each clause, spelling the
149/// value-less form either `asyncOnly`/`waitOnly` or `async`/`wait`.
150template <typename OpTy>
151using has_device_type_clauses_t =
152 decltype(std::declval<OpTy>().hasAsyncOnly(DeviceType::None));
153template <typename OpTy>
154using has_async_only_t = decltype(std::declval<OpTy>().getAsyncOnly());
155template <typename OpTy>
156using has_wait_only_t = decltype(std::declval<OpTy>().getWaitOnly());
157} // namespace detail
158
159/// Returns the value the `async` clause of \p op names for \p deviceType, and
160/// sets \p asyncOnly when the clause names no queue. The value is the one the
161/// operation holds, so a caller in a conversion has to remap it.
162template <typename OpTy>
163Value getAsyncClauseValue(OpTy op, DeviceType deviceType, bool &asyncOnly) {
164 asyncOnly = false;
165 if constexpr (llvm::is_detected<detail::has_device_type_clauses_t,
166 OpTy>::value) {
167 for (DeviceType candidate : getDeviceTypesByPrecedence(deviceType)) {
168 if (op.hasAsyncOnly(candidate)) {
169 asyncOnly = true;
170 return {};
171 }
172 if (Value asyncValue = op.getAsyncValue(candidate))
173 return asyncValue;
174 }
175 return {};
176 } else if constexpr (llvm::is_detected<detail::has_async_only_t,
177 OpTy>::value) {
178 asyncOnly = op.getAsyncOnly();
179 return op.getAsyncOperand();
180 } else {
181 asyncOnly = op.getAsync();
182 return op.getAsyncOperand();
183 }
184}

186/// Appends to \p waitValues the queues the `wait` clause of \p op names for
187/// \p deviceType, and returns the device type the clause naming them is given
188/// for - a clause naming no queue waits for every one of them. Returns
189/// std::nullopt when \p op gives no such clause for \p deviceType. The values
190/// are the ones the operation holds, so a caller in a conversion has to remap
191/// them.
192template <typename OpTy>
193std::optional<DeviceType>
194getWaitClauseValues(OpTy op, DeviceType deviceType,
195 SmallVectorImpl<Value> &waitValues) {
196 if constexpr (llvm::is_detected<detail::has_device_type_clauses_t,
197 OpTy>::value) {
198 for (DeviceType candidate : getDeviceTypesByPrecedence(deviceType)) {
199 if (op.hasWaitOnly(candidate))
200 return candidate;
201 auto values = op.getWaitValues(candidate);
202 if (!values.empty()) {
203 llvm::append_range(waitValues, values);
204 return candidate;
205 }
206 }
207 return std::nullopt;
208 } else {
209 llvm::append_range(waitValues, op.getWaitOperands());
210 bool waitsForEveryQueue;
211 if constexpr (llvm::is_detected<detail::has_wait_only_t, OpTy>::value)
212 waitsForEveryQueue = op.getWaitOnly();
213 else
214 waitsForEveryQueue = op.getWait();
215 if (!waitsForEveryQueue && waitValues.empty())
216 return std::nullopt;
217 // The operation holds a single wait clause, which no device_type clause
218 // narrows to a device type.
219 return DeviceType::None;
220 }
221}

223/// Returns whether the `wait` clause \p op gives for \p deviceType carries a
224/// devnum modifier, which selects the device the queues belong to. Only the
225/// clause given for that device type is asked, so a caller passes the device
226/// type getWaitClauseValues took the queues from.
227template <typename OpTy>
228bool hasWaitDevnum(OpTy op, DeviceType deviceType) {
229 if constexpr (llvm::is_detected<detail::has_device_type_clauses_t,
230 OpTy>::value) {
231 return static_cast<bool>(op.getWaitDevnum(deviceType));
232 } else {
233 return static_cast<bool>(op.getWaitDevnum());
234 }
235}

237/// Returns the queue that the runtime calls of \p op run on, from the `async`
238/// clause it gives for \p deviceType.
239template <typename OpTy>
240Value getAsyncQueue(OpTy op, DeviceType deviceType,
241 ConversionPatternRewriter &rewriter,
242 const ACCRuntimeCallConfig &config) {
243 bool asyncOnly = false;
244 Value asyncValue = getAsyncClauseValue(op, deviceType, asyncOnly);
245 if (asyncValue)
246 asyncValue = rewriter.getRemappedValue(asyncValue);
247 return getAsyncQueue(op.getLoc(), asyncValue, asyncOnly, rewriter, config);
248}

250/// Emits the wait that a `wait` clause on \p op asks for before the runtime
251/// calls of the construct, waiting on \p asyncQueue for the queues the clause
252/// names, or for every queue when it names none. Nothing is emitted when there
253/// is no such clause for \p deviceType. A devnum modifier, which selects the
254/// device the queues belong to, is reported through \p accSupport as not yet
255/// implemented.
256template <typename OpTy>
257LogicalResult
258emitWaitClause(OpTy op, DeviceType deviceType, Value asyncQueue,
259 ConversionPatternRewriter &rewriter, OpenACCSupport &accSupport,
260 Region &globalSymbolRegion, SymbolTable &symbolTable,
261 const ACCRuntimeCallConfig &config) {
262 SmallVector<Value> waitValues;
263 std::optional<DeviceType> clauseDeviceType =
264 getWaitClauseValues(op, deviceType, waitValues);
265 if (!clauseDeviceType)
266 return success();
267 if (hasWaitDevnum(op, *clauseDeviceType)) {
268 (void)accSupport.emitNYI(op.getLoc(), "wait clause with a devnum modifier");
269 return failure();
270 }
271 for (Value &waitValue : waitValues)
272 waitValue = rewriter.getRemappedValue(waitValue);
273 return emitWaitCall(op.getLoc(), waitValues, asyncQueue, rewriter,
274 globalSymbolRegion, symbolTable, config);
275}
