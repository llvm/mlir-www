38bool LLVM::DbgValueOp::canUsesBeRemoved(
39 const SmallPtrSetImpl<OpOperand *> &blockingUses,
40 SmallVectorImpl<OpOperand *> &newBlockingUses,
41 const DataLayout &dataLayout) {
42 // There is only one operand that we can remove the use of.
43 if (blockingUses.size() != 1)
44 return false;
45
46 return (*blockingUses.begin())->get() == getValue();
47}
48
49DeletionKind LLVM::DbgValueOp::removeBlockingUses(
50 const SmallPtrSetImpl<OpOperand *> &blockingUses, OpBuilder &builder) {
51 // builder by default is after '*this', but we need it before '*this'.
52 builder.setInsertionPoint(*this);
53
54 // Rather than dropping the debug value, replace it with undef to preserve the
55 // debug local variable info. This allows the debugger to inform the user that
56 // the variable has been optimized out.
57 auto undef =
58 UndefOp::create(builder, getValue().getLoc(), getValue().getType());
59 getValueMutable().assign(undef);
60 return DeletionKind::Keep;
61}
