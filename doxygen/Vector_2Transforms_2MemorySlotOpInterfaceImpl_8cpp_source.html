9// This file implements Mem2Reg-related interfaces that let a statically-shaped
10// memref be promoted into a single vector SSA value, provided every access to
11// the buffer is a whole-buffer read or write (or a whole-sub-region access of
12// such a buffer via a subview). With these models, Mem2Reg replaces the memory
13// slot with a vector value, threading it as the reaching definition:
14//
15// * `vector.transfer_read` of the whole buffer becomes a use of the current
16// vector value; `vector.transfer_write` of the whole buffer becomes a new
17// definition of it (see the `PromotableMemOpInterface` models below).
18//
19// Accesses that are not whole-(sub-)buffer -- masked or partial transfers,
20// non-zero transfer indices -- are left untouched, so the buffer is not
21// promoted.
22//
23//===----------------------------------------------------------------------===//

39/// Returns whether `xferOp` accesses exactly the whole contents of `slot`, so
40/// it can act as a plain whole-buffer load/store during Mem2Reg.
41static bool
42isWholeBufferTransfer(VectorTransferOpInterface xferOp, const MemorySlot &slot,
43 const SmallPtrSetImpl<OpOperand *> &blockingUses) {
44 // The sole blocking use must be the slot pointer as the transfer's base.
45 if (blockingUses.size() != 1)
46 return false;
47 Value blockingUse = (*blockingUses.begin())->get();
48 if (blockingUse != slot.ptr || xferOp.getBase() != slot.ptr)
49 return false;
50
51 // Reject the tensor form (already implied, since slot pointers are memrefs).
52 if (!isa<MemRefType>(xferOp.getBase().getType()))
53 return false;
54
55 // Exact type match pins rank/extents/element type/scalable dims.
56 if (xferOp.getVectorType() != slot.valueType)
57 return false;
58
59 // Access must start at the buffer origin in every dimension.
60 for (Value index : xferOp.getIndices()) {
61 std::optional<int64_t> constIndex = getConstantIntValue(index);
62 if (!constIndex || *constIndex != 0)
63 return false;
64 }
65
66 // Identity map: no broadcast or transpose.
67 if (!xferOp.getPermutationMap().isIdentity())
68 return false;
69
70 // All dimensions must be in bounds. An out-of-bounds dimension means the
71 // transfer reaches past the buffer, so a read would materialize padding
72 // rather than buffer contents and a write would only cover part of the
73 // buffer: in neither case does the transfer stand in for the whole slot.
74 if (xferOp.hasOutOfBoundsDim())
75 return false;
76
77 // A mask could make the access partial.
78 if (xferOp.getMask())
79 return false;
80
81 return true;
82}

90struct TransferReadOpMemOpModel
91 : public PromotableMemOpInterface::ExternalModel<TransferReadOpMemOpModel,
92 vector::TransferReadOp> {
93 bool loadsFrom(Operation *op, const MemorySlot &slot) const {
94 return cast<vector::TransferReadOp>(op).getBase() == slot.ptr;
95 }
96
97 bool storesTo(Operation *op, const MemorySlot &slot) const { return false; }
98
99 Value getStored(Operation *op, const MemorySlot &slot, OpBuilder &builder,
100 Value reachingDef, const DataLayout &dataLayout) const {
101 llvm_unreachable("getStored should not be called on TransferReadOp");
102 }
103
104 bool canUsesBeRemoved(Operation *op, const MemorySlot &slot,
105 const SmallPtrSetImpl<OpOperand *> &blockingUses,
106 SmallVectorImpl<OpOperand *> &newBlockingUses,
107 const DataLayout &dataLayout) const {
108 return isWholeBufferTransfer(cast<VectorTransferOpInterface>(op), slot,
109 blockingUses);
110 }
111
112 DeletionKind
113 removeBlockingUses(Operation *op, const MemorySlot &slot,
114 const SmallPtrSetImpl<OpOperand *> &blockingUses,
115 OpBuilder &builder, Value reachingDefinition,
116 const DataLayout &dataLayout) const {
117 // Whole-buffer read: replace the loaded vector with the reaching
118 // definition.
119 cast<vector::TransferReadOp>(op).getVector().replaceAllUsesWith(
120 reachingDefinition);
121 return DeletionKind::Delete;
122 }
123};
124
125struct TransferWriteOpMemOpModel
126 : public PromotableMemOpInterface::ExternalModel<TransferWriteOpMemOpModel,
127 vector::TransferWriteOp> {
128 bool loadsFrom(Operation *op, const MemorySlot &slot) const { return false; }
129
130 bool storesTo(Operation *op, const MemorySlot &slot) const {
131 return cast<vector::TransferWriteOp>(op).getBase() == slot.ptr;
132 }
133
134 Value getStored(Operation *op, const MemorySlot &slot, OpBuilder &builder,
135 Value reachingDef, const DataLayout &dataLayout) const {
136 return cast<vector::TransferWriteOp>(op).getValueToStore();
137 }
138
139 bool canUsesBeRemoved(Operation *op, const MemorySlot &slot,
140 const SmallPtrSetImpl<OpOperand *> &blockingUses,
141 SmallVectorImpl<OpOperand *> &newBlockingUses,
142 const DataLayout &dataLayout) const {
143 // No self-store guard needed: a vector value can never equal a memref slot.
144 return isWholeBufferTransfer(cast<VectorTransferOpInterface>(op), slot,
145 blockingUses);
146 }
147
148 DeletionKind
149 removeBlockingUses(Operation *op, const MemorySlot &slot,
150 const SmallPtrSetImpl<OpOperand *> &blockingUses,
151 OpBuilder &builder, Value reachingDefinition,
152 const DataLayout &dataLayout) const {
153 return DeletionKind::Delete;
154 }
155};
