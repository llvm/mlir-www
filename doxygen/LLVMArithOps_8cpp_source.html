28/// Returns a boolean type that has the same shape as `type`. It supports both
29/// fixed size vectors as well as scalable vectors.
30Type mlir::LLVM::getI1SameShape(Type type) {
31 Type i1Type = IntegerType::get(type.getContext(), 1);
32 if (LLVM::isCompatibleVectorType(type))
33 return LLVM::getVectorType(i1Type, LLVM::getVectorNumElements(type));
34 return i1Type;
35}

41template <typename PredicateAttr, typename Predicate>
42static ParseResult parseCmpPredicateImpl(
43 OpAsmParser &parser, PredicateAttr &predicate,
44 function_ref<std::optional<Predicate>(StringRef)> symbolize) {
45 std::string spelling;
46 SMLoc loc = parser.getCurrentLocation();
47 if (parser.parseString(&spelling))
48 return failure();
49 std::optional<Predicate> value = symbolize(spelling);
50 if (!value)
51 return parser.emitError(loc)
52 << "'" << spelling
53 << "' is an incorrect value of the 'predicate' attribute";
54 predicate = PredicateAttr::get(parser.getContext(), *value);
55 return success();
56}

91OpFoldResult ICmpOp::fold(FoldAdaptor adaptor) {
92 if (getPredicate() != ICmpPredicate::eq &&
93 getPredicate() != ICmpPredicate::ne)
94 return {};
95
96 // cmpi(eq/ne, x, x) -> true/false
97 if (getLhs() == getRhs())
98 return getBoolAttribute(getType(), getContext(),
99 getPredicate() == ICmpPredicate::eq);
100
101 // cmpi(eq/ne, alloca, null) -> false/true
102 if (getLhs().getDefiningOp<AllocaOp>() && getRhs().getDefiningOp<ZeroOp>())
103 return getBoolAttribute(getType(), getContext(),
104 getPredicate() == ICmpPredicate::ne);
105
106 // cmpi(eq/ne, null, alloca) -> cmpi(eq/ne, alloca, null)
107 if (getLhs().getDefiningOp<ZeroOp>() && getRhs().getDefiningOp<AllocaOp>()) {
108 Value lhs = getLhs();
109 Value rhs = getRhs();
110 getLhsMutable().assign(rhs);
111 getRhsMutable().assign(lhs);
112 return getResult();
113 }
114
115 return {};
116}

122OpFoldResult LLVM::ShlOp::fold(FoldAdaptor adaptor) {
123 auto rhs = dyn_cast_or_null<IntegerAttr>(adaptor.getRhs());
124 if (!rhs)
125 return {};
126
127 if (rhs.getValue().uge(getLhs().getType().getIntOrFloatBitWidth()))
128 return {}; // TODO: Fold into poison.
129
130 auto lhs = dyn_cast_or_null<IntegerAttr>(adaptor.getLhs());
131 if (!lhs)
132 return {};
133
134 return IntegerAttr::get(getType(), lhs.getValue().shl(rhs.getValue()));
135}

141OpFoldResult LLVM::OrOp::fold(FoldAdaptor adaptor) {
142 auto lhs = dyn_cast_or_null<IntegerAttr>(adaptor.getLhs());
143 if (!lhs)
144 return {};
145
146 auto rhs = dyn_cast_or_null<IntegerAttr>(adaptor.getRhs());
147 if (!rhs)
148 return {};
149
150 return IntegerAttr::get(getType(), lhs.getValue() | rhs.getValue());
151}

156Speculation::Speculatability UDivOp::getSpeculatability() {
157 // X / 0 => UB
158 Value divisor = getRhs();
159 if (matchPattern(divisor, m_IntRangeWithoutZeroU()))
160 return Speculation::Speculatable;
161
162 return Speculation::NotSpeculatable;
163}

165//===----------------------------------------------------------------------===//
166// SDivOp
167//===----------------------------------------------------------------------===//
168Speculation::Speculatability SDivOp::getSpeculatability() {
169 // This function conservatively assumes that all signed division by -1 are
170 // not speculatable.
171 // X / 0 => UB
172 // INT_MIN / -1 => UB
173 Value divisor = getRhs();
174 if (matchPattern(divisor, m_IntRangeWithoutZeroS()) &&
175 matchPattern(divisor, m_IntRangeWithoutNegOneS()))
176 return Speculation::Speculatable;
177
178 return Speculation::NotSpeculatable;
179}
