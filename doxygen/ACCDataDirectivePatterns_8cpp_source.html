9// Lowering of the OpenACC data constructs - data, enter data, exit data,
10// update and declare - to the data entry points of the OpenACC runtime. Each
11// construct becomes one call per point where it establishes or tears down its
12// mappings, which carries the argument arrays its operands are packed into,
13// the queue its async clause selects and, before them, the wait its wait
14// clause asks for. An if clause guards the calls it applies to, and the body
15// of a structured construct is left in place of the construct.
16//
17//===----------------------------------------------------------------------===//

33/// Emits the call to the mapping entry point \p fn for \p mappingOperands, the
34/// data-clause operands of a data construct, paired with their type converted
35/// values in \p convertedOperands.
36static LogicalResult emitDataRuntimeCall(
37 Location loc, RuntimeFunction fn, ValueRange mappingOperands,
38 ValueRange convertedOperands, ACCDataCallKind callKind,
39 ConversionPatternRewriter &rewriter, OpenACCSupport &accSupport,
40 Region &globalSymbolRegion, SymbolTable &symbolTable,
41 const ACCRuntimeCallConfig &config, ArrayRef<Value> trailingArguments) {
42 if (mappingOperands.empty())
43 return success();
44
45 Operation *firstMapOp = mappingOperands.front().getDefiningOp();
46 ACCDataRuntimeArgs runtimeArgs;
47 if (failed(emitACCDataRuntimeArgs(
48 loc, mappingOperands, convertedOperands, rewriter, globalSymbolRegion,
49 accSupport, config, runtimeArgs, callKind, &symbolTable)))
50 return rewriter.notifyMatchFailure(
51 firstMapOp, "unsupported OpenACC data-clause operand");
52
53 SmallVector<Value> arguments = runtimeArgs.getCallArgs();
54 arguments.append(trailingArguments.begin(), trailingArguments.end());
55 return createRuntimeCall(loc, rewriter, globalSymbolRegion, symbolTable, fn,
56 config, arguments);
57}

59/// Returns the action that the declare-action recipe holding \p op carries
60/// out, which the recipe states by naming itself in the matching slot of its
61/// `acc.declare_action` attribute. Returns a null attribute when \p op is not
62/// in such a recipe.
63static DeclareActionAttr getEnclosingDeclareAction(Operation *op) {
64 auto func = dyn_cast<FunctionOpInterface>(op);
65 if (!func)
66 func = op->getParentOfType<FunctionOpInterface>();
67 if (!func)
68 return {};
69 return func->getAttrOfType<DeclareActionAttr>(getDeclareActionAttrName());
70}

76 config.getAsyncSyncRuntimeValue());

125struct DataOpLowering : public ACCDataDirectivePattern<DataOp> {
126 using ACCDataDirectivePattern<DataOp>::ACCDataDirectivePattern;
127
128 LogicalResult
129 matchAndRewrite(DataOp op, DataOp::Adaptor adaptor,
130 ConversionPatternRewriter &rewriter) const override {
131 // The mappings are established where the clause operations are, and torn
132 // down where the exit operations of the construct were.
133 ValueRange mappingOperands = op.getDataClauseOperands();
134 Location beginLoc = mappingOperands.empty()
135 ? op.getLoc()
136 : mappingOperands.front().getLoc();
137 Location endLoc =
138 acc::getMappingExitLoc(mappingOperands).value_or(beginLoc);
139 Value asyncQueue = getAsyncQueue(op, rewriter);
140 if (failed(emitWaitClause(op, asyncQueue, rewriter)))
141 return failure();
142
143 auto emitBegin = [&]() {
144 return emitMappingCall(op, adaptor.getDataClauseOperands(), beginLoc,
145 RuntimeFunction::ACCRTL_tgt_acc_data_begin,
146 ACCDataCallKind::DataEnter, asyncQueue, rewriter);
147 };
148 if (failed(acc::emitGuardedByIfCond(beginLoc, adaptor.getIfCond(), rewriter,
149 emitBegin)))
150 return failure();
151
152 rewriter.setInsertionPointAfter(op);
153 auto emitEnd = [&]() {
154 return emitMappingCall(op, adaptor.getDataClauseOperands(), endLoc,
155 RuntimeFunction::ACCRTL_tgt_acc_data_end,
156 ACCDataCallKind::DataExit, asyncQueue, rewriter);
157 };
158 if (failed(acc::emitGuardedByIfCond(endLoc, adaptor.getIfCond(), rewriter,
159 emitEnd)))
160 return failure();
161
162 acc::spliceConstructRegion(op, op.getRegion(), rewriter);
163 rewriter.eraseOp(op);
164 return success();
165 }
166};

168/// Lowering shared by the directives that map their operands with a single
169/// call: enter data, exit data and update.
170template <typename OpTy, RuntimeFunction fn, ACCDataCallKind callKind>
171struct ACCDataCallLowering : public ACCDataDirectivePattern<OpTy> {
172 using ACCDataDirectivePattern<OpTy>::ACCDataDirectivePattern;
173
174 LogicalResult
175 matchAndRewrite(OpTy op, typename OpTy::Adaptor adaptor,
176 ConversionPatternRewriter &rewriter) const override {
177 Location loc = op.getLoc();
178 Value asyncQueue = this->getAsyncQueue(op, rewriter);
179 if (failed(this->emitWaitClause(op, asyncQueue, rewriter)))
180 return failure();
181
182 auto emit = [&]() {
183 return this->emitMappingCall(op, adaptor.getDataClauseOperands(), loc, fn,
184 callKind, asyncQueue, rewriter);
185 };
186 if (failed(
187 acc::emitGuardedByIfCond(loc, adaptor.getIfCond(), rewriter, emit)))
188 return failure();
189 rewriter.eraseOp(op);
190 return success();
191 }
192};

204/// Structured `acc.declare_enter` (token used by a matching exit) maps like
205/// `acc.data` enter. In a declare-action recipe the object the directive
206/// names is allocated by the host, so only the device mirror of it is
207/// allocated here. An unstructured enter registers the objects it names with
208/// `__tgt_acc_declare`.
209struct DeclareEnterOpLowering : public ConvertOpToLLVMPattern<DeclareEnterOp> {
210 DeclareEnterOpLowering(const LLVMTypeConverter &converter,
211 OpenACCSupport &accSupport, Region &globalSymbolRegion,
212 SymbolTable &symbolTable,
213 const ACCRuntimeCallConfig &config,
214 PatternBenefit benefit = 1)
215 : ConvertOpToLLVMPattern<DeclareEnterOp>(converter, benefit),
216 accSupport(accSupport), globalSymbolRegion(globalSymbolRegion),
217 symbolTable(symbolTable), config(config) {}
218
219 LogicalResult
220 matchAndRewrite(DeclareEnterOp op, DeclareEnterOp::Adaptor adaptor,
221 ConversionPatternRewriter &rewriter) const override {
222 Location loc = op.getLoc();
223 ValueRange mappingOperands = op.getDataClauseOperands();
224 ValueRange convertedOperands = adaptor.getDataClauseOperands();
225 Value asyncQueue = getDeclareAsyncQueue(loc, rewriter, config);
226
227 LogicalResult emitted = success();
228 if (getEnclosingDeclareAction(op)) {
229 emitted = emitDataRuntimeCall(
230 loc, RuntimeFunction::ACCRTL_tgt_acc_mirror_alloc, mappingOperands,
231 convertedOperands, ACCDataCallKind::DataEnter, rewriter, accSupport,
232 globalSymbolRegion, symbolTable, config, {});
233 } else if (!op.getToken().use_empty()) {
234 emitted = emitDataRuntimeCall(
235 loc, RuntimeFunction::ACCRTL_tgt_acc_data_begin, mappingOperands,
236 convertedOperands, ACCDataCallKind::DataEnter, rewriter, accSupport,
237 globalSymbolRegion, symbolTable, config, {asyncQueue});
238 } else {
239 Value binaryDescriptor =
240 config.createDeclareBinaryDescriptor(loc, rewriter);
241 emitted = emitDataRuntimeCall(
242 loc, RuntimeFunction::ACCRTL_tgt_acc_declare, mappingOperands,
243 convertedOperands, ACCDataCallKind::DataEnter, rewriter, accSupport,
244 globalSymbolRegion, symbolTable, config,
245 {asyncQueue, binaryDescriptor});
246 }
247 if (failed(emitted))
248 return failure();
249
250 if (op.getToken().use_empty()) {
251 rewriter.eraseOp(op);
252 } else {
253 // The token only ties a structured enter to its exit. Keep a converted
254 // placeholder until the exit is erased; it carries no runtime state.
255 Type tokenTy = getTypeConverter()->convertType(op.getToken().getType());
256 rewriter.replaceOp(op, LLVM::ZeroOp::create(rewriter, loc, tokenTy));
257 }
258 return success();
259 }
260
261 OpenACCSupport &accSupport;
262 Region &globalSymbolRegion;
263 SymbolTable &symbolTable;
264 ACCRuntimeCallConfig config;
265};

313/// A data clause operation only describes a mapping; the runtime calls of the
314/// construct carry everything it said, so the operation itself goes away once
315/// they are in place. Whatever else referred to the mapped object is given the
316/// address of that object, which is what the clause result stood for.
317template <typename OpTy>
318struct DataEntryOpLowering : public ConvertOpToLLVMPattern<OpTy> {
319 using ConvertOpToLLVMPattern<OpTy>::ConvertOpToLLVMPattern;
320 using OpAdaptor = typename OpTy::Adaptor;
321
322 LogicalResult
323 matchAndRewrite(OpTy op, OpAdaptor adaptor,
324 ConversionPatternRewriter &rewriter) const override {
325 rewriter.replaceOp(op, adaptor.getVar());
326 return success();
327 }
328};

330/// An `acc.getdeviceptr` clause stands for the device address of an object
331/// whose mapping was established elsewhere. Where it names the object of a
332/// data exit operation, the runtime call of that construct carries the host
333/// address and looks the device one up itself, so the clause is replaced by
334/// the address of the object as any other data entry clause is. Read as a
335/// value, it is the runtime that has to be asked for the address.
336struct GetDevicePtrOpLowering
337 : public ConvertOpToLLVMPattern<acc::GetDevicePtrOp> {
338 GetDevicePtrOpLowering(const LLVMTypeConverter &converter,
339 OpenACCSupport &accSupport, Region &globalSymbolRegion,
340 SymbolTable &symbolTable,
341 const ACCRuntimeCallConfig &config,
342 PatternBenefit benefit = 1)
343 : ConvertOpToLLVMPattern<acc::GetDevicePtrOp>(converter, benefit),
344 accSupport(accSupport), globalSymbolRegion(globalSymbolRegion),
345 symbolTable(symbolTable), config(config) {}
346
347 /// Returns whether \p user only states the mapping the clause describes,
348 /// which is what a data exit operation and the construct holding the clause
349 /// do.
350 static bool isMappingUse(Operation *user) {
351 return isa<ACC_DATA_EXIT_OPS, ACC_COMPUTE_AND_DATA_CONSTRUCT_OPS,
352 acc::KernelEnvironmentOp>(user);
353 }
354
355 LogicalResult
356 matchAndRewrite(acc::GetDevicePtrOp op, OpAdaptor adaptor,
357 ConversionPatternRewriter &rewriter) const override {
358 bool readAsValue = llvm::any_of(
359 op->getUsers(), [](Operation *user) { return !isMappingUse(user); });
360 if (!readAsValue) {
361 rewriter.replaceOp(op, adaptor.getVar());
362 return success();
363 }
364 // The mapping hands the host address of the object to the runtime call of
365 // the construct, so the clause cannot stand for the device address at the
366 // same time.
367 if (llvm::any_of(op->getUsers(), isMappingUse)) {
368 (void)accSupport.emitNYI(
369 op.getLoc(), "device address read from a clause that also states a "
370 "mapping of the object");
371 return failure();
372 }
373
374 FailureOr<Value> devicePtr = acc::emitGetDevicePtrCall(
375 op, adaptor.getVar(), /*ifPresent=*/false, rewriter, globalSymbolRegion,
376 symbolTable, config);
377 if (failed(devicePtr))
378 return failure();
379 rewriter.replaceOp(op, *devicePtr);
380 return success();
381 }
382
383 OpenACCSupport &accSupport;
384 Region &globalSymbolRegion;
385 SymbolTable &symbolTable;
386 ACCRuntimeCallConfig config;
387};

406void mlir::configureACCDataDirectiveConversionLegality(
407 ConversionTarget &target) {
408 // The map entries themselves stay legal: a construct removes the ones it
409 // consumes, and the ones describing a construct that is lowered elsewhere
410 // have to survive this conversion. An `acc.getdeviceptr` read as a value is
411 // the exception, as the address it stands for is only known to the runtime,
412 // so leaving it behind would state a mapping that nothing carries out.
413 target.addIllegalOp<acc::DataOp, acc::EnterDataOp, acc::ExitDataOp,
414 acc::UpdateOp, acc::DeclareEnterOp, acc::DeclareExitOp,
415 acc::GetDevicePtrOp>();
416}
