28/// Returns whether `ptr` is a pointer into the address space `targetAS`.
29inline bool isPtrInAddrSpace(mlir::Value ptr, NVVMMemorySpace targetAS) {
30 auto ptrTy = llvm::cast<LLVM::LLVMPointerType>(ptr.getType());
31 return ptrTy.getAddressSpace() == static_cast<unsigned>(targetAS);
32}
33
34/// Returns whether `ptr` is a pointer into the shared memory of the CTA.
35inline bool isPtrInSharedCTASpace(mlir::Value ptr) {
36 return isPtrInAddrSpace(ptr, NVVMMemorySpace::Shared);
37}
