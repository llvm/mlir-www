14// so include this header only if most of the headers below are needed anyway.
15// Otherwise include:
16// - LLVMDialectDecl.h for the dialect, its attributes and its types,
17// - LLVMInterfaces.h for its interfaces,
18// - LLVM<Partition>Ops.h for the ops of an op partition, e.g. LLVMArithOps.h
19// for `llvm.add` and LLVMMemoryOps.h for `llvm.load`.
