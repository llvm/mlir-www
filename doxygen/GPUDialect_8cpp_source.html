86/// 8 4 0
87/// Example mask : 0 0 0 1 1 0 1 0 0
88///
89/// Active physical (resp. logical) is 2 (0), 4 (1) and 5 (2).
90/// Logical id for e.g. 5 (2) constructs filter (1 << 5 - 1).
91///
92/// Example mask : 0 0 0 1 1 0 1 0 0
93/// Example filter: 0 0 0 0 1 1 1 1 1
94/// Intersection : 0 0 0 0 1 0 1 0 0
95/// PopCnt : 2
96Value GPUMappingMaskAttr::createLogicalLinearMappingId(
97 OpBuilder &b, Value physicalLinearMappingId) const {
98 Location loc = physicalLinearMappingId.getLoc();
99 Value mask =
100 arith::ConstantOp::create(b, loc, b.getI64IntegerAttr(getMask()));
101 Value one = arith::ConstantOp::create(b, loc, b.getI64IntegerAttr(1));
102 Value filter = arith::ShLIOp::create(b, loc, one, physicalLinearMappingId);
103 filter = arith::SubIOp::create(b, loc, filter, one);
104 Value filteredId = arith::AndIOp::create(b, loc, mask, filter);
105 return math::CtPopOp::create(b, loc, filteredId);
106}

108/// 8 4 0
109/// Example mask : 0 0 0 1 1 0 1 0 0
110///
111/// Active physical (resp. logical) is 2 (0), 4 (1) and 5 (2).
112/// Logical id for e.g. 5 (2) constructs filter (1 << 5).
113///
114/// Example mask : 0 0 0 1 1 0 1 0 0
115/// Example filter: 0 0 0 1 0 0 0 0 0
116/// Intersection : 0 0 0 1 0 0 0 0 0
117/// Cmp : 1
118Value GPUMappingMaskAttr::createIsActiveIdPredicate(
119 OpBuilder &b, Value physicalLinearMappingId) const {
120 Location loc = physicalLinearMappingId.getLoc();
121 Value mask =
122 arith::ConstantOp::create(b, loc, b.getI64IntegerAttr(getMask()));
123 Value one = arith::ConstantOp::create(b, loc, b.getI64IntegerAttr(1));

126 Value zero = arith::ConstantOp::create(b, loc, b.getI64IntegerAttr(0));
127 return arith::CmpIOp::create(b, loc, arith::CmpIPredicate::ne, filtered,
128 zero);

198Type GPUDialect::parseType(DialectAsmParser &parser) const {
199 // Parse the main keyword for the type.
200 StringRef keyword;
201 if (parser.parseKeyword(&keyword))
202 return Type();
203 MLIRContext *context = getContext();
204
205 // Handle 'async token' types.
206 if (keyword == "async.token")
207 return AsyncTokenType::get(context);
208
209 if (keyword == "mma_matrix") {
210 SMLoc beginLoc = parser.getNameLoc();
211
212 // Parse '<'.
213 if (parser.parseLess())
214 return nullptr;
215
216 // Parse the size and elementType.
217 SmallVector<int64_t> shape;
218 Type elementType;
219 if (parser.parseDimensionList(shape, /*allowDynamic=*/false) ||
220 parser.parseType(elementType))
221 return nullptr;
222
223 // Parse ','
224 if (parser.parseComma())
225 return nullptr;
226
227 // Parse operand.
228 std::string operand;
229 if (failed(parser.parseOptionalString(&operand)))
230 return nullptr;
231
232 // Parse '>'.
233 if (parser.parseGreater())
234 return nullptr;
235
236 return MMAMatrixType::getChecked(mlir::detail::getDefaultDiagnosticEmitFn(
237 parser.getEncodedSourceLoc(beginLoc)),
238 shape, elementType, operand);
239 }
240
241 if (keyword == "named_barrier")
242 return NamedBarrierType::get(context);
243
244 if (keyword == getSparseHandleKeyword(SparseHandleKind::DnTensor))
245 return SparseDnTensorHandleType::get(context);
246 if (keyword == getSparseHandleKeyword(SparseHandleKind::SpMat))
247 return SparseSpMatHandleType::get(context);
248 if (keyword == getSparseHandleKeyword(SparseHandleKind::SpGEMMOp))
249 return SparseSpGEMMOpHandleType::get(context);
250
251 parser.emitError(parser.getNameLoc(), "unknown gpu type: " + keyword);
252 return Type();
253}

256void GPUDialect::printType(Type type, DialectAsmPrinter &os) const {
257 TypeSwitch<Type>(type)
258 .Case<AsyncTokenType>([&](Type) { os << "async.token"; })
259 .Case<NamedBarrierType>([&](Type) { os << "named_barrier"; })
260 .Case<SparseDnTensorHandleType>([&](Type) {
261 os << getSparseHandleKeyword(SparseHandleKind::DnTensor);
262 })
263 .Case<SparseSpMatHandleType>(
264 [&](Type) { os << getSparseHandleKeyword(SparseHandleKind::SpMat); })
265 .Case<SparseSpGEMMOpHandleType>([&](Type) {
266 os << getSparseHandleKeyword(SparseHandleKind::SpGEMMOp);
267 })
268 .Case([&](MMAMatrixType fragTy) {
269 os << "mma_matrix<";
270 auto shape = fragTy.getShape();
271 for (auto dim = shape.begin(), e = shape.end() - 1; dim != e; ++dim)
272 os << *dim << 'x';
273 os << shape.back() << 'x' << fragTy.getElementType();
274 os << ", \"" << fragTy.getOperand() << "\"" << '>';
275 })
276 .DefaultUnreachable("unexpected 'gpu' type kind");
277}

311/// Parses an optional list of async operands with an optional leading keyword.
312/// (`async`)? (`[` ssa-id-list `]`)?
313///
314/// This method is used by the tablegen assembly format for async ops as well.
315static ParseResult parseAsyncDependencies(
316 OpAsmParser &parser, Type &asyncTokenType,
317 SmallVectorImpl<OpAsmParser::UnresolvedOperand> &asyncDependencies) {
318 auto loc = parser.getCurrentLocation();
319 if (succeeded(parser.parseOptionalKeyword("async"))) {
320 if (parser.getNumResults() == 0)
321 return parser.emitError(loc, "needs to be named when marked 'async'");
322 asyncTokenType = parser.getBuilder().getType<AsyncTokenType>();
323 }
324 return parser.parseOperandList(asyncDependencies,
325 OpAsmParser::Delimiter::OptionalSquare);
326}

343// GPU Memory attributions functions shared by LaunchOp and GPUFuncOp.
344/// Parses a GPU function memory attribution.
345///
346/// memory-attribution ::= (`workgroup` `(` ssa-id-and-type-list `)`)?
347/// (`private` `(` ssa-id-and-type-list `)`)?
348///
349/// Note that this function parses only one of the two similar parts, with the
350/// keyword provided as argument.
351static ParseResult
352parseAttributions(OpAsmParser &parser, StringRef keyword,
353 SmallVectorImpl<OpAsmParser::Argument> &args) {
354 // If we could not parse the keyword, just assume empty list and succeed.
355 if (failed(parser.parseOptionalKeyword(keyword)))
356 return success();
357
358 return parser.parseArgumentList(args, OpAsmParser::Delimiter::Paren,
359 /*allowType=*/true);
360}

362static void printAttributions(OpAsmPrinter &p, StringRef keyword,
363 ArrayRef<BlockArgument> values,
364 ArrayAttr attributes = {}) {
365 if (values.empty())
366 return;
367
368 p << ' ' << keyword << '(';
369 llvm::interleaveComma(
370 llvm::enumerate(values), p, [&p, attributes](auto pair) {
371 BlockArgument v = pair.value();
372 p << v << " : " << v.getType();
373
374 size_t attributionIndex = pair.index();
375 DictionaryAttr attrs;
376 if (attributes && attributionIndex < attributes.size())
377 attrs = llvm::cast<DictionaryAttr>(attributes[attributionIndex]);
378 if (attrs)
379 p.printOptionalAttrDict(attrs.getValue());
380 });
381 p << ')';
382}

384/// Verifies a GPU function memory attribution.
385static LogicalResult verifyAttributions(Operation *op,
386 ArrayRef<BlockArgument> attributions,
387 gpu::AddressSpace memorySpace) {
388 for (Value v : attributions) {
389 auto type = llvm::dyn_cast<MemRefType>(v.getType());
390 if (!type)
391 return op->emitOpError() << "expected memref type in attribution";
392
393 // We can only verify the address space if it hasn't already been lowered
394 // from the AddressSpaceAttr to a target-specific numeric value.
395 auto addressSpace =
396 llvm::dyn_cast_or_null<gpu::AddressSpaceAttr>(type.getMemorySpace());
397 if (!addressSpace)
398 continue;
399 if (addressSpace.getValue() != memorySpace)
400 return op->emitOpError()
401 << "expected memory space " << stringifyAddressSpace(memorySpace)
402 << " in attribution";
403 }
404 return success();
405}

411static LogicalResult verifyReduceOpAndType(gpu::AllReduceOperation opName,
412 Type resType) {
413 using Kind = gpu::AllReduceOperation;
414 if (llvm::is_contained({Kind::MINNUMF, Kind::MAXNUMF, Kind::MINIMUMF,
415 Kind::MAXIMUMF, Kind::MINIMUMNUMF, Kind::MAXIMUMNUMF},
416 opName)) {
417 if (!isa<FloatType>(resType))
418 return failure();
419 }
420
421 if (llvm::is_contained({Kind::MINSI, Kind::MINUI, Kind::MAXSI, Kind::MAXUI,
422 Kind::AND, Kind::OR, Kind::XOR},
423 opName)) {
424 if (!isa<IntegerType>(resType))
425 return failure();
426 }
427
428 return success();
429}

431LogicalResult gpu::AllReduceOp::verifyRegions() {
432 if (getBody().empty() != getOp().has_value())
433 return emitError("expected either an op attribute or a non-empty body");
434 if (!getBody().empty()) {
435 if (getBody().getNumArguments() != 2)
436 return emitError("expected two region arguments");
437 for (auto argument : getBody().getArguments()) {
438 if (argument.getType() != getType())
439 return emitError("incorrect region argument type");
440 }
441 unsigned yieldCount = 0;
442 for (Block &block : getBody()) {
443 if (auto yield = dyn_cast<gpu::YieldOp>(block.getTerminator())) {
444 if (yield.getNumOperands() != 1)
445 return emitError("expected one gpu.yield operand");
446 if (yield.getOperand(0).getType() != getType())
447 return emitError("incorrect gpu.yield type");
448 ++yieldCount;
449 }
450 }
451 if (yieldCount == 0)
452 return emitError("expected gpu.yield op in region");
453 } else {
454 gpu::AllReduceOperation opName = *getOp();
455 if (failed(verifyReduceOpAndType(opName, getType()))) {
456 return emitError() << '`' << gpu::stringifyAllReduceOperation(opName)
457 << "` reduction operation is not compatible with type "
458 << getType();
459 }
460 }
461
462 return success();
463}

490LogicalResult gpu::SubgroupReduceOp::verify() {
491 Type elemType = getType();
492 if (auto vecTy = dyn_cast<VectorType>(elemType)) {
493 if (vecTy.isScalable())
494 return emitOpError() << "is not compatible with scalable vector types";
495
496 elemType = vecTy.getElementType();
497 }
498
499 gpu::AllReduceOperation opName = getOp();
500 if (failed(verifyReduceOpAndType(opName, elemType))) {
501 return emitError() << '`' << gpu::stringifyAllReduceOperation(opName)
502 << "` reduction operation is not compatible with type "
503 << getType();
504 }
505
506 auto clusterSize = getClusterSize();
507 if (clusterSize) {
508 uint32_t size = *clusterSize;
509 if (!llvm::isPowerOf2_32(size)) {
510 return emitOpError() << "cluster size " << size
511 << " is not a power of two";
512 }
513 }
514
515 uint32_t stride = getClusterStride();
516 if (stride != 1 && !clusterSize) {
517 return emitOpError() << "cluster stride can only be specified if cluster "
518 "size is specified";
519 }
520 if (!llvm::isPowerOf2_32(stride)) {
521 return emitOpError() << "cluster stride " << stride
522 << " is not a power of two";
523 }
524
525 return success();
526}

544void gpu::addAsyncDependency(Operation *op, Value token) {
545 op->insertOperands(0, {token});
546 if (!op->template hasTrait<OpTrait::AttrSizedOperandSegments>())
547 return;
548 auto attrName =
549 OpTrait::AttrSizedOperandSegments<void>::getOperandSegmentSizeAttr();
550 auto sizeAttr = dyn_cast_or_null<DenseI32ArrayAttr>(
551 op->getInherentAttr(attrName).value_or(Attribute{}));
552
553 // Async dependencies is the only variadic operand.
554 if (!sizeAttr)
555 return;
556
557 SmallVector<int32_t, 8> sizes(sizeAttr.asArrayRef());
558 ++sizes.front();
559 op->setInherentAttr(StringAttr::get(op->getContext(), attrName),
560 Builder(op->getContext()).getDenseI32ArrayAttr(sizes));
561}

567void LaunchOp::build(OpBuilder &builder, OperationState &result,
568 Value gridSizeX, Value gridSizeY, Value gridSizeZ,
569 Value getBlockSizeX, Value getBlockSizeY,
570 Value getBlockSizeZ, Value dynamicSharedMemorySize,
571 Type asyncTokenType, ValueRange asyncDependencies,
572 Value asyncObject, TypeRange workgroupAttributions,
573 TypeRange privateAttributions, Value clusterSizeX,
574 Value clusterSizeY, Value clusterSizeZ,
575 FlatSymbolRefAttr module, FlatSymbolRefAttr function) {
576 OpBuilder::InsertionGuard g(builder);
577
578 if (!workgroupAttributions.empty())
579 result.addAttribute(
580 getWorkgroupAttributionsAttrName(result.name),
581 builder.getI64IntegerAttr(workgroupAttributions.size()));
582
583 // Add Op operands.
584 result.addOperands(asyncDependencies);
585 if (asyncTokenType)
586 result.types.push_back(builder.getType<AsyncTokenType>());
587
588 // Add grid and block sizes as op operands, followed by the data operands.
589 result.addOperands({gridSizeX, gridSizeY, gridSizeZ, getBlockSizeX,
590 getBlockSizeY, getBlockSizeZ});
591 if (clusterSizeX)
592 result.addOperands(clusterSizeX);
593 if (clusterSizeY)
594 result.addOperands(clusterSizeY);
595 if (clusterSizeZ)
596 result.addOperands(clusterSizeZ);
597 if (dynamicSharedMemorySize)
598 result.addOperands(dynamicSharedMemorySize);
599 if (asyncObject)
600 result.addOperands(asyncObject);
601
602 // Add optional module and function attributes.
603 if (module)
604 result.addAttribute(getModuleAttrName(result.name), module);
605 if (function)
606 result.addAttribute(getFunctionAttrName(result.name), function);
607
608 // Create a kernel body region with kNumConfigRegionAttributes + N memory
609 // attributions, where the first kNumConfigRegionAttributes arguments have
610 // `index` type and the rest have the same types as the data operands.
611 Region *kernelRegion = result.addRegion();
612 Block *body = builder.createBlock(kernelRegion);
613 // TODO: Allow passing in proper locations here.
614 for (unsigned i = 0; i < kNumConfigRegionAttributes; ++i)
615 body->addArgument(builder.getIndexType(), result.location);
616 // Add WorkGroup & Private attributions to the region arguments.
617 for (Type argTy : workgroupAttributions)
618 body->addArgument(argTy, result.location);
619 for (Type argTy : privateAttributions)
620 body->addArgument(argTy, result.location);
621 // Fill OperandSegmentSize Attribute.
622 SmallVector<int32_t, 12> segmentSizes(12, 1);
623 segmentSizes.front() = asyncDependencies.size();
624 segmentSizes[7] = clusterSizeX ? 1 : 0;
625 segmentSizes[8] = clusterSizeY ? 1 : 0;
626 segmentSizes[9] = clusterSizeZ ? 1 : 0;
627 segmentSizes[10] = dynamicSharedMemorySize ? 1 : 0;
628 segmentSizes[11] = asyncObject ? 1 : 0;
629 result.addAttribute(getOperandSegmentSizeAttr(),
630 builder.getDenseI32ArrayAttr(segmentSizes));
631}

713LogicalResult LaunchOp::verifyRegions() {
714 // Kernel launch takes kNumConfigOperands leading operands for grid/block
715 // sizes and transforms them into kNumConfigRegionAttributes region arguments
716 // for block/thread identifiers and grid/block sizes.
717 if (getBody().empty()) {
718 return emitOpError("body region is empty");
719 }
720 unsigned actualNumRegionArgs = getBody().getNumArguments();
721 unsigned expectedNumRegionArgs =
722 getNumConfigRegionAttributes() + getNumWorkgroupAttributions();
723 if (actualNumRegionArgs < expectedNumRegionArgs) {
724 return emitOpError("expected at least ")
725 << expectedNumRegionArgs << " region arguments, but got "
726 << actualNumRegionArgs;
727 }
728
729 // Verify Attributions Address Spaces.
730 if (failed(verifyAttributions(getOperation(), getWorkgroupAttributionBBArgs(),
731 GPUDialect::getWorkgroupAddressSpace())) ||
732 failed(verifyAttributions(getOperation(), getPrivateAttributions(),
733 GPUDialect::getPrivateAddressSpace())))
734 return failure();
735
736 // Block terminators without successors are expected to exit the kernel region
737 // and must be `gpu.terminator`.
738 for (Block &block : getBody()) {
739 if (block.empty())
740 continue;
741 if (block.back().getNumSuccessors() != 0)
742 continue;
743 if (!isa<gpu::TerminatorOp>(&block.back())) {
744 return block.back()
745 .emitError()
746 .append("expected '", gpu::TerminatorOp::getOperationName(),
747 "' or a terminator with successors")
748 .attachNote(getLoc())
749 .append("in '", LaunchOp::getOperationName(), "' body region");
750 }
751 }
752
753 return success();
754}

825// Parse the size assignment blocks for blocks and threads. These have the form
826// (%region_arg, %region_arg, %region_arg) in
827// (%region_arg = %operand, %region_arg = %operand, %region_arg = %operand)
828// where %region_arg are percent-identifiers for the region arguments to be
829// introduced further (SSA defs), and %operand are percent-identifiers for the
830// SSA value uses.
831static ParseResult
832parseSizeAssignment(OpAsmParser &parser,
833 MutableArrayRef<OpAsmParser::UnresolvedOperand> sizes,
834 MutableArrayRef<OpAsmParser::UnresolvedOperand> regionSizes,
835 MutableArrayRef<OpAsmParser::UnresolvedOperand> indices,
836 StringRef keyword) {
837 assert(indices.size() == 3 && "space for three indices expected");
838 SmallVector<OpAsmParser::UnresolvedOperand, 3> args;
839 if (parser.parseOperandList(args, OpAsmParser::Delimiter::Paren,
840 /*allowResultNumber=*/false) ||
841 parser.parseKeyword("in") || parser.parseLParen())
842 return failure();
843
844 if (args.size() != 3) {
845 return parser.emitError(parser.getNameLoc())
846 << keyword << " expects 3 arguments, but got " << args.size();
847 }
848 std::move(args.begin(), args.end(), indices.begin());
849
850 for (int i = 0; i < 3; ++i) {
851 if (i != 0 && parser.parseComma())
852 return failure();
853 if (parser.parseOperand(regionSizes[i], /*allowResultNumber=*/false) ||
854 parser.parseEqual() || parser.parseOperand(sizes[i]))
855 return failure();
856 }
857
858 return parser.parseRParen();
859}

861/// Parses a Launch operation.
862/// operation ::= `gpu.launch` (`<` ssa-use `:` type `>`)?
863/// (`async` `[` ssa-id-list `]`)?
864/// (`clusters` `(` ssa-id-list `)` `in` ssa-reassignment)?
865/// `blocks` `(` ssa-id-list `)` `in` ssa-reassignment
866/// `threads` `(` ssa-id-list `)` `in` ssa-reassignment
867/// (`dynamic_shared_memory_size` ssa-use)?
868/// (`module(` symbol-ref-id `)`)?
869/// (`function(` symbol-ref-id `)`)?
870/// memory-attribution
871/// region attr-dict?
872/// ssa-reassignment ::= `(` ssa-id `=` ssa-use (`,` ssa-id `=` ssa-use)* `)`
873ParseResult LaunchOp::parse(OpAsmParser &parser, OperationState &result) {
874 // Sizes of the grid and block.
875 SmallVector<OpAsmParser::UnresolvedOperand, LaunchOp::kNumConfigOperands>
876 sizes(LaunchOp::kNumConfigOperands);
877
878 // Region arguments to be created.
879 SmallVector<OpAsmParser::UnresolvedOperand, 16> regionArgs(
880 LaunchOp::kNumConfigRegionAttributes);
881
882 // Parse optional asyncObject: < value : type >
883 OpAsmParser::UnresolvedOperand asyncObjectOperand;
884 Type asyncObjectType;
885 bool hasAsyncObject = false;
886 if (succeeded(parser.parseOptionalLess())) {
887 hasAsyncObject = true;
888 if (parser.parseOperand(asyncObjectOperand) || parser.parseColon() ||
889 parser.parseType(asyncObjectType) || parser.parseGreater())
890 return failure();
891 }
892
893 // Parse optional async dependencies.
894 SmallVector<OpAsmParser::UnresolvedOperand, 4> asyncDependencies;
895 Type asyncTokenType;
896 if (failed(
897 parseAsyncDependencies(parser, asyncTokenType, asyncDependencies)) ||
898 parser.resolveOperands(asyncDependencies, asyncTokenType,
899 result.operands))
900 return failure();
901 if (parser.getNumResults() > 0) {
902 if (!asyncTokenType)
903 return parser.emitError(
904 parser.getNameLoc(),
905 "gpu.launch requires 'async' keyword to return a value");
906 result.types.push_back(asyncTokenType);
907 }
908
909 bool hasCluster = false;
910 if (succeeded(parser.parseOptionalKeyword(LaunchOp::getClustersKeyword()))) {
911 hasCluster = true;
912 sizes.resize(9);
913 regionArgs.resize(18);
914 }
915 MutableArrayRef<OpAsmParser::UnresolvedOperand> sizesRef(sizes);
916 MutableArrayRef<OpAsmParser::UnresolvedOperand> regionArgsRef(regionArgs);
917
918 // Last three segment assigns the cluster size. In the region argument
919 // list, this is last 6 arguments.
920 if (hasCluster) {
921 if (parseSizeAssignment(
922 parser, sizesRef.drop_front(6), regionArgsRef.slice(15, 3),
923 regionArgsRef.slice(12, 3), LaunchOp::getClustersKeyword()))
924 return failure();
925 }
926 // Parse the size assignment segments: the first segment assigns grid sizes
927 // and defines values for block identifiers; the second segment assigns block
928 // sizes and defines values for thread identifiers. In the region argument
929 // list, identifiers precede sizes, and block-related values precede
930 // thread-related values.
931 if (parser.parseKeyword(LaunchOp::getBlocksKeyword()) ||
932 parseSizeAssignment(parser, sizesRef.take_front(3),
933 regionArgsRef.slice(6, 3), regionArgsRef.slice(0, 3),
934 LaunchOp::getBlocksKeyword()) ||
935 parser.parseKeyword(LaunchOp::getThreadsKeyword()) ||
936 parseSizeAssignment(parser, sizesRef.drop_front(3),
937 regionArgsRef.slice(9, 3), regionArgsRef.slice(3, 3),
938 LaunchOp::getThreadsKeyword()) ||
939 parser.resolveOperands(sizes, parser.getBuilder().getIndexType(),
940 result.operands))
941 return failure();
942
943 OpAsmParser::UnresolvedOperand dynamicSharedMemorySize;
944 bool hasDynamicSharedMemorySize = false;
945 if (!parser.parseOptionalKeyword(
946 LaunchOp::getDynamicSharedMemorySizeKeyword())) {
947 hasDynamicSharedMemorySize = true;
948 if (parser.parseOperand(dynamicSharedMemorySize) ||
949 parser.resolveOperand(dynamicSharedMemorySize,
950 parser.getBuilder().getI32Type(),
951 result.operands))
952 return failure();
953 }
954
955 // Resolve the asyncObject operand
956 if (hasAsyncObject && parser.resolveOperand(asyncObjectOperand,
957 asyncObjectType, result.operands))
958 return failure();
959
960 // Parse optional module attribute.
961 StringRef moduleAttrName = getModuleAttrName(result.name);
962 if (succeeded(parser.parseOptionalKeyword(moduleAttrName))) {
963 FlatSymbolRefAttr moduleSymbol;
964 if (parser.parseLParen() ||
965 parser.parseAttribute(moduleSymbol, Type(), moduleAttrName,
966 result.attributes) ||
967 parser.parseRParen())
968 return failure();
969 }
970 // Parse optional function attribute.
971 StringRef functionAttrName = getFunctionAttrName(result.name);
972 if (succeeded(parser.parseOptionalKeyword(functionAttrName))) {
973 FlatSymbolRefAttr funcSymbol;
974 if (parser.parseLParen() ||
975 parser.parseAttribute(funcSymbol, Type(), functionAttrName,
976 result.attributes) ||
977 parser.parseRParen())
978 return failure();
979 }
980
981 // Parse optional cooperative keyword.
982 if (succeeded(parser.parseOptionalKeyword("cooperative")))
983 result.addAttribute("cooperative", parser.getBuilder().getUnitAttr());
984
985 // Create the region arguments: fixed launch-config args (`index`), then
986 // workgroup / private attribution args. The workgroup count is stored in the
987 // inherent `workgroup_attributions` attribute when non-zero.
988 Type index = parser.getBuilder().getIndexType();
989 SmallVector<Type, LaunchOp::kNumConfigRegionAttributes> dataTypes(
990 LaunchOp::kNumConfigRegionAttributes + 6, index);
991
992 SmallVector<OpAsmParser::Argument> regionArguments;
993 for (auto ssaValueAndType : llvm::zip(regionArgs, dataTypes)) {
994 OpAsmParser::Argument arg;
995 arg.ssaName = std::get<0>(ssaValueAndType);
996 arg.type = std::get<1>(ssaValueAndType);
997 regionArguments.push_back(arg);
998 }
999
1000 Builder &builder = parser.getBuilder();
1001 // Parse workgroup memory attributions.
1002 if (failed(parseAttributions(parser, LaunchOp::getWorkgroupKeyword(),
1003 regionArguments)))
1004 return failure();
1005
1006 // Store the number of operands we just parsed as the number of workgroup
1007 // memory attributions.
1008 unsigned numWorkgroupAttrs = regionArguments.size() -
1009 LaunchOp::kNumConfigRegionAttributes -
1010 (hasCluster ? 6 : 0);
1011 if (numWorkgroupAttrs != 0)
1012 result.addAttribute(LaunchOp::getWorkgroupAttributionsAttrName(result.name),
1013 builder.getI64IntegerAttr(numWorkgroupAttrs));
1014
1015 // Parse private memory attributions.
1016 if (failed(parseAttributions(parser, LaunchOp::getPrivateKeyword(),
1017 regionArguments)))
1018 return failure();
1019
1020 // Introduce the body region and parse it. The region has
1021 // kNumConfigRegionAttributes arguments that correspond to
1022 // block/thread identifiers and grid/block sizes, all having `index` type.
1023 Region *body = result.addRegion();
1024 if (parser.parseRegion(*body, regionArguments) ||
1025 parser.parseOptionalAttrDict(result.attributes))
1026 return failure();
1027
1028 SmallVector<int32_t, 12> segmentSizes(12, 1);
1029 segmentSizes.front() = asyncDependencies.size();
1030
1031 if (!hasCluster) {
1032 segmentSizes[7] = 0;
1033 segmentSizes[8] = 0;
1034 segmentSizes[9] = 0;
1035 }
1036 segmentSizes[10] = hasDynamicSharedMemorySize ? 1 : 0;
1037 segmentSizes[11] = hasAsyncObject ? 1 : 0;
1038 result.addAttribute(LaunchOp::getOperandSegmentSizeAttr(),
1039 parser.getBuilder().getDenseI32ArrayAttr(segmentSizes));
1040 return success();
1041}

1043/// Simplify the gpu.launch when the range of a thread or block ID is
1044/// trivially known to be one.
1045struct FoldLaunchArguments : public OpRewritePattern<LaunchOp> {
1046 using OpRewritePattern<LaunchOp>::OpRewritePattern;
1047 LogicalResult matchAndRewrite(LaunchOp op,
1048 PatternRewriter &rewriter) const override {
1049 // If the range implies a single value for `id`, replace `id`'s uses by
1050 // zero.
1051 Value zero;
1052 bool simplified = false;
1053 auto constPropIdUses = [&](Value id, Value size) {
1054 // Check if size is trivially one.
1055 if (!matchPattern(size, m_One()))
1056 return;
1057 if (id.getUses().empty())
1058 return;
1059 if (!simplified) {
1060 // Create a zero value the first time.
1061 OpBuilder::InsertionGuard guard(rewriter);
1062 rewriter.setInsertionPointToStart(&op.getBody().front());
1063 zero =
1064 arith::ConstantIndexOp::create(rewriter, op.getLoc(), /*value=*/0);
1065 }
1066 rewriter.replaceAllUsesWith(id, zero);
1067 simplified = true;
1068 };
1069 constPropIdUses(op.getBlockIds().x, op.getGridSizeX());
1070 constPropIdUses(op.getBlockIds().y, op.getGridSizeY());
1071 constPropIdUses(op.getBlockIds().z, op.getGridSizeZ());
1072 constPropIdUses(op.getThreadIds().x, op.getBlockSizeX());
1073 constPropIdUses(op.getThreadIds().y, op.getBlockSizeY());
1074 constPropIdUses(op.getThreadIds().z, op.getBlockSizeZ());
1075
1076 return success(simplified);
1077 }
1078};

1081 MLIRContext *context) {
1082 rewrites.add<FoldLaunchArguments>(context);
1083}

1219LogicalResult
1220LaunchFuncOp::verifySymbolUses(SymbolTableCollection &symbolTable) {
1221 LaunchFuncOp launchOp = *this;
1222 Operation *table = SymbolTable::getNearestSymbolTable(launchOp);
1223 // GPU modules cannot be nested within each other, escape to resolve the name.
1224 if (isa<GPUModuleOp>(table))
1225 table = SymbolTable::getNearestSymbolTable(table->getParentOp());
1226
1227 // Ignore launches that are nested more or less deep than functions in the
1228 // module we are currently checking.
1229 if (!launchOp->getParentOp() ||
1230 launchOp->getParentOp()->getParentOp() != table)
1231 return success();
1232
1233 // Ignore launch ops with missing attributes here. The errors will be
1234 // reported by the verifiers of those ops.
1235 if (!launchOp.getKernelAttr())
1236 return success();
1237
1238 // Check that `launch_func` refers to a well-formed GPU kernel container.
1239 StringAttr kernelContainerName = launchOp.getKernelModuleName();
1240 Operation *kernelContainer =
1241 symbolTable.lookupNearestSymbolFrom(table, kernelContainerName);
1242 if (!kernelContainer)
1243 return launchOp.emitOpError()
1244 << "kernel container '" << kernelContainerName.getValue()
1245 << "' is undefined";
1246
1247 // If the container is a GPU binary op return success.
1248 if (isa<BinaryOp>(kernelContainer))
1249 return success();
1250
1251 auto kernelModule = dyn_cast<GPUModuleOp>(kernelContainer);
1252 if (!kernelModule)
1253 return launchOp.emitOpError()
1254 << "kernel module '" << kernelContainerName.getValue()
1255 << "' is undefined";
1256
1257 // Check that `launch_func` refers to a well-formed kernel function.
1258 Operation *kernelFunc = symbolTable.lookupNearestSymbolFrom(
1259 kernelModule, launchOp.getKernelName());
1260 if (!kernelFunc)
1261 return launchOp.emitOpError("kernel function '")
1262 << launchOp.getKernel() << "' is undefined";
1263 auto kernelConvertedFunction = dyn_cast<FunctionOpInterface>(kernelFunc);
1264 if (!kernelConvertedFunction) {
1265 InFlightDiagnostic diag = launchOp.emitOpError()
1266 << "referenced kernel '" << launchOp.getKernel()
1267 << "' is not a function";
1268 diag.attachNote(kernelFunc->getLoc()) << "see the kernel definition here";
1269 return diag;
1270 }
1271
1272 if (!GPUDialect::isKernel(kernelFunc))
1273 return launchOp.emitOpError("kernel function is missing the '")
1274 << GPUDialect::getKernelFuncAttrName() << "' attribute";
1275
1276 // TODO: If the kernel isn't a GPU function (which happens during separate
1277 // compilation), do not check type correspondence as it would require the
1278 // verifier to be aware of the type conversion.
1279 auto kernelGPUFunction = dyn_cast<gpu::GPUFuncOp>(kernelFunc);
1280 if (!kernelGPUFunction)
1281 return success();
1282

1293 return launchOp.emitOpError("type of function argument ")
1294 << i << " does not match";
1295 }
1296 }
1297
1298 return success();
1299}

1395/// Remove gpu.barrier after gpu.barrier, the threads are already synchronized!
1396static LogicalResult eraseRedundantGpuBarrierOps(BarrierOp op,
1397 PatternRewriter &rewriter) {
1398 auto nextOp = dyn_cast_or_null<BarrierOp>(op->getNextNode());
1399 if (!nextOp)
1400 return failure();
1401
1402 // Cannot merge barriers of different scopes.
1403 if (op.getScope() != nextOp.getScope())
1404 return failure();
1405
1406 // Cannot merge named barriers unless both refer to the same handle.
1407 if (op.getNamedBarrier() != nextOp.getNamedBarrier())
1408 return failure();
1409
1410 std::optional<ArrayAttr> thisMemfence = op.getAddressSpaces();
1411 std::optional<ArrayAttr> nextMemfence = nextOp.getAddressSpaces();
1412
1413 if (thisMemfence) {
1414 rewriter.modifyOpInPlace(op, [&]() {
1415 if (!nextMemfence) {
1416 op.removeAddressSpacesAttr();
1417 return;
1418 }
1419 // Fast path - merge where the two barriers fence the same spaces.
1420 if (*thisMemfence == *nextMemfence) {
1421 return;
1422 }
1423
1424 llvm::SmallSetVector<Attribute, 4> mergedSpaces;
1425 for (Attribute attr : *thisMemfence)
1426 mergedSpaces.insert(attr);
1427 for (Attribute attr : *nextMemfence)
1428 mergedSpaces.insert(attr);
1429 op.setAddressSpacesAttr(rewriter.getArrayAttr(mergedSpaces.takeVector()));
1430 });
1431 }
1432
1433 rewriter.eraseOp(nextOp);
1434 return success();
1435}
1436
1437void BarrierOp::getCanonicalizationPatterns(RewritePatternSet &results,
1438 MLIRContext *context) {
1439 results.add(eraseRedundantGpuBarrierOps);
1440}

1515/// Parses a GPU function memory attribution.
1516///
1517/// memory-attribution ::= (`workgroup` `(` ssa-id-and-type-list `)`)?
1518/// (`private` `(` ssa-id-and-type-list `)`)?
1519///
1520/// Note that this function parses only one of the two similar parts, with the
1521/// keyword provided as argument.
1522static ParseResult
1523parseAttributions(OpAsmParser &parser, StringRef keyword,
1524 SmallVectorImpl<OpAsmParser::Argument> &args,
1525 Attribute &attributionAttrs) {
1526 // If we could not parse the keyword, just assume empty list and succeed.
1527 if (failed(parser.parseOptionalKeyword(keyword)))
1528 return success();
1529
1530 size_t existingArgs = args.size();
1531 ParseResult result =
1532 parser.parseArgumentList(args, OpAsmParser::Delimiter::Paren,
1533 /*allowType=*/true, /*allowAttrs=*/true);
1534 if (failed(result))
1535 return result;
1536
1537 bool hadAttrs = llvm::any_of(ArrayRef(args).drop_front(existingArgs),
1538 [](const OpAsmParser::Argument &arg) -> bool {
1539 return arg.attrs && !arg.attrs.empty();
1540 });
1541 if (!hadAttrs) {
1542 attributionAttrs = nullptr;
1543 return result;
1544 }
1545
1546 Builder &builder = parser.getBuilder();
1547 SmallVector<Attribute> attributionAttrsVec;
1548 for (const auto &argument : ArrayRef(args).drop_front(existingArgs)) {
1549 if (!argument.attrs)
1550 attributionAttrsVec.push_back(builder.getDictionaryAttr({}));
1551 else
1552 attributionAttrsVec.push_back(argument.attrs);
1553 }
1554 attributionAttrs = builder.getArrayAttr(attributionAttrsVec);
1555 return result;
1556}

1558/// Parses a GPU function.
1559///
1560/// <operation> ::= `gpu.func` symbol-ref-id `(` argument-list `)`
1561/// (`->` function-result-list)? memory-attribution `kernel`?
1562/// function-attributes? region
1563ParseResult GPUFuncOp::parse(OpAsmParser &parser, OperationState &result) {
1564 SmallVector<OpAsmParser::Argument> entryArgs;
1565 SmallVector<DictionaryAttr> resultAttrs;
1566 SmallVector<Type> resultTypes;
1567 bool isVariadic;
1568
1569 // Parse the function name.
1570 StringAttr nameAttr;
1571 if (parser.parseSymbolName(nameAttr, getSymNameAttrName(result.name),
1572 result.attributes))
1573 return failure();
1574
1575 auto signatureLocation = parser.getCurrentLocation();
1576 if (failed(function_interface_impl::parseFunctionSignatureWithArguments(
1577 parser, /*allowVariadic=*/false, entryArgs, isVariadic, resultTypes,
1578 resultAttrs)))
1579 return failure();
1580
1581 if (!entryArgs.empty() && entryArgs[0].ssaName.name.empty())
1582 return parser.emitError(signatureLocation)
1583 << "gpu.func requires named arguments";
1584
1585 // Construct the function type. More types will be added to the region, but
1586 // not to the function type.
1587 Builder &builder = parser.getBuilder();
1588
1589 SmallVector<Type> argTypes;
1590 for (auto &arg : entryArgs)
1591 argTypes.push_back(arg.type);
1592 auto type = builder.getFunctionType(argTypes, resultTypes);
1593 result.addAttribute(getFunctionTypeAttrName(result.name),
1594 TypeAttr::get(type));
1595
1596 call_interface_impl::addArgAndResultAttrs(
1597 builder, result, entryArgs, resultAttrs, getArgAttrsAttrName(result.name),
1598 getResAttrsAttrName(result.name));
1599
1600 Attribute workgroupAttributionAttrs;
1601 // Parse workgroup memory attributions.
1602 if (failed(parseAttributions(parser, GPUFuncOp::getWorkgroupKeyword(),
1603 entryArgs, workgroupAttributionAttrs)))
1604 return failure();
1605
1606 // Store the number of operands we just parsed as the number of workgroup
1607 // memory attributions.
1608 unsigned numWorkgroupAttrs = entryArgs.size() - type.getNumInputs();
1609 if (numWorkgroupAttrs != 0)
1610 result.addAttribute(
1611 GPUFuncOp::getWorkgroupAttributionsAttrName(result.name),
1612 builder.getI64IntegerAttr(numWorkgroupAttrs));
1613 if (workgroupAttributionAttrs)
1614 result.addAttribute(GPUFuncOp::getWorkgroupAttribAttrsAttrName(result.name),
1615 workgroupAttributionAttrs);
1616
1617 Attribute privateAttributionAttrs;
1618 // Parse private memory attributions.
1619 if (failed(parseAttributions(parser, GPUFuncOp::getPrivateKeyword(),
1620 entryArgs, privateAttributionAttrs)))
1621 return failure();
1622 if (privateAttributionAttrs)
1623 result.addAttribute(GPUFuncOp::getPrivateAttribAttrsAttrName(result.name),
1624 privateAttributionAttrs);
1625
1626 // Parse the kernel attribute if present.
1627 if (succeeded(parser.parseOptionalKeyword(GPUFuncOp::getKernelKeyword())))
1628 result.addAttribute(GPUFuncOp::getKernelAttrName(result.name),
1629 builder.getUnitAttr());
1630
1631 // Parse attributes.
1632 if (failed(parser.parseOptionalAttrDictWithKeyword(result.attributes)))
1633 return failure();
1634
1635 // Parse the region. If no argument names were provided, take all names
1636 // (including those of attributions) from the entry block.
1637 auto *body = result.addRegion();
1638 return parser.parseRegion(*body, entryArgs);
1639}

1741static void setAttributionAttr(GPUFuncOp op, unsigned index, StringAttr name,
1742 Attribute value, StringAttr attrsName) {
1743 MLIRContext *ctx = op.getContext();
1744 SmallVector<NamedAttribute> elems;
1745 DictionaryAttr oldDict = getAttributionAttrs(op, index, attrsName);
1746 if (oldDict)
1747 elems.append(oldDict.getValue().begin(), oldDict.getValue().end());
1748
1749 bool found = false;
1750 bool mustSort = true;
1751 for (unsigned i = 0, e = elems.size(); i < e; ++i) {
1752 if (elems[i].getName() == name) {
1753 found = true;
1754 if (!value) {
1755 std::swap(elems[i], elems[elems.size() - 1]);
1756 elems.pop_back();
1757 } else {
1758 mustSort = false;
1759 elems[i] = NamedAttribute(elems[i].getName(), value);
1760 }
1761 break;
1762 }
1763 }
1764 if (!found) {
1765 if (!value)
1766 return;
1767 elems.emplace_back(name, value);
1768 }
1769 if (mustSort) {
1770 DictionaryAttr::sortInPlace(elems);
1771 }
1772 auto newDict = DictionaryAttr::getWithSorted(ctx, elems);
1773 setAttributionAttrs(op, index, newDict, attrsName);
1774}

1820 if (failed(verifyAttributions(getOperation(), getWorkgroupAttributionBBArgs(),
1821 GPUDialect::getWorkgroupAddressSpace())) ||
1822 failed(verifyAttributions(getOperation(), getPrivateAttributions(),
1823 GPUDialect::getPrivateAddressSpace())))
1824 return failure();
1825
1826 return success();
