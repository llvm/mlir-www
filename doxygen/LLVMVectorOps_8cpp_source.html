27void ShuffleVectorOp::build(OpBuilder &builder, OperationState &state, Value v1,
28 Value v2, DenseI32ArrayAttr mask,
29 ArrayRef<NamedAttribute> attrs) {
30 auto containerType = v1.getType();
31 auto vType = LLVM::getVectorType(
32 cast<VectorType>(containerType).getElementType(), mask.size(),
33 LLVM::isScalableVectorType(containerType));
34 build(builder, state, vType, v1, v2, mask);
35 state.addAttributes(attrs);
36}

43/// Build the result type of a shuffle vector operation.
44ParseResult mlir::LLVM::parseShuffleType(AsmParser &parser, Type v1Type,
45 Type &resType,
46 DenseI32ArrayAttr mask) {
47 if (!LLVM::isCompatibleVectorType(v1Type))
48 return parser.emitError(parser.getCurrentLocation(),
49 "expected an LLVM compatible vector type");
50 resType =
51 LLVM::getVectorType(cast<VectorType>(v1Type).getElementType(),
52 mask.size(), LLVM::isScalableVectorType(v1Type));
53 return success();
54}

61LogicalResult ShuffleVectorOp::verify() {
62 if (LLVM::isScalableVectorType(getV1().getType()) &&
63 llvm::any_of(getMask(), [](int32_t v) { return v != 0; }))
64 return emitOpError("expected a splat operation for scalable vectors");
65 return success();
66}

68// Folding for shufflevector op when v1 is single element 1D vector
69// and the mask is a single zero. OpFoldResult will be v1 in this case.
70OpFoldResult ShuffleVectorOp::fold(FoldAdaptor adaptor) {
71 // Check if operand 0 is a single element vector.
72 auto vecType = llvm::dyn_cast<VectorType>(getV1().getType());
73 if (!vecType || vecType.getRank() != 1 || vecType.getNumElements() != 1)
74 return {};
75 // Check if the mask is a single zero.
76 // Note: The mask is guaranteed to be non-empty.
77 if (getMask().size() != 1 || getMask()[0] != 0)
78 return {};
79 return getV1();
80}
