35template <typename OpType>
36static LogicalResult verifyFloatBinaryOp(OpType op, StringRef opKind) {
37 mlir::NVVM::FPRoundingMode rndMode = op.getRnd();
38 mlir::NVVM::SaturationMode satMode = op.getSat();
39 bool isFTZ = op.getFtz();
40
41 mlir::Type opType = op.getRes().getType();
42 mlir::Type opBaseType = isa<VectorType>(opType)
43 ? cast<VectorType>(opType).getElementType()
44 : opType;
45
46 if (opBaseType.isF64() && (satMode != NVVM::SaturationMode::NONE || isFTZ))
47 return op.emitOpError("FTZ and saturation are not supported for ")
48 << opKind << " involving f64 type";
49
50 if (opBaseType.isF16() && !(rndMode == NVVM::FPRoundingMode::RN ||
51 rndMode == NVVM::FPRoundingMode::NONE))
52 return op.emitOpError("only RN rounding mode is supported for f16 and "
53 "vector<2xf16> ")
54 << opKind;
55
56 if (opBaseType.isBF16()) {
57 if (rndMode != NVVM::FPRoundingMode::RN &&
58 rndMode != NVVM::FPRoundingMode::NONE)
59 return op.emitOpError("only RN rounding mode is supported for bf16 and "
60 "vector<2xbf16> ")
61 << opKind;
62 if (satMode != NVVM::SaturationMode::NONE || isFTZ)
63 return op.emitOpError("FTZ and saturation are not supported for bf16 and "
64 "vector<2xbf16> ")
65 << opKind;
66 }
67
68 return success();
69}

83LogicalResult NVVM::FmaOp::verify() {
84 auto opType = getRes().getType();
85 mlir::NVVM::FPRoundingMode rndMode = getRnd();
86 mlir::NVVM::SaturationMode satMode = getSat();
87 bool isFTZ = getFtz();
88 bool isRelu = getRelu();
89 bool hasOOB = getOob();
90
91 auto getBaseFType = [](Type type) -> Type {
92 if (isa<VectorType>(type))
93 return cast<VectorType>(type).getElementType();
94 return type;
95 };
96
97 auto opBaseType = getBaseFType(opType);
98
99 if (rndMode == NVVM::FPRoundingMode::NONE)
100 return emitOpError("rounding mode must be specified");
101
102 if (isRelu && satMode == NVVM::SaturationMode::SAT)
103 return emitOpError("relu and saturation are not supported together");
104
105 if (hasOOB && (satMode == NVVM::SaturationMode::SAT || isFTZ))
106 return emitOpError("oob is not supported with saturation or FTZ");
107
108 if (!(opBaseType.isF16() || opBaseType.isBF16()) && (isRelu || hasOOB))
109 return emitOpError("relu and oob are only supported for f16 and bf16");
110
111 if (opBaseType.isF64() && (satMode != NVVM::SaturationMode::NONE || isFTZ))
112 return emitOpError("FTZ and saturation are not supported for f64 type");
113
114 if (opBaseType.isF16() && rndMode != NVVM::FPRoundingMode::RN)
115 return emitOpError(
116 "only RN rounding mode is supported for f16 and vector<2xf16>");
117
118 if (opBaseType.isBF16()) {
119 if (rndMode != NVVM::FPRoundingMode::RN)
120 return emitOpError(
121 "only RN rounding mode is supported for bf16 and vector<2xbf16>");
122 if (satMode != NVVM::SaturationMode::NONE || isFTZ)
123 return emitOpError(
124 "FTZ and saturation are not supported for bf16 and vector<2xbf16>");
125 }
126
127 return success();
128}

140LogicalResult NVVM::DivFOp::verify() {
141 bool isApprox = getApprox();
142 bool isFull = getFull();
143 bool isF64 = getRes().getType().isF64();
144 bool isFtz = getFtz();
145 NVVM::FPRoundingMode rndMode = getRnd();
146
147 if (isApprox && isFull)
148 return emitOpError("'approx' and 'full' are mutually exclusive");
149
150 if (isApprox || isFull) {
151 if (isF64)
152 return emitOpError("'approx' and 'full' forms are f32-only");
153 if (rndMode != NVVM::FPRoundingMode::NONE)
154 return emitOpError(
155 "'approx' and 'full' forms do not accept a rounding mode");
156 return success();
157 }
158
159 // Rounded form below.
160 if (rndMode == NVVM::FPRoundingMode::NONE)
161 return emitOpError("rounding mode cannot be None for the rounded divide");
162 if (isF64 && isFtz)
163 return emitOpError("FTZ is not supported for f64");
164
165 return success();
166}

172struct ConvertFsubToFnegFadd : public OpRewritePattern<SubFOp> {
173 using OpRewritePattern<SubFOp>::OpRewritePattern;
174
175 LogicalResult matchAndRewrite(SubFOp op,
176 PatternRewriter &rewriter) const override {
177 Location loc = op.getLoc();
178 Value negRhs =
179 LLVM::FNegOp::create(rewriter, loc, op.getRhs().getType(), op.getRhs());
180
181 rewriter.replaceOpWithNewOp<AddFOp>(op, op.getType(), op.getLhs(), negRhs,
182 op.getRnd(), op.getSat(), op.getFtz());
183 return success();
184 }
185};

187void SubFOp::getCanonicalizationPatterns(RewritePatternSet &patterns,
188 MLIRContext *context) {
189 patterns.add<ConvertFsubToFnegFadd>(context);
190}

237 auto vectorType = dyn_cast<VectorType>(getSrc().getType());
238 if (!vectorType)
239 return success();
240
241 if (vectorType.getElementType().isF16() && getFtz())
242 return emitOpError("FTZ is not supported for vector<2xf16>");
243 if (vectorType.getElementType().isBF16() && !getFtz())
244 return emitOpError("FTZ is required for vector<2xbf16>");
245 return success();

249RsqrtOp::getIntrinsicIDAndArgs(Operation &op, LLVM::ModuleTranslation &mt,

254
255 llvm::Intrinsic::ID id = [&] {

257 return isFtz ? llvm::Intrinsic::nvvm_rsqrt_approx_ftz_f
258 : llvm::Intrinsic::nvvm_rsqrt_approx_f;
259 }

261 return isFtz ? llvm::Intrinsic::nvvm_rsqrt_approx_ftz_d
262 : llvm::Intrinsic::nvvm_rsqrt_approx_d;
263 }();
264
265 return {id, {mt.lookupValue(thisOp.getSrc())}};

268mlir::NVVM::IDArgPair
269SqrtOp::getIntrinsicIDAndArgs(Operation &op, LLVM::ModuleTranslation &mt,
270 llvm::IRBuilderBase &builder) {
271 auto thisOp = cast<NVVM::SqrtOp>(op);
272 Type t = thisOp.getRes().getType();
273 NVVM::FPRoundingMode rndMode = thisOp.getRnd();
274 bool isFtz = thisOp.getFtz();
275
276 // RM is one of RN/RM/RP/RZ (verifier rejects NONE).
277 // Subtracting 1 maps RN=1..RZ=4 to 0..3.
278 unsigned rndIndex = static_cast<unsigned>(rndMode) - 1;
279
280 static constexpr llvm::Intrinsic::ID f32IDs[] = {
281 llvm::Intrinsic::nvvm_sqrt_rn_f,
282 llvm::Intrinsic::nvvm_sqrt_rm_f,
283 llvm::Intrinsic::nvvm_sqrt_rp_f,
284 llvm::Intrinsic::nvvm_sqrt_rz_f,
285 };
286 static constexpr llvm::Intrinsic::ID f32FTZIDs[] = {
287 llvm::Intrinsic::nvvm_sqrt_rn_ftz_f,
288 llvm::Intrinsic::nvvm_sqrt_rm_ftz_f,
289 llvm::Intrinsic::nvvm_sqrt_rp_ftz_f,
290 llvm::Intrinsic::nvvm_sqrt_rz_ftz_f,
291 };
292 static constexpr llvm::Intrinsic::ID f64IDs[] = {
293 llvm::Intrinsic::nvvm_sqrt_rn_d,
294 llvm::Intrinsic::nvvm_sqrt_rm_d,
295 llvm::Intrinsic::nvvm_sqrt_rp_d,
296 llvm::Intrinsic::nvvm_sqrt_rz_d,
297 };
298
299 llvm::Intrinsic::ID id =
300 t.isF32() ? (isFtz ? f32FTZIDs[rndIndex] : f32IDs[rndIndex])
301 : f64IDs[rndIndex];
302
303 return {id, {mt.lookupValue(thisOp.getSrc())}};
304}

316mlir::NVVM::IDArgPair
317DivFOp::getIntrinsicIDAndArgs(Operation &op, LLVM::ModuleTranslation &mt,
318 llvm::IRBuilderBase &builder) {
319 auto thisOp = cast<NVVM::DivFOp>(op);
320 bool isFtz = thisOp.getFtz();
321
322 llvm::Intrinsic::ID id;
323
324 if (thisOp.getApprox()) {
325 id = isFtz ? llvm::Intrinsic::nvvm_div_approx_ftz_f
326 : llvm::Intrinsic::nvvm_div_approx_f;
327 } else if (thisOp.getFull()) {
328 // Intrinsic Naming quirk: int_nvvm_div_full has no `_f` suffix (unlike
329 // approx).
330 id = isFtz ? llvm::Intrinsic::nvvm_div_full_ftz
331 : llvm::Intrinsic::nvvm_div_full;
332 } else {
333 // Rounded form — three 4-entry tables indexed by (rndMode - 1).
334 unsigned rndIndex = static_cast<unsigned>(thisOp.getRnd()) - 1;
335
336 static constexpr llvm::Intrinsic::ID f32IDs[] = {
337 llvm::Intrinsic::nvvm_div_rn_f,
338 llvm::Intrinsic::nvvm_div_rm_f,
339 llvm::Intrinsic::nvvm_div_rp_f,
340 llvm::Intrinsic::nvvm_div_rz_f,
341 };
342 static constexpr llvm::Intrinsic::ID f32FTZIDs[] = {
343 llvm::Intrinsic::nvvm_div_rn_ftz_f,
344 llvm::Intrinsic::nvvm_div_rm_ftz_f,
345 llvm::Intrinsic::nvvm_div_rp_ftz_f,
346 llvm::Intrinsic::nvvm_div_rz_ftz_f,
347 };
348 static constexpr llvm::Intrinsic::ID f64IDs[] = {
349 llvm::Intrinsic::nvvm_div_rn_d,
350 llvm::Intrinsic::nvvm_div_rm_d,
351 llvm::Intrinsic::nvvm_div_rp_d,
352 llvm::Intrinsic::nvvm_div_rz_d,
353 };
354 Type t = thisOp.getRes().getType();
355 id = t.isF32() ? (isFtz ? f32FTZIDs[rndIndex] : f32IDs[rndIndex])
356 : f64IDs[rndIndex];
357 }
358
359 return {id,
360 {mt.lookupValue(thisOp.getLhs()), mt.lookupValue(thisOp.getRhs())}};
361}

369NVVM::IDArgPair DotAccumulate4WayOp::getIntrinsicIDAndArgs(
370 Operation &op, LLVM::ModuleTranslation &mt, llvm::IRBuilderBase &builder) {
371 auto curOp = cast<NVVM::DotAccumulate4WayOp>(op);
372
373 llvm::SmallVector<llvm::Value *> args;
374 args.push_back(getAsPackedI32(mt.lookupValue(curOp.getA()), builder));
375 args.push_back(getAsPackedI32(mt.lookupValue(curOp.getB()), builder));
376 args.push_back(mt.lookupValue(curOp.getC()));
377
378 bool isASigned = curOp.getAType() == NVVM::DotAccumulateType::SIGNED;
379 bool isBSigned = curOp.getBType() == NVVM::DotAccumulateType::SIGNED;
380 unsigned type = (isASigned << 1) | isBSigned;
381 const llvm::Intrinsic::ID ids[] = {
382 llvm::Intrinsic::nvvm_idp4a_u_u,
383 llvm::Intrinsic::nvvm_idp4a_u_s,
384 llvm::Intrinsic::nvvm_idp4a_s_u,
385 llvm::Intrinsic::nvvm_idp4a_s_s,
386 };
387 return {ids[type], args};
388}
