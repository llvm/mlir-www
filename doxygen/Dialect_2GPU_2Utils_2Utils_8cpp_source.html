47FailureOr<SubgroupReduceClusterInfo>
48getSubgroupReduceClusterInfo(SubgroupReduceOp op, unsigned subgroupSize) {
49 assert(llvm::isPowerOf2_32(subgroupSize));
50
51 std::optional<uint32_t> clusterSize = op.getClusterSize();
52 assert(!clusterSize ||
53 llvm::isPowerOf2_32(*clusterSize)); // Verifier should've caught this.
54 if (clusterSize && *clusterSize > subgroupSize)
55 return op.emitOpError()
56 << "cluster size " << *clusterSize
57 << " is greater than subgroup size " << subgroupSize;
58 unsigned effectiveClusterSize = clusterSize.value_or(subgroupSize);
59
60 auto clusterStride = op.getClusterStride();
61 assert(llvm::isPowerOf2_32(clusterStride)); // Verifier should've caught this.
62 if (clusterStride >= subgroupSize)
63 return op.emitOpError()
64 << "cluster stride " << clusterStride
65 << " is not less than subgroup size " << subgroupSize;
66
67 return SubgroupReduceClusterInfo{clusterStride, effectiveClusterSize,
68 subgroupSize};
69}
