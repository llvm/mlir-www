38void ConvertOpenMPToLLVMPass::runOnOperation() {
39 auto module = getOperation();
40
41 // Convert to OpenMP operations with LLVM IR dialect
42 RewritePatternSet patterns(&getContext());
43 LLVMTypeConverter converter(&getContext());
44 arith::populateArithToLLVMConversionPatterns(converter, patterns);
45 cf::populateControlFlowToLLVMConversionPatterns(converter, patterns);
46 cf::populateAssertToLLVMConversionPattern(converter, patterns);
47 populateFinalizeMemRefToLLVMConversionPatterns(converter, patterns);
48 populateFuncToLLVMConversionPatterns(converter, patterns);
49 populateOpenMPToLLVMConversionPatterns(converter, patterns);
50
51 LLVMConversionTarget target(getContext());
52 target.addLegalOp<omp::BarrierOp, omp::FlushOp, omp::TaskwaitOp,
53 omp::TaskyieldOp, omp::TerminatorOp>();
54 configureOpenMPToLLVMConversionLegality(target, converter);
55 if (failed(applyPartialConversion(module, target, std::move(patterns))))
56 signalPassFailure();
57}
