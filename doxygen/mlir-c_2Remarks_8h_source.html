30/// An opaque reference to a remark. A remark is only valid for the duration of
31/// the callback it is passed to; it must not be stored.
32struct MlirRemark {
33 const void *ptr;
34};
35typedef struct MlirRemark MlirRemark;
36
37/// The outcome a remark describes (see mlir::remark::RemarkKind).
38enum MlirRemarkKind {
39 MlirRemarkKindUnknown,
40 /// An optimization was applied.
41 MlirRemarkKindPassed,
42 /// A profitable optimization opportunity was found but not applied.
43 MlirRemarkKindMissed,
44 /// The optimization was attempted but failed.
45 MlirRemarkKindFailure,
46 /// Informational context without a pass/fail outcome.
47 MlirRemarkKindAnalysis
48};
49typedef enum MlirRemarkKind MlirRemarkKind;
50
51/// The emitting policy of the remark engine.
52enum MlirRemarkPolicy {
53 /// Report every remark as it is emitted.
54 MlirRemarkPolicyAll,
55 /// Postpone remarks and report the final set when the engine is finalized,
56 /// grouping related remarks under their parents.
57 MlirRemarkPolicyFinal
58};
59typedef enum MlirRemarkPolicy MlirRemarkPolicy;
60
61/// The serialization format of a remark output file.
62enum MlirRemarkFileFormat {
63 MlirRemarkFileFormatYAML,
64 MlirRemarkFileFormatBitstream
65};
66typedef enum MlirRemarkFileFormat MlirRemarkFileFormat;
67
68/// Regular expressions (llvm::Regex syntax, anchored by the engine) selecting
69/// the remark categories to report: `all` applies to every kind, the others to
70/// one kind each. A string of length 0 means the filter is not set, so the
71/// corresponding kind is not reported unless `all` matches.
72struct MlirRemarkCategories {
73 MlirStringRef all;
74 MlirStringRef passed;
75 MlirStringRef missed;
76 MlirStringRef analysis;
77 MlirStringRef failed;
78};
79typedef struct MlirRemarkCategories MlirRemarkCategories;
80
81/// Callback receiving every remark the engine reports.
82typedef void (*MlirRemarkCallback)(MlirRemark remark, void *userData);

84//===----------------------------------------------------------------------===//
85// Enabling and finalizing the engine of a context.
86//===----------------------------------------------------------------------===//
87
88/// Enables the remark engine on the context without a streamer; when
89/// `printAsEmitRemarks` is set, every reported remark is emitted as an MLIR
90/// remark diagnostic at its location. Fails when the context already has a
91/// remark engine (finalize it first).
92MLIR_CAPI_EXPORTED MlirLogicalResult mlirContextEnableOptimizationRemarks(
93 MlirContext context, MlirRemarkCategories categories,
94 MlirRemarkPolicy policy, bool printAsEmitRemarks);
95
96/// Enables the remark engine on the context, streaming the reported remarks to
97/// the file at `path` in `format`. The file is written when the engine is
98/// finalized. Fails when the file cannot be created or the context already has
99/// a remark engine.
100MLIR_CAPI_EXPORTED MlirLogicalResult mlirContextEnableOptimizationRemarksToFile(
101 MlirContext context, MlirRemarkCategories categories,
102 MlirRemarkPolicy policy, MlirRemarkFileFormat format, MlirStringRef path,
103 bool printAsEmitRemarks);
104
105/// Enables the remark engine on the context, delivering every reported remark
106/// to `callback`. `userData` is passed to the callback unchanged and released
107/// through `deleteUserData` (may be null) when the engine is finalized or the
108/// context is destroyed. Fails when the context already has a remark engine.
109MLIR_CAPI_EXPORTED MlirLogicalResult
110mlirContextEnableOptimizationRemarksWithCallback(
111 MlirContext context, MlirRemarkCategories categories,
112 MlirRemarkPolicy policy, MlirRemarkCallback callback, void *userData,
113 void (*deleteUserData)(void *), bool printAsEmitRemarks);
114
115/// Finalizes and removes the remark engine of the context: postponed remarks
116/// are reported, the output file is written and the callback's user data is
117/// released. Does nothing when no engine is enabled.
118MLIR_CAPI_EXPORTED void
119mlirContextFinalizeOptimizationRemarks(MlirContext context);
120
121/// Returns whether the context has a remark engine enabled.
122MLIR_CAPI_EXPORTED bool mlirContextHasRemarkEngine(MlirContext context);

171/// location's context. `message` (when non-empty) and the `numArgs` key/value
