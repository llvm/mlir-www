117std::vector<EnumCase> EnumInfo::getAllCases() const {
118 const auto *inits = def->getValueAsListInit("enumerants");

120 std::vector<EnumCase> cases;
121 cases.reserve(inits->size());

155EnumInfo::BitEnumSeparatorKind EnumInfo::getBitEnumSeparatorKind() const {
156 StringRef separator = getBitEnumSeparator().trim();
157 if (separator == "|")
158 return BitEnumSeparatorKind::VerticalBar;
159 if (separator == ",")
160 return BitEnumSeparatorKind::Comma;
161 if (separator.empty())
162 return BitEnumSeparatorKind::Space;
163 PrintFatalError(def->getLoc(),
164 "bit enum separator must be '|', ',', or a space ' '");
165}
166
167StringRef EnumInfo::getBitEnumSeparatorParser() const {
168 switch (getBitEnumSeparatorKind()) {
169 case BitEnumSeparatorKind::VerticalBar:
170 return "parseOptionalVerticalBar";
171 case BitEnumSeparatorKind::Comma:
172 return "parseOptionalComma";
173 case BitEnumSeparatorKind::Space:
174 // Whitespace is not a token, so there is nothing to parse between cases.
175 PrintFatalError(def->getLoc(),
176 "a bit enum with a space separator has no separator "
177 "parser");
178 }
179 llvm_unreachable("unknown bit enum separator kind");
180}
