105class ProfileInfoDepot {
106public:
107 ProfileInfoDepot(Operation *op) {
108 if (failed(populatationDispatch(op)))
109 op->emitOpError() << "fail to populate the profile info\n";
110 }
111
112 void addType(Type t) { tyInfo.push_back(convertTypeToInfo(t)); }
113 void addValue(Value v) { tyInfo.push_back(convertValueToInfo(v)); }
114 SmallVector<TypeInfo> getInfo() { return tyInfo; }
115
116private:
117 TypeInfo convertTypeToInfo(Type type) {
118 if (auto blockScaledTy = dyn_cast<tosa::BlockScaledType>(type)) {
119 Type valueTy = blockScaledTy.getValueType();
120 Type scaleTy = blockScaledTy.getScaleType();
121 return {type.getTypeID(), tosa::getBitWidth(valueTy), valueTy.getTypeID(),
122 scaleTy.getTypeID(), blockScaledTy.getBlockShape()};
123 }
124 return {type.getTypeID(), tosa::getBitWidth(type)};
125 }
126
127 TypeInfo convertValueToInfo(Value value) {
128 return convertTypeToInfo(getElementTypeOrSelf(value.getType()));
129 }
130
131 LogicalResult populatationDispatch(Operation *op);
132
133 // Add input operands and output results to the profile type info list
134 LogicalResult populateProfileInfo(ValueRange operands, ValueRange results);
135
136 // Base
137 template <typename T>
138 LogicalResult populateProfileInfo(T op) {
139 return op->emitOpError()
140 << "profile requirement for this op has not been defined";
141 }
142 // For conv2d, conv3d, transpose_conv2d, and depthwise_conv2d.
143 template <typename T>
144 LogicalResult populateProfileInfoConv(T op);
145
146 // For reshape, slice, tile, and transpose.
147 template <typename T>
148 LogicalResult populateProfileInfoDataLayout(T op);
149
150private:
151 SmallVector<TypeInfo> tyInfo;
152};
