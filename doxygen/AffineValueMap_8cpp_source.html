36void AffineValueMap::combine(const AffineValueMap &a, const AffineValueMap &b,
37 AffineExpr expr, AffineValueMap *res,
38 bool pairwise) {
39 assert((!pairwise || a.getNumResults() == b.getNumResults()) &&
40 "matching by position requires equal result counts");
41 bool validExpr = true;
42 expr.walk([&](AffineExpr subExpr) {
43 if (auto dim = llvm::dyn_cast<AffineDimExpr>(subExpr))
44 validExpr &= dim.getPosition() < 2;
45 else if (llvm::isa<AffineSymbolExpr>(subExpr))
46 validExpr = false;
47 });
48 assert(validExpr && "combining expression must use only d0 and d1");
49
50 SmallVector<Value, 4> allOperands;
51 allOperands.reserve(a.getNumOperands() + b.getNumOperands());
52 auto aDims = a.getOperands().take_front(a.getNumDims());
53 auto bDims = b.getOperands().take_front(b.getNumDims());
54 auto aSyms = a.getOperands().take_back(a.getNumSymbols());
55 auto bSyms = b.getOperands().take_back(b.getNumSymbols());
56 allOperands.append(aDims.begin(), aDims.end());
57 allOperands.append(bDims.begin(), bDims.end());
58 allOperands.append(aSyms.begin(), aSyms.end());
59 allOperands.append(bSyms.begin(), bSyms.end());
60
61 // Shift dims and symbols of b's map.
62 auto bMap = b.getAffineMap()
63 .shiftDims(a.getNumDims())
64 .shiftSymbols(a.getNumSymbols());
65
66 // Substitute each pair of results into the combining expression.
67 auto aMap = a.getAffineMap();
68 SmallVector<AffineExpr> resultExprs;
69 if (pairwise) {
70 resultExprs.reserve(a.getNumResults());
71 for (auto [aResult, bResult] :
72 llvm::zip_equal(aMap.getResults(), bMap.getResults())) {
73 AffineExpr replacements[] = {aResult, bResult};
74 resultExprs.push_back(expr.replaceDims(replacements));
75 }
76 } else {
77 resultExprs.reserve(a.getNumResults() * b.getNumResults());
78 for (AffineExpr aResult : aMap.getResults())
79 for (AffineExpr bResult : bMap.getResults()) {
80 AffineExpr replacements[] = {aResult, bResult};
81 resultExprs.push_back(expr.replaceDims(replacements));
82 }
83 }
84
85 auto resultMap = AffineMap::get(bMap.getNumDims(), bMap.getNumSymbols(),
86 resultExprs, bMap.getContext());
87 fullyComposeAffineMapAndOperands(&resultMap, &allOperands);
88 canonicalizeMapAndOperands(&resultMap, &allOperands);
89 resultMap = simplifyAffineMap(resultMap);
90 res->reset(resultMap, allOperands);
91}

93void AffineValueMap::difference(const AffineValueMap &a,
94 const AffineValueMap &b, AffineValueMap *res) {
95 MLIRContext *context = a.getAffineMap().getContext();
96 AffineExpr lhs = getAffineDimExpr(0, context);
97 AffineExpr rhs = getAffineDimExpr(1, context);
98 combine(a, b, lhs - rhs, res);
99}

110// Returns true and sets 'indexOfMatch' if 'valueToMatch' is found in
111// 'valuesToSearch' beginning at 'indexStart'. Returns false otherwise.
112static bool findIndex(Value valueToMatch, ArrayRef<Value> valuesToSearch,
113 unsigned indexStart, unsigned *indexOfMatch) {
114 unsigned size = valuesToSearch.size();
115 for (unsigned i = indexStart; i < size; ++i) {
116 if (valueToMatch == valuesToSearch[i]) {
117 *indexOfMatch = i;
118 return true;
119 }
120 }
121 return false;
122}

128/// This method uses the invariant that operands are always positionally aligned
129/// with the AffineDimExpr in the underlying AffineMap.
130bool AffineValueMap::isFunctionOf(unsigned idx, Value value) const {
131 unsigned index;
132 if (!findIndex(value, operands, /*indexStart=*/0, &index)) {
133 return false;
134 }
135 auto expr = const_cast<AffineValueMap *>(this)->getAffineMap().getResult(idx);
136 // TODO: this is better implemented on a flattened representation.
137 // At least for now it is conservative.
138 return expr.isFunctionOfDim(index);
139}

151bool AffineValueMap::operator==(const AffineValueMap &other) const {
152 if (getNumResults() != other.getNumResults())
153 return false;
154 AffineValueMap diff;
155 AffineValueMap::difference(*this, other, &diff);
156 return llvm::all_of(diff.getAffineMap().getResults(), [](AffineExpr e) {
157 return e == getAffineConstantExpr(0, e.getContext());
158 });
159}
