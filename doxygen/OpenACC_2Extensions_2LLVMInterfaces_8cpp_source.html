25 mlir::Value genLoad(Type pointer, OpBuilder &builder, Location loc,
26 TypedValue<PointerLikeType> srcPtr,
27 Type valueType) const {
28 // For LLVM pointers, we need the valueType to determine what to load
29 if (!valueType)
30 return {};
31
32 return LLVM::LoadOp::create(builder, loc, valueType, srcPtr);
33 }

35 bool genStore(Type pointer, OpBuilder &builder, Location loc,
36 Value valueToStore, TypedValue<PointerLikeType> destPtr) const {
37 LLVM::StoreOp::create(builder, loc, valueToStore, destPtr);
38 return true;
39 }

41 Value genCast(Type, OpBuilder &builder, Location loc, Value value,
42 Type resultType) const {
43 if (value.getType() == resultType)
44 return value;
45
46 auto srcPtrTy = dyn_cast<LLVM::LLVMPointerType>(value.getType());
47 auto dstPtrTy = dyn_cast<LLVM::LLVMPointerType>(resultType);
48 if (srcPtrTy && dstPtrTy) {
49 if (srcPtrTy.getAddressSpace() != dstPtrTy.getAddressSpace())
50 return LLVM::AddrSpaceCastOp::create(builder, loc, resultType, value);
51 return value;
52 }
53
54 if (srcPtrTy && isa<IntegerType>(resultType))
55 return LLVM::PtrToIntOp::create(builder, loc, resultType, value);
56
57 if (dstPtrTy) {
58 Value intVal = value;
59 if (isa<IndexType>(value.getType()))
60 intVal = arith::IndexCastUIOp::create(builder, loc,
61 builder.getI64Type(), value);
62 if (isa<IntegerType>(intVal.getType()))
63 return LLVM::IntToPtrOp::create(builder, loc, resultType, intVal);
64 }
65
66 if (auto resPtrLike = dyn_cast<PointerLikeType>(resultType))
67 if (!isa<LLVM::LLVMPointerType>(resPtrLike))
68 if (Value v = resPtrLike.genCast(builder, loc, value, resultType))
69 return v;
70 if (auto valPtrLike = dyn_cast<PointerLikeType>(value.getType()))
71 if (!isa<LLVM::LLVMPointerType>(valPtrLike))
72 if (Value v = valPtrLike.genCast(builder, loc, value, resultType))
73 return v;
74
75 return UnrealizedConversionCastOp::create(builder, loc,
76 TypeRange(resultType), value)
77 .getResult(0);
78 }
79};
