52/// libdevice's __nv_powi and __nv_powif take an i32 exponent, so math.fpowi
53/// with a wider or index exponent is not lowered.
54static bool exponentFitsInI32(math::FPowIOp op) {
55 auto type =
56 dyn_cast<IntegerType>(getElementTypeOrSelf(op.getRhs().getType()));
57 return type && type.getWidth() <= 32;
58}

74struct FPowIOpLowering : public OpToFuncCallLowering<math::FPowIOp> {
75 using OpToFuncCallLowering::OpToFuncCallLowering;
76
77 LogicalResult
78 matchAndRewrite(math::FPowIOp op, OpAdaptor adaptor,
79 ConversionPatternRewriter &rewriter) const override {
80 if (!exponentFitsInI32(op))
81 return rewriter.notifyMatchFailure(op, "exponent does not fit in i32");
82 // Return before creating anything in the cases OpToFuncCallLowering
83 // does not handle.
84 if (!op->getParentOfType<FunctionOpInterface>() ||
85 !isa<Float16Type, BFloat16Type, Float32Type, Float64Type>(
86 adaptor.getLhs().getType()))
87 return rewriter.notifyMatchFailure(op, "cannot be lowered to libdevice");
88 // Sign-extend a narrower exponent.
89 Value exponent = adaptor.getRhs();
90 if (!exponent.getType().isInteger(32))
91 exponent = LLVM::SExtOp::create(rewriter, op.getLoc(),
92 rewriter.getI32Type(), exponent);
93 Value operands[] = {adaptor.getLhs(), exponent};
94 return OpToFuncCallLowering::matchAndRewrite(
95 op, OpAdaptor(ValueRange(operands), op), rewriter);
96 }
97};

100// Custom pattern for sincos since it returns two values
101struct SincosOpLowering : public ConvertOpToLLVMPattern<math::SincosOp> {
102 using ConvertOpToLLVMPattern<math::SincosOp>::ConvertOpToLLVMPattern;
103
104 LogicalResult
105 matchAndRewrite(math::SincosOp op, OpAdaptor adaptor,
106 ConversionPatternRewriter &rewriter) const override {
107 Location loc = op.getLoc();
108 Value input = adaptor.getOperand();
109 Type inputType = input.getType();
110 auto convertedInput = maybeExt(input, rewriter);
111 auto computeType = convertedInput.getType();
112
113 StringRef sincosFunc;
114 if (isa<Float32Type>(computeType)) {
115 const arith::FastMathFlags flag = op.getFastmath();
116 const bool useApprox =
117 mlir::arith::bitEnumContainsAny(flag, arith::FastMathFlags::afn);
118 sincosFunc = useApprox ? "__nv_fast_sincosf" : "__nv_sincosf";
119 } else if (isa<Float64Type>(computeType)) {
120 sincosFunc = "__nv_sincos";
121 } else {
122 return rewriter.notifyMatchFailure(op,
123 "unsupported operand type for sincos");
124 }
125
126 auto ptrType = LLVM::LLVMPointerType::get(rewriter.getContext());
127
128 Value sinPtr, cosPtr;
129 {
130 OpBuilder::InsertionGuard guard(rewriter);
131 auto *scope =
132 op->getParentWithTrait<mlir::OpTrait::AutomaticAllocationScope>();
133 assert(scope && "Expected op to be inside automatic allocation scope");
134 rewriter.setInsertionPointToStart(&scope->getRegion(0).front());
135 auto one = LLVM::ConstantOp::create(rewriter, loc, rewriter.getI32Type(),
136 rewriter.getI32IntegerAttr(1));
137 sinPtr =
138 LLVM::AllocaOp::create(rewriter, loc, ptrType, computeType, one, 0);
139 cosPtr =
140 LLVM::AllocaOp::create(rewriter, loc, ptrType, computeType, one, 0);
141 }
142
143 createSincosCall(rewriter, loc, sincosFunc, convertedInput, sinPtr, cosPtr,
144 op);
145
146 auto sinResult = LLVM::LoadOp::create(rewriter, loc, computeType, sinPtr);
147 auto cosResult = LLVM::LoadOp::create(rewriter, loc, computeType, cosPtr);
148
149 rewriter.replaceOp(op, {maybeTrunc(sinResult, inputType, rewriter),
150 maybeTrunc(cosResult, inputType, rewriter)});
151 return success();
152 }
153
154private:
155 Value maybeExt(Value operand, PatternRewriter &rewriter) const {
156 if (isa<Float16Type, BFloat16Type>(operand.getType()))
157 return LLVM::FPExtOp::create(rewriter, operand.getLoc(),
158 Float32Type::get(rewriter.getContext()),
159 operand);
160 return operand;
161 }
162
163 Value maybeTrunc(Value operand, Type type, PatternRewriter &rewriter) const {
164 if (operand.getType() != type)
165 return LLVM::FPTruncOp::create(rewriter, operand.getLoc(), type, operand);
166 return operand;
167 }
168
169 void createSincosCall(ConversionPatternRewriter &rewriter, Location loc,
170 StringRef funcName, Value input, Value sinPtr,
171 Value cosPtr, Operation *op) const {
172 auto voidType = LLVM::LLVMVoidType::get(rewriter.getContext());
173 auto ptrType = sinPtr.getType();
174
175 SmallVector<Type> operandTypes = {input.getType(), ptrType, ptrType};
176 auto funcType = LLVM::LLVMFunctionType::get(voidType, operandTypes);
177
178 auto funcAttr = StringAttr::get(op->getContext(), funcName);
179 auto funcOp =
180 SymbolTable::lookupNearestSymbolFrom<LLVM::LLVMFuncOp>(op, funcAttr);
181
182 if (!funcOp) {
183 auto parentFunc = op->getParentOfType<FunctionOpInterface>();
184 assert(parentFunc && "expected there to be a parent function");
185 OpBuilder b(parentFunc);
186
187 auto globalloc = loc->findInstanceOfOrUnknown<FileLineColLoc>();
188 funcOp = LLVM::LLVMFuncOp::create(b, globalloc, funcName, funcType);
189 }
190
191 SmallVector<Value> callOperands = {input, sinPtr, cosPtr};
192 LLVM::CallOp::create(rewriter, loc, funcOp, callOperands);
193 }
194};
