31mlir::Value mlir::acc::getACCOperandForBlockArg(mlir::Value v) {
32 auto barg = mlir::dyn_cast<mlir::BlockArgument>(v);
33 if (!barg)
34 return nullptr;
35
36 mlir::Block *block = barg.getOwner();
37 auto computeReg =
38 mlir::dyn_cast<mlir::acc::ComputeRegionOp>(block->getParentOp());
39 if (!computeReg)
40 return nullptr;
41 assert(block == computeReg.getBody() &&
42 "block must be the body of acc.compute_region");
43 return computeReg.getOperand(barg);
44}

54template <typename OpTy>
55static bool isOnlyUsedByOpClauses(mlir::Value val, mlir::Region &region) {
56 auto checkIfUsedOnlyByOpInside = [&](mlir::Operation *user) {
57 // For any users which are not in the current acc region, we can ignore.
58 // Return true so that it can be used in a `all_of` check.
59 if (!region.isAncestor(user->getParentRegion()))
60 return true;
61 return mlir::isa<OpTy>(user);
62 };
63
64 return llvm::all_of(val.getUsers(), checkIfUsedOnlyByOpInside);
65}

77std::optional<mlir::acc::ClauseDefaultValue>
78mlir::acc::getDefaultAttr(Operation *op) {
79 std::optional<mlir::acc::ClauseDefaultValue> defaultAttr;
80 Operation *currOp = op;
81
82 // Iterate outwards until a default clause is found (since OpenACC
83 // specification notes that a visible default clause is the nearest default
84 // clause appearing on the compute construct or a lexically containing data
85 // construct.
86 while (!defaultAttr.has_value() && currOp) {
87 defaultAttr =
88 llvm::TypeSwitch<mlir::Operation *,
89 std::optional<mlir::acc::ClauseDefaultValue>>(currOp)
90 .Case<ACC_COMPUTE_CONSTRUCT_OPS, mlir::acc::DataOp>(
91 [&](auto op) { return op.getDefaultAttr(); })
92 .Default([&](Operation *) { return std::nullopt; });
93 currOp = currOp->getParentOp();
94 }
95
96 return defaultAttr;
97}

116std::string mlir::acc::getVariableName(mlir::Value v) {
117 Value current = v;
118
119 // Walk through view operations until a name is found or can't go further
120 while (Operation *definingOp = current.getDefiningOp()) {
121 // For integer constants, return their value as a string.
122 if (std::optional<int64_t> constVal = getConstantIntValue(current))
123 return std::to_string(*constVal);
124
125 // Check for `acc.var_name` attribute
126 if (auto varNameAttr = definingOp->getDiscardableAttrOfType<VarNameAttr>(
127 getVarNameAttrName()))
128 return varNameAttr.getName().str();
129
130 // If it is a data entry operation, get name via getVarName
131 if (isa<ACC_DATA_ENTRY_OPS>(definingOp))
132 if (auto name = acc::getVarName(definingOp))
133 return name->str();
134
135 // If it's a view operation, continue to the source
136 if (auto viewOp = dyn_cast<ViewLikeOpInterface>(definingOp)) {
137 current = viewOp.getViewSource();
138 continue;
139 }
140
141 break;
142 }
143
144 return "";
145}

147std::string mlir::acc::getRecipeName(mlir::acc::RecipeKind kind,
148 mlir::Type type) {
149 assert(kind == mlir::acc::RecipeKind::private_recipe ||
150 kind == mlir::acc::RecipeKind::firstprivate_recipe ||
151 kind == mlir::acc::RecipeKind::reduction_recipe);
152 if (!llvm::isa<mlir::acc::PointerLikeType, mlir::acc::MappableType>(type))
153 return "";
154
155 std::string recipeName;
156 llvm::raw_string_ostream ss(recipeName);
157 ss << (kind == mlir::acc::RecipeKind::private_recipe ? "privatization_"
158 : kind == mlir::acc::RecipeKind::firstprivate_recipe
159 ? "firstprivatization_"
160 : "reduction_");
161
162 // Print the type using its dialect-defined textual format.
163 type.print(ss);
164 ss.flush();
165
166 // Replace invalid characters (anything that's not a letter, number, or
167 // period) since this needs to be a valid MLIR identifier.
168 for (char &c : recipeName) {
169 if (!std::isalnum(static_cast<unsigned char>(c)) && c != '.' && c != '_') {
170 if (c == '?')
171 c = 'U';
172 else if (c == '*')
173 c = 'Z';
174 else if (c == '(' || c == ')' || c == '[' || c == ']' || c == '{' ||
175 c == '}' || c == '<' || c == '>')
176 c = '_';
177 else
178 c = 'X';
179 }
180 }
181
182 return recipeName;
183}

195/// Look up `symbol` in the `gpu.module`s of the enclosing module. A
196/// `gpu.module` is its own symbol table, so `lookupNearestSymbolFrom` from a
197/// user outside of it does not find definitions placed inside.
198static mlir::Operation *lookupSymbolInGPUModules(mlir::Operation *user,
199 mlir::SymbolRefAttr symbol) {
200 auto moduleOp = user->getParentOfType<mlir::ModuleOp>();
201 if (!moduleOp)
202 return nullptr;
203 for (auto gpuModule : moduleOp.getOps<mlir::gpu::GPUModuleOp>()) {
204 if (mlir::Operation *op = mlir::SymbolTable::lookupSymbolIn(
205 gpuModule, symbol.getRootReference()))
206 return op;
207 }
208 return nullptr;
209}

211bool mlir::acc::isValidSymbolUse(mlir::Operation *user,
212 mlir::SymbolRefAttr symbol,
213 mlir::Operation **definingOpPtr) {
214 // A pass may prepare the device-side definition of a symbol inside a
215 // `gpu.module` while the use is still a reference from host IR. Such a
216 // symbol is meant to be used on device, so the use is valid.
217 if (mlir::Operation *gpuOp = lookupSymbolInGPUModules(user, symbol)) {
218 if (definingOpPtr)
219 *definingOpPtr = gpuOp;
220 return true;
221 }
222
223 mlir::Operation *definingOp =
224 mlir::SymbolTable::lookupNearestSymbolFrom(user, symbol);
225
226 // If there are no defining ops, we have no way to ensure validity because
227 // we cannot check for any attributes.
228 if (!definingOp)
229 return false;
230
231 if (definingOpPtr)
232 *definingOpPtr = definingOp;
233
234 // Check if the defining op is a recipe.
235 // Recipes are valid as they get materialized before being offloaded to
236 // device. They are only instructions for how to materialize.
237 if (mlir::isa<mlir::accomp::RecipeInterface>(definingOp))
238 return true;
239
240 // Check if the defining op is a global variable that is device data.
241 // Device data is already resident on the device and does not need mapping.
242 if (auto globalVar =
243 mlir::dyn_cast<mlir::acc::GlobalVariableOpInterface>(definingOp))
244 if (globalVar.isDeviceData())
245 return true;
246
247 // Check if the defining op is a function
248 if (auto func =
249 mlir::dyn_cast_if_present<mlir::FunctionOpInterface>(definingOp)) {
250 // If this symbol is actually an acc routine or a specialized acc routine -
251 // then it is expected for it to be offloaded - therefore it is valid.
252 if (func->hasDiscardableAttr(mlir::acc::getRoutineInfoAttrName()) ||
253 func->hasDiscardableAttr(mlir::acc::getSpecializedRoutineAttrName()))
254 return true;
255
256 // If this symbol is a call to an LLVM intrinsic, then it is likely valid.
257 // Check the following:
258 // 1. The function is private
259 // 2. The function has no body
260 // 3. Name starts with "llvm."
261 // 4. The function's name is a valid LLVM intrinsic name
262 if (func.getVisibility() == mlir::SymbolTable::Visibility::Private &&
263 func.getFunctionBody().empty() && func.getName().starts_with("llvm.") &&
264 llvm::Intrinsic::lookupIntrinsicID(func.getName()) !=
265 llvm::Intrinsic::not_intrinsic)
266 return true;
267 }
268
269 // A declare attribute is needed for symbol references.
270 bool hasDeclare =
271 definingOp->hasDiscardableAttr(mlir::acc::getDeclareAttrName());
272 return hasDeclare;
273}

275bool mlir::acc::isDeviceValue(mlir::Value val) {
276 // Check if the value is device data via type interfaces.
277 // Device data is already resident on the device and does not need mapping.
278 if (auto mappableTy = dyn_cast<mlir::acc::MappableType>(val.getType()))
279 if (mappableTy.isDeviceData(val))
280 return true;
281
282 if (auto pointerLikeTy = dyn_cast<mlir::acc::PointerLikeType>(val.getType()))
283 if (pointerLikeTy.isDeviceData(val))
284 return true;
285
286 mlir::Operation *defOp = val.getDefiningOp();
287 if (!defOp)
288 return false;
289
290 // `acc.declare` with deviceptr marks data that is already associated with
291 // the device.
292 if (auto declareAttr =
293 defOp->getDiscardableAttrOfType<mlir::acc::DeclareAttr>(
294 mlir::acc::getDeclareAttrName()))
295 if (declareAttr.getDataClause().getValue() ==
296 mlir::acc::DataClause::acc_deviceptr)
297 return true;
298
299 // Handle operations that access a partial entity - check if the base entity
300 // is device data.
301 if (auto partialAccess =
302 dyn_cast<mlir::acc::PartialEntityAccessOpInterface>(defOp)) {
303 if (mlir::Value base = partialAccess.getBaseEntity())
304 return isDeviceValue(base);
305 }
306
307 // Handle address_of - check if the referenced global is device data.
308 if (auto addrOfIface =
309 dyn_cast<mlir::acc::AddressOfGlobalOpInterface>(defOp)) {
310 auto symbol = addrOfIface.getSymbol();
311 if (auto global = mlir::SymbolTable::lookupNearestSymbolFrom<
312 mlir::acc::GlobalVariableOpInterface>(defOp, symbol))
313 return global.isDeviceData();
314 }
315
316 return false;
317}

319bool mlir::acc::isValidValueUse(mlir::Value val, mlir::Region &region) {
320 // Types that can be passed by value are legal.
321 Type type = val.getType();
322 if (type.isIntOrIndexOrFloat() || isa<mlir::ComplexType>(type) ||
323 llvm::isa<mlir::VectorType>(type))
324 return true;
325
326 // If this is produced by an ACC data entry operation, it is valid.
327 if (isa_and_nonnull<ACC_DATA_ENTRY_OPS>(val.getDefiningOp()))
328 return true;
329
330 // If the value is only used by private clauses, it is not a live-in.
331 if (isOnlyUsedByPrivateClauses(val, region))
332 return true;
333
334 // If this is device data, it is valid.
335 if (isDeviceValue(val))
336 return true;
337
338 // Arguments of an enclosing acc routine are already on the device.
339 if (mlir::Operation *parent = region.getParentOp()) {
340 if (auto func = parent->getParentOfType<mlir::FunctionOpInterface>()) {
341 if ((mlir::acc::isAccRoutine(func) ||
342 mlir::acc::isSpecializedAccRoutine(func)) &&
343 llvm::is_contained(func.getArguments(), val))
344 return true;
345 }
346 }
347
348 return false;
349}

351llvm::SmallVector<mlir::Value>
352mlir::acc::getDominatingDataClauses(mlir::Operation *computeConstructOp,
353 mlir::DominanceInfo &domInfo,
354 mlir::PostDominanceInfo &postDomInfo) {
355 llvm::SmallSetVector<mlir::Value, 8> dominatingDataClauses;
356
357 llvm::TypeSwitch<mlir::Operation *>(computeConstructOp)
358 .Case<mlir::acc::ParallelOp, mlir::acc::KernelsOp, mlir::acc::SerialOp>(
359 [&](auto op) {
360 for (auto dataClause : op.getDataClauseOperands()) {
361 dominatingDataClauses.insert(dataClause);
362 }
363 })
364 .Default([](mlir::Operation *) {});
365
366 // Collect the data clauses from enclosing data constructs.
367 mlir::Operation *currParentOp = computeConstructOp->getParentOp();
368 while (currParentOp) {
369 if (mlir::isa<mlir::acc::DataOp>(currParentOp)) {
370 for (auto dataClause : mlir::dyn_cast<mlir::acc::DataOp>(currParentOp)
371 .getDataClauseOperands()) {
372 dominatingDataClauses.insert(dataClause);
373 }
374 }
375 currParentOp = currParentOp->getParentOp();
376 }
377
378 // Find the enclosing function/subroutine
379 auto funcOp =
380 computeConstructOp->getParentOfType<mlir::FunctionOpInterface>();
381 if (!funcOp)
382 return dominatingDataClauses.takeVector();
383
384 // Walk the function to find `acc.declare_enter`/`acc.declare_exit` pairs that
385 // dominate and post-dominate the compute construct and add their data
386 // clauses to the list.
387 funcOp->walk([&](mlir::acc::DeclareEnterOp declareEnterOp) {
388 if (domInfo.dominates(declareEnterOp.getOperation(), computeConstructOp)) {
389 // Collect all `acc.declare_exit` ops for this token.
390 llvm::SmallVector<mlir::acc::DeclareExitOp> exits;
391 for (auto *user : declareEnterOp.getToken().getUsers())
392 if (auto declareExit = mlir::dyn_cast<mlir::acc::DeclareExitOp>(user))
393 exits.push_back(declareExit);
394
395 // Only add clauses if every `acc.declare_exit` op post-dominates the
396 // compute construct.
397 if (!exits.empty() &&
398 llvm::all_of(exits, [&](mlir::acc::DeclareExitOp exitOp) {
399 return postDomInfo.postDominates(exitOp, computeConstructOp);
400 })) {
401 for (auto dataClause : declareEnterOp.getDataClauseOperands())
402 dominatingDataClauses.insert(dataClause);
403 }
404 }
405 });
406
407 return dominatingDataClauses.takeVector();
408}

410mlir::remark::detail::InFlightRemark
411mlir::acc::emitRemark(mlir::Operation *op,
412 const std::function<std::string()> &messageFn,
413 llvm::StringRef category) {
414 using namespace mlir::remark;
415 mlir::Location loc = op->getLoc();
416 auto *engine = loc->getContext()->getRemarkEngine();
417 if (!engine)
418 return remark::detail::InFlightRemark{};
419
420 llvm::StringRef funcName;
421 if (auto func = dyn_cast<mlir::FunctionOpInterface>(op))
422 funcName = func.getName();
423 else if (auto funcOp = op->getParentOfType<mlir::FunctionOpInterface>())
424 funcName = funcOp.getName();
425
426 auto opts = RemarkOpts::name("openacc").category(category);
427 if (!funcName.empty())
428 opts = opts.function(funcName);
429
430 auto remark = engine->emitOptimizationRemark(loc, opts);
431 if (remark)
432 remark << messageFn();
433 return remark;
434}
