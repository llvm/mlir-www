45DictionaryAttr NVVMAttachTarget::getFlags(OpBuilder &builder) const {
46 UnitAttr unitAttr = builder.getUnitAttr();
47 SmallVector<NamedAttribute, 3> flags;
48 auto addFlag = [&](StringRef flag) {
49 flags.push_back(builder.getNamedAttr(flag, unitAttr));
50 };
51 if (fastFlag)
52 addFlag("fast");
53 if (ftzFlag)
54 addFlag("ftz");
55 if (compilerDiagnosticsFlag)
56 addFlag("collect-compiler-diagnostics");
57
58 // Tokenize and set the optional command line options.
59 if (!cmdOptions.empty()) {
60 auto options = gpu::TargetOptions::tokenizeCmdOptions(cmdOptions);
61 if (!options.second.empty()) {
62 llvm::SmallVector<mlir::Attribute> nvvmOptionAttrs;
63 for (const char *opt : options.second) {
64 nvvmOptionAttrs.emplace_back(
65 mlir::StringAttr::get(builder.getContext(), StringRef(opt)));
66 }
67 flags.push_back(builder.getNamedAttr(
68 "ptxas-cmd-options",
69 mlir::ArrayAttr::get(builder.getContext(), nvvmOptionAttrs)));
70 }
71 }
72
73 if (!flags.empty())
74 return builder.getDictionaryAttr(flags);
75 return nullptr;
76}

78void NVVMAttachTarget::runOnOperation() {
79 OpBuilder builder(&getContext());
80 ArrayRef<std::string> libs(linkLibs);
81 SmallVector<StringRef> filesToLink(libs);
82 auto target = builder.getAttr<NVVMTargetAttr>(
83 optLevel, triple, chip, features, getFlags(builder),
84 filesToLink.empty() ? nullptr : builder.getStrArrayAttr(filesToLink),
85 verifyTarget);
86 llvm::Regex matcher(moduleMatcher);
87 for (Region &region : getOperation()->getRegions())
88 for (Block &block : region.getBlocks())
89 for (auto module : block.getOps<gpu::GPUModuleOp>()) {
90 // Check if the name of the module matches.
91 if (!moduleMatcher.empty() && !matcher.match(module.getName()))
92 continue;
93 // Create the target array.
94 SmallVector<Attribute> targets;
95 if (std::optional<ArrayAttr> attrs = module.getTargets())
96 targets.append(attrs->getValue().begin(), attrs->getValue().end());
97 targets.push_back(target);
98 // Remove any duplicate targets.
99 targets.erase(llvm::unique(targets), targets.end());
100 // Update the target attribute array.
101 module.setTargetsAttr(builder.getArrayAttr(targets));
102 }
