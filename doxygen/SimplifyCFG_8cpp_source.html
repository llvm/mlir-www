27// Running the branch patterns standalone is profitable where full
28// canonicalization is not wanted: the generic canonicalizer runs
29// every pattern and folds the whole module, which costs more than
30// merging blocks and changes operations beyond the branches. Today
31// the pass applies only the branch patterns, with folding off, so its
32// only effect is block merging, and pipelines that never run
33// canonicalization at all can still use the transform through the
34// same pattern set.
35// Merge forwarding blocks in `func` by applying the llvm.br
36// canonicalization patterns.
37static LogicalResult mergeBranchBlocks(LLVM::LLVMFuncOp func) {
38 RewritePatternSet patterns(func.getContext());
39 LLVM::BrOp::getCanonicalizationPatterns(patterns, func.getContext());
40 SmallVector<Operation *> ops;
41 func.walk([&](LLVM::BrOp op) { ops.push_back(op); });
42 // Only the collected branches are rewritten, and folding is off:
43 // otherwise the driver would fold operations exposed by a merge and
44 // delete dead operand definitions, changing IR beyond block merging.
45 GreedyRewriteConfig config;
46 config.setStrictness(GreedyRewriteStrictness::ExistingOps);
47 config.enableFolding(false);
48 return applyOpPatternsGreedily(ops, std::move(patterns), config);
49}

51// This pass is the home for control-flow simplifications in the LLVM
52// dialect. Simplifications should be added here.
53struct SimplifyCFGPass
54 : public LLVM::impl::LLVMSimplifyCFGPassBase<SimplifyCFGPass> {
55 void runOnOperation() override {
56 LLVM::LLVMFuncOp func = getOperation();
57 if (func.isOptimizeNone())
58 return;
59 if (failed(mergeBranchBlocks(func)))
60 signalPassFailure();
61 // Further transforms can be called here.
62 }
63};
