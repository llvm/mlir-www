9// This pass lowers `acc.compute_region` to the GPU dialect. For host-side
10// kernels it wraps the region in `gpu.launch`; for specialized acc routines
11// already inside a `gpu.func`, the body is lowered in place without emitting
12// a launch.
13//
14// Overview:
15// ---------
16// `acc.compute_region` is the compute-body representation produced after
17// OpenACC compute constructs are decomposed and parallelism has been assigned.
18// This pass is the final ACC-to-GPU lowering step for that body: it converts
19// nested `scf.parallel` / `scf.for` loops marked with `acc.par_dims` into GPU
20// block and thread parallelism, materializes privatization and reductions for
21// the device, inserts synchronization where shared state is observed across
22// threads, and erases the ACC scaffolding (`acc.compute_region`,
23// `acc.par_width`).
24//
25// Transformations:
26// ----------------
27// 1. Launch creation: outside a `gpu.func`, each `acc.compute_region` becomes a
28// `gpu.launch` whose grid and block sizes come from `acc.par_width` launch
29// operands (defaulting to 1). Kernel/module name attributes are preserved.
30// Inside a `gpu.func` (specialized acc routine), no launch is emitted.
31//
32// 2. Parallel loops: `scf.parallel` with a single `acc.par_dims` entry is
33// mapped to the corresponding GPU dimension (`block_*` or `thread_*`).
34// Sequential dimensions remain as `scf.parallel`/`scf.for` loops in the
35// generated kernel body.
36//
37// 3. Privatization: `acc.privatize` / `acc.private_local` storage is
38// materialized as one of: a per-thread `memref.alloca` (thread-private
39// arrays within the stack budget), an `acc.gpu_shared_memory` buffer
40// (gang-/worker-private arrays that fit the shared-memory budget), or a
41// `memref.alloc` whose pointer is broadcast to the block through a small
42// shared-memory slot (the data lives in global memory; shared memory only
43// holds the broadcast pointer).
44//
45// 4. Predication: `acc.predicate_region` becomes `scf.if` guarded by active
46// thread/block indices derived from `acc.par_dims` and launch dimensions.
47//
48// 5. Reductions: `acc.reduction_*` ops are lowered to GPU reduction and
49// synchronization primitives according to each reduction's parallel
50// dimensions and accumulator storage class.
51//
52// Example:
53// --------
54// Before:
55// %c128 = arith.constant 128 : index
56// %tx = acc.par_width %c128 {par_dim = #acc.par_dim<thread_x>}
57// acc.compute_region launch(%arg0 = %tx) {
58// %c0 = arith.constant 0 : index
59// %c1 = arith.constant 1 : index
60// scf.parallel (%iv) = (%c0) to (%c128) step (%c1) {
61// ...
62// scf.reduce
63// } {acc.par_dims = #acc<par_dims[thread_x]>}
64// acc.yield
65// } <{origin = "acc.parallel"}>
66//
67// After:
68// gpu.launch blocks(%bidx, %bidy, %bidz) in (%gdimx = %c1, ...)
69// threads(%tidx, %tidy, %tidz) in (%bdimx = %c128, ...) {
70// ...
71// }
72//
73// Requirements:
74// -------------
75// - Must run on a GPU device type (`device-type` option); host and multicore
76// targets are rejected.
77// - Input must already be in the `acc.compute_region` form: nested SCF loops
78// carry `acc.par_dims`, privatization is expressed via `acc.privatize` /
79// `acc.private_local`, and reductions use the `acc.reduction_*` ops.
80// - Each `scf.parallel` processed by this pass is expected to have exactly
81// one parallel dimension and one induction variable.
82// - For acc routines, the `acc.compute_region` must live inside a `gpu.func`
83// in the GPU module.
84// - Uses `acc::OpenACCSupport` for NYI reporting and compiler remarks.
85// - Pass options: `max-workgroup-shared-memory`, `max-thread-private-stack`,
86// and `subgroup-size` (used for reductions and block-dimension alignment).
87//

172/// Maps an acc.routine's parallelism clauses to a GPU parallel dimension.
173static GPUParallelDimAttr
174getAccRoutineParDim(RoutineOp routineOp, MLIRContext *ctx,
175 const ACCToGPUMappingPolicy &policy) {
176 if (routineOp.getGangDimValue() ||
177 routineOp.getGangDimValue(DeviceType::Nvidia)) {
178 int64_t gangDimValue = routineOp.getGangDimValue(DeviceType::Nvidia)
179 ? *routineOp.getGangDimValue(DeviceType::Nvidia)
180 : *routineOp.getGangDimValue();
181 ParLevel gangLevel = getGangParLevel(gangDimValue);
182 return policy.gangDim(ctx, gangLevel);
183 }
184 if (routineOp.hasGang() || routineOp.hasGang(DeviceType::Nvidia))
185 return policy.gangDim(ctx, ParLevel::gang_dim1);
186 if (routineOp.hasWorker() || routineOp.hasWorker(DeviceType::Nvidia))
187 return policy.workerDim(ctx);
188 if (routineOp.hasVector() || routineOp.hasVector(DeviceType::Nvidia))
189 return policy.vectorDim(ctx);
190 return policy.seqDim(ctx);
191}

194static RoutineOp getRoutineOpForAccRoutineFunction(FunctionOpInterface funcOp,

222/// Returns the parallelism dimension of a callee acc routine, if any.
223static GPUParallelDimAttr
224getAccRoutineCallParDim(CallOpInterface callOp,
225 const ACCToGPUMappingPolicy &policy) {
226 std::optional<CallInterfaceCallable> callee = callOp.getCallableForCallee();
227 if (!callee)
228 return nullptr;
229 SymbolRefAttr calleeSymbolRef = dyn_cast<SymbolRefAttr>(*callee);
230 if (!calleeSymbolRef)
231 return nullptr;
232 ModuleOp moduleOp = callOp->getParentOfType<ModuleOp>();
233 if (!moduleOp)
234 return nullptr;
235
236 SymbolTable symTab(moduleOp);
237 FunctionOpInterface funcOp =
238 symTab.lookup<FunctionOpInterface>(calleeSymbolRef.getLeafReference());
239 if (!funcOp)
240 return nullptr;
241
242 if (isSpecializedAccRoutine(funcOp))
243 return getSpecializedRoutineDim(funcOp, policy);
244 if (RoutineOp routineOp = getRoutineOpForAccRoutineFunction(funcOp, symTab))
245 return getAccRoutineParDim(routineOp, funcOp.getContext(), policy);
246 return nullptr;
247}

249/// Collects parallel dimensions from enclosing loops and the compute region.
250static SmallVector<GPUParallelDimAttr> getAncestorParDims(Operation *op) {
251 SmallVector<GPUParallelDimAttr> parDimsArray;
252 ComputeRegionOp computeRegion = op->getParentOfType<ComputeRegionOp>();
253 assert(computeRegion && "missing enclosing acc.compute_region");
254 scf::ParallelOp parentLoop = op->getParentOfType<scf::ParallelOp>();
255 // True while parentLoop is the innermost parallel ancestor of op.
256 bool isInnermostParallelParent = true;
257 while (parentLoop) {
258 bool hasNonSeqParDim = false;
259 if (GPUParallelDimsAttr parDimsAttr = getParDimsAttr(parentLoop)) {
260 for (GPUParallelDimAttr parDim : parDimsAttr.getArray()) {
261 insertParDim(parDimsArray, parDim);
262 if (!parDim.isSeq())
263 hasNonSeqParDim = true;
264 }
265 }
266 // Include launch dims for a block-redundant ancestor when:
267 // - it is itself worksharing (e.g. vector + gpu_block_redundant), or
268 // - it is the innermost parallel parent (sequential remnant after
269 // partition-kernel-loops), so the body is not predicated on blockIdx.
270 // Do not include them for an outer sequential block-redundant wrapper
271 // around nested gang/worker/vector worksharing: that widens gang-private
272 // storage to per-thread.
273 if (hasGPUBlockRedundantAttr(parentLoop) &&
274 (hasNonSeqParDim || isInnermostParallelParent))
275 for (GPUParallelDimAttr parDim : computeRegion.getLaunchParDims())
276 insertParDim(parDimsArray, parDim);
277 isInnermostParallelParent = false;
278 parentLoop = parentLoop->getParentOfType<scf::ParallelOp>();
279 }
280
281 if (GPUParallelDimsAttr parDimsAttr = getParDimsAttr(computeRegion))
282 for (GPUParallelDimAttr parDim : parDimsAttr.getArray())
283 insertParDim(parDimsArray, parDim);
284 return parDimsArray;
285}

287/// Strips index casts to reach the underlying defining value.
288static Value stripIndexCastsFromValue(Value x) {
289 Operation *op = x.getDefiningOp();
290 if (!op)
291 return x;
292 while (arith::IndexCastOp castOp = dyn_cast<arith::IndexCastOp>(op)) {
293 op = castOp->getOperand(0).getDefiningOp();
294 if (!op)
295 return x;
296 }
297 return op->getResult(0);
298}

300/// Extracts a compile-time integer constant from \p x, when known.
301static FailureOr<int64_t> extractIntConst(Value x,
302 bool stripIndexCasts = false) {
303 if (stripIndexCasts)
304 x = stripIndexCastsFromValue(x);
305 Operation *op = x.getDefiningOp();
306 if (op) {
307 if (arith::ConstantIntOp constOp = dyn_cast<arith::ConstantIntOp>(op)) {
308 assert(constOp.getType().getIntOrFloatBitWidth() <= 64);
309 return constOp.value();
310 }
311 if (arith::ConstantIndexOp constOp = dyn_cast<arith::ConstantIndexOp>(op))
312 return constOp.value();
313 }
314 return failure();
315}

317/// True when \p x is a constant equal to \p y (modulo index casts).
318static bool sameEffectiveValue(Value x, int64_t y) {
319 x = stripIndexCastsFromValue(x);
320 FailureOr<int64_t> conX = extractIntConst(x);
321 if (failed(conX))
322 return false;
323 return *conX == y;
324}

326/// Continues tracking a memref through view-like and partial-access ops.
327static bool getPassThroughResults(Operation *userOp, Value trackedOperand,
328 SmallVectorImpl<Value> &passThroughResults) {
329 if (ViewLikeOpInterface viewLikeOp = dyn_cast<ViewLikeOpInterface>(userOp)) {
330 if (viewLikeOp.getViewSource() == trackedOperand) {
331 passThroughResults.push_back(viewLikeOp.getViewDest());
332 return true;
333 }
334 return false;
335 }
336
337 // Partial-entity accesses (e.g. array element or field access) forward the
338 // base entity through to their results, so treat them as pass-through when
339 // the base entity is the value being tracked.
340 if (acc::PartialEntityAccessOpInterface partialAccess =
341 dyn_cast<acc::PartialEntityAccessOpInterface>(userOp)) {
342 if (partialAccess.getBaseEntity() == trackedOperand) {
343 passThroughResults.append(userOp->result_begin(), userOp->result_end());
344 return true;
345 }
346 return false;
347 }
348 return false;
349}

351/// Skips memref view/cast chains to reach the underlying buffer.
352static Value unwrapMemRefConversion(Value v) {
353 while (Operation *op = v.getDefiningOp()) {
354 if (ViewLikeOpInterface viewLike = dyn_cast<ViewLikeOpInterface>(op)) {
355 if (isa<MemRefType>(viewLike.getViewSource().getType()) ||
356 isa<MemRefType>(viewLike.getViewDest().getType())) {
357 v = viewLike.getViewSource();
358 continue;
359 }
360 }
361 break;
362 }
363 return v;
364}

412/// Emits a barrier over the lanes of this thread's ThreadY row. \p rowWidth
413/// (blockDim.x) must divide \p subgroupSize so the rows tile a subgroup. A
414/// subgroup-wide barrier would also tie together the other rows sharing it.
415static void emitGPUBarrierRow(OpBuilder &builder, Location loc,
416 int64_t rowWidth, int64_t subgroupSize) {
417 assert(rowWidth > 0 && subgroupSize % rowWidth == 0 &&
418 "row must tile the subgroup");
419 Type i32Ty = builder.getI32Type();
420 int64_t rowsPerSubgroup = subgroupSize / rowWidth;
421
422 // laneBase = (tid.y % rowsPerSubgroup) * rowWidth. Both operands are powers
423 // of two here, since rowWidth divides the subgroup size.
424 Value threadY = gpu::ThreadIdOp::create(builder, loc, builder.getIndexType(),
425 gpu::Dimension::y);
426 Value threadY32 = arith::IndexCastOp::create(builder, loc, i32Ty, threadY);
427 Value rowMask =
428 arith::ConstantIntOp::create(builder, loc, i32Ty, rowsPerSubgroup - 1);
429 Value rowIndex = arith::AndIOp::create(builder, loc, threadY32, rowMask);
430 Value width = arith::ConstantIntOp::create(builder, loc, i32Ty, rowWidth);
431 Value laneBase = arith::MulIOp::create(builder, loc, rowIndex, width);
432
433 Value laneBits = arith::ConstantIntOp::create(builder, loc, i32Ty,
434 (int64_t{1} << rowWidth) - 1);
435 Value mask = arith::ShLIOp::create(builder, loc, laneBits, laneBase);
436 NVVM::SyncWarpOp::create(builder, loc, mask);
437}

750/// True if the accumulate spans a block dim or is nested in a block-mapped
751/// loop, i.e. each block owns the elements it reduces across threads. A
752/// thread-only accumulate with no block context grid-strides its element loop
753/// onto blocks, so per-thread partials would be dropped; such reductions must
754/// stay shared.
755static bool reductionHasBlockContext(acc::ReductionAccumulateArrayOp accArr) {
756 auto hasBlock = [](mlir::acc::GPUParallelDimsAttr parDims) {
757 return parDims && llvm::any_of(parDims.getArray(),
758 [](auto pd) { return pd.isAnyBlock(); });
759 };
760 if (hasBlock(accArr.getParDimsAttr()))
761 return true;
762 for (scf::ParallelOp loop = accArr->getParentOfType<scf::ParallelOp>(); loop;
763 loop = loop->getParentOfType<scf::ParallelOp>()) {
764 if (hasBlock(mlir::acc::getParDimsAttr(loop)))
765 return true;
766 }
767 return false;
768}

770/// Returns the array reduction accumulate (through cast/view ops) that \p v
771/// feeds if it needs per-thread storage: its par_dims include a thread dim
772/// and it has block context so the cross-thread all_reduce is well defined.
773static acc::ReductionAccumulateArrayOp perThreadArrayReductionAccum(Value v) {
774 SmallVector<Value> worklist{v};
775 DenseSet<Value> seen;
776 while (!worklist.empty()) {
777 Value cur = worklist.pop_back_val();
778 if (!seen.insert(cur).second)
779 continue;
780 for (Operation *user : cur.getUsers()) {
781 if (acc::ReductionAccumulateArrayOp accArr =
782 dyn_cast<acc::ReductionAccumulateArrayOp>(user)) {
783 bool hasThread = false;
784 for (auto pd : accArr.getParDims().getArray())
785 hasThread |= pd.isAnyThread();
786 if (hasThread && reductionHasBlockContext(accArr))
787 return accArr;
788 continue;
789 }
790 SmallVector<Value> through;
791 if (getPassThroughResults(user, cur, through))
792 worklist.append(through.begin(), through.end());
793 else if (isa<ViewLikeOpInterface>(user))
794 worklist.append(user->result_begin(), user->result_end());
795 }
796 }
797 return nullptr;
798}

800/// Store the reduction identity to every element of a freshly allocated
801/// per-thread array accumulator so all lanes start from identity (the original
802/// init loop may only run on one lane).
803static void initPerThreadArrayAccum(OpBuilder &b, Location loc, Value alloca,
804 MemRefType baseTy,
805 arith::AtomicRMWKind kind) {
806 assert(baseTy.getRank() > 0 && baseTy.hasStaticShape() &&
807 "per-thread array reduction accumulator must be static ranked");
808 Value ident = createIdentityValue(b, loc, baseTy.getElementType(), kind,
809 /*useOnlyFiniteValue=*/true);
810 Value lb = arith::ConstantIndexOp::create(b, loc, 0);
811 Value step = arith::ConstantIndexOp::create(b, loc, 1);
812 SmallVector<Value> indices;
813 auto buildLoopNest = [&](auto &&self, unsigned dim) -> void {
814 if (dim == baseTy.getRank()) {
815 memref::StoreOp::create(b, loc, ident, alloca, indices);
816 return;
817 }
818
819 Value ub = arith::ConstantIndexOp::create(b, loc, baseTy.getShape()[dim]);
820 auto forOp = scf::ForOp::create(b, loc, lb, ub, step);
821 OpBuilder::InsertionGuard g(b);
822 b.setInsertionPoint(forOp.getBody()->getTerminator());
823 indices.push_back(forOp.getInductionVar());
824 self(self, dim + 1);
825 indices.pop_back();
826 };
827 buildLoopNest(buildLoopNest, 0);
828}

831ACCCGToGPULowering::isEligibleForSharedMemory(acc::PrivateLocalOp privateLocal,
832 MemRefType baseTy) {
833 // Cross-thread array reduction accumulators must stay per-thread when their
834 // storage scope includes thread_x. Gang-/worker-scoped array temps remain
835 // eligible for shared memory.
836 if (perThreadArrayReductionAccum(privateLocal.getResult()) &&
837 storageHasThreadX(privateLocal, computeRegion))

839 ModuleOp module = computeRegion->getParentOfType<ModuleOp>();
840 FailureOr<bool> isCandidate = isPrivateLocalSharedMemoryCandidate(
841 privateLocal, computeRegion, module, defaultPolicy, &accSupport);
842 if (failed(isCandidate)) {

844 return std::nullopt;

847 return std::nullopt;

849 getPrivateLocalSharedMemoryUpperBoundBytes(privateLocal, computeRegion,
850 module, defaultPolicy);
851 assert(upperBound && "candidate private_local must have an upper bound");
852 int64_t elementSize =
853 getElementSizeInBytes(privateLocal.getLoc(), baseTy.getElementType());
854 int64_t numElements = 1;
855 for (int64_t dim : baseTy.getShape())
856 numElements *= dim;

878LogicalResult ACCCGToGPULowering::rewrite() {
879
880 // Pre-compute if thread-level reductions exist. ThreadY reduction generates
881 // shuffles which require subgroup alignment (blockDim.x = subgroupSize),
882 // meaning ThreadX lanes exist even without explicit ThreadX parallelism.
883 // ThreadX is tracked too, to tell worker-only reductions from row shuffles.
884 computeRegion->walk([&](acc::ReductionAccumulateOp op) {
885 for (auto parDim : op.getParDimsAttr().getArray()) {
886 hasThreadXReduction |= parDim.isThreadX();
887 hasThreadYReduction |= parDim.isThreadY();
888 }
889 });
890 computeRegion->walk([&](acc::ReductionAccumulateArrayOp op) {
891 for (auto parDim : op.getParDimsAttr().getArray()) {
892 if (!parDim.isAnyBlock())
893 hasThreadLevelArrayReduction = true;
894 }
895 });
896
897 // Pre-compute if any thread-level (vector or worker) routine call exists.
898 // Such routines partition work across ThreadX/ThreadY and emit workgroup-wide
899 // barriers internally (e.g. for shared memory alloca synchronization), so all
900 // workgroup threads must reach the call site for those barriers to converge.
901 computeRegion->walk([&](CallOpInterface callOp) -> WalkResult {
902 if (mlir::acc::GPUParallelDimAttr parDim =
903 getAccRoutineCallParDim(callOp, defaultPolicy)) {
904 if (parDim.isThreadX() || parDim.isThreadY()) {
905 hasThreadLevelRoutineCall = true;
906 return WalkResult::interrupt();
907 }
908 }
909 return WalkResult::advance();
910 });
911
912 Location loc = computeRegion->getLoc();
913 Value constantOne = arith::ConstantIndexOp::create(rewriter, loc, 1);
914
915 auto launchArgument = [&](gpu::Processor processor) -> Value {
916 mlir::acc::GPUParallelDimAttr parDim = mlir::acc::GPUParallelDimAttr::get(
917 computeRegion->getContext(), processor);
918 std::optional<Value> maybeLaunchArg =
919 computeRegion.getKnownLaunchArg(parDim);
920 LLVM_DEBUG(llvm::dbgs() << "ACCCGToGPU: launch-arg: "
921 << " parDim: " << parDim << " gpu: " << processor
922 << " widthValue: "
923 << maybeLaunchArg.value_or(constantOne) << "\n");
924
925 return getValueOrCreateCastToIndexLike(
926 rewriter, loc, rewriter.getIndexType(),
927 maybeLaunchArg.value_or(constantOne));
928 };
929 LLVM_DEBUG(llvm::dbgs() << "ACCCGToGPU: creating gpu launch op: \n");
930
931 // acc.compute_region keeps launch argument as block argument, for rewriting
932 // we now replace these with gpu.launch dimensions.
933 auto mapLaunchArguments = [&](gpu::Processor processor, Value launchArg) {
934 mlir::acc::GPUParallelDimAttr parDim = mlir::acc::GPUParallelDimAttr::get(
935 computeRegion->getContext(), processor);
936 std::optional<Value> kernelArg = computeRegion.getLaunchArg(parDim);
937 if (kernelArg)
938 mapping.map(computeRegion.gpuParWidth(processor), launchArg);
939 };
940
941 llvm::StringRef blockDimXName = "blockDim.x";
942 llvm::StringRef blockDimYName = "blockDim.y";
943 std::string deviceLabel = getDeviceRemarkQualifier(options.deviceType);
944
945 if (!computeRegion->getParentOfType<gpu::GPUFuncOp>()) {
946 Value blockDimX = launchArgument(gpu::Processor::ThreadX);
947 APInt bdxVal;
948 if (matchPattern(blockDimX, m_ConstantInt(&bdxVal)))
949 staticBlockDimX = bdxVal.getSExtValue();
950 Value blockDimY = launchArgument(gpu::Processor::ThreadY);
951 APInt bdyVal;
952 if (matchPattern(blockDimY, m_ConstantInt(&bdyVal)))
953 staticBlockDimY = bdyVal.getSExtValue();
954 Value blockDimZ = launchArgument(gpu::Processor::ThreadZ);
955 APInt bdzVal;
956 if (matchPattern(blockDimZ, m_ConstantInt(&bdzVal)))
957 staticBlockDimZ = bdzVal.getSExtValue();
958 Value gridDimX = launchArgument(gpu::Processor::BlockX);
959 Value gridDimY = launchArgument(gpu::Processor::BlockY);
960 Value gridDimZ = launchArgument(gpu::Processor::BlockZ);
961
962 // The format of the message is:
963 // Generating [serial] {deviceLabel} code with gridDim=32x1x1
964 // blockDim=256x1x1
965 accSupport.emitRemark(computeRegion, [&]() {
966 auto getName = [&](Value val) -> std::string {
967 std::string name = accSupport.getVariableName(val);
968 return name.empty() ? "(*)" : name;
969 };
970 bool isEffectivelySerial =
971 sameEffectiveValue(blockDimX, 1) &&
972 sameEffectiveValue(blockDimY, 1) &&
973 sameEffectiveValue(blockDimZ, 1) && sameEffectiveValue(gridDimX, 1) &&
974 sameEffectiveValue(gridDimY, 1) && sameEffectiveValue(gridDimZ, 1);
975 return (llvm::Twine("Generating ") +
976 llvm::Twine(isEffectivelySerial ? "serial " : "") + deviceLabel +
977 " code with gridDim=" + getName(gridDimX) + "x" +
978 getName(gridDimY) + "x" + getName(gridDimZ) +
979 " blockDim=" + getName(blockDimX) + "x" + getName(blockDimY) +
980 "x" + getName(blockDimZ))
981 .str();
982 });
983
984 // Check if kernel has a stream operand for async execution
985 if (mlir::Value streamValue = computeRegion.getStream()) {
986 LLVM_DEBUG(llvm::dbgs()
987 << "\nDEBUG: Creating async gpu.launch with stream: "
988 << streamValue << "\n");
989 launch = gpu::LaunchOp::create(
990 rewriter, loc, gridDimX, gridDimY, gridDimZ, blockDimX, blockDimY,
991 blockDimZ,
992 /*dynamicSharedMemorySize=*/mlir::Value{},
993 /*asyncTokenType=*/
994 mlir::gpu::AsyncTokenType::get(rewriter.getContext()));
995 // Add the stream as an async dependency
996 launch.getAsyncDependenciesMutable().append(streamValue);
997 } else {
998 LLVM_DEBUG(llvm::dbgs()
999 << "\nDEBUG: No stream, creating sync gpu.launch\n");
1000 launch = gpu::LaunchOp::create(rewriter, loc, gridDimX, gridDimY,
1001 gridDimZ, blockDimX, blockDimY, blockDimZ);
1002 }
1003
1004 // Transfer kernel function name and module name from acc.compute_region to
1005 // gpu.launch if present
1006 if (auto kernelFuncName = computeRegion.getKernelFuncNameAttr())
1007 launch.setFunctionAttr(kernelFuncName);
1008 if (auto kernelModuleName = computeRegion.getKernelModuleNameAttr())
1009 launch.setModuleAttr(kernelModuleName);
1010
1011 rewriter.setInsertionPointToEnd(&launch.getBody().front());
1012 gpu::TerminatorOp::create(rewriter, loc);
1013 rewriter.setInsertionPointToStart(&launch.getBody().front());
1014 mapLaunchArguments(gpu::Processor::BlockX,
1015 gpu::GridDimOp::create(rewriter, loc,
1016 rewriter.getIndexType(),
1017 gpu::Dimension::x));
1018 mapLaunchArguments(gpu::Processor::BlockY,
1019 gpu::GridDimOp::create(rewriter, loc,
1020 rewriter.getIndexType(),
1021 gpu::Dimension::y));
1022 mapLaunchArguments(gpu::Processor::BlockZ,
1023 gpu::GridDimOp::create(rewriter, loc,
1024 rewriter.getIndexType(),
1025 gpu::Dimension::z));
1026 mapLaunchArguments(gpu::Processor::ThreadX,
1027 gpu::BlockDimOp::create(rewriter, loc,
1028 rewriter.getIndexType(),
1029 gpu::Dimension::x));
1030 mapLaunchArguments(gpu::Processor::ThreadY,
1031 gpu::BlockDimOp::create(rewriter, loc,
1032 rewriter.getIndexType(),
1033 gpu::Dimension::y));
1034 mapLaunchArguments(gpu::Processor::ThreadZ,
1035 gpu::BlockDimOp::create(rewriter, loc,
1036 rewriter.getIndexType(),
1037 gpu::Dimension::z));
1038 } else {
1039 // Do not create gpu.launch for acc routine and map
1040 // to block/thread index and block/grid size instead
1041 // of launch arguments, using created maps.
1042 OpBuilder::InsertionGuard guard(rewriter);
1043 rewriter.setInsertionPointToStart(computeRegion->getBlock());
1044 createForAllDimensions(rewriter, loc, threadIdMap, dimensionMap);
1045 mapLaunchArguments(gpu::Processor::BlockX,
1046 dimensionMap[gpu::Processor::BlockX]);
1047 mapLaunchArguments(gpu::Processor::BlockY,
1048 dimensionMap[gpu::Processor::BlockY]);
1049 mapLaunchArguments(gpu::Processor::BlockZ,
1050 dimensionMap[gpu::Processor::BlockZ]);
1051 mapLaunchArguments(gpu::Processor::ThreadX,
1052 dimensionMap[gpu::Processor::ThreadX]);
1053 mapLaunchArguments(gpu::Processor::ThreadY,
1054 dimensionMap[gpu::Processor::ThreadY]);
1055 mapLaunchArguments(gpu::Processor::ThreadZ,
1056 dimensionMap[gpu::Processor::ThreadZ]);
1057 }
1058
1059 // Map input arguments for compute region; we go from an IsolatedFromAbove
1060 // operation to gpu.launch which is not IsolatedFromAbove.
1061 preparePrivatizeExtentInsOperands();
1062 Block *body = computeRegion.getBody();
1063 unsigned numLaunchArgs = computeRegion.getLaunchArgs().size();
1064 ValueRange inputArgs = computeRegion.getInputArgs();
1065 for (unsigned i = numLaunchArgs; i < body->getNumArguments(); ++i)
1066 mapping.map(body->getArgument(i), inputArgs[i - numLaunchArgs]);
1067
1068 assert(computeRegion.getRegion().hasOneBlock() &&
1069 "compute region only supports one block region for now");
1070 // process all operations inside kernel region
1071 for (auto &op : computeRegion.getRegion().getBlocks().front().getOperations())
1072 processOp(&op);
1073
1074 for (auto &parLoop : loopReductions)
1075 postprocessLoopReduction(parLoop);
1076
1077 // Replace combine reloads of a reduction slot with the block-reduced value.
1078 // Only when it dominates the reload; otherwise keep the reload.
1079 if (!pendingCombineReloads.empty() && launch) {
1080 DominanceInfo domInfo(launch);
1081 for (auto &[slot, loadOp] : pendingCombineReloads) {
1082 llvm::DenseMap<Value, Value>::iterator it =
1083 reductionAccumValue.find(slot);
1084 if (it == reductionAccumValue.end())
1085 continue;
1086 if (!domInfo.dominates(it->second, loadOp.getOperation()))
1087 continue;
1088 rewriter.replaceOp(loadOp, ValueRange{it->second});
1089 }
1090 }
1091
1092 if (launch) {
1093 const int64_t subgroupSize = options.subgroupSize;
1094 const int64_t subgroupAlignMask = subgroupSize - 1;
1095
1096 // Adjust blockDim.x to be a multiple of subgroupSize. This is required
1097 // because:
1098 // - Subgroup reductions (gpu.all_reduce) require full subgroups
1099 // - Per-row workgroup barriers require blockDim.x aligned to subgroupSize
1100 // Tracked apart because only a ThreadX reduction shuffles within the row,
1101 // so only it needs the row to start at a subgroup boundary.
1102 bool needsThreadXAlign = false;
1103 bool needsThreadYAlign = false;
1104 bool alignThreadXReduction =
1105 getConstantIntValue(launch.getBlockSizeY()) != 1 ||
1106 getConstantIntValue(launch.getBlockSizeZ()) != 1;
1107
1108 // A reduction over both dimensions counts as a ThreadX one: it still
1109 // shuffles ThreadX lanes within the row.
1110 auto classifyAllReduce = [&](gpu::AllReduceOp allReduce) {
1111 bool hasThreadX = false;
1112 bool hasThreadY = false;
1113 for (auto parDim : mlir::acc::getParDimsAttr(allReduce).getArray()) {
1114 hasThreadX |= parDim.isThreadX();
1115 hasThreadY |= parDim.isThreadY();
1116 }
1117 if (hasThreadX && alignThreadXReduction)
1118 needsThreadXAlign = true;
1119 else if (hasThreadY)
1120 needsThreadYAlign = true;
1121 };
1122
1123 launch.walk(classifyAllReduce);
1124 // Also check if called routines have thread-level reductions.
1125 launch.walk([&](func::CallOp callOp) {
1126 if (gpu::GPUFuncOp callee =
1127 callOp->getParentOfType<ModuleOp>().lookupSymbol<gpu::GPUFuncOp>(
1128 callOp.getCallee()))
1129 callee.walk(classifyAllReduce);
1130 });
1131
1132 bool isShuffleEnabled = needsThreadXAlign || needsThreadYAlign;
1133
1134 std::optional<int64_t> constBlockDimX =
1135 getConstantIntValue(launch.getBlockSizeX());
1136 std::optional<int64_t> constBlockDimY =
1137 getConstantIntValue(launch.getBlockSizeY());
1138 std::optional<int64_t> constBlockDimZ =
1139 getConstantIntValue(launch.getBlockSizeZ());
1140
1141 // Skip subgroup alignment only when the total thread count is already
1142 // below a subgroup (constant blockDim.x in 2..subgroupSize-1 and
1143 // constant blockDim.y/z == 1). If blockDim.y/z > 1 or is unknown,
1144 // padding blockDim.x to a subgroup is still required so subgroups don't
1145 // cross row boundaries for row-local shuffle/ThreadY-barrier reductions.
1146 bool skipAlign = constBlockDimX && constBlockDimY && constBlockDimZ &&
1147 *constBlockDimX > 1 && *constBlockDimX < subgroupSize &&
1148 *constBlockDimY == 1 && *constBlockDimZ == 1;
1149
1150 // Padding a worker-only launch divides blockDim.y by the same factor and
1151 // leaves the folded-away workers as ThreadX lanes redoing each other's
1152 // work, so keep its shape instead.
1153 if (isSingleThreadWorkerLaunch())
1154 skipAlign = true;
1155 if (needsThreadYAlign && !needsThreadXAlign && !hasThreadYBarrier &&
1156 isWorkerOnlyShuffleLaunch())
1157 skipAlign = true;
1158
1159 if ((isShuffleEnabled || hasThreadYBarrier) && !skipAlign) {
1160 rewriter.setInsertionPoint(launch);
1161
1162 Value curBlockDimX = launch.getBlockSizeX();
1163 Value curBlockDimY = launch.getBlockSizeY();
1164 Value curBlockDimZ = launch.getBlockSizeZ();
1165
1166 // Emit a report on changing parallelism.
1167 accSupport.emitRemark(computeRegion, [&]() {
1168 auto getName = [&](Value val) -> std::string {
1169 std::string name = accSupport.getVariableName(val);
1170 return name.empty() ? "(*)" : name;
1171 };
1172 std::string blockDimXValStr = getName(curBlockDimX);
1173 std::string blockDimYValStr = getName(curBlockDimY);
1174 llvm::StringRef kind =
1175 isShuffleEnabled ? "Shuffle reduction" : "ThreadY barrier";
1176 return (llvm::Twine(kind) +
1177 " is generated while adjusting the number of threads into "
1178 "groups of " +
1179 llvm::Twine(subgroupSize) + ".\n\t" + blockDimXName + ": `" +
1180 blockDimXValStr + "` to `((" + blockDimXValStr + " + " +
1181 llvm::Twine(subgroupAlignMask) + ") / " +
1182 llvm::Twine(subgroupSize) + ") * " + llvm::Twine(subgroupSize) +
1183 "`\n" + "\t" + blockDimYName + ": `" + blockDimYValStr +
1184 "` to `max(1, (new-" + blockDimXName + " * " + blockDimYValStr +
1185 ") / new-" + blockDimXName + ")`")
1186 .str();
1187 });
1188
1189 // Update the ThreadX length and the numbers of ThreadY and ThreadZ.
1190 // When the original block dimensions are compile-time
1191 // constants, compute the adjusted dimensions as constants directly so
1192 // that the GpuKernelOutliningPass can set `known_block_size` on the
1193 // outlined gpu.func.
1194 Value newBlockDimX, newBlockDimY, newBlockDimZ;
1195 if (constBlockDimX && constBlockDimY && constBlockDimZ) {
1196 int64_t bdx = *constBlockDimX;
1197 int64_t bdy = *constBlockDimY;
1198 int64_t bdz = *constBlockDimZ;
1199 int64_t alignedBdx =
1200 ((bdx + subgroupAlignMask) / subgroupSize) * subgroupSize;
1201 int64_t numXYThreads = bdx * bdy;
1202 int64_t numThreads = numXYThreads * bdz;
1203 int64_t newBdy = std::max<int64_t>(1, numXYThreads / alignedBdx);
1204 int64_t newBdz =
1205 std::max<int64_t>(1, numThreads / (alignedBdx * newBdy));
1206 newBlockDimX =
1207 arith::ConstantIndexOp::create(rewriter, loc, alignedBdx);
1208 newBlockDimY = arith::ConstantIndexOp::create(rewriter, loc, newBdy);
1209 newBlockDimZ = arith::ConstantIndexOp::create(rewriter, loc, newBdz);
1210 } else {
1211 // numXYThreads = blockDim.x * blockDim.y
1212 Value numXYThreads =
1213 arith::MulIOp::create(rewriter, loc, curBlockDimX, curBlockDimY);
1214 Value numThreads =
1215 arith::MulIOp::create(rewriter, loc, numXYThreads, curBlockDimZ);
1216 // blockDim.x = ((blockDim.x + mask) / subgroupSize) * subgroupSize
1217 Value cstMask =
1218 arith::ConstantIndexOp::create(rewriter, loc, subgroupAlignMask);
1219 Value cstSubgroupSize =
1220 arith::ConstantIndexOp::create(rewriter, loc, subgroupSize);
1221 Value padded =
1222 arith::AddIOp::create(rewriter, loc, curBlockDimX, cstMask);
1223 Value subgroupsRequired =
1224 arith::DivUIOp::create(rewriter, loc, padded, cstSubgroupSize);
1225 newBlockDimX = arith::MulIOp::create(rewriter, loc, subgroupsRequired,
1226 cstSubgroupSize);
1227 // blockDim.y = max(1, numXYThreads / blockDim.x)
1228 Value quotient =
1229 arith::DivUIOp::create(rewriter, loc, numXYThreads, newBlockDimX);
1230 Value cst1 = arith::ConstantIndexOp::create(rewriter, loc, 1);
1231 newBlockDimY = arith::MaxUIOp::create(rewriter, loc, cst1, quotient);
1232 // blockDim.z = max(1, numThreads / (blockDim.x * blockDim.y))
1233 Value newNumXYThreads =
1234 arith::MulIOp::create(rewriter, loc, newBlockDimX, newBlockDimY);
1235 quotient =
1236 arith::DivUIOp::create(rewriter, loc, numThreads, newNumXYThreads);
1237 newBlockDimZ = arith::MaxUIOp::create(rewriter, loc, cst1, quotient);
1238 }
1239
1240 launch.getBlockSizeXMutable().assign(newBlockDimX);
1241 launch.getBlockSizeYMutable().assign(newBlockDimY);
1242 launch.getBlockSizeZMutable().assign(newBlockDimZ);
1243 }
1244 }
1245
1246 if (hasFailed)
1247 return failure();
1248
1249 if (!sharedMemPrivateVarNames.empty()) {
1250 accSupport.emitRemark(computeRegion, [&]() {
1251 return (llvm::Twine("GPU shared memory used for ") +
1252 llvm::join(sharedMemPrivateVarNames, ","))
1253 .str();
1254 });
1255 }
1256
1257 rewriter.eraseOp(computeRegion);
1258 return success();
1259}

1261/// True when this accumulate is redundant in a nested reduction chain: the
1262/// value is a load of the destination memref and a sibling
1263/// acc.reduction_combine with block par_dims has already reduced %M across
1264/// threads in the block.
1265///
1266/// %v = memref.load %M[]
1267/// acc.reduction_accumulate %v to %M ...
1268/// acc.reduction_combine %M into %parent ... {block par_dims}
1269///
1270/// Lowering the accumulate again would double-count. Detection is structural;
1271/// nested reductions into per-thread privates do not match because their
1272/// combines are not block-scoped.
1273static bool isRedundantChainAccumulate(acc::ReductionAccumulateOp op) {
1274 Value memref = op.getMemref();
1275 memref::LoadOp loadOp = op.getValue().getDefiningOp<memref::LoadOp>();
1276 if (!loadOp || loadOp.getMemRef() != memref)
1277 return false;
1278 for (Operation *user : memref.getUsers()) {
1279 acc::ReductionCombineOp combineOp = dyn_cast<acc::ReductionCombineOp>(user);
1280 if (!combineOp || combineOp.getDestMemref() != memref)
1281 continue;
1282 SmallVector<mlir::acc::GPUParallelDimAttr> parDims =
1283 getReductionCombineParDims(combineOp);
1284 if (llvm::any_of(parDims, [](mlir::acc::GPUParallelDimAttr d) {
1285 return d.isAnyBlock();
1286 })) {
1287 return true;
1288 }
1289 }
1290 return false;
1291}

1320 bool hasExplicitInactiveCombine = false;
1321 bool hasBroadeningConflict = false;

1325 hasActiveWorkerCombine |= other.hasActiveWorkerCombine;
1326 hasExplicitInactiveCombine |= other.hasExplicitInactiveCombine;
1327 hasBroadeningConflict |= other.hasBroadeningConflict;
1328 if (!diagnosticOp)
1329 diagnosticOp = other.diagnosticOp;

1333/// True when executing \p op on additional ThreadY rows may change behavior.
1334static bool hasUnsafeEffectsWhenBroadening(Operation *op) {
1335 if (auto effectOp = dyn_cast<MemoryEffectOpInterface>(op)) {
1336 SmallVector<MemoryEffects::EffectInstance> effects;
1337 effectOp.getEffects(effects);
1338 return llvm::any_of(effects, [](const auto &effect) {
1339 return !isa<MemoryEffects::Read>(effect.getEffect());

1345/// True when \p accumulator is the destination of a separate block-scoped
1346/// combine, so it holds a distinct per-worker partial rather than a broadcast.
1347static bool isFedByInnerBlockCombine(acc::PrivateLocalOp accumulator,
1348 Operation *selfCombine) {
1349 if (!accumulator)
1350 return false;
1351 for (Operation *user : accumulator.getResult().getUsers()) {
1352 if (user == selfCombine)
1353 continue;
1354 auto combineOp = dyn_cast<acc::ReductionCombineOp>(user);
1355 if (!combineOp ||
1356 unwrapMemRefConversion(combineOp.getDestMemref()).getDefiningOp() !=
1357 accumulator.getOperation())
1358 continue;
1359 SmallVector<mlir::acc::GPUParallelDimAttr> parDims =
1360 getReductionCombineParDims(combineOp);
1361 if (llvm::any_of(parDims, [](mlir::acc::GPUParallelDimAttr d) {
1362 return d.isAnyBlock();
1363 }))
1364 return true;
1365 }
1366 return false;
1367}

1369/// Records whether \p combineOp requires ThreadY to remain active.
1370static void classifyThreadYCombine(ThreadYBroadeningInfo &info,
1371 Operation *combineOp, Value src, Value dest,
1372 ArrayRef<GPUParallelDimAttr> parDims,
1373 acc::ComputeRegionOp computeRegion) {
1374 bool hasThreadY = llvm::any_of(
1375 parDims, [](GPUParallelDimAttr parDim) { return parDim.isThreadY(); });
1376 bool hasBlock = llvm::any_of(
1377 parDims, [](GPUParallelDimAttr parDim) { return parDim.isAnyBlock(); });
1378 acc::PrivateLocalOp srcPrivate =
1379 unwrapMemRefConversion(src).getDefiningOp<acc::PrivateLocalOp>();
1380 acc::PrivateLocalOp destPrivate =
1381 unwrapMemRefConversion(dest).getDefiningOp<acc::PrivateLocalOp>();
1382 bool hasPrivateDest = isa<acc::ReductionCombineOp>(combineOp) && srcPrivate &&
1383 destPrivate &&
1384 getPrivatizeOp(srcPrivate, computeRegion) !=
1385 getPrivatizeOp(destPrivate, computeRegion);
1386 if (hasThreadY && hasBlock &&
1387 isThreadYPrivate(srcPrivate, hasPrivateDest, computeRegion)) {
1388 info.hasActiveWorkerCombine = true;
1389 return;
1390 }
1391
1392 // A block_y+thread_y accumulator fed by an inner block-scoped combine holds
1393 // a distinct partial per worker row, so its combine runs on every row; a
1394 // plain worker accumulate broadcasts via all_reduce and stays row-zero.
1395 if (hasThreadY && hasBlock &&
1396 isThreadYPrivate(srcPrivate, /*allowBlock=*/true, computeRegion) &&
1397 isFedByInnerBlockCombine(srcPrivate, combineOp)) {
1398 info.hasActiveWorkerCombine = true;
1399 return;
1400 }
1401
1402 info.hasExplicitInactiveCombine = true;
1403 if (!info.diagnosticOp)
1404 info.diagnosticOp = combineOp;
1405}

1407/// ThreadY must remain active for hierarchical combines over worker rows.
1408/// Broadening is rejected when siblings require inactive ThreadY or have side
1409/// effects; nested predicates enforce their own safety.
1410static ThreadYBroadeningInfo
1411analyzeThreadYBroadening(Block &predicateBlock,
1412 acc::ComputeRegionOp computeRegion) {
1413 ThreadYBroadeningInfo info;
1414 for (Operation &nestedOp : predicateBlock) {
1415 if (acc::PredicateRegionOp nestedPredicate =
1416 dyn_cast<acc::PredicateRegionOp>(nestedOp)) {
1417 ThreadYBroadeningInfo nestedInfo = analyzeThreadYBroadening(
1418 nestedPredicate.getRegion().front(), computeRegion);
1419 info.hasActiveWorkerCombine |= nestedInfo.hasActiveWorkerCombine;
1420 continue;
1421 }
1422 if (acc::ReductionCombineOp combineOp =
1423 dyn_cast<acc::ReductionCombineOp>(nestedOp)) {
1424 classifyThreadYCombine(
1425 info, combineOp, combineOp.getSrcMemref(), combineOp.getDestMemref(),
1426 getReductionCombineParDims(combineOp), computeRegion);
1427 continue;
1428 }
1429 if (acc::ReductionCombineRegionOp combineRegionOp =
1430 dyn_cast<acc::ReductionCombineRegionOp>(nestedOp)) {
1431 classifyThreadYCombine(info, combineRegionOp, combineRegionOp.getSrcVar(),
1432 combineRegionOp.getDestVar(),
1433 getReductionCombineParDims(combineRegionOp),
1434 computeRegion);
1435 continue;
1436 }
1437 if (nestedOp.getNumRegions() != 0) {
1438 if (hasUnsafeEffectsWhenBroadening(&nestedOp)) {
1439 info.hasBroadeningConflict = true;
1440 if (!info.diagnosticOp)
1441 info.diagnosticOp = &nestedOp;
1442 }
1443 for (Region &region : nestedOp.getRegions())
1444 for (Block &nestedBlock : region)
1445 info.merge(analyzeThreadYBroadening(nestedBlock, computeRegion));
1446 continue;
1447 }
1448 if (hasUnsafeEffectsWhenBroadening(&nestedOp)) {
1449 info.hasBroadeningConflict = true;
1450 if (!info.diagnosticOp)
1451 info.diagnosticOp = &nestedOp;
1452 }
1453 }
1454 return info;
1455}

1457std::pair<SmallVector<mlir::acc::GPUParallelDimAttr>,
1458 SmallVector<mlir::acc::GPUParallelDimAttr>>
1459ACCCGToGPULowering::computeActiveAndInactiveParDims(Operation *op,
1460 Block *block) {
1461 MLIRContext *ctx = computeRegion->getContext();
1462
1463 mlir::acc::GPUParallelDimAttr routineParDim;
1464 if (isInsideACCSpecializedRoutine(computeRegion)) {
1465 FunctionOpInterface funcOp =
1466 computeRegion->getParentOfType<FunctionOpInterface>();
1467 routineParDim = getSpecializedRoutineDim(funcOp, defaultPolicy);
1468 }
1469
1470 // The active set is precomputed separately from the ownership par_dims for
1471 // privatizations; prefer that attribute when present. The inactive dims are
1472 // the launch dims which are not active.
1473 if (isa<acc::PrivateLocalOp, acc::PrivatizeOp>(op)) {
1474 if (mlir::acc::ActiveParDimsAttr precomputedActiveParDims =
1475 getActiveParDimsAttr(op)) {
1476 mlir::acc::GPUParallelDimAttr lowestParDim =
1477 mlir::acc::GPUParallelDimAttr::threadXDim(ctx);
1478
1479 SmallVector<mlir::acc::GPUParallelDimAttr> launchParDims;
1480 if (routineParDim) {
1481 for (mlir::acc::GPUParallelDimAttr parDim = routineParDim;
1482 parDim.getOrder() >= lowestParDim.getOrder();
1483 parDim = parDim.getOneLower()) {
1484 mlir::acc::insertParDim(launchParDims, parDim);
1485 }
1486 } else {
1487 launchParDims = computeRegion.getLaunchParDims();
1488 }
1489
1490 SmallVector<mlir::acc::GPUParallelDimAttr> activeParDims(
1491 precomputedActiveParDims.getArray());
1492 SmallVector<mlir::acc::GPUParallelDimAttr> inactiveParDims;
1493 for (mlir::acc::GPUParallelDimAttr launchParDim : launchParDims) {
1494 if (launchParDim.getOrder() < lowestParDim.getOrder())
1495 break;
1496 if (!llvm::is_contained(activeParDims, launchParDim))
1497 inactiveParDims.push_back(launchParDim);
1498 }
1499 return std::pair{activeParDims, inactiveParDims};
1500 }
1501 }
1502
1503 SmallVector<mlir::acc::GPUParallelDimAttr> ancestorParDims =
1504 getAncestorParDims(op);
1505 // Preserve whether there were any structural ancestor par-dims before
1506 // we start augmenting them based on inner uses (e.g. private_local).
1507 // This is needed for gang redundancy check - stores to worker-indexed
1508 // private_local should not disable redundant gang execution.
1509 bool noStructuralAncestorParDims =
1510 llvm::none_of(ancestorParDims, [](auto pd) { return !pd.isSeq(); });
1511
1512 if (routineParDim) {
1513 if (routineParDim.isThreadX()) {
1514 mlir::acc::insertParDim(ancestorParDims,
1515 mlir::acc::GPUParallelDimAttr::threadYDim(ctx));
1516 }
1517 mlir::acc::insertParDim(ancestorParDims,
1518 mlir::acc::GPUParallelDimAttr::blockXDim(ctx));
1519 }
1520
1521 // acc.private_local should use the same par_dims as acc.reduction_accumulate.
1522 if (acc::PrivateLocalOp privateLocalOp = dyn_cast<acc::PrivateLocalOp>(op)) {
1523 for (Operation *user : privateLocalOp.getResult().getUsers()) {
1524 if (acc::ReductionAccumulateOp accumulateOp =
1525 dyn_cast<acc::ReductionAccumulateOp>(user)) {
1526 if (accumulateOp.getMemref() == privateLocalOp.getResult()) {
1527 for (mlir::acc::GPUParallelDimAttr parDim :
1528 accumulateOp.getParDims().getArray()) {
1529 mlir::acc::insertParDim(ancestorParDims, parDim);
1530 }
1531 }
1532 }
1533 // For decomposed complex reductions, the private_local is consumed
1534 // by an acc.reduction_combine{,_region} (no acc.reduction_accumulate
1535 // user). Mirror the par_dims so this private_local is treated as the
1536 // accumulator at the same parallelism level as a scalar reduction
1537 // would be (per-thread, not block-shared).
1538 if (acc::ReductionCombineOp combineOp =
1539 dyn_cast<acc::ReductionCombineOp>(user)) {
1540 if (combineOp.getSrcMemref() == privateLocalOp.getResult()) {
1541 for (mlir::acc::GPUParallelDimAttr parDim :
1542 getReductionCombineParDims(combineOp)) {
1543 mlir::acc::insertParDim(ancestorParDims, parDim);
1544 }
1545 }
1546 }
1547 if (auto combineRegionOp =
1548 dyn_cast<acc::ReductionCombineRegionOp>(user)) {
1549 if (combineRegionOp.getSrcVar() == privateLocalOp.getResult()) {
1550 for (mlir::acc::GPUParallelDimAttr parDim :
1551 getReductionCombineParDims(combineRegionOp)) {
1552 mlir::acc::insertParDim(ancestorParDims, parDim);
1553 }
1554 }
1555 }
1556 }
1557 // A dynamically-shaped privatization is materialized as a strided view
1558 // whose type does not reveal its parallel scope. Keep its own par_dims
1559 // active so the view is materialized (and predicated) per owning
1560 // thread/block. Statically-shaped privatizations are left to the stack-fit
1561 // decision so only do this for dynamic shapes.
1562 acc::PrivateType privTy =
1563 cast<acc::PrivateType>(privateLocalOp.getPrivatized().getType());
1564 MemRefType baseTy = getPrivateBaseMemRefType(
1565 privTy.getBaseTy(), computeRegion->getParentOfType<ModuleOp>());
1566 if (!baseTy.hasStaticShape()) {
1567 GPUParallelDimsAttr ownParDims =
1568 getPrivateParDims(privateLocalOp, computeRegion);
1569 if (ownParDims)
1570 for (GPUParallelDimAttr parDim : ownParDims.getArray())
1571 mlir::acc::insertParDim(ancestorParDims, parDim);
1572 }
1573 }
1574
1575 bool hasBlock = false;
1576 for (mlir::acc::GPUParallelDimAttr parDim : ancestorParDims)
1577 if (parDim.isAnyBlock())
1578 hasBlock = true;
1579
1580 mlir::acc::GPUParallelDimAttr lowestParDim =
1581 mlir::acc::GPUParallelDimAttr::threadXDim(ctx);
1582 if (block) {
1583 ThreadYBroadeningInfo threadYInfo =
1584 analyzeThreadYBroadening(*block, computeRegion);
1585
1586 auto applyCombineParDims =
1587 [&](ArrayRef<mlir::acc::GPUParallelDimAttr> combineParDims) {
1588 for (mlir::acc::GPUParallelDimAttr parDim : combineParDims)
1589 mlir::acc::removeParDim(ancestorParDims, parDim);
1590 return success();
1591 };
1592 block->walk([&](Operation *op) -> WalkResult {
1593 // Writes to acc.private_local must keep the privatization's par_dims
1594 // active so the write runs on all owning threads instead of being
1595 // predicated to a single lane.
1596 auto addPrivateStoreParDims = [&](Value target) {
1597 if (auto privateLocalOp = getPrivateLocalForMemref(target)) {
1598 GPUParallelDimsAttr parDimsAttr =
1599 getPrivateParDims(privateLocalOp, computeRegion);
1600 if (parDimsAttr)
1601 for (auto parDim : parDimsAttr.getArray())
1602 mlir::acc::insertParDim(ancestorParDims, parDim);
1603 }
1604 };
1605 if (auto memEffects = dyn_cast<MemoryEffectOpInterface>(op)) {
1606 SmallVector<MemoryEffects::EffectInstance> effects;
1607 memEffects.getEffects(effects);
1608 for (const MemoryEffects::EffectInstance &effect : effects) {
1609 if (isa<MemoryEffects::Write>(effect.getEffect()) &&
1610 effect.getValue())
1611 addPrivateStoreParDims(effect.getValue());
1612 }
1613 }
1614 // Consider ACC routine calls; routine calls should be predicated up to
1615 // one level above the parallel dimension of the callee.
1616 if (CallOpInterface callOp = dyn_cast<CallOpInterface>(op)) {
1617 if (mlir::acc::GPUParallelDimAttr parDim =
1618 getAccRoutineCallParDim(callOp, defaultPolicy)) {
1619 if (parDim.isBlockZ())
1620 lowestParDim = parDim;
1621 else
1622 lowestParDim = parDim.getOneHigher();
1623 }
1624 }
1625 // acc.reduction_combine_region should be predicated with the par_dims of
1626 // acc.reduction_accumulate. This is required when using combine between
1627 // kernel and loop in combined constructs.
1628 if (acc::ReductionCombineOp reductionCombineOp =
1629 dyn_cast<acc::ReductionCombineOp>(op)) {
1630 if (failed(applyCombineParDims(
1631 getReductionCombineParDims(reductionCombineOp))))
1632 return WalkResult::interrupt();
1633 }
1634 if (acc::ReductionCombineRegionOp combineRegionOp =
1635 dyn_cast<acc::ReductionCombineRegionOp>(op)) {
1636 if (failed(applyCombineParDims(
1637 getReductionCombineParDims(combineRegionOp))))
1638 return WalkResult::interrupt();
1639 }
1640 // An array accumulate reduces across its par_dims via gpu.all_reduce, so
1641 // all those threads must execute it - treat them as active (unlike the
1642 // scalar accumulate, which is active through its enclosing scf.parallel).
1643 if (acc::ReductionAccumulateArrayOp accArrayOp =
1644 dyn_cast<acc::ReductionAccumulateArrayOp>(op)) {
1645 for (mlir::acc::GPUParallelDimAttr parDim :
1646 accArrayOp.getParDims().getArray()) {
1647 mlir::acc::insertParDim(ancestorParDims, parDim);
1648 }
1649 }
1650 return WalkResult::advance();
1651 });
1652 mlir::acc::GPUParallelDimAttr threadY =
1653 mlir::acc::GPUParallelDimAttr::threadYDim(ctx);
1654 bool baselineThreadYActive = llvm::is_contained(ancestorParDims, threadY);
1655 if (threadYInfo.hasActiveWorkerCombine && !baselineThreadYActive) {
1656 if (threadYInfo.hasExplicitInactiveCombine ||
1657 threadYInfo.hasBroadeningConflict) {
1658 Operation *diagnosticOp =
1659 threadYInfo.diagnosticOp ? threadYInfo.diagnosticOp : op;
1660 (void)accSupport.emitNYI(
1661 diagnosticOp->getLoc(),
1662 "operations in the same predicate region require incompatible "
1663 "ThreadY predication");
1664 hasFailed = true;
1665 return {};
1666 }
1667 mlir::acc::insertParDim(ancestorParDims, threadY);
1668 }
1669 }
1670
1671 // Obtain launch dimensions
1672 SmallVector<mlir::acc::GPUParallelDimAttr> launchParDims;
1673 if (routineParDim) {
1674 for (mlir::acc::GPUParallelDimAttr parDim = routineParDim;
1675 parDim.getOrder() >= lowestParDim.getOrder();
1676 parDim = parDim.getOneLower()) {
1677 mlir::acc::insertParDim(launchParDims, parDim);
1678 }
1679 } else {
1680 launchParDims = computeRegion.getLaunchParDims();
1681 }
1682
1683 // Compute dimensions that execute op
1684 SmallVector<mlir::acc::GPUParallelDimAttr> activeParDims, inactiveParDims;
1685 for (mlir::acc::GPUParallelDimAttr launchParDim : launchParDims) {
1686 if (launchParDim.getOrder() < lowestParDim.getOrder())
1687 break;
1688 if (llvm::find(ancestorParDims, launchParDim) != ancestorParDims.end() ||
1689 (launchParDim.isAnyBlock() &&
1690 (noStructuralAncestorParDims || hasBlock))) {
1691 activeParDims.push_back(launchParDim);
1692 } else {
1693 inactiveParDims.push_back(launchParDim);
1694 }

1718 Location loc, mlir::acc::GPUParallelDimsAttr parDimsAttr) {
1719 bool hasAnyBlock = false, hasThreadY = false, hasThreadX = false;
1720 for (auto parDim : parDimsAttr.getArray()) {
1721 if (parDim.isAnyBlock())
1722 hasAnyBlock = true;
1723 if (parDim.isThreadY())
1724 hasThreadY = true;
1725 if (parDim.isThreadX())

1735void ACCCGToGPULowering::createPerRowBarrier(Location loc) {
1736 // A row of a (1, N, 1) launch is a single thread; a subgroup barrier would
1737 // instead synchronize N workers.
1738 if (isSingleThreadWorkerLaunch())
1739 return;
1740
1741 // A worker-only row narrower than a subgroup covers only part of one, so
1742 // synchronize just its lanes. Both paths leave hasThreadYBarrier unset, so
1743 // the launch is not padded on account of this barrier.
1744 if (isWorkerOnlyShuffleLaunch() && staticBlockDimX < options.subgroupSize) {
1745 emitGPUBarrierRow(rewriter, loc, staticBlockDimX, options.subgroupSize);
1746 return;
1747 }
1748
1749 hasThreadYBarrier = true;
1750
1751 if (staticBlockDimX <= options.subgroupSize) {
1752 emitGPUBarrierSubgroup(rewriter, loc);
1753 return;
1754 }
1755
1756 if (options.deviceType != mlir::acc::DeviceType::Nvidia) {
1757 (void)accSupport.emitNYI(
1758 loc,
1759 "per-row barrier to support worker parallelism on non-NVIDIA device");
1760 }
1761
1762 // Per-row barrier with fully runtime branching.
1763 // Three mutually exclusive paths:
1764 // blockDim.y == 1 -> gpu.barrier (workgroup-wide, only one worker)
1765 // blockDim.x <= subgroupSize -> gpu.barrier scope<subgroup>
1766 // blockDim.x > subgroupSize -> nvvm.barrier (tid.y + 1), blockDim.x
1767 // (named)
1768 //
1769 // Per-row barriers use tid.y+1 so IDs start at 1, avoiding clash with
1770 // barrier 0. When blockDim.y >= 16, worker 15's ID (16) wraps to
1771 // physical barrier 0; this is safe because named barriers are reusable
1772 // resources - workgroup-wide and per-row barriers on the same physical
1773 // barrier execute at different program points and never overlap.
1774 Value blockDimX = gpu::BlockDimOp::create(
1775 rewriter, loc, rewriter.getIndexType(), gpu::Dimension::x);
1776 Value blockDimY = gpu::BlockDimOp::create(
1777 rewriter, loc, rewriter.getIndexType(), gpu::Dimension::y);
1778 Value cst1 = arith::ConstantIndexOp::create(rewriter, loc, 1);
1779 Value isSingleWorker = arith::CmpIOp::create(
1780 rewriter, loc, arith::CmpIPredicate::eq, blockDimY, cst1);
1781
1782 auto outerIf = scf::IfOp::create(rewriter, loc, isSingleWorker,
1783 /*withElseRegion=*/true);
1784
1785 // Then: blockDim.y == 1 -> workgroup-wide barrier (safe, only one worker)
1786 rewriter.setInsertionPointToStart(&outerIf.getThenRegion().front());
1787 emitGPUBarrierWorkgroup(rewriter, loc);
1788
1789 // Else: blockDim.y > 1 - choose between subgroup sync and named barrier
1790 rewriter.setInsertionPointToStart(&outerIf.getElseRegion().front());
1791 Value cstSubgroupSize =
1792 arith::ConstantIndexOp::create(rewriter, loc, options.subgroupSize);
1793 Value isSubgroupSized = arith::CmpIOp::create(
1794 rewriter, loc, arith::CmpIPredicate::ule, blockDimX, cstSubgroupSize);
1795
1796 auto innerIf = scf::IfOp::create(rewriter, loc, isSubgroupSized,
1797 /*withElseRegion=*/true);
1798
1799 // Then: blockDim.x <= subgroupSize -> subgroup barrier (one worker per
1800 // subgroup)
1801 rewriter.setInsertionPointToStart(&innerIf.getThenRegion().front());
1802 emitGPUBarrierSubgroup(rewriter, loc);
1803
1804 // Else: blockDim.x > subgroupSize -> per-row named barrier with tid.y + 1.
1805 // The 1024-thread-per-block hardware limit with subgroup-aligned blockDim.x
1806 // (>= 64 here) guarantees blockDim.y <= 16, so IDs span at most 16
1807 // physical barriers (0-15) with no aliasing across workers.
1808 rewriter.setInsertionPointToStart(&innerIf.getElseRegion().front());
1809 Value threadYId = gpu::ThreadIdOp::create(
1810 rewriter, loc, rewriter.getIndexType(), gpu::Dimension::y);
1811 Value barrierId = arith::AddIOp::create(rewriter, loc, threadYId, cst1);
1812 Type i32Ty = rewriter.getI32Type();
1813 Value barrierId32 =
1814 arith::IndexCastOp::create(rewriter, loc, i32Ty, barrierId);
1815 Value numberOfThreads32 =
1816 arith::IndexCastOp::create(rewriter, loc, i32Ty, blockDimX);
1817
1818 // GPU dialect named barriers do not have a means to create a custom barrier
1819 // id. Thus use nvvm directly.
1820 assert(options.deviceType == mlir::acc::DeviceType::Nvidia);
1821 NVVM::BarrierOp::create(rewriter, loc, barrierId32, numberOfThreads32);
1822
1823 rewriter.setInsertionPointAfter(outerIf);
1824}

1834 return true;

1838 nested = true;
1839 return WalkResult::interrupt();
1840 }
1841 return WalkResult::advance();

1844 return true;

1846 return false;

1872// A sequential loop that uses gang-private shared memory needs a
1873// workgroup-wide barrier afterward so every thread in the block observes the
1874// same state. When more work still lies between the loop and the next
1875// thread-reconvergence point in the same block, that barrier must follow that
1876// work; not sit immediately after the loop.

1881/// Marks the end of a loop body's iteration in the current block.

1883 return isa<scf::ReduceOp, scf::YieldOp, acc::YieldOp>(op);

1886/// Thread-reconvergence point where any postponed post-loop barrier for earlier
1887/// loops in this block must be inserted before proceeding.
1888static bool isDeferredBarrierFlushPoint(Operation *op) {

1890 return true;

1898 mlir::acc::getParDimsAttr(parallelOp)) {
1899 if (parDims.getArray().size() == 1 &&
1900 parDims.getArray().front().isSeq()) {
1901 return true;

1903 }
1904 }
1905 }
1906 return false;

1920// Insert a barrier after a sequential loop when the surrounding parallel
1921// structure requires it, so that block threads observe the loop's writes to
1922// gang-/block-private shared memory before a later loop reads them. The barrier
1923// must land at an enclosing collective (block- or worker-level) point where all
1924// threads converge, so it cannot deadlock.
1925//
1926// The function walks the parent hierarchy of the loop to decide where (if at
1927// all) to place the barrier:
1928// - No enclosing parallel loop (top-level worksharing loop): emit a
1929// workgroup barrier when the loop writes shared memory and a later sibling
1930// loop may read it.
1931// - Nearest scf.parallel parent is itself sequential: a barrier *after* it
1932// would be unsafe when threads have varying iteration counts. However, when
1933// the loop has thread-level sub-loops collaborating on block-private memory
1934// (all threads participate every iteration), walk up to the block-level
1935// ancestor and insert the barrier there.
1936// - Nearest scf.parallel parent is a non-sequential parallel loop: walk up to
1937// the block-level (or worker/thread-y) ancestor and insert the barrier
1938// there, handling gang-redundant init loops and grid-stride remainders.
1939void ACCCGToGPULowering::createBarrierAfterSeqLoop(Operation *loopOp) {
1940 scf::ParallelOp wsLoop = loopOp->getParentOfType<scf::ParallelOp>();
1941 if (!wsLoop) {
1942 // loopOp is a worksharing loop at the kernel-body top level (no enclosing
1943 // parallel loop). When it writes gang-private shared memory
1944 // that a later sibling loop reads, a workgroup barrier is needed between
1945 // them. The point just after a top-level loop is uniform (all threads reach
1946 // it), so a workgroup-wide barrier here cannot deadlock. Loops writing only
1947 // global/thread-private memory, or with no subsequent reader, get none.
1948 if (mayWriteSharedMemory(loopOp) && hasSubsequentLoopSibling(loopOp))
1949 emitGPUBarrierWorkgroup(rewriter, loopOp->getLoc());
1950 return;
1951 }
1952
1953 bool parentIsSeq = false;
1954 if (mlir::acc::GPUParallelDimsAttr wsParDims =
1955 mlir::acc::getParDimsAttr(wsLoop)) {
1956 if (wsParDims.getArray().size() == 1 &&
1957 wsParDims.getArray().front().isSeq()) {
1958 parentIsSeq = true;
1959 }
1960 }
1961
1962 if (parentIsSeq) {
1963 // loopOp is nested inside a sequential parent loop.
1964 // When the loop contains thread-level sub-loops,
1965 // multiple threads collaborate on shared (block-private) memory
1966 // within each iteration. Insert a block-level barrier.
1967 bool hasThreadSubLoop = false;
1968 loopOp->walk([&](scf::ParallelOp innerPar) -> WalkResult {
1969 if (innerPar.getOperation() == loopOp)
1970 return WalkResult::advance();
1971 if (mlir::acc::GPUParallelDimsAttr dims =
1972 mlir::acc::getParDimsAttr(innerPar)) {
1973 for (auto d : dims.getArray()) {
1974 if (d.isThreadX() || d.isThreadY()) {
1975 hasThreadSubLoop = true;
1976 return WalkResult::interrupt();
1977 }
1978 }
1979 }
1980 return WalkResult::advance();
1981 });
1982 if (!hasThreadSubLoop)
1983 return;
1984 scf::ParallelOp threadLoop = wsLoop->getParentOfType<scf::ParallelOp>();
1985 if (!threadLoop)
1986 return;
1987 scf::ParallelOp blockLoop = threadLoop->getParentOfType<scf::ParallelOp>();
1988 if (!blockLoop)
1989 return;
1990 mlir::acc::GPUParallelDimsAttr parDimsAttr =
1991 mlir::acc::getParDimsAttr(blockLoop);
1992 if (parDimsAttr.hasOnlyBlockLevel())
1993 createBarrier(loopOp->getLoc(), parDimsAttr);
1994 return;
1995 }
1996
1997 // Parent is a non-sequential parallel loop. Walk up to find the block-level
1998 // ancestor and insert a barrier there.
1999 scf::ParallelOp seqLoop = wsLoop->getParentOfType<scf::ParallelOp>();
2000 if (!seqLoop) {
2001 // wsLoop is a worksharing loop directly under the compute region with no
2002 // gang ancestor: a gang-redundant init loop (e.g. a thread-level loop that
2003 // every gang runs to fill its own copy of gang-private shared memory). If
2004 // it writes gang-private (block-only) memory that a later sibling loop
2005 // reads, insert a workgroup barrier between them. The barrier sits just
2006 // after the top-level loop where all threads converge, so it cannot
2007 // deadlock.
2008 if (mayWriteSharedMemory(loopOp) && hasSubsequentLoopSibling(wsLoop))
2009 emitGPUBarrierWorkgroup(rewriter, loopOp->getLoc());
2010 return;
2011 }
2012 if (scf::ParallelOp outerParLoop =
2013 seqLoop->getParentOfType<scf::ParallelOp>()) {
2014 mlir::acc::GPUParallelDimsAttr parDimsAttr =
2015 mlir::acc::getParDimsAttr(outerParLoop);
2016 if (parDimsAttr.hasOnlyBlockLevel()) {
2017 createBarrier(loopOp->getLoc(), parDimsAttr);
2018 } else if (parDimsAttr.hasOnlyThreadYLevel()) {
2019 createPerRowBarrier(loopOp->getLoc());
2020 } else if (parDimsAttr && parDimsAttr.isSeq()) {
2021 // outerParLoop is a sequential grid-stride remainder of a partitioned
2022 // gang loop, not the gang. Walk past the remainder(s) to the block-level
2023 // gang and barrier there.
2024 for (Operation *gangLoop =
2025 outerParLoop->getParentOfType<scf::ParallelOp>();
2026 gangLoop; gangLoop = gangLoop->getParentOfType<scf::ParallelOp>()) {
2027 mlir::acc::GPUParallelDimsAttr gangDims =
2028 mlir::acc::getParDimsAttr(gangLoop);
2029 if (!gangDims)
2030 break;
2031 if (gangDims.hasOnlyBlockLevel()) {
2032 createBarrier(loopOp->getLoc(), gangDims);
2033 break;
2034 }
2035 if (!gangDims.isSeq())
2036 break;
2037 }
2038 }
2039 return;
2040 }
2041
2042 // No block-level ancestor above seqLoop: the gang loop is seqLoop itself, a
2043 // gang(+vector) loop directly under the compute region. The fixed-depth
2044 // lookup above misses it (issue: a gang+vector init loop that writes
2045 // gang-private shared memory must be followed by a barrier before another
2046 // loop reads it). Only needed when the loop writes such shared memory; loops
2047 // writing global/thread-private memory (e.g. a plain gang-vector saxpy) do
2048 // not need one here.
2049 mlir::acc::GPUParallelDimsAttr parDimsAttr =
2050 mlir::acc::getParDimsAttr(seqLoop);
2051 if (parDimsAttr && parDimsAttr.hasOnlyBlockLevel() &&
2052 mayWriteSharedMemory(loopOp)) {
2053 createBarrier(loopOp->getLoc(), parDimsAttr);
2054 }
2055}

2057bool ACCCGToGPULowering::mayWriteSharedMemory(Operation *loopOp) {
2058 bool found = false;
2059 loopOp->walk([&](memref::StoreOp storeOp) {
2060 // Trace the store target back to its backing acc.private_local through
2061 // view/cast/box ops, then check the privatization is gang-level.
2062 llvm::SmallVector<Value, 8> worklist{storeOp.getMemref()};
2063 llvm::SmallPtrSet<Value, 8> seen;
2064 while (!worklist.empty()) {
2065 Value v = worklist.pop_back_val();
2066 if (!seen.insert(v).second)
2067 continue;
2068 Operation *def = v.getDefiningOp();
2069 if (!def)
2070 continue;
2071 if (acc::PrivateLocalOp privateLocal =
2072 dyn_cast<acc::PrivateLocalOp>(def)) {
2073 acc::PrivatizeOp privatizeOp =
2074 getPrivatizeOp(privateLocal, computeRegion);
2075 // Only gang-level (block, no thread) private memory is a single copy
2076 // shared across the workgroup's threads, so a write needs a workgroup
2077 // barrier before another thread reads it. A [block_x, thread_x]
2078 // privatization is thread-private (one copy per thread); barriering on
2079 // it would deadlock when threads take different grid-stride iteration
2080 // counts.
2081 if (mlir::acc::GPUParallelDimsAttr parDims =
2082 privatizeOp.getParDimsAttr()) {
2083 bool hasBlock = false, hasThread = false;
2084 for (mlir::acc::GPUParallelDimAttr d : parDims.getArray()) {
2085 if (d.isAnyBlock())
2086 hasBlock = true;
2087 if (d.isThreadX() || d.isThreadY())
2088 hasThread = true;
2089 }
2090 if (hasBlock && !hasThread) {
2091 found = true;
2092 return WalkResult::interrupt();
2093 }
2094 }
2095 continue;
2096 }
2097 worklist.append(def->getOperands().begin(), def->getOperands().end());
2098 }
2099 return WalkResult::advance();
2100 });
2101 return found;
2102}

2242void ACCCGToGPULowering::processPredicateRegion(
2243 acc::PredicateRegionOp interOp) {
2244 LLVM_DEBUG(llvm::dbgs() << "processing predicate region: ";
2245 interOp->print(llvm::dbgs()); llvm::dbgs() << "\n");
2246 Location loc = interOp->getLoc();
2247
2248 std::pair<SmallVector<mlir::acc::GPUParallelDimAttr>,
2249 SmallVector<mlir::acc::GPUParallelDimAttr>>
2250 parDimsPair = computeActiveAndInactiveParDims(
2251 interOp, &interOp.getRegion().front());
2252 if (hasFailed)
2253 return;
2254
2255 // If ThreadY reduction exists, subgroup alignment is applied
2256 // (blockDim.x = subgroupSize), so ThreadX lanes exist even without explicit
2257 // ThreadX parallelism. Add ThreadX to inactiveParDims if not already present.
2258 // Exception: if this region contains a thread-level (vector or worker)
2259 // routine call, all ThreadX threads must reach the call so the routine's
2260 // workgroup-wide barriers (e.g. shared memory alloca sync) converge.
2261 // A single-thread worker launch is left unaligned, so it has no such lanes.

2263 MLIRContext *ctx = computeRegion->getContext();

2265 mlir::acc::GPUParallelDimAttr::threadXDim(ctx);

2267 llvm::any_of(parDimsPair.first, [](mlir::acc::GPUParallelDimAttr pd) {
2268 return pd.isThreadX();
2269 });
2270 bool hasThreadXInInactive =
2271 llvm::any_of(parDimsPair.second, [](mlir::acc::GPUParallelDimAttr pd) {
2272 return pd.isThreadX();
2273 });
2274
2275 // Check if THIS predicate region contains a thread-level routine call.
2276 // We use the pre-computed hasThreadLevelRoutineCall as an early-out
2277 // optimization.
2278 bool regionHasThreadLevelRoutineCall = false;
2279 if (hasThreadLevelRoutineCall) {
2280 interOp.getRegion().walk([&](CallOpInterface callOp) {
2281 if (mlir::acc::GPUParallelDimAttr parDim =
2282 getAccRoutineCallParDim(callOp, defaultPolicy)) {
2283 if (parDim.isThreadX() || parDim.isThreadY()) {
2284 regionHasThreadLevelRoutineCall = true;
2285 return WalkResult::interrupt();
2286 }
2287 }
2288 return WalkResult::advance();
2289 });
2290 }
2291
2292 if (!hasThreadXInActive && !hasThreadXInInactive &&
2293 !regionHasThreadLevelRoutineCall) {
2294 parDimsPair.second.push_back(threadXParDim);
2295 }
2296 }

2298 if (Value predicate = emitPredicate(loc, parDimsPair.second)) {
2299 LLVM_DEBUG(llvm::dbgs() << "predicate: " << predicate << "\n");
2300 bool isInsideThreadXLoop = false;
2301 bool isInsideThreadYLoop = false;
2302 for (auto parDim : parDimsPair.first) {
2303 if (parDim.isThreadX())
2304 isInsideThreadXLoop = true;
2305 if (parDim.isThreadY())
2306 isInsideThreadYLoop = true;
2307 }
2308 // Emits the reconvergence barrier matching this region's predication level,
2309 // at the current insertion point. Called for both before and after the
2310 // predicated store. Below is the pre-predicate barrier; the post-predicate
2311 // barrier is emitted after the ifOp.
2312 auto emitReconvergenceBarrier = [&]() {
2313 if (isInsideThreadXLoop) {
2314 // Inside ThreadX loop - skip barrier
2315 } else if (isInsideThreadYLoop) {
2316 // Inside ThreadY loop
2317 if (!isInsideACCSpecializedRoutine(computeRegion)) {
2318 // Add barrier if ThreadX is predicated (lane 0 writes must be
2319 // visible to all lanes before they read).
2320 bool predicatesThreadX = llvm::any_of(
2321 parDimsPair.second,
2322 [](mlir::acc::GPUParallelDimAttr pd) { return pd.isThreadX(); });
2323 if (predicatesThreadX) {
2324 createBarrier(loc, mlir::acc::GPUParallelDimsAttr::get(
2325 interOp->getContext(), parDimsPair.second));
2326 }
2327 }
2328 // For acc routine ThreadY routines, skip barrier
2329 } else if (!parDimsPair.first.empty()) {
2330 // Inside block loop
2331 createBarrier(loc, mlir::acc::GPUParallelDimsAttr::get(
2332 interOp->getContext(), parDimsPair.first));
2333 } else {
2334 // Top level
2335 createBarrier(loc, mlir::acc::GPUParallelDimsAttr::get(
2336 interOp->getContext(), parDimsPair.second));
2337 }
2338 };
2339
2340 // A gang-level (block, no thread) shared slot written here and reused on
2341 // the next iteration of an enclosing sequential loop must not be
2342 // overwritten before all threads read the current value. Emit the
2343 // reconvergence barrier before the store too (the post-store barrier below
2344 // only orders this iteration's store->read). Restricted to the block-level
2345 // case: such loops are run uniformly by every workgroup thread, so the
2346 // workgroup barrier cannot deadlock; worker/vector (thread_y/thread_x)
2347 // loops may have divergent trip counts.
2348 PrivateMemScope scope = needsPreStoreReuseBarrier(interOp);
2349 if (scope == PrivateMemScope::Gang)
2350 emitGPUBarrierWorkgroup(rewriter, loc);
2351 else if (scope == PrivateMemScope::Worker)
2352 createPerRowBarrier(loc);
2353
2354 auto ifOp = scf::IfOp::create(rewriter, loc, predicate,
2355 /*withElseRegion=*/false);
2356 Region &thenRegion = ifOp.getThenRegion();
2357 Block &thenBlock = thenRegion.back();
2358 rewriter.setInsertionPoint(thenBlock.getTerminator());
2359 // Ops in a predicate region may need to be further processed, recurse
2360 for (auto &bodyOp : interOp.getRegion().front().getOperations()) {
2361 // If the store's value loads from a block-level reduction
2362 // memref, convert to atomic for cross-block correctness.
2363 // Only at kernel top level (no active thread dims).
2364 if (memref::StoreOp storeOp = dyn_cast<memref::StoreOp>(&bodyOp)) {
2365 std::optional<arith::AtomicRMWKind> blockReduceKind;
2366 bool failedReductionKind = false;
2367 Value storeVal = storeOp.getValueToStore();
2368 if (storeVal.getDefiningOp()) {
2369 // Walk the epilogue def-chain within the enclosing block to find
2370 // the block-level accumulate load feeding this store. Epilogue ops
2371 // (type conversions, arithmetic, etc.) are traversed transparently.
2372 // Non-acc loads and values defined outside the block are treated as
2373 // loop-invariant and return nullopt, bounding the search naturally.
2374 Block *epilogueBlock = interOp->getBlock();
2375 auto findBlockAccLoad =
2376 [&](auto &self,
2377 Value val) -> std::optional<arith::AtomicRMWKind> {
2378 Operation *def = val.getDefiningOp();
2379 if (!def || def->getBlock() != epilogueBlock)
2380 return std::nullopt;
2381 if (memref::LoadOp loadOp = dyn_cast<memref::LoadOp>(def)) {
2382 for (auto *user : loadOp.getMemRef().getUsers()) {
2383 if (acc::ReductionAccumulateOp accOp =
2384 dyn_cast<acc::ReductionAccumulateOp>(user)) {
2385 if (llvm::any_of(accOp.getParDims().getArray(),
2386 [](mlir::acc::GPUParallelDimAttr pd) {
2387 return pd.isAnyBlock();
2388 })) {
2389 FailureOr<arith::AtomicRMWKind> kind = getReductionKind(
2390 accOp.getReductionOperator(),
2391 accOp.getValue().getType(), accOp.getLoc());
2392 if (failed(kind)) {
2393 failedReductionKind = true;
2394 return std::nullopt;
2395 }
2396 return *kind;
2397 }
2398 }
2399 }
2400 return std::nullopt;
2401 }
2402 for (Value operand : def->getOperands())
2403 if (auto kind = self(self, operand))
2404 return kind;
2405 return std::nullopt;
2406 };
2407 blockReduceKind = findBlockAccLoad(findBlockAccLoad, storeVal);
2408 }
2409 if (failedReductionKind)
2410 return;
2411 if (blockReduceKind) {
2412 Value input = mapping.lookupOrDefault(storeOp.getValueToStore());
2413 Value memref = mapping.lookupOrDefault(storeOp.getMemref());
2414 bool threadIsActive = llvm::any_of(
2415 parDimsPair.first, [](mlir::acc::GPUParallelDimAttr pd) {
2416 return !pd.isAnyBlock();
2417 });
2418 if (!threadIsActive &&
2419 !isa_and_nonnull<memref::AllocaOp>(
2420 unwrapMemRefConversion(memref).getDefiningOp())) {
2421 // Initialize the destination to the reduction identity before
2422 // cross-block atomics so that the final result reflects pure
2423 // assignment semantics (e.g. r = sum(a)), not accumulation
2424 // on top of the pre-kernel value.
2425 if (launch) {
2426 MemRefType memrefTy = cast<MemRefType>(memref.getType());
2427 // Map the store indices for ranked memrefs.
2428 SmallVector<Value> initIndices;
2429 for (Value idx : storeOp.getIndices())
2430 initIndices.push_back(mapping.lookupOrDefault(idx));
2431
2432 OpBuilder::InsertionGuard guard(rewriter);
2433 Block &launchBody = launch.getBody().front();
2434 Operation *insertBefore = nullptr;
2435
2436 launchBody.walk([&](scf::ParallelOp parOp) -> WalkResult {
2437 for (Operation *parent = parOp->getParentOp(); parent;
2438 parent = parent->getParentOp()) {
2439 if (parent == launch.getOperation())
2440 break;
2441 if (isa<scf::ParallelOp>(parent))
2442 return WalkResult::advance();
2443 }
2444 insertBefore = parOp.getOperation();
2445 return WalkResult::interrupt();
2446 });
2447 if (insertBefore)
2448 rewriter.setInsertionPoint(insertBefore);
2449 else
2450 rewriter.setInsertionPointToStart(&launchBody);
2451 // Recursively re-materialize operations whose definitions
2452 // do not dominate the insertion point. A single-level clone
2453 // is insufficient when the value is produced by a chain of
2454 // operations (e.g. reinterpret_cast depending on box_dims,
2455 // divsi, convert, etc.) that are all defined after the
2456 // insertion point.
2457 DominanceInfo domInfo(launch);
2458 IRMapping initMapping;
2459 std::function<Value(Value)> materialize =
2460 [&](Value val) -> Value {
2461 Operation *defOp = val.getDefiningOp();
2462 if (!defOp)
2463 return val;
2464 if (domInfo.dominates(defOp, &*rewriter.getInsertionPoint()))
2465 return val;
2466 if (auto mapped = initMapping.lookupOrNull(val))
2467 return mapped;
2468 // Recurse on operands; the recursive call seeds
2469 // initMapping for any operand it clones, which the
2470 // subsequent rewriter.clone(..., initMapping) picks up.
2471 for (Value operand : defOp->getOperands())
2472 materialize(operand);
2473 Operation *cloned = rewriter.clone(*defOp, initMapping);
2474 for (auto [orig, clonedRes] :
2475 llvm::zip(defOp->getResults(), cloned->getResults())) {
2476 initMapping.map(orig, clonedRes);
2477 }
2478 return initMapping.lookup(val);
2479 };
2480 Value initMemref = materialize(memref);
2481 for (auto &idx : initIndices)
2482 idx = materialize(idx);
2483 Value identityVal = createIdentityValue(
2484 rewriter, loc, memrefTy.getElementType(), *blockReduceKind,
2485 /*useOnlyFiniteValue=*/true);
2486 Value blockId = gpu::BlockIdOp::create(
2487 rewriter, loc, rewriter.getIndexType(), gpu::Dimension::x);
2488 Value threadId = gpu::ThreadIdOp::create(
2489 rewriter, loc, rewriter.getIndexType(), gpu::Dimension::x);
2490 Value zero = arith::ConstantIndexOp::create(rewriter, loc, 0);
2491 Value isBlock0 = arith::CmpIOp::create(
2492 rewriter, loc, arith::CmpIPredicate::eq, blockId, zero);
2493 Value isThread0 = arith::CmpIOp::create(
2494 rewriter, loc, arith::CmpIPredicate::eq, threadId, zero);
2495 Value isFirstThread =
2496 arith::AndIOp::create(rewriter, loc, isBlock0, isThread0);
2497 auto initIf = scf::IfOp::create(rewriter, loc, isFirstThread,
2498 /*withElseRegion=*/false);
2499 rewriter.setInsertionPoint(
2500 initIf.getThenRegion().back().getTerminator());
2501 memref::StoreOp::create(rewriter, loc, identityVal, initMemref,
2502 initIndices);
2503 rewriter.setInsertionPointAfter(initIf);
2504 gpu::BarrierOp::create(rewriter, loc);
2505 }
2506 SmallVector<Value> atomicIndices;
2507 for (Value idx : storeOp.getIndices())
2508 atomicIndices.push_back(mapping.lookupOrDefault(idx));
2509 constructAtomicAccumulation(loc, memref, atomicIndices, input,
2510 *blockReduceKind);
2511 continue;
2512 }
2513 }
2514 }
2515 processOp(&bodyOp);
2516 }
2517 rewriter.setInsertionPointAfter(ifOp);
2518 emitReconvergenceBarrier();
2519 } else {
2520 // Ops in a predicate region may need to be further processed, recurse
2521 for (auto &bodyOp : interOp.getRegion().front().getOperations())
2522 processOp(&bodyOp);
2523 }
2524}

2526// clang-format off
2527//
2528// Allocates private storage and broadcasts its pointer through a shared
2529// memref-of-memref slot, using the same predication utility as predicate
2530// regions.
2531//
2532// %0 = arith.cmpi eq, %thread_id_y, %c0 : index
2533// %1 = arith.cmpi eq, %thread_id_x, %c0 : index
2534// %2 = arith.andi %1, %0 : i1
2535// scf.if %2 {
2536// %alloc = memref.alloc() : memref<10xi32>
2537// memref.store %alloc, %arg1[] : memref<memref<10xi32>, #gpu.address_space<workgroup>>
2538// }
2539// gpu.barrier scope<subgroup>
2540// %4 = memref.load %arg1[] : memref<memref<10xi32>, #gpu.address_space<workgroup>>
2541//
2542// clang-format on
2543
2544Value ACCCGToGPULowering::processPrivatize(acc::PrivatizeOp privatize) {
2545 LLVM_DEBUG(llvm::dbgs() << "processing privatize: ";
2546 privatize->print(llvm::dbgs()); llvm::dbgs() << "\n");
2547 Value tracked = privatize.getResult();
2548 if (acc::ComputeRegionOp insUser =
2549 dyn_cast<acc::ComputeRegionOp>(getOnlyUser(tracked))) {
2550 assert(privatize->hasOneUse() &&
2551 "expected acc.privatize op to have one use");
2552 tracked = insUser.getBody()->getArgument(
2553 privatize->use_begin()->getOperandNumber());
2554 }
2555 Operation *privatizeUser = getOnlyUser(tracked);
2556 assert(privatizeUser && "expected PrivateLocalOp user for privatize");
2557
2558 std::pair<SmallVector<mlir::acc::GPUParallelDimAttr>,
2559 SmallVector<mlir::acc::GPUParallelDimAttr>>
2560 parDimsPair = computeActiveAndInactiveParDims(privatizeUser, nullptr);
2561 // Set `par_dims` only when this `acc.privatize` does not already carry it.
2562 if (!privatize.getParDimsAttr()) {
2563 privatize.setParDimsAttr(mlir::acc::GPUParallelDimsAttr::get(
2564 rewriter.getContext(), parDimsPair.first));
2565 }
2566
2567 Location loc = privatize->getLoc();
2568 acc::PrivateType privTy = cast<acc::PrivateType>(privatize.getType());
2569 ModuleOp module = computeRegion->getParentOfType<ModuleOp>();
2570 MemRefType baseTy = getPrivateBaseMemRefType(privTy.getBaseTy(), module);
2571
2572 gpu::GPUFuncOp gpuFuncOp = computeRegion->getParentOfType<gpu::GPUFuncOp>();
2573 // acc.privatize is outside this compute_region (e.g. passed via ins).
2574 // Leave the op unchanged here; processPrivateLocal materializes
2575 // storage when acc.private_local is lowered.
2576 if (!gpuFuncOp &&
2577 privatize->getParentOfType<acc::ComputeRegionOp>() != computeRegion) {
2578 return privatize.getResult();
2579 }
2580
2581 for (mlir::acc::GPUParallelDimAttr parDim : parDimsPair.first) {
2582 if (parDim.isThreadX() &&
2583 canUseStackAlloca(baseTy, loc, options.maxThreadPrivateStack)) {
2584 auto alloca = memref::AllocaOp::create(rewriter, loc, baseTy);
2585 mapping.map(privatize.getResult(), alloca.getResult());
2586 return alloca.getResult();
2587 }
2588 }
2589
2590 if (!gpuFuncOp)
2591 return privatize.getResult();
2592
2593 // When ThreadY is active, shared memory must be indexed by ThreadY ID to
2594 // avoid races between ThreadY threads.
2595 // For ThreadX acc routines, assume ThreadY may be active since the routine
2596 // can be called from a ThreadY loop at runtime.
2597 // For ThreadY/block acc routines, do not force ThreadY indexing - variables
2598 // outside the ThreadY loop should be shared across ThreadY threads.
2599 bool threadYIsActive =
2600 llvm::any_of(parDimsPair.first, [](mlir::acc::GPUParallelDimAttr parDim) {
2601 return parDim.isThreadY();
2602 });
2603 // For routines with multiple ThreadY threads, need a workgroup barrier
2604 // instead of a subgroup barrier.
2605 // - ThreadX routines: called independently by different ThreadY threads, need
2606 // per-ThreadY slots (override threadYIsActive)
2607 // - ThreadY routines: workgroup barrier, but respect parDimsPair for
2608 // threadYIsActive
2609 // (variables before ThreadY loop are shared, inside are per-ThreadY)
2610 // - Block routines: single call, ThreadY threads cooperate within the
2611 // routine, so
2612 // variables at routine level are shared across ThreadY threads (single
2613 // slot)
2614 bool needsWorkgroupBarrier = false;
2615 if (isInsideACCSpecializedRoutine(computeRegion)) {
2616 FunctionOpInterface funcOp =
2617 computeRegion->getParentOfType<FunctionOpInterface>();
2618 mlir::acc::GPUParallelDimAttr routineParDim =
2619 getSpecializedRoutineDim(funcOp, defaultPolicy);
2620 if (routineParDim.isThreadX()) {
2621 // ThreadX routine: always per-ThreadY slots since called from ThreadY
2622 // loops
2623 threadYIsActive = true;
2624 } else if (routineParDim.isThreadY()) {
2625 // ThreadY routine: workgroup barrier, but keep original threadYIsActive
2626 // (variables before ThreadY loop shared, inside per-ThreadY)
2627 needsWorkgroupBarrier = true;
2628 } else if (routineParDim.isAnyBlock()) {
2629 needsWorkgroupBarrier = true;
2630 }
2631 }
2632
2633 llvm::SmallVector<Value> mappedDynamicSizes;
2634 for (auto dynamicSize : privatize.getDynamicSizes()) {
2635 Value mappedDynamicSize = mapping.lookupOrDefault(dynamicSize);
2636 mappedDynamicSizes.push_back(mappedDynamicSize);
2637 }
2638 if (isInsideACCSpecializedRoutine(computeRegion) &&
2639 computeRegion.isEffectivelySerial()) {
2640 if (mappedDynamicSizes.empty()) {
2641 // Static sizes: use alloca (stack allocation)
2642 auto alloca =
2643 memref::AllocaOp::create(rewriter, privatize->getLoc(), baseTy);
2644 mapping.map(privatize.getResult(), alloca.getResult());
2645 return alloca.getResult();
2646 }
2647 // Dynamic sizes: use alloc (heap allocation) with dealloc
2648 auto alloc = memref::AllocOp::create(rewriter, privatize->getLoc(), baseTy,
2649 mappedDynamicSizes);
2650
2651 // Insert dealloc (free) before the function return
2652 OpBuilder::InsertPoint currentInsertPoint = rewriter.saveInsertionPoint();
2653 Block &parentBlock = *alloc->getBlock();
2654 if (parentBlock.mightHaveTerminator()) {
2655 rewriter.setInsertionPoint(parentBlock.getTerminator());
2656 memref::DeallocOp::create(rewriter, privatize->getLoc(), alloc);
2657 }
2658 rewriter.restoreInsertionPoint(currentInsertPoint);
2659
2660 mapping.map(privatize.getResult(), alloc.getResult());
2661 return alloc.getResult();
2662 }
2663
2664 // Predication - when threadYIsActive, don't predicate on ThreadY dimension
2665 // since each ThreadY needs to execute the allocation for its own slot
2666 SmallVector<mlir::acc::GPUParallelDimAttr> predicateDims;
2667 for (auto parDim : parDimsPair.second) {
2668 // Skip ThreadY if threadYIsActive - each ThreadY needs to allocate
2669 if (threadYIsActive && parDim.isThreadY())
2670 continue;
2671 predicateDims.push_back(parDim);
2672 }
2673 Value predicate = emitPredicate(loc, predicateDims);
2674 if (!predicate) {
2675 predicate = arith::ConstantOp::create(
2676 rewriter, loc, rewriter.getIntegerAttr(rewriter.getI1Type(), 1));
2677 }
2678 auto ifOp = scf::IfOp::create(rewriter, loc, predicate,
2679 /*withElseRegion=*/false);
2680 Region &thenRegion = ifOp.getThenRegion();
2681 Block &thenBlock = thenRegion.back();
2682 rewriter.setInsertionPoint(thenBlock.getTerminator());
2683 auto mem = memref::AllocOp::create(rewriter, privatize->getLoc(), baseTy,
2684 mappedDynamicSizes);
2685 // Shared memory allocation
2686 gpu::AddressSpaceAttr sharedMemoryAddressSpace = gpu::AddressSpaceAttr::get(
2687 computeRegion->getContext(), gpu::GPUDialect::getWorkgroupAddressSpace());
2688 // When ThreadY is active, create a shared memory array indexed by ThreadY ID.
2689 // Each ThreadY stores to its own slot; ThreadX lanes within a ThreadY share
2690 // it.
2691 constexpr int64_t kMaxThreadY = 32;
2692 MemRefType sharedMemTy =
2693 threadYIsActive
2694 ? MemRefType::get({kMaxThreadY}, baseTy, MemRefLayoutAttrInterface{},
2695 sharedMemoryAddressSpace)
2696 : MemRefType::get({}, baseTy, MemRefLayoutAttrInterface{},
2697 sharedMemoryAddressSpace);
2698 // The slot only transiently broadcasts the storage pointer, so reuse one per
2699 // type across privatizes, barriering (before the predicated store) on reuse.
2700 bool reuseBroadcast = !gpuFuncOp.isKernel();
2701 Value alloca;
2702 llvm::DenseMap<Type, Value>::iterator cachedSlot =
2703 reuseBroadcast ? privatizeBroadcastCache.find(sharedMemTy)
2704 : privatizeBroadcastCache.end();
2705 if (reuseBroadcast && cachedSlot != privatizeBroadcastCache.end()) {
2706 alloca = cachedSlot->second;
2707 OpBuilder::InsertionGuard guard(rewriter);
2708 rewriter.setInsertionPoint(ifOp);
2709 mlir::acc::GPUParallelDimAttr dim =
2710 needsWorkgroupBarrier
2711 ? mlir::acc::GPUParallelDimAttr::threadYDim(rewriter.getContext())
2712 : mlir::acc::GPUParallelDimAttr::threadXDim(rewriter.getContext());
2713 createBarrier(
2714 loc, mlir::acc::GPUParallelDimsAttr::get(rewriter.getContext(), {dim}));
2715 } else {
2716 alloca = gpuFuncOp.addWorkgroupAttribution(sharedMemTy,
2717 rewriter.getUnknownLoc());
2718 // Setting the alignment to 16 because of a bug in the gpu toolchain.
2719 // The default alignment is 8, but optimizations create a packed store of 16
2720 // bytes which cause a misalignment error at runtime.
2721 unsigned index = gpuFuncOp.getNumWorkgroupAttributions() - 1;
2722 gpuFuncOp.setWorkgroupAttributionAttr(index,
2723 LLVM::LLVMDialect::getAlignAttrName(),
2724 rewriter.getI32IntegerAttr(16));
2725 if (reuseBroadcast)
2726 privatizeBroadcastCache[sharedMemTy] = alloca;
2727 }
2728 // Store to shared memory, indexed by ThreadY ID when ThreadY is active.
2729 if (threadYIsActive) {
2730 Value threadYId = getThreadId(loc, gpu::Dimension::y);
2731 memref::StoreOp::create(rewriter, privatize->getLoc(), mem, alloca,
2732 ValueRange{threadYId});
2733 } else {
2734 memref::StoreOp::create(rewriter, privatize->getLoc(), mem, alloca);
2735 }
2736
2737 // Sync and load - use workgroup barrier for Block/ThreadY routines,
2738 // ThreadX barrier for ThreadX-only routines
2739 rewriter.setInsertionPointAfter(ifOp);
2740 if (needsWorkgroupBarrier) {
2741 mlir::acc::GPUParallelDimsAttr threadYDimsAttr =
2742 mlir::acc::GPUParallelDimsAttr::get(
2743 rewriter.getContext(),
2744 {mlir::acc::GPUParallelDimAttr::threadYDim(rewriter.getContext())});
2745 createBarrier(loc, threadYDimsAttr);
2746 } else {
2747 // ThreadX-only: use per-ThreadY barrier
2748 mlir::acc::GPUParallelDimsAttr threadXDimsAttr =
2749 mlir::acc::GPUParallelDimsAttr::get(
2750 rewriter.getContext(),
2751 {mlir::acc::GPUParallelDimAttr::threadXDim(rewriter.getContext())});
2752 createBarrier(loc, threadXDimsAttr);
2753 }
2754 // Load from shared memory, indexed by ThreadY ID when ThreadY is active.
2755 Value load;
2756 if (threadYIsActive) {
2757 Value threadYId = getThreadId(loc, gpu::Dimension::y);
2758 load = memref::LoadOp::create(rewriter, privatize->getLoc(), baseTy, alloca,
2759 ValueRange{threadYId});
2760 } else {
2761 load =
2762 memref::LoadOp::create(rewriter, privatize->getLoc(), baseTy, alloca);
2763 }
2764 rewriter.setInsertionPointAfter(load.getDefiningOp());
2765 mapping.map(privatize.getResult(), load);
2766
2767 // Operations inside the kernel are all rewritten from scratch.
2768 // But if the privatize op is outside the kernel, it needs to be replaced.
2769 if (!privatize->getParentOfType<acc::ComputeRegionOp>())
2770 rewriter.replaceOp(privatize, load);
2771 // Deallocate
2772 rewriter.setInsertionPoint(ifOp->getBlock()->getTerminator());
2773 if (needsWorkgroupBarrier) {
2774 mlir::acc::GPUParallelDimsAttr workerDimsAttr =
2775 mlir::acc::GPUParallelDimsAttr::get(
2776 rewriter.getContext(),
2777 {mlir::acc::GPUParallelDimAttr::threadYDim(rewriter.getContext())});
2778 createBarrier(loc, workerDimsAttr);
2779 } else {
2780 mlir::acc::GPUParallelDimsAttr vectorDimsAttr =
2781 mlir::acc::GPUParallelDimsAttr::get(
2782 rewriter.getContext(),
2783 {mlir::acc::GPUParallelDimAttr::threadXDim(rewriter.getContext())});
2784 createBarrier(loc, vectorDimsAttr);
2785 }
2786 auto ifOp2 = scf::IfOp::create(rewriter, loc, predicate,
2787 /*withElseRegion=*/false);
2788 Region &thenRegion2 = ifOp2.getThenRegion();
2789 Block &thenBlock2 = thenRegion2.back();
2790 rewriter.setInsertionPoint(thenBlock2.getTerminator());
2791 memref::DeallocOp::create(rewriter, privatize->getLoc(), load);
2792
2793 // Return the private memory
2794 rewriter.setInsertionPointAfter(load.getDefiningOp());
2795
2796 return load;
2797}

3074 toFlush.push_back(loopOp);

3083 deferredBarrierSeqLoops.end(),

3302void ACCCGToGPULowering::constructAtomicAccumulation(
3303 Location loc, Value memref, ValueRange indices, Value input,
3304 arith::AtomicRMWKind kind) {
3305 assert(!memref.getDefiningOp<memref::AllocaOp>() &&
3306 "cannot lower atomic accumulation on an stack variable");
3307
3308 // acc.atomic.update derives the element address from the memref descriptor's
3309 // base pointer and offset field; it has no subscript operand. When the store
3310 // being lowered targets a specific array element (e.g. result(idx) =
3311 // max(...)), fold the indices into the descriptor offset with a subview so
3312 // the atomic updates the intended element. Otherwise the atomic always hits
3313 // element 0, so any reduction whose destination index is non-zero is
3314 // miscompiled (the result lands in element 0 while the intended element keeps
3315 // its identity-init value).
3316 Value target = memref;

3321 SmallVector<OpFoldResult> offsets(indices.begin(), indices.end());
3322 SmallVector<OpFoldResult> sizes(rank, rewriter.getIndexAttr(1));
3323 SmallVector<OpFoldResult> strides(rank, rewriter.getIndexAttr(1));
3324 target = memref::SubViewOp::create(rewriter, loc, memref, offsets, sizes,
3325 strides);
3326 }
3327
3328 auto atomicUpdateOp =
3329 acc::AtomicUpdateOp::create(rewriter, loc, target, /*ifCond=*/Value());
3330 Region &region = atomicUpdateOp->getRegion(0);
3331 Block *block =
3332 rewriter.createBlock(&region, region.begin(), {input.getType()}, {loc});
3333 rewriter.setInsertionPointToStart(block);
3334 Value reductionExpr =
3335 generateReductionOp(rewriter, loc, input, block->getArgument(0), kind);
3336 acc::YieldOp::create(rewriter, loc, reductionExpr);
3337 rewriter.setInsertionPointAfter(atomicUpdateOp);

3340void ACCCGToGPULowering::createGPUAllReduceOp(
3341 Location loc, Value input, Value memref, arith::AtomicRMWKind kind,
3342 mlir::acc::GPUParallelDimsAttr parDimsAttr, ValueRange indices,
3343 bool isPerThreadPrivateTarget) {
3344 gpu::AllReduceOperationAttr attr = gpu::AllReduceOperationAttr::get(
3345 computeRegion->getContext(), getAllReduceOperation(kind));
3346 auto allReduceOp = gpu::AllReduceOp::create(rewriter, loc, input, attr, true);
3347 mlir::acc::setParDimsAttr(allReduceOp, parDimsAttr);
3348 // Predicate the store on the thread-level dimensions being reduced so that
3349 // only one thread per reduced group writes the result. Only dimensions in
3350 // parDimsAttr are included; sweeping over all dimensions between the highest
3351 // par_dim and thread_x would incorrectly add unrelated dimensions (e.g.
3352 // thread_y for a thread_x-only reduction), preventing other rows from
3353 // storing their independent results.
3354 SmallVector<mlir::acc::GPUParallelDimAttr> inactiveParDims;
3355 MLIRContext *ctx = computeRegion->getContext();
3356 bool hasThreadX = false;
3357 for (auto parDim : parDimsAttr.getArray()) {
3358 if (parDim.isAnyBlock())
3359 continue;
3360 if (parDim.isThreadX())
3361 hasThreadX = true;
3362 if (computeRegion.getLaunchArg(parDim) ||
3363 isInsideACCSpecializedRoutine(computeRegion)) {
3364 inactiveParDims.push_back(parDim);
3365 }
3366 }
3367 // Subgroup alignment may introduce extra ThreadX lanes even when ThreadX is
3368 // not part of the reduction. Predicate on ThreadX so only one lane stores.
3369 // A single-thread worker launch is left unaligned, so it has no such lanes.
3370 if (!hasThreadX && !isSingleThreadWorkerLaunch())
3371 inactiveParDims.push_back(mlir::acc::GPUParallelDimAttr::threadXDim(ctx));
3372 Value predicate = emitPredicate(loc, inactiveParDims);
3373 // Predication is only needed when the store target is visible to
3374 // multiple threads (shared/global memory). Per-thread targets like
3375 // memref.alloca are thread-private: gpu.all_reduce returns the same
3376 // value on all threads, so each can safely store to its own copy.
3377 // Detect per-thread storage by walking through conversion ops to
3378 // find the underlying allocation.
3379 bool isPerThreadPrivate = isPerThreadPrivateTarget ||
3380 isa_and_nonnull<memref::AllocaOp>(
3381 unwrapMemRefConversion(memref).getDefiningOp());
3382 // combine
3383 scf::IfOp ifOp;
3384 if (predicate && !isPerThreadPrivate) {
3385 ifOp =
3386 scf::IfOp::create(rewriter, loc, predicate, /*withElseRegion=*/false);
3387 Region &thenRegion = ifOp.getThenRegion();
3388 Block &thenBlock = thenRegion.back();
3389 rewriter.setInsertionPoint(thenBlock.getTerminator());
3390 }
3391 memref::StoreOp::create(rewriter, loc, allReduceOp, memref, indices);
3392 if (predicate && !isPerThreadPrivate)
3393 rewriter.setInsertionPointAfter(ifOp);
3394 // A later block combine reuses this instead of reloading the slot. This only
3395 // applies to scalar accumulators; array elements are indexed individually.
3396 if (indices.empty())
3397 reductionAccumValue[memref] = allReduceOp;
3398}

3400void ACCCGToGPULowering::postprocessAccumulateOp(
3401 acc::ReductionAccumulateOp op) {
3402 Location loc = op->getLoc();
3403
3404 rewriter.setInsertionPoint(op);
3405
3406 // Check whether this accumulate has only block-level par dims (no thread
3407 // dims). gpu.all_reduce reduces across threads within a block, which is
3408 // wrong for block-only reductions - the loop result is already per-thread
3409 // and only needs a predicated store to the memref.
3410 bool hasThreadDim = false;
3411 SmallVector<mlir::acc::GPUParallelDimAttr> threadParDims;
3412 for (auto parDim : op.getParDims().getArray()) {
3413 if (!parDim.isAnyBlock()) {
3414 hasThreadDim = true;
3415 threadParDims.push_back(parDim);
3416 }
3417 }
3418
3419 std::optional<arith::AtomicRMWKind> kind;
3420 if (hasThreadDim) {
3421 FailureOr<arith::AtomicRMWKind> kindOr = getReductionKind(
3422 op.getReductionOperator(), op.getValue().getType(), loc);
3423 if (failed(kindOr))
3424 return;
3425 kind = *kindOr;
3426 }
3427
3428 if (hasThreadDim && reductionSharedBuf &&
3429 op.getValue().getType() ==
3430 cast<MemRefType>(reductionSharedBuf.getType()).getElementType()) {
3431 Value val = op.getValue();
3432 Value mem = op.getMemref();
3433 Value tidY = getThreadId(loc, gpu::Dimension::y);
3434 memref::AtomicRMWOp::create(rewriter, loc, *kind, val, reductionSharedBuf,
3435 ValueRange{tidY});
3436 createPerRowBarrier(loc);
3437 Value result =
3438 memref::LoadOp::create(rewriter, loc, reductionSharedBuf, tidY);
3439 memref::StoreOp::create(rewriter, loc, result, mem);
3440 reductionAccumValue[mem] = result;
3441 } else if (hasThreadDim) {
3442 createGPUAllReduceOp(loc, op.getValue(), op.getMemref(), *kind,
3443 op.getParDims());
3444 } else {
3445 // Block-only: no gpu.all_reduce needed (all threads have the same
3446 // value after broadcast). Just store the value to the memref.
3447 // Predicate on all thread dims when the target is shared memory.
3448 Value val = mapping.lookupOrDefault(op.getValue());
3449 Value mem = mapping.lookupOrDefault(op.getMemref());
3450 bool isPerThreadPrivate = isa_and_nonnull<memref::AllocaOp>(
3451 unwrapMemRefConversion(mem).getDefiningOp());
3452 if (!isPerThreadPrivate) {
3453 SmallVector<mlir::acc::GPUParallelDimAttr> predDims;
3454 for (auto parDim : computeRegion.getLaunchParDims())
3455 if (!parDim.isAnyBlock())
3456 predDims.push_back(parDim);
3457 if (predDims.empty()) {
3458 predDims.push_back(mlir::acc::GPUParallelDimAttr::threadXDim(
3459 computeRegion->getContext()));
3460 }
3461 Value predicate = emitPredicate(loc, predDims);
3462 auto ifOp =
3463 scf::IfOp::create(rewriter, loc, predicate, /*withElseRegion=*/false);
3464 rewriter.setInsertionPoint(ifOp.getThenRegion().back().getTerminator());
3465 memref::StoreOp::create(rewriter, loc, val, mem);
3466 rewriter.setInsertionPointAfter(ifOp);
3467 } else {
3468 memref::StoreOp::create(rewriter, loc, val, mem);
3469 }
3470 }
3471
3472 // erase acc.reduction_accumulate
3473 rewriter.eraseOp(op);
3474}

3491void ACCCGToGPULowering::processExecuteRegion(scf::ExecuteRegionOp op) {
3492 LLVM_DEBUG(llvm::dbgs() << "processing execute region op: ";
3493 op->print(llvm::dbgs()); llvm::dbgs() << "\n");
3494 Location loc = op->getLoc();
3495 auto types = op.getResultTypes();
3496 Region &oldRegion = op.getRegion();
3497 // create the executeRegion op inside gpu launch
3498 auto executeRegionOp = scf::ExecuteRegionOp::create(rewriter, loc, types);
3499 Region &region = executeRegionOp.getRegion();
3500 rewriter.createBlock(&region);
3501 rewriter.setInsertionPointToEnd(&region.front());
3502
3503 llvm::DenseMap<Block *, Block *> blockMap;
3504 blockMap[&oldRegion.front()] = &region.front();
3505
3506 // Create blocks in the new operation corresponding to all blocks in the
3507 // original op
3508 for (auto &oldBlock : llvm::drop_begin(oldRegion.getBlocks())) {
3509 TypeRange argTypes = oldBlock.getArgumentTypes();
3510 size_t numArgs = argTypes.size();
3511 // Create new block with same argument types
3512 Block *newBlock = rewriter.createBlock(&region, region.end(), argTypes,
3513 SmallVector<Location>(numArgs, loc));
3514 blockMap[&oldBlock] = newBlock;
3515 // Map block arguments
3516 mapping.map(oldBlock.getArguments(), newBlock->getArguments());
3517 }
3518
3519 // Iterate over all blocks of oldRegion and all operations inside them
3520 // process all the ops except the terminator
3521 for (auto [oldBlock, newBlock] :
3522 llvm::zip(oldRegion.getBlocks(), region.getBlocks())) {
3523 OpBuilder::InsertionGuard blockGuard(rewriter);
3524 rewriter.setInsertionPointToStart(&newBlock);
3525 for (auto &bodyOp : oldBlock.getOperations()) {
3526 // Skip terminators during normal iteration - handle them separately
3527 if (bodyOp.hasTrait<OpTrait::IsTerminator>())
3528 continue;
3529 processOp(&bodyOp);
3530 }
3531
3532 // Copy the terminator from old block to new block
3533 Operation *oldTerminator = oldBlock.getTerminator();
3534 rewriter.setInsertionPointToEnd(&newBlock);
3535 Operation *newTerminator = rewriter.clone(*oldTerminator, mapping);
3536
3537 // Replace successors with mapped blocks
3538 for (unsigned i = 0; i < oldTerminator->getNumSuccessors(); ++i) {
3539 Block *oldDest = oldTerminator->getSuccessor(i);
3540 Block *newDest = blockMap.lookup(oldDest);
3541 assert(newDest && "Successor block must be in blockMap");
3542 newTerminator->setSuccessor(newDest, i);
3543 }
3544 }
3545 mapping.map(op->getResults(), executeRegionOp->getResults());
3546 rewriter.setInsertionPointAfter(executeRegionOp);
3547}

3595/// True when \p v takes a distinct value per thread: it derives from a thread
3596/// id, either directly or through the bounds of an enclosing loop. By this
3597/// point a thread-mapped loop carries its mapping in its bounds rather than in
3598/// `acc.par_dims`, so the bounds are what must be inspected.
3599static bool isThreadVarying(Value v, ArrayRef<Value> threadIds,
3600 DenseSet<Value> &visited) {
3601 if (!v || !visited.insert(v).second)
3602 return false;
3603 if (llvm::is_contained(threadIds, v))
3604 return true;
3605 if (auto arg = dyn_cast<BlockArgument>(v)) {
3606 Operation *owner = arg.getOwner()->getParentOp();
3607 unsigned dim = arg.getArgNumber();
3608 if (auto loop = dyn_cast<scf::ParallelOp>(owner)) {
3609 if (dim >= loop.getLowerBound().size())
3610 return false;
3611 return isThreadVarying(loop.getLowerBound()[dim], threadIds, visited) ||
3612 isThreadVarying(loop.getStep()[dim], threadIds, visited);
3613 }
3614 if (auto loop = dyn_cast<scf::ForOp>(owner))
3615 return dim == 0 &&
3616 (isThreadVarying(loop.getLowerBound(), threadIds, visited) ||
3617 isThreadVarying(loop.getStep(), threadIds, visited));
3618 return false;
3619 }
3620 Operation *def = v.getDefiningOp();
3621 if (!def)
3622 return false;
3623 if (isa<gpu::ThreadIdOp, gpu::LaneIdOp>(def))
3624 return true;
3625 return llvm::any_of(def->getOperands(), [&](Value o) {
3626 return isThreadVarying(o, threadIds, visited);
3627 });
3628}

3630/// Strips view and memory-space-cast chains to the underlying buffer.
3631static Value accumulatorRoot(Value v) {
3632 while (Operation *op = v.getDefiningOp()) {
3633 if (auto cast = dyn_cast<memref::MemorySpaceCastOp>(op)) {
3634 v = cast.getSource();
3635 continue;
3636 }
3637 if (auto viewLike = dyn_cast<ViewLikeOpInterface>(op)) {
3638 if (isa<MemRefType>(viewLike.getViewSource().getType())) {
3639 v = viewLike.getViewSource();
3640 continue;
3641 }
3642 }
3643 break;
3644 }
3645 return v;
3646}

3648/// Matches `%l = load %m[%i]` / `%c = combine(%l, %x)` / `store %c, %m[%i]` on
3649/// the accumulator \p accum and returns the contributed value `%x`.
3650static Value matchAccumulatorUpdate(memref::StoreOp store, Value accum) {
3651 if (accumulatorRoot(store.getMemRef()) != accum)
3652 return {};
3653 Operation *combine = store.getValueToStore().getDefiningOp();
3654 if (!combine || combine->getNumOperands() != 2)
3655 return {};
3656 for (unsigned i = 0; i != 2; ++i) {
3657 auto load = combine->getOperand(i).getDefiningOp<memref::LoadOp>();
3658 if (!load || accumulatorRoot(load.getMemRef()) != accum)
3659 continue;
3660 if (!llvm::equal(load.getIndices(), store.getIndices()))
3661 continue;
3662 return combine->getOperand(1 - i);
3663 }
3664 return {};
3665}

3667/// A block-shared accumulator is updated in place by the loop body, so several
3668/// threads may hit the same element. Make those updates atomic unless the
3669/// element index provably varies across the participating threads.
3670static void atomicizeSharedAccumulatorUpdates(Value accum,
3671 arith::AtomicRMWKind kind,
3672 ArrayRef<Value> threadIds,
3673 RewriterBase &rewriter) {
3674 OpBuilder::InsertionGuard guard(rewriter);
3675 SmallVector<memref::StoreOp> stores;
3676 SmallVector<Value> worklist{accum};
3677 DenseSet<Value> seen;
3678 while (!worklist.empty()) {
3679 Value cur = worklist.pop_back_val();
3680 if (!seen.insert(cur).second)
3681 continue;
3682 for (Operation *user : cur.getUsers()) {
3683 if (auto store = dyn_cast<memref::StoreOp>(user))
3684 stores.push_back(store);
3685 else if (isa<ViewLikeOpInterface, memref::MemorySpaceCastOp>(user))
3686 llvm::append_range(worklist, user->getResults());
3687 }
3688 }
3689
3690 for (memref::StoreOp store : stores) {
3691 Value contribution = matchAccumulatorUpdate(store, accum);
3692 if (!contribution)
3693 continue;
3694 // A thread-varying index means each thread owns its element, so the
3695 // existing plain update is already race-free.
3696 if (llvm::any_of(store.getIndices(), [&](Value idx) {
3697 DenseSet<Value> visited;
3698 return isThreadVarying(idx, threadIds, visited);
3699 }))
3700 continue;
3701 Operation *combine = store.getValueToStore().getDefiningOp();
3702 rewriter.setInsertionPoint(store);
3703 memref::AtomicRMWOp::create(rewriter, store.getLoc(), kind, contribution,
3704 store.getMemRef(), store.getIndices());
3705 rewriter.eraseOp(store);
3706 if (combine && combine->use_empty())
3707 rewriter.eraseOp(combine);
3708 }

3711void ACCCGToGPULowering::processAccumulateArrayOp(
3712 acc::ReductionAccumulateArrayOp op) {
3713 LLVM_DEBUG(llvm::dbgs() << "processing accumulate array op: " << *op << "\n");
3714 Location loc = op.getLoc();
3715
3716 Value memref = mapping.lookupOrDefault(op.getMemref());
3717 MemRefType memrefTy = dyn_cast<MemRefType>(memref.getType());
3718 if (!memrefTy) {
3719 (void)accSupport.emitNYI(loc, "reduction: non-MemRefTy accumulate array");
3720 return;
3721 }
3722 FailureOr<arith::AtomicRMWKind> kindOr = getReductionKind(
3723 op.getReductionOperator(), memrefTy.getElementType(), loc);
3724 if (failed(kindOr))
3725 return;
3726 arith::AtomicRMWKind kind = *kindOr;
3727
3728 // The (already mapped/cloned) acc.bounds op describes the element range; it
3729 // is dead after lowering since we read its operands directly.
3730 acc::DataBoundsOp boundsOp = mapping.lookupOrDefault(op.getBounds())
3731 .getDefiningOp<acc::DataBoundsOp>();
3732 assert(boundsOp && "expected acc.bounds defining op for array accumulate");
3733 auto eraseDeadBounds = [&] {
3734 if (boundsOp->use_empty())
3735 rewriter.eraseOp(boundsOp);
3736 };
3737
3738 bool hasThreadDim = false;
3739 bool hasBlockDim = false;
3740 for (auto pd : op.getParDims().getArray()) {
3741 hasThreadDim |= pd.isAnyThread();
3742 hasBlockDim |= pd.isAnyBlock();
3743 }
3744
3745 // Block-only (gang) reduction: each element is produced by one gang, so the
3746 // per-gang copy already holds the result and the combine does the rest.
3747 if (hasBlockDim && !hasThreadDim) {
3748 eraseDeadBounds();
3749 return;
3750 }
3751
3752 // A thread-only accumulate merges with a within-block all_reduce, which is
3753 // complete only in one block: block context, or a launch with no block dim.
3754 // Multi-block thread-only still grid-strides across blocks, so stays NYI.
3755 bool regionLaunchesBlocks = llvm::any_of(
3756 computeRegion.getLaunchParDims(),
3757 [](mlir::acc::GPUParallelDimAttr d) { return d.isAnyBlock(); });
3758 if (!reductionHasBlockContext(op) && regionLaunchesBlocks) {
3759 (void)accSupport.emitNYI(
3760 loc, "reduction: thread-only array reduction accumulate");
3761 return;
3762 }
3763
3764 // Per-element gpu.all_reduce is only correct when each thread owns its own
3765 // accumulator copy. For a statically-shaped accumulator, classify from the
3766 // operand: an explicit shared/heap allocation is block-shared regardless of
3767 // size, and a stack alloca (or a view over one) is per-thread when it fits
3768 // the per-thread stack budget. Storage `acc.par_dims` without thread_x means
3769 // gang-/worker-scoped privacy (shared among vector lanes) even if the type
3770 // would fit on the stack. For a dynamically-shaped accumulator the type
3771 // conveys no size, so classify from storage/accumulate thread_x dims.
3772 auto storageIsThreadXPrivate = [&](Value v) -> bool {
3773 acc::PrivateLocalOp privateLocal = getPrivateLocalForMemref(v);
3774 GPUParallelDimsAttr dims =
3775 privateLocal ? getPrivateParDims(privateLocal, computeRegion)
3776 : GPUParallelDimsAttr();
3777 if (!dims) {
3778 if (Operation *root = unwrapMemRefConversion(v).getDefiningOp())
3779 dims = getParDimsAttr(root);
3780 }
3781 return !dims ||
3782 llvm::any_of(dims.getArray(), [](auto d) { return d.isThreadX(); });
3783 };
3784 Operation *rootOp = unwrapMemRefConversion(memref).getDefiningOp();
3785 bool isSharedStorage = isa_and_nonnull<memref::AllocOp>(rootOp) ||
3786 isa_and_nonnull<acc::GPUSharedMemoryOp>(rootOp);
3787 if (auto addrSpace = dyn_cast_if_present<gpu::AddressSpaceAttr>(
3788 memrefTy.getMemorySpace())) {
3789 isSharedStorage |=
3790 addrSpace.getValue() == gpu::GPUDialect::getWorkgroupAddressSpace();
3791 }
3792 bool isPerThreadPrivate =
3793 !isSharedStorage && storageIsThreadXPrivate(op.getMemref()) &&
3794 (memrefTy.hasStaticShape()
3795 ? canUseStackAlloca(memrefTy, loc, options.maxThreadPrivateStack)
3796 : llvm::any_of(op.getParDims().getArray(),
3797 [](mlir::acc::GPUParallelDimAttr d) {
3798 return d.isThreadX();
3799 }));
3800 if (!isPerThreadPrivate) {
3801 // Block-shared accumulator: the body already updated it in place, so the
3802 // block partial is complete and the atomic combine finishes across blocks.
3803 // Threads that share an element must not race, so their in-place updates
3804 // become atomic.
3805 SmallVector<Value> threadIds;
3806 if (Value xId = getGPUThreadIdFor(gpu::Processor::ThreadX))
3807 threadIds.push_back(xId);
3808 if (Value yId = getGPUThreadIdFor(gpu::Processor::ThreadY))
3809 threadIds.push_back(yId);
3810 if (Value zId = getGPUThreadIdFor(gpu::Processor::ThreadZ))
3811 threadIds.push_back(zId);
3812 atomicizeSharedAccumulatorUpdates(accumulatorRoot(memref), kind, threadIds,
3813 rewriter);
3814 eraseDeadBounds();
3815 return;
3816 }
3817
3818 // Bounds are normalized to be zero-based.
3819 auto toIndex = [&](Value v) -> Value {
3820 if (v.getType().isIndex())
3821 return v;
3822 return arith::IndexCastOp::create(rewriter, loc, rewriter.getIndexType(),
3823 v);
3824 };
3825
3826 Value zero = arith::ConstantIndexOp::create(rewriter, loc, 0);
3827 Value one = arith::ConstantIndexOp::create(rewriter, loc, 1);
3828 Value lb =
3829 boundsOp.getLowerbound() ? toIndex(boundsOp.getLowerbound()) : zero;
3830 Value step = boundsOp.getStride() ? toIndex(boundsOp.getStride()) : one;
3831 // Exclusive upper bound. `extent` counts elements, so the span is
3832 // `extent * step` (for the common unit-stride case step is 1); fall back to
3833 // the inclusive upperbound when no extent is given.
3834 Value ub;
3835 if (boundsOp.getExtent()) {
3836 Value span = arith::MulIOp::create(rewriter, loc,
3837 toIndex(boundsOp.getExtent()), step);
3838 ub = arith::AddIOp::create(rewriter, loc, lb, span);
3839 } else {
3840 assert(boundsOp.getUpperbound() &&
3841 "acc.bounds must specify an extent or upperbound");
3842 ub = arith::AddIOp::create(rewriter, loc, toIndex(boundsOp.getUpperbound()),
3843 one);
3844 }
3845
3846 // Reduce each array element across the requested parallel dimensions.
3847 auto forOp = scf::ForOp::create(rewriter, loc, lb, ub, step);
3848 {
3849 OpBuilder::InsertionGuard guard(rewriter);
3850 rewriter.setInsertionPoint(forOp.getBody()->getTerminator());
3851 Value iv = forOp.getInductionVar();
3852 SmallVector<Value> indices{iv};
3853 if (memrefTy.getRank() > 1) {
3854 indices.resize(memrefTy.getRank());
3855 Value linearIndex = iv;
3856 for (int64_t dim = memrefTy.getRank() - 1; dim >= 0; --dim) {
3857 Value dimSize =
3858 memrefTy.isDynamicDim(dim)
3859 ? memref::DimOp::create(rewriter, loc, memref, dim).getResult()
3860 : arith::ConstantIndexOp::create(rewriter, loc,
3861 memrefTy.getDimSize(dim))
3862 .getResult();
3863 indices[dim] =
3864 arith::RemUIOp::create(rewriter, loc, linearIndex, dimSize);
3865 if (dim != 0)
3866 linearIndex =
3867 arith::DivUIOp::create(rewriter, loc, linearIndex, dimSize);
3868 }
3869 }
3870 Value elem = memref::LoadOp::create(rewriter, loc, memref, indices);
3871 createGPUAllReduceOp(loc, elem, memref, kind, op.getParDims(), indices,
3872 /*isPerThreadPrivateTarget=*/true);
3873 }
3874
3875 eraseDeadBounds();
3876}

3913 // must be updated atomically. But when the destination resolves to a
3914 // thread-private stack alloca (e.g. an inner-loop reduction combining one
3915 // private accumulator into another private accumulator for the same thread),
3916 // the update is not visible to other threads and must not be atomic. Treat
3917 // such destinations as a plain load/combine/store below.

3999 /*indices=*/{}, reductionResult, kind);
