31std::optional<FileLineColLoc>
32acc::getFileLineColLoc(Location loc, bool errorOnInvalidLocation) {
33 Location unfusedLoc = unfuseLoc(loc);
34
35 if (auto fileLoc = dyn_cast<FileLineColLoc>(unfusedLoc))
36 return fileLoc;
37
38 if (auto callSiteLoc = dyn_cast<CallSiteLoc>(unfusedLoc)) {
39 if (auto calleeFileLoc = getFileLineColLoc(callSiteLoc.getCallee(), false))
40 return calleeFileLoc;
41 if (auto callerFileLoc =
42 getFileLineColLoc(callSiteLoc.getCaller(), errorOnInvalidLocation))
43 return callerFileLoc;
44 }
45
46 if (errorOnInvalidLocation)
47 llvm_unreachable(
48 "cannot get file:line information: invalid Location information");
49 return std::nullopt;
50}

75std::string acc::getInternalGlobalName(StringRef kind, StringRef detail) {
76 SmallString<64> name("acc.");
77 name += kind;
78 if (detail.empty())
79 return name.str().str();
80
81 name += '.';
82 for (unsigned char c : detail)
83 name += (llvm::isAlnum(c) || c == '_' || c == '$' || c == '.') ? c : '_';
84 return name.str().str();

87/// Creates or reuses a module-internal null-terminated string global and
88/// returns the GlobalOp.
89static LLVM::GlobalOp getOrCreateGlobalStringOp(Location loc,
90 OpBuilder &builder,
91 StringRef name, StringRef value,
92 Region &globalSymbolRegion,
93 SymbolTable *symbolTable) {
94 SmallString<32> nullTermStr(value);
95 nullTermStr.push_back('\0');
96 StringAttr valueAttr = builder.getStringAttr(nullTermStr);
97
98 // A name says what the global holds, so a global of that name is the one to
99 // reuse as long as it holds this very value. Names come out of detail the
100 // source spells and are not one to one with it, so a name can also lead to a
101 // global holding something else, which a suffix then keeps apart.
102 SmallString<64> uniqueName(name);
103 if (symbolTable) {
104 unsigned suffix = 0;
105 while (Operation *taken = symbolTable->lookup(uniqueName)) {
106 auto global = dyn_cast<LLVM::GlobalOp>(taken);
107 if (global && global.getValueOrNull() == valueAttr)
108 return global;
109 uniqueName.clear();
110 (name + "." + Twine(suffix++)).toVector(uniqueName);
111 }
112 }
113
114 // Materialize the global through the incoming builder so that it stays
115 // tracked when the caller is a dialect conversion rewriter.
116 OpBuilder::InsertionGuard guard(builder);
117 builder.setInsertionPointToStart(&globalSymbolRegion.front());
118 auto arrayTy = LLVM::LLVMArrayType::get(builder.getI8Type(),
119 nullTermStr.size_in_bytes());
120 auto global =
121 LLVM::GlobalOp::create(builder, loc, arrayTy, /*isConstant=*/true,
122 LLVM::Linkage::Internal, uniqueName, valueAttr,
123 /*alignment=*/0);
124 if (symbolTable)
125 symbolTable->insert(global);
126 return global;
127}

129Value acc::getOrCreateGlobalString(Location loc, OpBuilder &builder,
130 StringRef name, StringRef value,
131 Region &globalSymbolRegion,
132 SymbolTable *symbolTable) {
133 Type i64Ty = builder.getI64Type();
134 Type ptrTy = LLVM::LLVMPointerType::get(builder.getContext());
135 LLVM::GlobalOp global = getOrCreateGlobalStringOp(
136 loc, builder, name, value, globalSymbolRegion, symbolTable);
137
138 Value globalPtr = LLVM::AddressOfOp::create(builder, loc, global);
139 Value cst0 = LLVM::ConstantOp::create(builder, loc, i64Ty,
140 builder.getI64IntegerAttr(0));
141 return LLVM::GEPOp::create(builder, loc, ptrTy, global.getType(), globalPtr,
142 ArrayRef<Value>({cst0, cst0}));
143}

145Value acc::createIdent(Location loc, StringRef functionName, OpBuilder &builder,
146 Region &globalSymbolRegion,
147 const ACCRuntimeCallConfig &config,
148 SymbolTable *symbolTable) {
149 MLIRContext *ctx = builder.getContext();
150 Type i32Ty = builder.getI32Type();
151 Type i64Ty = builder.getI64Type();
152 Type ptrTy = LLVM::LLVMPointerType::get(ctx);
153 Type structTy = LLVM::LLVMStructType::getLiteral(
154 ctx, {i32Ty, i32Ty, i32Ty, i32Ty, ptrTy});
155
156 std::string source;
157 // The position the globals describe is what names them, so a global found
158 // under such a name holds this very source string or ident.
159 std::string position;
160 if (auto fileLineColLoc =
161 getFileLineColLoc(loc, /*errorOnInvalidLocation=*/false)) {
162 std::string filename = fileLineColLoc->getFilename().str();
163 std::string line = std::to_string(fileLineColLoc->getLine());
164 std::string column = std::to_string(fileLineColLoc->getColumn());
165 std::string functionDisplayName =
166 functionName.empty() ? std::string()
167 : config.getFunctionDisplayName(functionName);
168 source = ";";
169 source += filename + ";";
170 source += functionDisplayName + ";";
171 source += line + ";";
172 source += column + ";";
173 source += ";";
174 position = line + "." + column + ".";
175 position += std::to_string(static_cast<uint64_t>(llvm::hash_value(source)));
176 } else {
177 source = ";unknown;unknown;0;0;;";
178 position = "unknown";
179 }
180
181 std::string identGlobalName = getInternalGlobalName("ident", position);
182 LLVM::GlobalOp identGlobal =
183 symbolTable ? symbolTable->lookup<LLVM::GlobalOp>(identGlobalName)
184 : LLVM::GlobalOp();
185 if (!identGlobal) {
186 LLVM::GlobalOp sourceGlobal = getOrCreateGlobalStringOp(
187 loc, builder, getInternalGlobalName("loc", position), source,
188 globalSymbolRegion, symbolTable);
189
190 OpBuilder::InsertionGuard guard(builder);
191 builder.setInsertionPointAfter(sourceGlobal);
192 identGlobal = LLVM::GlobalOp::create(
193 builder, loc, structTy, /*isConstant=*/true, LLVM::Linkage::Internal,
194 identGlobalName, /*value=*/Attribute(), /*alignment=*/0);
195 if (symbolTable)
196 symbolTable->insert(identGlobal);
197
198 Block *block = builder.createBlock(&identGlobal.getInitializerRegion());
199 builder.setInsertionPointToStart(block);
200 Value ident = LLVM::ZeroOp::create(builder, loc, structTy);
201 Value sourceBase = LLVM::AddressOfOp::create(builder, loc, sourceGlobal);
202 Value cst0 = LLVM::ConstantOp::create(builder, loc, i64Ty,
203 builder.getI64IntegerAttr(0));
204 Value sourcePtr =
205 LLVM::GEPOp::create(builder, loc, ptrTy, sourceGlobal.getType(),
206 sourceBase, ArrayRef<Value>({cst0, cst0}));
207 ident = LLVM::InsertValueOp::create(builder, loc, structTy, ident,
208 sourcePtr, ArrayRef<int64_t>{4});
209 LLVM::ReturnOp::create(builder, loc, ident);
210 }
211
212 return LLVM::AddressOfOp::create(builder, loc, identGlobal);
213}

215Value acc::castToI64(Location loc, Value value, OpBuilder &builder) {
216 Type i64Ty = builder.getI64Type();
217 if (value.getType() == i64Ty)
218 return value;
219 if (isa<IndexType>(value.getType()))
220 return arith::IndexCastOp::create(builder, loc, i64Ty, value);
221 unsigned bitwidth = value.getType().getIntOrFloatBitWidth();
222 if (bitwidth > 64)
223 return arith::TruncIOp::create(builder, loc, i64Ty, value);
224 return arith::ExtSIOp::create(builder, loc, i64Ty, value);
225}

240LogicalResult acc::emitWaitCall(Location loc, ValueRange waitOperands,
241 Value asyncQueue, OpBuilder &builder,
242 Region &globalSymbolRegion,
243 SymbolTable &symbolTable,
244 const ACCRuntimeCallConfig &config) {
245 Type i32Ty = builder.getI32Type();
246 Type i64Ty = builder.getI64Type();
247 Type ptrTy = LLVM::LLVMPointerType::get(builder.getContext());
248
249 SmallVector<Value> queues;
250 queues.reserve(waitOperands.size());
251 for (Value waitOperand : waitOperands)
252 queues.push_back(castToI64(loc, waitOperand, builder));
253
254 unsigned size = queues.size();
255 Value waitNum = LLVM::ConstantOp::create(builder, loc, i32Ty, size);
256 Value waitList;
257 if (size == 0) {
258 waitList = LLVM::ZeroOp::create(builder, loc, ptrTy);
259 } else {
260 waitList = LLVM::AllocaOp::create(builder, loc, ptrTy, i64Ty, waitNum);
261 for (auto [index, queue] : llvm::enumerate(queues)) {
262 Value idx = LLVM::ConstantOp::create(builder, loc, i32Ty,
263 static_cast<int64_t>(index));
264 Value elementPtr = LLVM::GEPOp::create(builder, loc, ptrTy, i64Ty,
265 waitList, ArrayRef<Value>{idx});
266 LLVM::StoreOp::create(builder, loc, queue, elementPtr);
267 }
268 }
269
270 StringRef functionName = getParentFunctionName(waitOperands);
271 if (functionName.empty())
272 if (Block *block = builder.getInsertionBlock())
273 functionName = getParentFunctionName(block->getParentOp());
274 Value ident = createIdent(loc, functionName, builder, globalSymbolRegion,
275 config, &symbolTable);
276 Value flags = LLVM::ConstantOp::create(builder, loc, i64Ty, 0);
277 Value deviceType = LLVM::ConstantOp::create(
278 builder, loc, i64Ty, config.getDeviceTypeRuntimeValue(DeviceType::None));
279 Value deviceNum = LLVM::ConstantOp::create(builder, loc, i32Ty, 0);
280
281 return createRuntimeCall(
282 loc, builder, globalSymbolRegion, symbolTable,
283 RuntimeFunction::ACCRTL_tgt_acc_wait, config,
284 {ident, flags, deviceType, deviceNum, waitNum, waitList, asyncQueue});
285}

288 bool ifPresent, OpBuilder &builder,
289 Region &globalSymbolRegion,
290 SymbolTable &symbolTable,
291 const ACCRuntimeCallConfig &config) {
292 Location loc = clauseOp->getLoc();
293 Value ident = createIdent(loc, getParentFunctionName(clauseOp), builder,
294 globalSymbolRegion, config, &symbolTable);
295 Value flags = LLVM::ConstantOp::create(
296 builder, loc, builder.getI64Type(),
297 config.getMapFlagsRuntimeValue(ifPresent ? MapFlags::if_present
298 : MapFlags::none));
299 // Both slots hold the address of the object, a clause naming a section of
300 // one included: the entry point looks up the mapping that address belongs
301 // to rather than being told which one to ask about.
302 FailureOr<LLVM::CallOp> call = createRuntimeCall(
303 loc, builder, globalSymbolRegion, symbolTable,
304 RuntimeFunction::ACCRTL_tgt_acc_get_deviceptr, config,
305 {ident, /*basePtr=*/hostPtr, flags, /*hostPtr=*/hostPtr});
306 if (failed(call))
307 return failure();
308 return call->getResult();
309}

321LogicalResult acc::emitGuardedByIfCond(Location loc, Value ifCond,
322 RewriterBase &rewriter,
323 function_ref<LogicalResult()> emitFn) {
324 if (!ifCond)
325 return emitFn();
326
327 Block *parentBlock = rewriter.getInsertionBlock();
328 Block *continueBlock =
329 rewriter.splitBlock(parentBlock, rewriter.getInsertionPoint());
330 Block *thenBlock = rewriter.createBlock(
331 parentBlock->getParent(), std::next(Region::iterator(parentBlock)));
332
333 rewriter.setInsertionPointToEnd(parentBlock);
334 LLVM::CondBrOp::create(rewriter, loc, ifCond, thenBlock, ValueRange{},
335 continueBlock, ValueRange{});
336
337 rewriter.setInsertionPointToStart(thenBlock);
338 LogicalResult result = emitFn();
339 rewriter.setInsertionPointToEnd(thenBlock);
340 LLVM::BrOp::create(rewriter, loc, ValueRange{}, continueBlock);
341 rewriter.setInsertionPointToStart(continueBlock);
342 return result;
343}

345FailureOr<Value>
346acc::emitValueSelectedByIfCond(Location loc, Value ifCond,
347 RewriterBase &rewriter,
348 function_ref<FailureOr<Value>()> thenFn,
349 function_ref<FailureOr<Value>()> elseFn) {
350 if (!ifCond)
351 return thenFn();
352
353 Block *parentBlock = rewriter.getInsertionBlock();
354 Block *continueBlock =
355 rewriter.splitBlock(parentBlock, rewriter.getInsertionPoint());
356 Block *thenBlock = rewriter.createBlock(
357 parentBlock->getParent(), std::next(Region::iterator(parentBlock)));
358 Block *elseBlock = rewriter.createBlock(
359 parentBlock->getParent(), std::next(Region::iterator(thenBlock)));
360
361 rewriter.setInsertionPointToEnd(parentBlock);
362 LLVM::CondBrOp::create(rewriter, loc, ifCond, thenBlock, ValueRange{},
363 elseBlock, ValueRange{});
364
365 rewriter.setInsertionPointToStart(thenBlock);
366 FailureOr<Value> thenValue = thenFn();
367 if (failed(thenValue))
368 return failure();
369 rewriter.setInsertionPointToEnd(thenBlock);
370 LLVM::BrOp::create(rewriter, loc, ValueRange{*thenValue}, continueBlock);
371
372 rewriter.setInsertionPointToStart(elseBlock);
373 FailureOr<Value> elseValue = elseFn();
374 if (failed(elseValue))
375 return failure();
376 rewriter.setInsertionPointToEnd(elseBlock);
377 LLVM::BrOp::create(rewriter, loc, ValueRange{*elseValue}, continueBlock);
378
379 assert(thenValue->getType() == elseValue->getType() &&
380 "both paths of an if clause have to produce the same type");
381 BlockArgument selected =
382 continueBlock->addArgument(thenValue->getType(), loc);
383 rewriter.setInsertionPointToStart(continueBlock);
384 return Value(selected);
385}

387void acc::spliceConstructRegion(Operation *op, Region &region,
388 RewriterBase &rewriter) {
389 Block *prev = op->getBlock();
390 Block *succ = rewriter.splitBlock(prev, Block::iterator(op));
391 rewriter.setInsertionPointToEnd(prev);
392 LLVM::BrOp::create(rewriter, op->getLoc(), &region.getBlocks().front());
393 for (Block &block : region.getBlocks()) {
394 // A region that ends without a terminator, as an `acc.kernel_environment`
395 // body does, continues after the construct from the end of its block.
396 if (block.empty() || !block.back().hasTrait<OpTrait::IsTerminator>()) {
397 rewriter.setInsertionPointToEnd(&block);
398 LLVM::BrOp::create(rewriter, op->getLoc(), succ);
399 continue;
400 }
401 Operation *terminator = block.getTerminator();
402 if (isa<acc::TerminatorOp>(terminator)) {
403 rewriter.setInsertionPoint(terminator);
404 LLVM::BrOp::create(rewriter, op->getLoc(), succ);
405 rewriter.eraseOp(terminator);
406 }
407 }
408 rewriter.inlineRegionBefore(region, succ);
409}
