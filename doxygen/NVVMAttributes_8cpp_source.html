91LogicalResult
92NVVMTargetAttr::verify(function_ref<InFlightDiagnostic()> emitError,
93 int optLevel, StringRef triple, StringRef chip,
94 StringRef features, DictionaryAttr flags,
95 ArrayAttr files, bool verifyTarget) {
96 if (optLevel < 0 || optLevel > 3) {
97 emitError() << "The optimization level must be a number between 0 and 3.";
98 return failure();
99 }
100 if (triple.empty()) {
101 emitError() << "The target triple cannot be empty.";
102 return failure();
103 }
104 if (chip.empty()) {
105 emitError() << "The target chip cannot be empty.";
106 return failure();
107 }
108 if (files && !llvm::all_of(files, [](::mlir::Attribute attr) {
109 return mlir::isa_and_nonnull<StringAttr>(attr);
110 })) {
111 emitError() << "All the elements in the `link` array must be strings.";
112 return failure();
113 }
114 return success();
115}

117LogicalResult NVVMTargetAttr::verifyTarget(Operation *gpuModule) {
118 if (!getVerifyTarget())
119 return success();
120
121 auto gpuModuleOp = llvm::dyn_cast<gpu::GPUModuleOp>(gpuModule);
122 if (!gpuModuleOp) {
123 return emitError(gpuModule->getLoc(),
124 "NVVM target attribute must be attached to a GPU module");
125 }
126
127 std::optional<unsigned> targetFullSmVersion =
128 NVVMCheckSMVersion::getTargetFullSmVersionFromStr(getChip());
129 if (!targetFullSmVersion)
130 return emitError(gpuModule->getLoc())
131 << "invalid NVVM target chip \"" << getChip()
132 << "\", expected sm_<version>[a|f]";
133
134 if (!NVVMCheckSMVersion::isMinimumSMVersion(*targetFullSmVersion)) {
135 return emitError(gpuModule->getLoc(),
136 "Minimum NVVM target SM version is sm_20");
137 }
138
139 if (gpuModuleOp
140 ->walk([&](Operation *op) {
141 if (auto reqOp = llvm::dyn_cast<NVVM::RequiresSMInterface>(op)) {
142 const NVVMCheckSMVersion requirement =
143 reqOp.getRequiredMinSMVersion();
144 if (!requirement.isCompatibleWith(*targetFullSmVersion)) {
145 op->emitOpError() << "is not supported on " << getChip();
146 return WalkResult::interrupt();
147 }
148 }
149 return WalkResult::advance();
150 })
151 .wasInterrupted())
152 return failure();
153
154 return success();
155}
