46 TypeRange operandTypes, StringRef tag) {
47 p.printString(tag);
48 p << "(";
49
50 if (!operands.empty()) {

54 }
55
56 p << ")";

59void mlir::LLVM::printOpBundles(OpAsmPrinter &p, Operation *op,
60 OperandRangeRange opBundleOperands,
61 TypeRangeRange opBundleOperandTypes,
62 std::optional<ArrayAttr> opBundleTags) {
63 if (opBundleOperands.empty())
64 return;
65 assert(opBundleTags && "expect operand bundle tags");
66
67 p << "[";
68 llvm::interleaveComma(
69 llvm::zip(opBundleOperands, opBundleOperandTypes, *opBundleTags), p,
70 [&p](auto bundle) {
71 auto bundleTag = cast<StringAttr>(std::get<2>(bundle)).getValue();
72 printOneOpBundle(p, std::get<0>(bundle), std::get<1>(bundle),
73 bundleTag);
74 });
75 p << "]";
76}

78static ParseResult parseOneOpBundle(
79 OpAsmParser &p,
80 SmallVector<SmallVector<OpAsmParser::UnresolvedOperand>> &opBundleOperands,
81 SmallVector<SmallVector<Type>> &opBundleOperandTypes,
82 SmallVector<Attribute> &opBundleTags) {
83 SMLoc currentParserLoc = p.getCurrentLocation();
84 SmallVector<OpAsmParser::UnresolvedOperand> operands;
85 SmallVector<Type> types;
86 std::string tag;
87
88 if (p.parseString(&tag))
89 return p.emitError(currentParserLoc, "expect operand bundle tag");
90
91 if (p.parseLParen())
92 return failure();
93
94 if (p.parseOptionalRParen()) {
95 if (p.parseOperandList(operands) || p.parseColon() ||
96 p.parseTypeList(types) || p.parseRParen())
97 return failure();
98 }
99
100 opBundleOperands.push_back(std::move(operands));
101 opBundleOperandTypes.push_back(std::move(types));
102 opBundleTags.push_back(StringAttr::get(p.getContext(), tag));
103
104 return success();
105}

107std::optional<ParseResult> mlir::LLVM::parseOpBundles(
108 OpAsmParser &p,
109 SmallVector<SmallVector<OpAsmParser::UnresolvedOperand>> &opBundleOperands,
110 SmallVector<SmallVector<Type>> &opBundleOperandTypes,
111 ArrayAttr &opBundleTags) {
112 if (p.parseOptionalLSquare())
113 return std::nullopt;
114
115 if (succeeded(p.parseOptionalRSquare()))
116 return success();
117
118 SmallVector<Attribute> opBundleTagAttrs;
119 auto bundleParser = [&] {
120 return parseOneOpBundle(p, opBundleOperands, opBundleOperandTypes,
121 opBundleTagAttrs);
122 };
123 if (p.parseCommaSeparatedList(bundleParser))
124 return failure();
125
126 if (p.parseRSquare())
127 return failure();
128
129 opBundleTags = ArrayAttr::get(p.getContext(), opBundleTagAttrs);
130
131 return success();
132}

331/// Return the operands of `callOp` that correspond to the declared parameters
332/// of the callee, i.e., its `CallOpInterface` argument operands.
333///
334/// The variadic arguments of a call to a variadic callee are *not* included:
335/// they do not correspond to any argument of the callee. The callee does not
336/// receive them as block arguments but reads them with `llvm.intr.vastart` and
337/// friends, so in terms of `CallOpInterface` they are consumed operands rather
338/// than forwarded ones.
339template <typename OpTy>
340static Operation::operand_range getArgOperandsImpl(OpTy callOp) {
341 Operation::operand_range operands = getOperandsPassedToCallee(callOp);
342 if (std::optional<LLVMFunctionType> varCalleeType = callOp.getVarCalleeType())
343 return operands.take_front(varCalleeType->getNumParams());
344 return operands;
345}

347Operation::operand_range CallOp::getArgOperands() {
348 return getArgOperandsImpl(*this);
349}

358/// Verify that an inlinable callsite of a debug-info-bearing function in a
359/// debug-info-bearing function has a debug location attached to it. This
360/// mirrors an LLVM IR verifier.
361template <typename OpTy>
362static LogicalResult verifyCallOpDebugInfo(OpTy callOp, LLVMFuncOp callee) {
363 if (callee.isExternal())
364 return success();
365 auto parentFunc = callOp->template getParentOfType<FunctionOpInterface>();
366 if (!parentFunc)
367 return success();
368
369 auto hasSubprogram = [](Operation *op) {
370 return op->getLoc()
371 ->findInstanceOf<FusedLocWith<LLVM::DISubprogramAttr>>() !=
372 nullptr;
373 };
374 if (!hasSubprogram(parentFunc) || !hasSubprogram(callee))
375 return success();
376 bool containsLoc = !isa<UnknownLoc>(callOp->getLoc());
377 if (!containsLoc)
378 return callOp.emitError()
379 << "inlinable function call in a function with a DISubprogram "
380 "location must have a debug location";
381 return success();
382}

384/// Verify that the parameter and return types of the variadic callee type match
385/// the `callOp` argument and result types.
386template <typename OpTy>
387static LogicalResult verifyCallOpVarCalleeType(OpTy callOp) {
388 // An indirect call stores the callee in its first callee operand.
389 if (!callOp.getCallee().has_value() && callOp.getCalleeOperands().empty())
390 return callOp.emitOpError(
391 "must have either a `callee` attribute or at least an operand");
392
393 std::optional<LLVMFunctionType> varCalleeType = callOp.getVarCalleeType();
394 if (!varCalleeType)
395 return success();
396
397 // Verify the variadic callee type is a variadic function type.
398 if (!varCalleeType->isVarArg())
399 return callOp.emitOpError(
400 "expected var_callee_type to be a variadic function type");
401
402 // Note: `getArgOperands` is derived from `var_callee_type`, so the raw callee
403 // operands are used here instead.
404 Operation::operand_range passedOperands = getOperandsPassedToCallee(callOp);
405
406 // Verify the variadic callee type has at most as many parameters as the call
407 // has argument operands.
408 if (varCalleeType->getNumParams() > passedOperands.size())
409 return callOp.emitOpError("expected var_callee_type to have at most ")
410 << passedOperands.size() << " parameters";
411
412 // Verify the variadic callee type matches the call argument types.
413 for (auto [paramType, operand] :
414 llvm::zip(varCalleeType->getParams(), passedOperands))
415 if (paramType != operand.getType())
416 return callOp.emitOpError()
417 << "var_callee_type parameter type mismatch: " << paramType
418 << " != " << operand.getType();
419
420 // Verify the variadic callee type matches the call result type.
421 if (!callOp.getNumResults()) {
422 if (!isa<LLVMVoidType>(varCalleeType->getReturnType()))
423 return callOp.emitOpError("expected var_callee_type to return void");
424 } else {
425 if (callOp.getResult().getType() != varCalleeType->getReturnType())
426 return callOp.emitOpError("var_callee_type return type mismatch: ")
427 << varCalleeType->getReturnType()
428 << " != " << callOp.getResult().getType();
429 }
430 return success();
431}

435template <typename OpTy>
436static LogicalResult
437verifyCallOpSymbolUses(OpTy callOp, SymbolTableCollection &symbolTable) {
438 if (failed(verifyCallOpVarCalleeType(callOp)))
439 return failure();
440 // Type for the callee, we'll get it differently depending if it is a direct
441 // or indirect call.
442 Type fnType;
443
444 // If this is an indirect call, the callee attribute is missing.
445 FlatSymbolRefAttr calleeName = callOp.getCalleeAttr();
446 if (!calleeName) {
447 // Note: `verifyCallOpVarCalleeType` has already checked that there is a
448 // callee operand.
449 auto ptrType =
450 llvm::dyn_cast<LLVMPointerType>(callOp.getOperand(0).getType());
451 if (!ptrType)
452 return callOp.emitOpError("indirect call expects a pointer as callee: ")
453 << callOp.getOperand(0).getType();
454
455 // Nothing else to verify: an indirect callee cannot be resolved.
456 return success();
457 } else {
458 Operation *callee =
459 symbolTable.lookupNearestSymbolFrom(callOp, calleeName.getAttr());
460 if (!callee)
461 return callOp.emitOpError()
462 << "'" << calleeName.getValue()
463 << "' does not reference a symbol in the current scope";
464 if (auto fn = dyn_cast<LLVMFuncOp>(callee)) {
465 if (failed(verifyCallOpDebugInfo(callOp, fn)))
466 return failure();
467 fnType = fn.getFunctionType();
468 } else if (auto ifunc = dyn_cast<IFuncOp>(callee)) {
469 fnType = ifunc.getIFuncType();
470 } else if (isa<AliasOp>(callee)) {
471 // Aliases can alias functions, so calling through an alias is valid.
472 // The function type is determined by the call's operands and result
473 // types.
474 fnType = callOp.getCalleeFunctionType();
475 } else {
476 return callOp.emitOpError()
477 << "'" << calleeName.getValue()
478 << "' does not reference a valid LLVM function, IFunc, or alias";
479 }
480 }
481
482 LLVMFunctionType funcType = llvm::dyn_cast<LLVMFunctionType>(fnType);
483 if (!funcType)
484 return callOp.emitOpError("callee does not have a functional type: ")
485 << fnType;
486
487 if (funcType.isVarArg() && !callOp.getVarCalleeType())
488 return callOp.emitOpError()
489 << "missing var_callee_type attribute for vararg call";
490
491 // Verify the result types. These checks are more specific than what
492 // `verifyCallOpInterface` can report, so they are run first.
493 if (callOp.getNumResults() == 0 &&
494 !llvm::isa<LLVM::LLVMVoidType>(funcType.getReturnType()))
495 return callOp.emitOpError() << "expected function call to produce a value";
496
497 if (callOp.getNumResults() != 0 &&
498 llvm::isa<LLVM::LLVMVoidType>(funcType.getReturnType()))
499 return callOp.emitOpError()
500 << "calling function with void result must not produce values";
501
502 if (callOp.getNumResults() > 1)
503 return callOp.emitOpError()
504 << "expected LLVM function call to produce 0 or 1 result";
505
506 if (callOp.getNumResults() &&
507 callOp.getResult().getType() != funcType.getReturnType())
508 return callOp.emitOpError()
509 << "result type mismatch: " << callOp.getResult().getType()
510 << " != " << funcType.getReturnType();
511
512 // Verify that the operand types match the callee. Note that this does not
513 // need to special-case a variadic callee: the variadic arguments are not
514 // argument operands.
515 SmallVector<Type, 1> calleeResultTypes;
516 if (!llvm::isa<LLVM::LLVMVoidType>(funcType.getReturnType()))
517 calleeResultTypes.push_back(funcType.getReturnType());
518 return call_interface_impl::verifyCallOpInterface(
519 callOp, funcType.getParams(), calleeResultTypes);
520}

526void CallOp::print(OpAsmPrinter &p) {
527 auto callee = getCallee();
528 bool isDirect = callee.has_value();
529
530 p << ' ';
531
532 // Print calling convention.
533 if (getCConv() != LLVM::CConv::C)
534 p << stringifyCConv(getCConv()) << ' ';
535
536 if (getTailCallKind() != LLVM::TailCallKind::None)
537 p << tailcallkind::stringifyTailCallKind(getTailCallKind()) << ' ';
538
539 // Print the direct callee if present as a function attribute, or an indirect
540 // callee (first operand) otherwise.
541 if (isDirect)
542 p.printSymbolName(callee.value());
543 else
544 p << getOperand(0);
545
546 auto args = getCalleeOperands().drop_front(isDirect ? 0 : 1);
547 p << '(' << args << ')';
548
549 // Print the variadic callee type if the call is variadic.
550 if (std::optional<LLVMFunctionType> varCalleeType = getVarCalleeType())
551 p << " vararg(" << *varCalleeType << ")";
552
553 if (!getOpBundleOperands().empty()) {
554 p << " ";
555 printOpBundles(p, *this, getOpBundleOperands(),
556 getOpBundleOperands().getTypes(), getOpBundleTags());
557 }
558
559 p.printOptionalAttrDict(processFMFAttr(getAttrsForPrinting(*this).getAttrs()),
560 {getCalleeAttrName(), getTailCallKindAttrName(),
561 getVarCalleeTypeAttrName(), getCConvAttrName(),
562 getOperandSegmentSizesAttrName(),
563 getOpBundleSizesAttrName(),
564 getOpBundleTagsAttrName(), getArgAttrsAttrName(),
565 getResAttrsAttrName()});
566
567 p << " : ";
568 if (!isDirect)
569 p << getOperand(0).getType() << ", ";
570
571 // Reconstruct the MLIR function type from operand and result types.
572 call_interface_impl::printFunctionSignature(
573 p, args.getTypes(), getArgAttrsAttr(),
574 /*isVariadic=*/false, getResultTypes(), getResAttrsAttr());
575}

577/// Parses the type of a call operation and resolves the operands if the parsing
578/// succeeds. Returns failure otherwise.
579static ParseResult parseCallTypeAndResolveOperands(
580 OpAsmParser &parser, OperationState &result, bool isDirect,
581 ArrayRef<OpAsmParser::UnresolvedOperand> operands,
582 SmallVectorImpl<DictionaryAttr> &argAttrs,
583 SmallVectorImpl<DictionaryAttr> &resultAttrs) {
584 SMLoc trailingTypesLoc = parser.getCurrentLocation();
585 SmallVector<Type> types;
586 if (parser.parseColon())
587 return failure();
588 if (!isDirect) {
589 types.emplace_back();
590 if (parser.parseType(types.back()))
591 return failure();
592 if (parser.parseOptionalComma())
593 return parser.emitError(
594 trailingTypesLoc, "expected indirect call to have 2 trailing types");
595 }
596 SmallVector<Type> argTypes;
597 SmallVector<Type> resTypes;
598 if (call_interface_impl::parseFunctionSignature(parser, argTypes, argAttrs,
599 resTypes, resultAttrs)) {
600 if (isDirect)
601 return parser.emitError(trailingTypesLoc,
602 "expected direct call to have 1 trailing types");
603 return parser.emitError(trailingTypesLoc,
604 "expected trailing function type");
605 }
606
607 if (resTypes.size() > 1)
608 return parser.emitError(trailingTypesLoc,
609 "expected function with 0 or 1 result");
610 if (resTypes.size() == 1 && llvm::isa<LLVM::LLVMVoidType>(resTypes[0]))
611 return parser.emitError(trailingTypesLoc,
612 "expected a non-void result type");
613
614 // The head element of the types list matches the callee type for
615 // indirect calls, while the types list is emtpy for direct calls.
616 // Append the function input types to resolve the call operation
617 // operands.
618 llvm::append_range(types, argTypes);
619 if (parser.resolveOperands(operands, types, parser.getNameLoc(),
620 result.operands))
621 return failure();
622 if (!resTypes.empty())
623 result.addTypes(resTypes);
624
625 return success();
626}

628/// Parses an optional function pointer operand before the call argument list
629/// for indirect calls, or stops parsing at the function identifier otherwise.
630static ParseResult parseOptionalCallFuncPtr(
631 OpAsmParser &parser,
632 SmallVectorImpl<OpAsmParser::UnresolvedOperand> &operands) {
633 OpAsmParser::UnresolvedOperand funcPtrOperand;
634 OptionalParseResult parseResult = parser.parseOptionalOperand(funcPtrOperand);
635 if (parseResult.has_value()) {
636 if (failed(*parseResult))
637 return *parseResult;
638 operands.push_back(funcPtrOperand);
639 }
640 return success();
641}

643static ParseResult resolveOpBundleOperands(
644 OpAsmParser &parser, SMLoc loc, OperationState &state,
645 ArrayRef<SmallVector<OpAsmParser::UnresolvedOperand>> opBundleOperands,
646 ArrayRef<SmallVector<Type>> opBundleOperandTypes,
647 StringAttr opBundleSizesAttrName) {
648 unsigned opBundleIndex = 0;
649 for (const auto &[operands, types] :
650 llvm::zip_equal(opBundleOperands, opBundleOperandTypes)) {
651 if (operands.size() != types.size())
652 return parser.emitError(loc, "expected ")
653 << operands.size()
654 << " types for operand bundle operands for operand bundle #"
655 << opBundleIndex << ", but actually got " << types.size();
656 if (parser.resolveOperands(operands, types, loc, state.operands))
657 return failure();
658 }
659
660 SmallVector<int32_t> opBundleSizes;
661 opBundleSizes.reserve(opBundleOperands.size());
662 for (const auto &operands : opBundleOperands)
663 opBundleSizes.push_back(operands.size());
664
665 state.addAttribute(
666 opBundleSizesAttrName,
667 DenseI32ArrayAttr::get(parser.getContext(), opBundleSizes));
668
669 return success();
670}

672// <operation> ::= `llvm.call` (cconv)? (tailcallkind)? (function-id | ssa-use)
673// `(` ssa-use-list `)`
674// ( `vararg(` var-callee-type `)` )?
675// ( `[` op-bundles-list `]` )?
676// attribute-dict? `:` (type `,`)? function-type
677ParseResult CallOp::parse(OpAsmParser &parser, OperationState &result) {
678 SymbolRefAttr funcAttr;
679 TypeAttr varCalleeType;
680 SmallVector<OpAsmParser::UnresolvedOperand> operands;
681 SmallVector<SmallVector<OpAsmParser::UnresolvedOperand>> opBundleOperands;
682 SmallVector<SmallVector<Type>> opBundleOperandTypes;
683 ArrayAttr opBundleTags;
684
685 // Default to C Calling Convention if no keyword is provided.
686 result.addAttribute(
687 getCConvAttrName(result.name),
688 CConvAttr::get(parser.getContext(),
689 parseOptionalLLVMKeyword<CConv>(parser, LLVM::CConv::C)));
690
691 result.addAttribute(
692 getTailCallKindAttrName(result.name),
693 TailCallKindAttr::get(parser.getContext(),
694 parseOptionalLLVMKeyword<TailCallKind>(
695 parser, LLVM::TailCallKind::None)));
696
697 // Parse a function pointer for indirect calls.
698 if (parseOptionalCallFuncPtr(parser, operands))
699 return failure();
700 bool isDirect = operands.empty();
701
702 // Parse a function identifier for direct calls.
703 if (isDirect)
704 if (parser.parseAttribute(funcAttr, "callee", result.attributes))
705 return failure();
706
707 // Parse the function arguments.
708 if (parser.parseOperandList(operands, OpAsmParser::Delimiter::Paren))
709 return failure();
710
711 bool isVarArg = parser.parseOptionalKeyword("vararg").succeeded();
712 if (isVarArg) {
713 StringAttr varCalleeTypeAttrName =
714 CallOp::getVarCalleeTypeAttrName(result.name);
715 if (parser.parseLParen().failed() ||
716 parser
717 .parseAttribute(varCalleeType, varCalleeTypeAttrName,
718 result.attributes)
719 .failed() ||
720 parser.parseRParen().failed())
721 return failure();
722 }
723
724 SMLoc opBundlesLoc = parser.getCurrentLocation();
725 if (std::optional<ParseResult> result = parseOpBundles(
726 parser, opBundleOperands, opBundleOperandTypes, opBundleTags);
727 result && failed(*result))
728 return failure();
729 if (opBundleTags && !opBundleTags.empty())
730 result.addAttribute(CallOp::getOpBundleTagsAttrName(result.name).getValue(),
731 opBundleTags);
732
733 if (parser.parseOptionalAttrDict(result.attributes))
734 return failure();
735
736 // Parse the trailing type list and resolve the operands.
737 SmallVector<DictionaryAttr> argAttrs;
738 SmallVector<DictionaryAttr> resultAttrs;
739 if (parseCallTypeAndResolveOperands(parser, result, isDirect, operands,
740 argAttrs, resultAttrs))
741 return failure();
742 call_interface_impl::addArgAndResultAttrs(
743 parser.getBuilder(), result, argAttrs, resultAttrs,
744 getArgAttrsAttrName(result.name), getResAttrsAttrName(result.name));
745 if (resolveOpBundleOperands(parser, opBundlesLoc, result, opBundleOperands,
746 opBundleOperandTypes,
747 getOpBundleSizesAttrName(result.name)))
748 return failure();
749
750 int32_t numOpBundleOperands = 0;
751 for (const auto &operands : opBundleOperands)
752 numOpBundleOperands += operands.size();
753
754 result.addAttribute(
755 CallOp::getOperandSegmentSizeAttr(),
756 parser.getBuilder().getDenseI32ArrayAttr(
757 {static_cast<int32_t>(operands.size()), numOpBundleOperands}));
758 return success();
759}

988

1017LogicalResult LandingpadOp::verify() {
1018 Value value;
1019 if (LLVMFuncOp func = (*this)->getParentOfType<LLVMFuncOp>()) {
1020 if (!func.getPersonality())
1021 return emitError(
1022 "llvm.landingpad needs to be in a function with a personality");
1023 }
1024
1025 // Consistency of llvm.landingpad result types is checked in
1026 // LLVMFuncOp::verify().
1027
1028 if (!getCleanup() && getOperands().empty())
1029 return emitError("landingpad instruction expects at least one clause or "
1030 "cleanup attribute");
1031
1032 for (unsigned idx = 0, ie = getNumOperands(); idx < ie; idx++) {
1033 value = getOperand(idx);
1034 bool isFilter = llvm::isa<LLVMArrayType>(value.getType());
1035 if (isFilter) {
1036 // FIXME: Verify filter clauses when arrays are appropriately handled
1037 } else {
1038 // catch - global addresses only.
1039 // Bitcast ops should have global addresses as their args.
1040 if (auto bcOp = value.getDefiningOp<BitcastOp>()) {
1041 if (auto addrOp = bcOp.getArg().getDefiningOp<AddressOfOp>())
1042 continue;
1043 return emitError("constant clauses expected").attachNote(bcOp.getLoc())
1044 << "global addresses expected as operand to "
1045 "bitcast used in clauses for landingpad";
1046 }
1047 // ZeroOp and AddressOfOp allowed
1048 if (value.getDefiningOp<ZeroOp>())
1049 continue;
1050 if (value.getDefiningOp<AddressOfOp>())
1051 continue;
1052 return emitError("clause #")
1053 << idx << " is not a known constant - null, addressof, bitcast";
1054 }
1055 }
1056 return success();
1057}

1076// <operation> ::= `llvm.landingpad` `cleanup`?
1077// ((`catch` | `filter`) operand-type ssa-use)* attribute-dict?
1078ParseResult LandingpadOp::parse(OpAsmParser &parser, OperationState &result) {
1079 // Check for cleanup
1080 if (succeeded(parser.parseOptionalKeyword("cleanup")))
1081 result.addAttribute("cleanup", parser.getBuilder().getUnitAttr());
1082
1083 // Parse clauses with types
1084 while (succeeded(parser.parseOptionalLParen()) &&
1085 (succeeded(parser.parseOptionalKeyword("filter")) ||
1086 succeeded(parser.parseOptionalKeyword("catch")))) {
1087 OpAsmParser::UnresolvedOperand operand;
1088 Type ty;
1089 if (parser.parseOperand(operand) || parser.parseColon() ||
1090 parser.parseType(ty) ||
1091 parser.resolveOperand(operand, ty, result.operands) ||
1092 parser.parseRParen())
1093 return failure();
1094 }
1095
1096 Type type;
1097 if (parser.parseColon() || parser.parseType(type))
1098 return failure();
1099
1100 result.addTypes(type);
1101 return success();
1102}
