23/// Convert an array of integer attributes to a vector of integers that can be
24/// used as indices in LLVM operations.
25template <typename IntT = int64_t>
26SmallVector<IntT> convertArrayToIndices(ArrayRef<Attribute> attrs) {
27 SmallVector<IntT> indices;
28 indices.reserve(attrs.size());
29 for (Attribute attr : attrs)
30 indices.push_back(cast<IntegerAttr>(attr).getInt());
31 return indices;
32}
33
34/// Convert an `ArrayAttr` of integer attributes to a vector of integers that
35/// can be used as indices in LLVM operations.
36template <typename IntT = int64_t>
37SmallVector<IntT> convertArrayToIndices(ArrayAttr attrs) {
38 return convertArrayToIndices<IntT>(attrs.getValue());
39}
