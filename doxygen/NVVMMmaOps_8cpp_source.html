44template <typename AttrTy>
45static ParseResult parseEnumKeyword(OpAsmParser &parser, AttrTy &attr) {
46 SMLoc loc = parser.getCurrentLocation();
47 std::string keyword;
48 if (parser.parseKeywordOrString(&keyword))
49 return failure();
50
51 using EnumTy = decltype(attr.getValue());
52 std::optional<EnumTy> value = NVVM::symbolizeEnum<EnumTy>(keyword);
53 if (!value)
54 return parser.emitError(loc) << "unknown enum value '" << keyword << "'";
55
56 attr = AttrTy::get(parser.getContext(), *value);
57 return success();
58}

64// Given the element type of an operand and whether or not it is an accumulator,
65// this function returns the PTX type (`NVVM::MMATypes`) that corresponds to the
66// operand's element type.
67std::optional<mlir::NVVM::MMATypes>
68MmaOp::inferOperandMMAType(Type operandElType, bool isAccumulator) {
69 auto half2Type =
70 VectorType::get(2, Float16Type::get(operandElType.getContext()));
71 if (operandElType.isF64())
72 return NVVM::MMATypes::f64;
73 if (operandElType.isF16() || operandElType == half2Type)
74 return NVVM::MMATypes::f16;
75 if (operandElType.isF32() && isAccumulator)
76 return NVVM::MMATypes::f32;
77 if (operandElType.isF32() && !isAccumulator)
78 return NVVM::MMATypes::tf32;
79 if (llvm::isa<IntegerType>(operandElType)) {
80 if (isAccumulator)
81 return NVVM::MMATypes::s32;
82 return std::nullopt;
83 }
84
85 if (auto structType = llvm::dyn_cast<LLVM::LLVMStructType>(operandElType)) {
86 if (structType.getBody().empty())
87 return std::nullopt;
88 return inferOperandMMAType(structType.getBody()[0], isAccumulator);
89 }
90
91 return std::nullopt;
92}

103 return isInt4PtxType(type) || isInt8PtxType(type) || type == MMATypes::b1 ||

144static ParseResult parseMmaPropertyValue(OpAsmParser &parser,
145 NamedAttrList &attributes,
146 StringRef name) {
147 if (attributes.get(name))
148 return parser.emitError(parser.getCurrentLocation(),
149 "duplicate property '" + name + "'");

152 return failure();

180static ParseResult parseMmaProperties(OpAsmParser &parser,
181 NamedAttrList &attributes,
182 ArrayRef<StringRef> allowedKeywords,
183 ArrayRef<StringRef> requiredProperties) {
184 while (true) {
185 StringRef keyword;
186 if (parser.parseKeyword(&keyword))
187 return failure();
188 if (!llvm::is_contained(allowedKeywords, keyword))
189 return parser.emitError(parser.getCurrentLocation(),
190 "unknown MMA property '" + keyword + "'");
191
192 ParseResult parseResult = success();
193 if (keyword == "shape")
194 parseResult =
195 parseMmaPropertyValue<MMAShapeAttr>(parser, attributes, "shape");
196 else if (keyword == "b1_op")
197 parseResult =
198 parseMmaEnumPropertyValue<MMAB1OpAttr>(parser, attributes, "b1Op");
199 else if (keyword == "int_overflow")
200 parseResult = parseMmaEnumPropertyValue<MMAIntOverflowAttr>(
201 parser, attributes, "intOverflowBehavior");
202 else if (keyword == "layout_a")
203 parseResult = parseMmaEnumPropertyValue<MMALayoutAttr>(parser, attributes,
204 "layoutA");
205 else if (keyword == "layout_b")
206 parseResult = parseMmaEnumPropertyValue<MMALayoutAttr>(parser, attributes,
207 "layoutB");
208 else if (keyword == "multiplicand_a_ptx_type")
209 parseResult = parseMmaEnumPropertyValue<MMATypesAttr>(
210 parser, attributes, "multiplicandAPtxType");
211 else if (keyword == "multiplicand_b_ptx_type")
212 parseResult = parseMmaEnumPropertyValue<MMATypesAttr>(
213 parser, attributes, "multiplicandBPtxType");
214 else if (keyword == "kind") {
215 if (llvm::is_contained(allowedKeywords, "block_scale_format"))
216 parseResult = parseMmaEnumPropertyValue<MMABlockScaleKindAttr>(
217 parser, attributes, "kind");
218 else
219 parseResult =
220 parseMmaEnumPropertyValue<MMAKindAttr>(parser, attributes, "kind");
221 } else if (keyword == "scale_vec_size")
222 parseResult = parseMmaEnumPropertyValue<ScaleVecSizeAttr>(
223 parser, attributes, "scaleVecSize");
224 else if (keyword == "block_scale_format")
225 parseResult = parseMmaEnumPropertyValue<BlockScaleFormatAttr>(
226 parser, attributes, "blockScaleFormat");
227 else if (keyword == "ordered_metadata") {
228 if (attributes.get("orderedMetadata"))
229 return parser.emitError(parser.getCurrentLocation(),
230 "duplicate property 'orderedMetadata'");
231 attributes.append("orderedMetadata", parser.getBuilder().getUnitAttr());
232 } else {
233 return parser.emitError(parser.getCurrentLocation(),
234 "unknown MMA property '" + keyword + "'");
235 }
236 if (failed(parseResult))
237 return failure();
238 if (failed(parser.parseOptionalComma()))
239 break;
240 }
241
242 for (StringRef property : requiredProperties) {
243 if (!attributes.get(property))
244 return parser.emitError(parser.getCurrentLocation(),
245 "missing required property '" + property + "'");
246 }
247
248 NamedAttrList discardableAttributes;
249 if (parser.parseOptionalAttrDict(discardableAttributes))
250 return failure();
251 for (NamedAttribute attribute : discardableAttributes) {
252 if (isMmaPropertyName(attribute.getName()))
253 return parser.emitError(
254 parser.getCurrentLocation(),
255 "inherent property '" + attribute.getName().getValue() +
256 "' must be spelled directly in the operation syntax");
257 attributes.append(attribute);
258 }
259 return success();
260}

262void MmaOp::print(OpAsmPrinter &p) {
263 SmallVector<Type, 4> regTypes;
264 struct MMAOperandFragment {
265 StringRef operandName;
266 StringRef ptxTypeAttr;
267 SmallVector<Value, 4> regs;
268 explicit MMAOperandFragment(StringRef name, StringRef ptxTypeName)
269 : operandName(name), ptxTypeAttr(ptxTypeName) {}
270 };
271
272 std::array<MMAOperandFragment, 3> frags{
273 MMAOperandFragment("A", getMultiplicandAPtxTypeAttrName()),
274 MMAOperandFragment("B", getMultiplicandBPtxTypeAttrName()),
275 MMAOperandFragment("C", "")};
276 SmallVector<StringRef, 4> ignoreAttrNames{
277 mlir::NVVM::MmaOp::getOperandSegmentSizeAttr()};
278
279 for (unsigned fragIdx = 0; fragIdx < frags.size(); fragIdx++) {
280 auto &frag = frags[fragIdx];
281 auto varOperandSpec = getODSOperandIndexAndLength(fragIdx);
282 for (auto operandIdx = varOperandSpec.first;
283 operandIdx < varOperandSpec.first + varOperandSpec.second;
284 operandIdx++) {
285 frag.regs.push_back(this->getOperand(operandIdx));
286 if (operandIdx == 0) {
287 regTypes.push_back(this->getOperand(operandIdx).getType());
288 }
289 }
290 std::optional<MMATypes> inferredType = MmaOp::inferOperandMMAType(
291 regTypes.back(), /*isAccumulator=*/fragIdx >= 2);
292 if (inferredType)
293 ignoreAttrNames.push_back(frag.ptxTypeAttr);
294 }
295
296 auto printMmaOperand = [&](const MMAOperandFragment &frag) -> void {
297 p << " " << frag.operandName;
298 p << "[";
299 p.printOperands(frag.regs);
300 p << "] ";
301 };
302
303 for (const auto &frag : frags) {
304 printMmaOperand(frag);
305 }
306
307 bool isFirstProperty = true;
308 printMmaProperty(p, isFirstProperty, "shape", getShapeAttr());
309 if (getB1OpAttr())
310 printMmaEnumProperty(p, isFirstProperty, "b1_op", getB1OpAttr());
311 if (getIntOverflowBehaviorAttr())
312 printMmaEnumProperty(p, isFirstProperty, "int_overflow",
313 getIntOverflowBehaviorAttr());
314 printMmaEnumProperty(p, isFirstProperty, "layout_a", getLayoutAAttr());
315 printMmaEnumProperty(p, isFirstProperty, "layout_b", getLayoutBAttr());
316 if (getMultiplicandAPtxTypeAttr() &&
317 !llvm::is_contained(ignoreAttrNames, getMultiplicandAPtxTypeAttrName()))
318 printMmaEnumProperty(p, isFirstProperty, "multiplicand_a_ptx_type",
319 getMultiplicandAPtxTypeAttr());
320 if (getMultiplicandBPtxTypeAttr() &&
321 !llvm::is_contained(ignoreAttrNames, getMultiplicandBPtxTypeAttrName()))
322 printMmaEnumProperty(p, isFirstProperty, "multiplicand_b_ptx_type",
323 getMultiplicandBPtxTypeAttr());
324 llvm::append_range(ignoreAttrNames,
325 ArrayRef<StringRef>{getShapeAttrName(), getB1OpAttrName(),
326 getIntOverflowBehaviorAttrName(),
327 getLayoutAAttrName(),
328 getLayoutBAttrName(),
329 getMultiplicandAPtxTypeAttrName(),
330 getMultiplicandBPtxTypeAttrName()});
331 p.printOptionalAttrDict((*this)->getDiscardableAttrDictionary().getValue(),
332 ignoreAttrNames);
333
334 // Print the types of the operands and result.
335 p << " : " << "(";
336 llvm::interleaveComma(SmallVector<Type, 3>{frags[0].regs[0].getType(),
337 frags[1].regs[0].getType(),
338 frags[2].regs[0].getType()},
339 p);
340 p << ")";
341 p.printArrowTypeList(TypeRange{this->getRes().getType()});
342}

344void MmaOp::build(OpBuilder &builder, OperationState &result, Type resultType,
345 ValueRange operandA, ValueRange operandB, ValueRange operandC,
346 ArrayRef<int64_t> shape, std::optional<MMAB1Op> b1Op,
347 std::optional<MMAIntOverflow> intOverflow,
348 std::optional<std::array<MMATypes, 2>> multiplicandPtxTypes,
349 std::optional<std::array<MMALayout, 2>> multiplicandLayouts) {
350
351 assert(shape.size() == 3 && "expected shape to have size 3 (m, n, k)");
352 MLIRContext *ctx = builder.getContext();
353 result.addAttribute(
354 "shape", builder.getAttr<MMAShapeAttr>(shape[0], shape[1], shape[2]));
355
356 result.addOperands(operandA);
357 result.addOperands(operandB);
358 result.addOperands(operandC);
359
360 if (multiplicandPtxTypes) {
361 result.addAttribute("multiplicandAPtxType",
362 MMATypesAttr::get(ctx, (*multiplicandPtxTypes)[0]));
363 result.addAttribute("multiplicandBPtxType",
364 MMATypesAttr::get(ctx, (*multiplicandPtxTypes)[1]));
365 } else {
366 if (auto res = inferOperandMMAType(operandA[0].getType(), false))
367 result.addAttribute("multiplicandAPtxType", MMATypesAttr::get(ctx, *res));
368 if (auto res = inferOperandMMAType(operandB[0].getType(), false))
369 result.addAttribute("multiplicandBPtxType", MMATypesAttr::get(ctx, *res));
370 }
371
372 if (multiplicandLayouts) {
373 result.addAttribute("layoutA",
374 MMALayoutAttr::get(ctx, (*multiplicandLayouts)[0]));
375 result.addAttribute("layoutB",
376 MMALayoutAttr::get(ctx, (*multiplicandLayouts)[1]));
377 } else {
378 result.addAttribute("layoutA", MMALayoutAttr::get(ctx, MMALayout::row));
379 result.addAttribute("layoutB", MMALayoutAttr::get(ctx, MMALayout::col));
380 }
381
382 if (intOverflow.has_value())
383 result.addAttribute("intOverflowBehavior",
384 MMAIntOverflowAttr::get(ctx, *intOverflow));
385 if (b1Op.has_value())
386 result.addAttribute("b1Op", MMAB1OpAttr::get(ctx, *b1Op));
387
388 result.addTypes(resultType);
389 result.addAttribute(
390 MmaOp::getOperandSegmentSizeAttr(),
391 builder.getDenseI32ArrayAttr({static_cast<int32_t>(operandA.size()),
392 static_cast<int32_t>(operandB.size()),
393 static_cast<int32_t>(operandC.size())}));
394}

396// <operation> :=
397// A `[` $operandA `]` B `[` $operandB `]` C `[` $operandC `]`
398// properties attr-dict
399// : (type($operandA[0]), type($operandB[0]), type($operandC[0]))
400// `->` type($res)
401ParseResult MmaOp::parse(OpAsmParser &parser, OperationState &result) {
402 struct MMAOperandFragment {
403 std::optional<MMATypes> elemtype;
404 SmallVector<OpAsmParser::UnresolvedOperand, 4> regs;
405 SmallVector<Type> regTypes;
406 };
407
408 Builder &builder = parser.getBuilder();
409 std::array<MMAOperandFragment, 4> frags;
410
411 NamedAttrList namedAttributes;
412
413 // A helper to parse the operand segments.
414 auto parseMmaOperand = [&](StringRef operandName,
415 MMAOperandFragment &frag) -> LogicalResult {
416 if (parser.parseKeyword(operandName).failed())
417 return failure();
418 if (parser
419 .parseOperandList(frag.regs, OpAsmParser::Delimiter::OptionalSquare)
420 .failed())
421 return failure();
422 return success();
423 };
424
425 // Parse the operand segments.
426 if (parseMmaOperand("A", frags[0]).failed())
427 return failure();
428 if (parseMmaOperand("B", frags[1]).failed())
429 return failure();
430 if (parseMmaOperand("C", frags[2]).failed())
431 return failure();
432
433 if (parseMmaProperties(parser, namedAttributes,
434 {"shape", "b1_op", "int_overflow", "layout_a",
435 "layout_b", "multiplicand_a_ptx_type",
436 "multiplicand_b_ptx_type"},
437 {"shape", "layoutA", "layoutB"}))
438 return failure();
439
440 // Parse the type specification and resolve operands.
441 SmallVector<Type, 3> operandTypes;
442 if (failed(parser.parseColon()))
443 return failure();
444 if (failed(parser.parseLParen()))
445 return failure();
446 if (failed(parser.parseTypeList(operandTypes)))
447 return failure();
448 if (failed(parser.parseRParen()))
449 if (operandTypes.size() != 3)
450 return parser.emitError(
451 parser.getNameLoc(),
452 "expected one type for each operand segment but got " +
453 Twine(operandTypes.size()) + " types");
454 for (const auto &iter : llvm::enumerate(operandTypes)) {
455 auto &frag = frags[iter.index()];
456 frag.regTypes.resize(frag.regs.size(), iter.value());
457 if (failed(parser.resolveOperands(frag.regs, frag.regTypes,
458 parser.getNameLoc(), result.operands)))
459 return failure();
460 frag.elemtype = inferOperandMMAType(frag.regTypes[0],
461 /*isAccumulator*/ iter.index() < 2);
462 }
463
464 Type resultType;
465 if (parser.parseArrow() || parser.parseType(resultType))
466 return failure();
467 frags[3].elemtype = inferOperandMMAType(resultType, /*isAccumulator*/ true);
468
469 std::array<StringRef, 2> names{"multiplicandAPtxType",
470 "multiplicandBPtxType"};
471 for (unsigned idx = 0; idx < names.size(); idx++) {
472 const auto &frag = frags[idx];
473 std::optional<NamedAttribute> attr = namedAttributes.getNamed(names[idx]);
474 if (!frag.elemtype.has_value() && !attr.has_value()) {
475 return parser.emitError(
476 parser.getNameLoc(),
477 "attribute " + names[idx] +
478 " is not provided explicitly and cannot be inferred");
479 }
480 if (!attr.has_value())
481 result.addAttribute(
482 names[idx], MMATypesAttr::get(parser.getContext(), *frag.elemtype));
483 }
484
485 result.addTypes(resultType);
486 if (!namedAttributes.empty())
487 result.addAttributes(namedAttributes);
488 result.addAttribute(MmaOp::getOperandSegmentSizeAttr(),
489 builder.getDenseI32ArrayAttr({
490 static_cast<int32_t>(frags[0].regs.size()),
491 static_cast<int32_t>(frags[1].regs.size()),
492 static_cast<int32_t>(frags[2].regs.size()),
493 }));
494 return success();
495}

497LogicalResult MmaOp::verify() {
498 MLIRContext *context = getContext();
499 auto f16Ty = Float16Type::get(context);
500 auto i32Ty = IntegerType::get(context, 32);
501 auto f16x2Ty = VectorType::get(2, f16Ty);
502 auto f32Ty = Float32Type::get(context);
503 auto f16x2x4StructTy = LLVM::LLVMStructType::getLiteral(
504 context, {f16x2Ty, f16x2Ty, f16x2Ty, f16x2Ty});
505
506 auto s32x4StructTy =
507 LLVM::LLVMStructType::getLiteral(context, {i32Ty, i32Ty, i32Ty, i32Ty});
508 auto f32x8StructTy =
509 LLVM::LLVMStructType::getLiteral(context, SmallVector<Type>(8, f32Ty));
510 auto f16x2x2StructTy =
511 LLVM::LLVMStructType::getLiteral(context, {f16x2Ty, f16x2Ty});
512 auto f32x4StructTy =
513 LLVM::LLVMStructType::getLiteral(context, {f32Ty, f32Ty, f32Ty, f32Ty});
514 auto s32x2StructTy =
515 LLVM::LLVMStructType::getLiteral(context, {i32Ty, i32Ty});
516
517 std::array<int64_t, 3> mmaShape{getShapeAttr().getM(), getShapeAttr().getN(),
518 getShapeAttr().getK()};
519
520 // These variables define the set of allowed data types for matrices A, B, C,
521 // and result.
522 using AllowedShapes = SmallVector<std::array<int64_t, 3>, 2>;
523 using AllowedTypes = SmallVector<SmallVector<Type, 4>, 2>;
524 AllowedShapes allowedShapes;
525 AllowedTypes expectedA;
526 AllowedTypes expectedB;
527 AllowedTypes expectedC;
528 SmallVector<Type> expectedResult;
529
530 // When M = 16, we just need to calculate the number of 8xk tiles, where
531 // k is a factor that depends on the data type.
532 if (mmaShape[0] == 16) {
533 int64_t kFactor;
534 Type multiplicandFragType;
535 switch (*getMultiplicandAPtxType()) {
536 case MMATypes::tf32:
537 kFactor = 4;
538 multiplicandFragType = i32Ty;
539 expectedResult.push_back(LLVM::LLVMStructType::getLiteral(
540 context, {f32Ty, f32Ty, f32Ty, f32Ty}));
541 break;
542 case MMATypes::bf16:
543 kFactor = 8;
544 multiplicandFragType = i32Ty;
545 expectedResult.push_back(LLVM::LLVMStructType::getLiteral(
546 context, {f32Ty, f32Ty, f32Ty, f32Ty}));
547 break;
548 case MMATypes::f16:
549 kFactor = 8;
550 multiplicandFragType = f16x2Ty;
551 expectedResult.push_back(f16x2x2StructTy);
552 expectedResult.push_back(f32x4StructTy);
553 break;
554 case MMATypes::e4m3:
555 case MMATypes::e5m2:
556 // FP8 (m16n8k16 / m16n8k32) packs 4 values per 32-bit register, same
557 // as s8/u8, but the accumulator is f16 or f32 (not integer).
558 kFactor = 16;
559 multiplicandFragType = i32Ty;
560 expectedResult.push_back(f16x2x2StructTy);
561 expectedResult.push_back(f32x4StructTy);
562 break;
563 case MMATypes::s4:
564 case MMATypes::u4:
565 kFactor = 32;
566 break;
567 case MMATypes::b1:
568 kFactor = 128;
569 break;
570 case MMATypes::s8:
571 case MMATypes::u8:
572 kFactor = 16;
573 break;
574 default:
575 return emitError("invalid shape or multiplicand type: ")
576 << getMultiplicandAPtxType().value();
577 }
578
579 if (isIntegerPtxType(getMultiplicandAPtxType().value())) {
580 expectedResult.push_back(s32x4StructTy);
581 expectedC.emplace_back(4, i32Ty);
582 multiplicandFragType = i32Ty;
583 } else {
584 expectedC.emplace_back(2, f16x2Ty);
585 expectedC.emplace_back(4, f32Ty);
586 }
587
588 int64_t unitA = (mmaShape[0] / 8) * (mmaShape[2] / kFactor);
589 int64_t unitB = (mmaShape[1] / 8) * (mmaShape[2] / kFactor);
590 expectedA.emplace_back(unitA, multiplicandFragType);
591 expectedB.emplace_back(unitB, multiplicandFragType);
592 allowedShapes.push_back({16, 8, kFactor});
593 allowedShapes.push_back({16, 8, kFactor * 2});
594
595 if (resultPtxType() != accumPtxType())
596 return emitOpError("ctype does not match dtype");
597 }
598
599 // In the M=8 case, there is only 1 possible case per data type.
600 if (mmaShape[0] == 8) {
601 if (*getMultiplicandAPtxType() == MMATypes::f16) {
602 expectedA.emplace_back(2, f16x2Ty);
603 expectedB.emplace_back(2, f16x2Ty);
604 expectedResult.push_back(f16x2x4StructTy);
605 expectedResult.push_back(f32x8StructTy);
606 expectedC.emplace_back(4, f16x2Ty);
607 expectedC.emplace_back(8, f32Ty);
608 allowedShapes.push_back({8, 8, 4});
609 }
610 if (*getMultiplicandAPtxType() == MMATypes::f64) {
611 Type f64Ty = Float64Type::get(context);
612 expectedA.emplace_back(1, f64Ty);
613 expectedB.emplace_back(1, f64Ty);
614 expectedC.emplace_back(2, f64Ty);
615 expectedResult.emplace_back(LLVM::LLVMStructType::getLiteral(
616 context, SmallVector<Type>(2, f64Ty)));
617 allowedShapes.push_back({8, 8, 4});
618 }
619 if (isIntegerPtxType(getMultiplicandAPtxType().value())) {
620 expectedA.push_back({i32Ty});
621 expectedB.push_back({i32Ty});
622 expectedC.push_back({i32Ty, i32Ty});
623 expectedResult.push_back(s32x2StructTy);
624 if (isInt4PtxType(getMultiplicandAPtxType().value()))
625 allowedShapes.push_back({8, 8, 32});
626 if (isInt8PtxType(getMultiplicandAPtxType().value()))
627 allowedShapes.push_back({8, 8, 16});
628 if (getMultiplicandAPtxType().value() == MMATypes::b1)
629 allowedShapes.push_back({8, 8, 128});
630 }
631 }
632
633 std::string errorMessage;
634 llvm::raw_string_ostream errorStream(errorMessage);
635
636 // Check that we matched an existing shape/dtype combination.
637 if (expectedA.empty() || expectedB.empty() || expectedC.empty() ||
638 !llvm::is_contained(allowedShapes, mmaShape)) {
639 errorStream << "unimplemented variant for MMA shape <";
640 llvm::interleaveComma(mmaShape, errorStream);
641 errorStream << ">";
642 return emitOpError(errorMessage);
643 }
644
645 // Verify the operand types for segments of A, B, and C operands.
646 std::array<StringRef, 3> operandNames{"A", "B", "C"};
647 for (const auto &iter : llvm::enumerate(std::array<AllowedTypes, 3>{
648 std::move(expectedA), std::move(expectedB), std::move(expectedC)})) {
649 auto spec = this->getODSOperandIndexAndLength(iter.index());
650 SmallVector<Type, 4> operandTySeg(operand_type_begin() + spec.first,
651 operand_type_begin() + spec.first +
652 spec.second);
653 bool match = llvm::is_contained(iter.value(), operandTySeg);
654
655 if (!match) {
656 errorStream << "Could not match types for the "
657 << operandNames[iter.index()]
658 << " operands; expected one of ";
659 for (const auto &x : iter.value()) {
660 errorStream << x.size() << "x" << x[0] << " ";
661 }
662 errorStream << "but got ";
663 llvm::interleaveComma(operandTySeg, errorStream);
664 return emitOpError(errorMessage);
665 }
666 }
667
668 // Check the result type
669 if (!llvm::any_of(expectedResult, [&](Type expectedResultType) {
670 return expectedResultType == getResult().getType();
671 })) {
672 errorStream
673 << "Could not match allowed types for the result; expected one of ";
674 llvm::interleaveComma(expectedResult, errorStream);
675 errorStream << " but got " << getResult().getType();
676 return emitOpError(errorMessage);
677 }
678
679 // Ensure that binary MMA variants have a b1 MMA operation defined.
680 if (getMultiplicandAPtxType() == MMATypes::b1 && !getB1Op()) {
681 return emitOpError("op requires " + getB1OpAttrName().strref() +
682 " attribute");
683 }
684
685 // Ensure int4/int8 MMA variants specify the accum overflow behavior
686 // attribute.
687 if (isInt4PtxType(*getMultiplicandAPtxType()) ||
688 isInt8PtxType(*getMultiplicandAPtxType())) {
689 if (!getIntOverflowBehavior())
690 return emitOpError("op requires " +
691 getIntOverflowBehaviorAttrName().strref() +
692 " attribute");
693 }
694
695 // Validate layout combinations. According to the operation description, most
696 // MMA operations require layoutA=row and layoutB=col. Only m8n8k4 with f16
697 // can use other layout combinations.
698 bool isM8N8K4_F16 =
699 (mmaShape[0] == 8 && mmaShape[1] == 8 && mmaShape[2] == 4 &&
700 getMultiplicandAPtxType() == MMATypes::f16);
701
702 if (!isM8N8K4_F16) {
703 // For all other shapes/types, layoutA must be row and layoutB must be col
704 if (getLayoutA() != MMALayout::row || getLayoutB() != MMALayout::col) {
705 return emitOpError("requires layoutA = #nvvm.mma_layout<row> and "
706 "layoutB = #nvvm.mma_layout<col> for shape <")
707 << mmaShape[0] << ", " << mmaShape[1] << ", " << mmaShape[2]
708 << "> with element types " << *getMultiplicandAPtxType() << " and "
709 << *getMultiplicandBPtxType()
710 << ". Only m8n8k4 with f16 supports other layouts.";
711 }
712 }
713
714 return success();
715}

752void MmaSpOp::print(OpAsmPrinter &p) {
753 SmallVector<Type, 4> regTypes;
754 struct MMAOperandFragment {
755 StringRef operandName;
756 StringRef ptxTypeAttr;
757 SmallVector<Value, 4> regs;
758 explicit MMAOperandFragment(StringRef name, StringRef ptxTypeName)
759 : operandName(name), ptxTypeAttr(ptxTypeName) {}
760 };
761
762 std::array<MMAOperandFragment, 5> frags{
763 MMAOperandFragment("A", getMultiplicandAPtxTypeAttrName()),
764 MMAOperandFragment("B", getMultiplicandBPtxTypeAttrName()),
765 MMAOperandFragment("C", ""), MMAOperandFragment("sparseMetadata", ""),
766 MMAOperandFragment("selector", "")};
767 SmallVector<StringRef, 4> ignoreAttrNames{
768 mlir::NVVM::MmaSpOp::getOperandSegmentSizeAttr()};
769
770 // Handle variadic operands A, B, C
771 for (unsigned fragIdx = 0; fragIdx < 3; fragIdx++) {
772 auto &frag = frags[fragIdx];
773 auto varOperandSpec = getODSOperandIndexAndLength(fragIdx);
774 for (auto operandIdx = varOperandSpec.first;
775 operandIdx < varOperandSpec.first + varOperandSpec.second;
776 operandIdx++) {
777 frag.regs.push_back(this->getOperand(operandIdx));
778 if (operandIdx == varOperandSpec.first) {
779 regTypes.push_back(this->getOperand(operandIdx).getType());
780 }
781 }
782 std::optional<MMATypes> inferredType = MmaOp::inferOperandMMAType(
783 regTypes.back(), /*isAccumulator=*/fragIdx >= 2);
784 if (inferredType)
785 ignoreAttrNames.push_back(frag.ptxTypeAttr);
786 }
787
788 // Handle sparse metadata and selector (single operands)
789 frags[3].regs.push_back(getSparseMetadata());
790 frags[4].regs.push_back(getSparsitySelector());
791
792 auto printMmaSpOperand = [&](const MMAOperandFragment &frag) -> void {
793 p << " " << frag.operandName;
794 p << "[";
795 p.printOperands(frag.regs);
796 p << "]";
797 };
798
799 for (const auto &frag : frags)
800 printMmaSpOperand(frag);
801
802 bool isFirstProperty = true;
803 printMmaProperty(p, isFirstProperty, "shape", getShapeAttr());
804 if (getIntOverflowBehaviorAttr())
805 printMmaEnumProperty(p, isFirstProperty, "int_overflow",
806 getIntOverflowBehaviorAttr());
807 if (getMultiplicandAPtxTypeAttr() &&
808 !llvm::is_contained(ignoreAttrNames, getMultiplicandAPtxTypeAttrName()))
809 printMmaEnumProperty(p, isFirstProperty, "multiplicand_a_ptx_type",
810 getMultiplicandAPtxTypeAttr());
811 if (getMultiplicandBPtxTypeAttr() &&
812 !llvm::is_contained(ignoreAttrNames, getMultiplicandBPtxTypeAttrName()))
813 printMmaEnumProperty(p, isFirstProperty, "multiplicand_b_ptx_type",
814 getMultiplicandBPtxTypeAttr());
815 if (getOrderedMetadata())
816 printMmaUnitProperty(p, isFirstProperty, "ordered_metadata");
817 if (getKindAttr())
818 printMmaEnumProperty(p, isFirstProperty, "kind", getKindAttr());
819 llvm::append_range(
820 ignoreAttrNames,
821 ArrayRef<StringRef>{getShapeAttrName(), getIntOverflowBehaviorAttrName(),
822 getMultiplicandAPtxTypeAttrName(),
823 getMultiplicandBPtxTypeAttrName(),
824 getOrderedMetadataAttrName(), getKindAttrName()});
825 p.printOptionalAttrDict((*this)->getDiscardableAttrDictionary().getValue(),
826 ignoreAttrNames);
827 p << " : ";
828 p << "(";
829 for (int i = 0; i < 3; ++i) {
830 p << regTypes[i];
831 if (i < 2)
832 p << ", ";
833 }
834 p << ") -> " << getResult().getType();
835}

880ParseResult MmaSpOp::parse(OpAsmParser &parser, OperationState &result) {
881 struct MMAOperandFragment {
882 std::optional<MMATypes> elemtype;
883 SmallVector<OpAsmParser::UnresolvedOperand, 4> regs;
884 SmallVector<Type> regTypes;
885 };
886
887 Builder &builder = parser.getBuilder();
888 std::array<MMAOperandFragment, 6> frags; // A, B, C, sparseMetadata, selector
889
890 NamedAttrList namedAttributes;
891
892 // A helper to parse the operand segments.
893 auto parseMmaSpOperand = [&](StringRef operandName,
894 MMAOperandFragment &frag) -> LogicalResult {
895 if (parser.parseKeyword(operandName).failed())
896 return failure();
897 if (parser
898 .parseOperandList(frag.regs, OpAsmParser::Delimiter::OptionalSquare)
899 .failed())
900 return failure();
901 return success();
902 };
903
904 // Parse the operand segments.
905 if (parseMmaSpOperand("A", frags[0]).failed())
906 return failure();
907 if (parseMmaSpOperand("B", frags[1]).failed())
908 return failure();
909 if (parseMmaSpOperand("C", frags[2]).failed())
910 return failure();
911 if (parseMmaSpOperand("sparseMetadata", frags[3]).failed())
912 return failure();
913 if (parseMmaSpOperand("selector", frags[4]).failed())
914 return failure();
915
916 if (parseMmaProperties(parser, namedAttributes,
917 {"shape", "int_overflow", "multiplicand_a_ptx_type",
918 "multiplicand_b_ptx_type", "ordered_metadata",
919 "kind"},
920 {"shape"}))
921 return failure();
922
923 // Parse the type specification and resolve operands.
924 SmallVector<Type, 3> operandTypes;
925 if (failed(parser.parseColon()))
926 return failure();
927 if (failed(parser.parseLParen()))
928 return failure();
929 if (failed(parser.parseTypeList(operandTypes)))
930 return failure();
931 if (failed(parser.parseRParen()))
932 return failure();
933 if (operandTypes.size() != 3)
934 return parser.emitError(
935 parser.getNameLoc(),
936 "expected one type for each operand segment but got " +
937 Twine(operandTypes.size()) + " types");
938 for (const auto &iter : llvm::enumerate(operandTypes)) {
939 auto &frag = frags[iter.index()];
940 frag.regTypes.resize(frag.regs.size(), iter.value());
941 if (failed(parser.resolveOperands(frag.regs, frag.regTypes,
942 parser.getNameLoc(), result.operands)))
943 return failure();
944 frag.elemtype =
945 MmaOp::inferOperandMMAType(frag.regTypes[0],
946 /*isAccumulator*/ iter.index() >= 2);
947 }
948
949 Type resultType;
950 if (parser.parseArrow() || parser.parseType(resultType))
951 return failure();
952 frags[5].elemtype =
953 MmaOp::inferOperandMMAType(resultType, /*isAccumulator*/ true);
954
955 // Resolve sparse metadata and selector (assume i32 type)
956 Type i32Type = builder.getIntegerType(32);
957 if (parser
958 .resolveOperands(frags[3].regs, i32Type, parser.getCurrentLocation(),
959 result.operands)
960 .failed())
961 return failure();
962 if (parser
963 .resolveOperands(frags[4].regs, i32Type, parser.getCurrentLocation(),
964 result.operands)
965 .failed())
966 return failure();
967
968 std::array<StringRef, 2> names{"multiplicandAPtxType",
969 "multiplicandBPtxType"};
970 for (unsigned idx = 0; idx < names.size(); idx++) {
971 const auto &frag = frags[idx];
972 std::optional<NamedAttribute> attr = namedAttributes.getNamed(names[idx]);
973 if (!frag.elemtype.has_value() && !attr.has_value()) {
974 return parser.emitError(
975 parser.getNameLoc(),
976 "attribute " + names[idx] +
977 " is not provided explicitly and cannot be inferred");
978 }
979 if (!attr.has_value())
980 result.addAttribute(
981 names[idx], MMATypesAttr::get(parser.getContext(), *frag.elemtype));
982 }
983
984 result.addTypes(resultType);
985 if (!namedAttributes.empty())
986 result.addAttributes(namedAttributes);
987 result.addAttribute(MmaSpOp::getOperandSegmentSizeAttr(),
988 builder.getDenseI32ArrayAttr({
989 static_cast<int32_t>(frags[0].regs.size()),
990 static_cast<int32_t>(frags[1].regs.size()),
991 static_cast<int32_t>(frags[2].regs.size()),
992 1, // sparseMetadata
993 1 // sparsitySelector
994 }));
995 return success();
996}

998LogicalResult MmaSpOp::verify() {
999 MLIRContext *context = getContext();
1000 auto f16Ty = Float16Type::get(context);
1001 auto i32Ty = IntegerType::get(context, 32);
1002 auto f16x2Ty = VectorType::get(2, f16Ty);
1003 auto f32Ty = Float32Type::get(context);
1004 auto f16x2x4StructTy = LLVM::LLVMStructType::getLiteral(
1005 context, {f16x2Ty, f16x2Ty, f16x2Ty, f16x2Ty});
1006
1007 auto s32x4StructTy =
1008 LLVM::LLVMStructType::getLiteral(context, {i32Ty, i32Ty, i32Ty, i32Ty});
1009 auto f32x8StructTy =
1010 LLVM::LLVMStructType::getLiteral(context, SmallVector<Type>(8, f32Ty));
1011 auto f16x2x2StructTy =
1012 LLVM::LLVMStructType::getLiteral(context, {f16x2Ty, f16x2Ty});
1013 auto f32x4StructTy =
1014 LLVM::LLVMStructType::getLiteral(context, {f32Ty, f32Ty, f32Ty, f32Ty});
1015 auto s32x2StructTy =
1016 LLVM::LLVMStructType::getLiteral(context, {i32Ty, i32Ty});
1017
1018 std::array<int64_t, 3> mmaShape{getShapeAttr().getM(), getShapeAttr().getN(),
1019 getShapeAttr().getK()};
1020
1021 // These variables define the set of allowed data types for matrices A, B, C,
1022 // and result.
1023 using AllowedShapes = SmallVector<std::array<int64_t, 3>, 2>;
1024 using AllowedTypes = SmallVector<SmallVector<Type, 4>, 2>;
1025 AllowedShapes allowedShapes;
1026 AllowedTypes expectedA;
1027 AllowedTypes expectedB;
1028 AllowedTypes expectedC;
1029 SmallVector<Type> expectedResult;
1030
1031 // When M = 16, we just need to calculate the number of 8xk tiles, where
1032 // k is a factor that depends on the data type.
1033 if (mmaShape[0] == 16) {
1034 int64_t kFactor;
1035 Type multiplicandFragType;
1036 switch (*getMultiplicandAPtxType()) {
1037 case MMATypes::tf32:
1038 kFactor = 4;
1039 multiplicandFragType = i32Ty;
1040 expectedResult.push_back(LLVM::LLVMStructType::getLiteral(
1041 context, {f32Ty, f32Ty, f32Ty, f32Ty}));
1042 // Sparse MMA supports m16n8k8 and m16n8k16 for tf32
1043 allowedShapes.push_back({16, 8, 8});
1044 allowedShapes.push_back({16, 8, 16});
1045 break;
1046 case MMATypes::bf16:
1047 kFactor = 8;
1048 multiplicandFragType = i32Ty;
1049 expectedResult.push_back(LLVM::LLVMStructType::getLiteral(
1050 context, {f32Ty, f32Ty, f32Ty, f32Ty}));
1051 // Sparse MMA supports m16n8k16 and m16n8k32 for bf16
1052 allowedShapes.push_back({16, 8, 16});
1053 allowedShapes.push_back({16, 8, 32});
1054 break;
1055 case MMATypes::f16:
1056 kFactor = 8;
1057 multiplicandFragType = f16x2Ty;
1058 expectedResult.push_back(f16x2x2StructTy);
1059 expectedResult.push_back(f32x4StructTy);
1060 // Sparse MMA supports m16n8k16 and m16n8k32 for f16
1061 allowedShapes.push_back({16, 8, 16});
1062 allowedShapes.push_back({16, 8, 32});
1063 break;
1064 case MMATypes::s4:
1065 case MMATypes::u4:
1066 kFactor = 32;
1067 // Sparse MMA supports m16n8k64 and m16n8k128 for s4/u4
1068 allowedShapes.push_back({16, 8, 64});
1069 allowedShapes.push_back({16, 8, 128});
1070 break;
1071 case MMATypes::s8:
1072 case MMATypes::u8:
1073 kFactor = 16;
1074 // Sparse MMA supports m16n8k32 and m16n8k64 for s8/u8
1075 allowedShapes.push_back({16, 8, 32});
1076 allowedShapes.push_back({16, 8, 64});
1077 break;
1078 case MMATypes::e4m3:
1079 case MMATypes::e5m2:
1080 case MMATypes::e3m2:
1081 case MMATypes::e2m3:
1082 case MMATypes::e2m1:
1083 kFactor = 16;
1084 multiplicandFragType = i32Ty;
1085 expectedResult.push_back(f16x2x2StructTy);
1086 expectedResult.push_back(f32x4StructTy);
1087 // Sparse MMA supports m16n8k64 for FP8 types
1088 allowedShapes.push_back({16, 8, 64});
1089 break;
1090 default:
1091 return emitError("invalid shape or multiplicand type: ")
1092 << getMultiplicandAPtxType().value();
1093 }
1094
1095 if (isIntegerPtxType(getMultiplicandAPtxType().value())) {
1096 expectedResult.push_back(s32x4StructTy);
1097 expectedC.emplace_back(4, i32Ty);
1098 multiplicandFragType = i32Ty;
1099 } else if (*getMultiplicandAPtxType() >= MMATypes::e4m3 &&
1100 *getMultiplicandAPtxType() <= MMATypes::e2m1) {
1101 // FP8 types
1102 expectedC.emplace_back(2, f16x2Ty);
1103 expectedC.emplace_back(4, f32Ty);
1104 } else {
1105 expectedC.emplace_back(2, f16x2Ty);
1106 expectedC.emplace_back(4, f32Ty);
1107 }
1108
1109 // For sparse MMA, A operand is compressed (2:4 sparsity means half the
1110 // elements)
1111 int64_t unitA = (mmaShape[0] / 8) * (mmaShape[2] / kFactor) / 2;
1112 int64_t unitB = (mmaShape[1] / 8) * (mmaShape[2] / kFactor);
1113 expectedA.emplace_back(unitA, multiplicandFragType);
1114 expectedB.emplace_back(unitB, multiplicandFragType);
1115
1116 if (resultPtxType() != accumPtxType())
1117 return emitOpError("ctype does not match dtype");
1118 }
1119
1120 // In the M=8 case, there is only 1 possible case per data type.
1121 if (mmaShape[0] == 8) {
1122 if (*getMultiplicandAPtxType() == MMATypes::f16) {
1123 expectedA.emplace_back(2, f16x2Ty);
1124 expectedB.emplace_back(2, f16x2Ty);
1125 expectedResult.push_back(f16x2x4StructTy);
1126 expectedResult.push_back(f32x8StructTy);
1127 expectedC.emplace_back(4, f16x2Ty);
1128 expectedC.emplace_back(8, f32Ty);
1129 allowedShapes.push_back({8, 8, 4});
1130 }
1131 if (*getMultiplicandAPtxType() == MMATypes::f64) {
1132 Type f64Ty = Float64Type::get(context);
1133 expectedA.emplace_back(1, f64Ty);
1134 expectedB.emplace_back(1, f64Ty);
1135 expectedC.emplace_back(2, f64Ty);
1136 expectedResult.emplace_back(LLVM::LLVMStructType::getLiteral(
1137 context, SmallVector<Type>(2, f64Ty)));
1138 allowedShapes.push_back({8, 8, 4});
1139 }
1140 if (isIntegerPtxType(getMultiplicandAPtxType().value())) {
1141 expectedA.push_back({i32Ty});
1142 expectedB.push_back({i32Ty});
1143 expectedC.push_back({i32Ty, i32Ty});
1144 expectedResult.push_back(s32x2StructTy);
1145 if (isInt4PtxType(getMultiplicandAPtxType().value()))
1146 allowedShapes.push_back({8, 8, 32});
1147 if (isInt8PtxType(getMultiplicandAPtxType().value()))
1148 allowedShapes.push_back({8, 8, 16});
1149 }
1150 }
1151
1152 std::string errorMessage;
1153 llvm::raw_string_ostream errorStream(errorMessage);
1154
1155 // Check that we matched an existing shape/dtype combination.
1156 if (expectedA.empty() || expectedB.empty() || expectedC.empty() ||
1157 !llvm::is_contained(allowedShapes, mmaShape)) {
1158 errorStream << "unimplemented variant for MMA shape <";
1159 llvm::interleaveComma(mmaShape, errorStream);
1160 errorStream << ">";
1161 return emitOpError(errorMessage);
1162 }
1163
1164 // Verify the operand types for segments of A, B, and C operands.
1165 std::array<StringRef, 3> operandNames{"A", "B", "C"};
1166 for (const auto &iter : llvm::enumerate(std::array<AllowedTypes, 3>{
1167 std::move(expectedA), std::move(expectedB), std::move(expectedC)})) {
1168 auto spec = this->getODSOperandIndexAndLength(iter.index());
1169 SmallVector<Type, 4> operandTySeg(operand_type_begin() + spec.first,
1170 operand_type_begin() + spec.first +
1171 spec.second);
1172 bool match = llvm::is_contained(iter.value(), operandTySeg);
1173
1174 if (!match) {
1175 errorStream << "Could not match types for the "
1176 << operandNames[iter.index()]
1177 << " operands; expected one of ";
1178 for (const auto &x : iter.value()) {
1179 errorStream << x.size() << "x" << x[0] << " ";
1180 }
1181 errorStream << "but got ";
1182 llvm::interleaveComma(operandTySeg, errorStream);
1183 return emitOpError(errorMessage);
1184 }
1185 }
1186
1187 // Check the result type
1188 if (!llvm::any_of(expectedResult, [&](Type expectedResultType) {
1189 return expectedResultType == getResult().getType();
1190 })) {
1191 errorStream
1192 << "Could not match allowed types for the result; expected one of ";
1193 llvm::interleaveComma(expectedResult, errorStream);
1194 errorStream << " but got " << getResult().getType();
1195 return emitOpError(errorMessage);
1196 }
1197
1198 // Ensure int4/int8 MMA variants specify the accum overflow behavior
1199 // attribute.
1200 if (isInt4PtxType(*getMultiplicandAPtxType()) ||
1201 isInt8PtxType(*getMultiplicandAPtxType())) {
1202 if (!getIntOverflowBehavior())
1203 return emitOpError("op requires " +
1204 getIntOverflowBehaviorAttrName().strref() +
1205 " attribute");
1206 }
1207
1208 // Validate sparse metadata type (should be i32)
1209 if (!getSparseMetadata().getType().isInteger(32)) {
1210 return emitOpError() << "sparse metadata must be i32 type";
1211 }
1212
1213 // Validate sparsity selector type (should be i32)
1214 if (!getSparsitySelector().getType().isInteger(32)) {
1215 return emitOpError() << "sparsity selector must be i32 type";
1216 }
1217
1218 return success();
1219}

1256// Helper to process operand fragments and determine which attributes can be
1257// inferred
1258template <typename Op>
1259static void
1260processOperandFragments(Op &op, std::array<MMAOperandFragment, 3> &frags,
1261 SmallVectorImpl<Type> &regTypes,
1262 SmallVectorImpl<StringRef> &ignoreAttrNames) {
1263 for (unsigned fragIdx = 0; fragIdx < frags.size(); fragIdx++) {
1264 auto &frag = frags[fragIdx];
1265 auto varOperandSpec = op.getODSOperandIndexAndLength(fragIdx);
1266 for (auto operandIdx = varOperandSpec.first;
1267 operandIdx < varOperandSpec.first + varOperandSpec.second;
1268 operandIdx++) {
1269 frag.regs.push_back(op.getOperand(operandIdx));
1270 if (fragIdx == 0 && operandIdx == varOperandSpec.first) {
1271 regTypes.push_back(op.getOperand(operandIdx).getType());
1272 }
1273 }
1274 if (fragIdx < 2) {
1275 regTypes.push_back(frag.regs[0].getType());
1276 }
1277 std::optional<MMATypes> inferredType =
1278 MmaOp::inferOperandMMAType(regTypes.back(),
1279 /*isAccumulator=*/fragIdx >= 2);
1280 if (inferredType)
1281 ignoreAttrNames.push_back(frag.ptxTypeAttr);
1282 }
1283}

1285// Helper to parse type signature: (A_type, B_type, C_type)
1286static LogicalResult
1287parseMmaTypeSignature(OpAsmParser &parser,
1288 SmallVectorImpl<Type> &operandTypes) {
1289 if (parser.parseColon().failed() || parser.parseLParen().failed())
1290 return failure();
1291
1292 auto typeParser = [&]() {
1293 Type ty;
1294 if (parser.parseType(ty).failed())
1295 return failure();
1296 operandTypes.push_back(ty);
1297 return success();
1298 };
1299 if (parser.parseCommaSeparatedList(typeParser))
1300 return failure();
1301
1302 if (operandTypes.size() != 3)
1303 return parser.emitError(parser.getCurrentLocation(),
1304 "expected exactly 3 types");
1305
1306 return parser.parseRParen();
1307}

1504 1, // scaleAData

1506 1, // threadIdA

1581LogicalResult MmaBlockScaleOp::verify() {
1582 LogicalResult result = success();
1583 int m = getShape().getM();
1584 int n = getShape().getN();
1585 int k = getShape().getK();
1586
1587 if (m == 16 && n == 8 && k == 64) {
1588 if (getMultiplicandAPtxType() != NVVM::MMATypes::e2m1 ||
1589 getMultiplicandBPtxType() != NVVM::MMATypes::e2m1)
1590 result = emitOpError(

1593 if (getScaleVecSize() != NVVM::ScaleVecSize::X2)

1597 result = emitOpError(
1598 "unsupported BlockScaleFormat attribute for mma.m16n8k64.mxf4");
1599 } else if (getKind() == NVVM::MMABlockScaleKind::MXF4NVF4) {
1600 if (!((getScaleVecSize() == NVVM::ScaleVecSize::X2 &&
1601 getBlockScaleFormat() == NVVM::BlockScaleFormat::UE8M0) ||
1602 (getScaleVecSize() == NVVM::ScaleVecSize::X4 &&
1603 (getBlockScaleFormat() == NVVM::BlockScaleFormat::UE4M3 ||
1604 getBlockScaleFormat() == NVVM::BlockScaleFormat::UE8M0))))
1605 result = emitOpError("unsupported ScaleVecSize and BlockScaleFormat "
1606 "attributes for mma.m16n8k64.mxf4nvf4");
1607 } else {
1608 result = emitOpError("unsupported Kind attribute for mma.m16n8k64");
1609 }
1610 } else if (m == 16 && n == 8 && k == 32) {
1611 if (!(getKind() == NVVM::MMABlockScaleKind::MXF8F6F4 &&
1612 getScaleVecSize() == NVVM::ScaleVecSize::X1 &&
1613 getBlockScaleFormat() == NVVM::BlockScaleFormat::UE8M0))
1614 result =
1615 emitOpError("unsupported Kind, ScaleVecSize and BlockScaleFormat "
1616 "attributes for mma.m16n8k32");
1617 } else {
1618 result = emitOpError("unsupported Geom for mma with block scaling");
1619 }
1620 return result;
