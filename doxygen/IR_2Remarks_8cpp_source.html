55// Simple helper to print key=val list (sorted).
56static void printArgs(llvm::raw_ostream &os, llvm::ArrayRef<Remark::Arg> args) {
57 if (args.empty())
58 return;
59
60 // Preserve the relative order of repeated keys, including related IDs.
61 llvm::SmallVector<Remark::Arg, 8> sorted(args.begin(), args.end());
62 llvm::stable_sort(sorted, [](const Remark::Arg &a, const Remark::Arg &b) {
63 return a.key < b.key;
64 });
65
66 for (size_t i = 0; i < sorted.size(); ++i) {
67 const auto &a = sorted[i];
68 os << a.key << "=";
69
70 llvm::StringRef val(a.val);
71 bool needsQuote = val.contains(' ') || val.contains(',') ||
72 val.contains('{') || val.contains('}');
73 if (needsQuote)
74 os << '"' << val << '"';
75 else
76 os << val;
77
78 if (i + 1 < sorted.size())
79 os << ", ";
80 }
81}

83/// Print the remark to the given output stream.
84/// Example output:
85// clang-format off
86/// [Missed] Category: Loop | Pass:Unroller | Function=main | Reason="tripCount=4 < threshold=256"
87/// [Failure] LoopOptimizer | Reason="failed due to unsupported pattern"
88// clang-format on
89void Remark::print(llvm::raw_ostream &os, bool printLocation) const {
90 // Header: [Type] pass:remarkName
91 StringRef type = getRemarkTypeString();
92 StringRef categoryName = getCombinedCategoryName();
93 StringRef name = remarkName;
94
95 os << '[' << type << "] ";
96 os << name << " | ";
97 if (!categoryName.empty())
98 os << "Category:" << categoryName << " | ";
99 if (!functionName.empty())
100 os << "Function=" << getFunction() << " | ";
101
102 if (printLocation) {
103 if (auto flc = getLocation()->findInstanceOf<FileLineColLoc>()) {
104 os << " @" << flc.getFilename() << ":" << flc.getLine() << ":"
105 << flc.getColumn();
106 }
107 }
108
109 printArgs(os, getArgs());
110}

152llvm::remarks::Remark Remark::generateRemark() const {
153 auto locLambda = [&]() -> llvm::remarks::RemarkLocation {
154 // Remarks are often emitted at name, fused or call-site locations wrapping
155 // a file location; use the first file location found inside.
156 if (auto flc = getLocation()->findInstanceOf<FileLineColLoc>())
157 return {flc.getFilename(), flc.getLine(), flc.getColumn()};
158 return {"<unknown file>", 0, 0};
159 };
160
161 llvm::remarks::Remark r; // The result.
162 r.RemarkType = getRemarkType();
163 r.RemarkName = getRemarkName();
164 // MLIR does not use passes; instead, it has categories and sub-categories.
165 r.PassName = getCombinedCategoryName();
166 r.FunctionName = getFunction();
167 r.Loc = locLambda();
168 // Keep serialized strings owned by the source remark.
169 for (const Remark::Arg &arg : getArgs()) {
170 r.Args.emplace_back();
171 r.Args.back().Key = arg.key;
172 r.Args.back().Val = arg.val;
173 }
174 return r;
175}

306/// Returns true if filter is already anchored like ^...$
307static bool isAnchored(llvm::StringRef s) {
308 s = s.trim();
309 return s.starts_with("^") && s.ends_with("$"); // note: startswith/endswith
310}

312/// Anchor the entire pattern so it matches the whole string.
313static std::string anchorWhole(llvm::StringRef filter) {
314 if (isAnchored(filter))
315 return filter.str();
316 return (llvm::Twine("^(") + filter + ")$").str();
317}

319/// Build a combined filter from cats.all and a category-specific pattern.
320/// If neither is present, return std::nullopt. Otherwise "(all|specific)"
321/// and anchor once. Also validate before returning.
322static std::optional<llvm::Regex>
323buildFilter(const mlir::remark::RemarkCategories &cats,
324 const std::optional<std::string> &specific) {
325 llvm::SmallVector<llvm::StringRef, 2> parts;
326 if (cats.all && !cats.all->empty())
327 parts.emplace_back(*cats.all);
328 if (specific && !specific->empty())
329 parts.emplace_back(*specific);
330
331 if (parts.empty())
332 return std::nullopt;
333
334 std::string joined = llvm::join(parts, "|");
335 std::string anchored = anchorWhole(joined);
336
337 llvm::Regex rx(anchored);
338 std::string err;
339 if (!rx.isValid(err))
340 return std::nullopt;
341
342 return std::make_optional<llvm::Regex>(std::move(rx));
343}

385static void dropDiscardedLinks(detail::Remark &remark,
386 const llvm::DenseSet<uint64_t> &discardedIds) {
387 if (discardedIds.empty())
388 return;
389 auto ids = llvm::to_vector(remark.getRelatedRemarkIds());
390 llvm::erase_if(
391 ids, [&](RemarkId id) { return discardedIds.contains(id.getValue()); });
392 if (ids.size() != remark.getRelatedRemarkIds().size())
393 remark.setRelatedRemarkIds(ids);
394}

396void RemarkEmittingPolicyFinal::reportRemark(const detail::Remark &remark) {
397 detail::Remark copy = remark;
398 std::scoped_lock<std::mutex> lock(mutex);
399 auto previous = pendingRemarks.find(copy);
400 if (previous != pendingRemarks.end()) {
401 if (previous->getId() && previous->getId() != copy.getId())
402 discardedIds.insert(previous->getId().getValue());
403 pendingRemarks.erase(previous);
404 }
405 pendingRemarks.insert(std::move(copy));
406}
407
408void RemarkEmittingPolicyFinal::finalize() {
409 assert(reportImpl && "reportImpl is not set");
410
411 SmallVector<detail::Remark, 0> remarks;
412 llvm::DenseSet<uint64_t> discarded;
413 {
414 std::scoped_lock<std::mutex> lock(mutex);
415 remarks = llvm::to_vector<0>(pendingRemarks);
416 pendingRemarks.clear();
417 discarded.swap(discardedIds);
418 }
419
420 // A re-reported ID that survives is still a valid target.
421 for (const auto &remark : remarks)
422 if (remark.getId())
423 discarded.erase(remark.getId().getValue());
424 for (auto &remark : remarks)
425 dropDiscardedLinks(remark, discarded);
426
427 llvm::sort(remarks, [](const auto &lhs, const auto &rhs) {
428 if (!lhs.getId())
429 return false;
430 if (!rhs.getId())
431 return true;
432 return lhs.getId().getValue() < rhs.getId().getValue();
433 });
434
435 // Build ID -> Remark* lookup for resolving related remark references.
436 llvm::DenseMap<uint64_t, const detail::Remark *> idMap;
437 llvm::DenseSet<uint64_t> childIds; // IDs referenced as children
438
439 for (const auto &remark : remarks) {
440 if (remark.getId())
441 idMap[remark.getId().getValue()] = &remark;
442 for (auto relId : remark.getRelatedRemarkIds())
443 childIds.insert(relId.getValue());
444 }
445
446 // Visit by address so ID-less remarks remain distinct.
447 llvm::DenseSet<const detail::Remark *> emitted;
448 SmallVector<const detail::Remark *> worklist;
449 auto emit = [&](const detail::Remark &root) {
450 worklist.push_back(&root);
451 while (!worklist.empty()) {
452 const auto *remark = worklist.pop_back_val();
453 if (!emitted.insert(remark).second)
454 continue;
455 reportImpl(*remark);
456 for (RemarkId id : llvm::reverse(remark->getRelatedRemarkIds()))
457 if (const auto *related = idMap.lookup(id.getValue()))
458 worklist.push_back(related);
459 }
460 };
461
462 for (const auto &remark : remarks) {
463 if (remark.getId() && childIds.contains(remark.getId().getValue()))
464 continue;
465 emit(remark);
466 }
467 // Cycles have no root.
468 for (const auto &remark : remarks)
469 if (!emitted.contains(&remark))
470 emit(remark);
471}
