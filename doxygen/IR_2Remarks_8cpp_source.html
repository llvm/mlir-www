57// Simple helper to print key=val list (sorted).
58static void printArgs(llvm::raw_ostream &os, llvm::ArrayRef<Remark::Arg> args) {
59 if (args.empty())
60 return;
61
62 // Preserve the relative order of repeated keys, including related IDs.
63 llvm::SmallVector<Remark::Arg, 8> sorted(args.begin(), args.end());
64 llvm::stable_sort(sorted, [](const Remark::Arg &a, const Remark::Arg &b) {
65 return a.key < b.key;
66 });
67
68 for (size_t i = 0; i < sorted.size(); ++i) {
69 const auto &a = sorted[i];
70 os << a.key << "=";
71
72 llvm::StringRef val(a.val);
73 bool needsQuote = val.contains(' ') || val.contains(',') ||
74 val.contains('{') || val.contains('}');
75 if (needsQuote)
76 os << '"' << val << '"';
77 else
78 os << val;
79
80 if (i + 1 < sorted.size())
81 os << ", ";
82 }
83}

85/// Print the remark to the given output stream.
86/// Example output:
87// clang-format off
88/// [Missed] Category: Loop | Pass:Unroller | Function=main | Reason="tripCount=4 < threshold=256"
89/// [Failure] LoopOptimizer | Reason="failed due to unsupported pattern"
90// clang-format on
91void Remark::print(llvm::raw_ostream &os, bool printLocation) const {
92 // Header: [Type] pass:remarkName
93 StringRef type = getRemarkTypeString();
94 StringRef categoryName = getCombinedCategoryName();
95 StringRef name = remarkName;
96
97 os << '[' << type << "] ";
98 os << name << " | ";
99 if (!categoryName.empty())
100 os << "Category:" << categoryName << " | ";
101 if (!functionName.empty())
102 os << "Function=" << getFunction() << " | ";
103
104 if (printLocation) {
105 if (auto flc = getLocation()->findInstanceOf<FileLineColLoc>()) {
106 os << " @" << flc.getFilename() << ":" << flc.getLine() << ":"
107 << flc.getColumn();
108 }
109 }
110
111 printArgs(os, getArgs());
112}

154llvm::remarks::Remark Remark::generateRemark() const {
155 auto locLambda = [&]() -> llvm::remarks::RemarkLocation {
156 // Remarks are often emitted at name, fused or call-site locations wrapping
157 // a file location; use the first file location found inside.
158 if (auto flc = getLocation()->findInstanceOf<FileLineColLoc>())
159 return {flc.getFilename(), flc.getLine(), flc.getColumn()};
160 return {"<unknown file>", 0, 0};
161 };
162
163 llvm::remarks::Remark r; // The result.
164 r.RemarkType = getRemarkType();
165 r.RemarkName = getRemarkName();
166 // MLIR does not use passes; instead, it has categories and sub-categories.
167 r.PassName = getCombinedCategoryName();
168 r.FunctionName = getFunction();
169 r.Loc = locLambda();
170 // Keep serialized strings owned by the source remark.
171 for (const Remark::Arg &arg : getArgs()) {
172 r.Args.emplace_back();
173 r.Args.back().Key = arg.key;
174 r.Args.back().Val = arg.val;
175 }
176 return r;
177}

183InFlightRemark &InFlightRemark::operator<<(const RelatedRemarkQuery &related) {
184 if (!remark || !owner)
185 return *this;
186 // Link each target once, even if an explicit link or an earlier query
187 // already added it.
188 llvm::SmallDenseSet<uint64_t> linked;
189 for (RemarkId id : remark->getRelatedRemarkIds())
190 linked.insert(id.getValue());
191 for (RemarkId id : owner->findRemarks(related.query))
192 if (linked.insert(id.getValue()).second)
193 remark->addRelatedRemark(id);
194 return *this;
195}

207struct RemarkEngine::RemarkHistory {
208 struct Entry {
209 RemarkId id;
210 RemarkKind kind;
211 Location location;
212 StringRef name, category, subCategory, function;
213
214 static Entry from(const Remark &remark) {
215 return {remark.getId(), remark.getRemarkKind(),
216 remark.getLocation(), remark.getRawRemarkName(),
217 remark.getCategoryName(), remark.getSubCategoryName(),
218 remark.getRawFunction()};
219 }
220
221 static Entry from(const Remark &remark, llvm::UniqueStringSaver &strings) {
222 Entry entry = from(remark);
223 entry.name = strings.save(entry.name);
224 entry.category = strings.save(entry.category);
225 entry.subCategory = strings.save(entry.subCategory);
226 entry.function = strings.save(entry.function);
227 return entry;
228 }
229 };

231 struct Storage {
232 llvm::BumpPtrAllocator allocator;
233 llvm::UniqueStringSaver strings{allocator};
234 std::vector<Entry> entries;
235 // IDs already recorded.
236 llvm::DenseSet<uint64_t> recordedIds;
237 };

240 // destroy it while another thread is about to lock it. Storage is opt-in.
241 std::mutex mutex;
242 std::unique_ptr<Storage> storage;
243};

264SmallVector<RemarkId>
265RemarkEngine::findRemarks(const RemarkQuery &query) const {
266 SmallVector<RemarkId> result;
267 if (!historyEnabled.load(std::memory_order_acquire))
268 return result;
269
270 auto matches = [](const auto &filter, const auto &value) {
271 return !filter || *filter == value;
272 };
273 auto matchesQuery = [&](const RemarkHistory::Entry &entry) {
274 return matches(query.location, entry.location) &&
275 matches(query.remarkKind, entry.kind) &&
276 matches(query.remarkName, entry.name) &&
277 matches(query.categoryName, entry.category) &&
278 matches(query.subCategoryName, entry.subCategory) &&
279 matches(query.functionName, entry.function);
280 };
281 // Call policies without holding the history lock.
282 if (remarkEmittingPolicy)
283 remarkEmittingPolicy->forEachPendingRemark([&](const Remark &remark) {
284 if (remark.getId() && matchesQuery(RemarkHistory::Entry::from(remark)))
285 result.push_back(remark.getId());
286 });
287
288 {
289 std::scoped_lock<std::mutex> lock(history->mutex);
290 if (!history->storage)
291 return {};
292 for (const RemarkHistory::Entry &entry : history->storage->entries)
293 if (matchesQuery(entry))
294 result.push_back(entry.id);
295 }
296 llvm::sort(result, [](RemarkId lhs, RemarkId rhs) {
297 return lhs.getValue() < rhs.getValue();
298 });
299 result.erase(llvm::unique(result), result.end());
300 return result;
301}

397void RemarkEngine::reportImpl(const Remark &remark) {
398 if (historyEnabled.load(std::memory_order_acquire)) {
399 std::scoped_lock<std::mutex> lock(history->mutex);
400 if (history->storage && remark.getId()) {
401 auto &storage = *history->storage;
402 // A policy may deliver the same event repeatedly.
403 if (storage.recordedIds.insert(remark.getId().getValue()).second)
404 storage.entries.push_back(
405 RemarkHistory::Entry::from(remark, storage.strings));
406 }
407 }
408 // Stream the remark
409 if (remarkStreamer) {
410 remarkStreamer->streamOptimizationRemark(remark);
411 }
412
413 // Print using MLIR's diagnostic
414 if (printAsEmitRemarks)
415 emitRemark(remark.getLocation(), remark.getMsg());
416}

444/// Returns true if filter is already anchored like ^...$
445static bool isAnchored(llvm::StringRef s) {
446 s = s.trim();
447 return s.starts_with("^") && s.ends_with("$"); // note: startswith/endswith
448}
449
450/// Anchor the entire pattern so it matches the whole string.
451static std::string anchorWhole(llvm::StringRef filter) {
452 if (isAnchored(filter))
453 return filter.str();
454 return (llvm::Twine("^(") + filter + ")$").str();
455}
456
457/// Build a combined filter from cats.all and a category-specific pattern.
458/// If neither is present, return std::nullopt. Otherwise "(all|specific)"
459/// and anchor once. Also validate before returning.
460static std::optional<llvm::Regex>
461buildFilter(const mlir::remark::RemarkCategories &cats,
462 const std::optional<std::string> &specific) {
463 llvm::SmallVector<llvm::StringRef, 2> parts;
464 if (cats.all && !cats.all->empty())
465 parts.emplace_back(*cats.all);
466 if (specific && !specific->empty())
467 parts.emplace_back(*specific);
468
469 if (parts.empty())
470 return std::nullopt;
471
472 std::string joined = llvm::join(parts, "|");
473 std::string anchored = anchorWhole(joined);
474
475 llvm::Regex rx(anchored);
476 std::string err;
477 if (!rx.isValid(err))
478 return std::nullopt;
479
480 return std::make_optional<llvm::Regex>(std::move(rx));
481}

524static void dropDiscardedLinks(detail::Remark &remark,
525 const llvm::DenseSet<uint64_t> &discardedIds) {
526 if (discardedIds.empty())
527 return;
528 auto ids = llvm::to_vector(remark.getRelatedRemarkIds());
529 llvm::erase_if(
530 ids, [&](RemarkId id) { return discardedIds.contains(id.getValue()); });
531 if (ids.size() != remark.getRelatedRemarkIds().size())
532 remark.setRelatedRemarkIds(ids);
533}

535void RemarkEmittingPolicyFinal::reportRemark(const detail::Remark &remark) {
536 detail::Remark copy = remark;
537 std::scoped_lock<std::mutex> lock(mutex);
538 auto previous = pendingRemarks.find(copy);
539 if (previous != pendingRemarks.end()) {
540 if (previous->getId() && previous->getId() != copy.getId())
541 discardedIds.insert(previous->getId().getValue());
542 pendingRemarks.erase(previous);
543 }
544 pendingRemarks.insert(std::move(copy));
545}

547void RemarkEmittingPolicyFinal::forEachPendingRemark(
548 llvm::function_ref<void(const detail::Remark &)> callback) const {
549 std::scoped_lock<std::mutex> lock(mutex);
550 for (const auto &remark : pendingRemarks)
551 callback(remark);
552}

554void RemarkEmittingPolicyFinal::finalize() {
555 assert(reportImpl && "reportImpl is not set");
556
557 SmallVector<detail::Remark, 0> remarks;
558 llvm::DenseSet<uint64_t> discarded;
559 {
560 std::scoped_lock<std::mutex> lock(mutex);
561 remarks = llvm::to_vector<0>(pendingRemarks);
562 pendingRemarks.clear();
563 discarded.swap(discardedIds);
564 }
565
566 // A re-reported ID that survives is still a valid target.
567 for (const auto &remark : remarks)
568 if (remark.getId())
569 discarded.erase(remark.getId().getValue());
570 for (auto &remark : remarks)
571 dropDiscardedLinks(remark, discarded);
572
573 llvm::sort(remarks, [](const auto &lhs, const auto &rhs) {
574 if (!lhs.getId())
575 return false;
576 if (!rhs.getId())
577 return true;
578 return lhs.getId().getValue() < rhs.getId().getValue();
579 });
580
581 // Build ID -> Remark* lookup for resolving related remark references.
582 llvm::DenseMap<uint64_t, const detail::Remark *> idMap;
583 llvm::DenseSet<uint64_t> childIds; // IDs referenced as children
584
585 for (const auto &remark : remarks) {
586 if (remark.getId())
587 idMap[remark.getId().getValue()] = &remark;
588 for (auto relId : remark.getRelatedRemarkIds())
589 childIds.insert(relId.getValue());
590 }
591
592 // Visit by address so ID-less remarks remain distinct.
593 llvm::DenseSet<const detail::Remark *> emitted;
594 SmallVector<const detail::Remark *> worklist;
595 auto emit = [&](const detail::Remark &root) {
596 worklist.push_back(&root);
597 while (!worklist.empty()) {
598 const auto *remark = worklist.pop_back_val();
599 if (!emitted.insert(remark).second)
600 continue;
601 reportImpl(*remark);
602 for (RemarkId id : llvm::reverse(remark->getRelatedRemarkIds()))
603 if (const auto *related = idMap.lookup(id.getValue()))
604 worklist.push_back(related);
605 }
606 };
607
608 for (const auto &remark : remarks) {
609 if (remark.getId() && childIds.contains(remark.getId().getValue()))
610 continue;
611 emit(remark);
612 }
613 // Cycles have no root.
614 for (const auto &remark : remarks)
615 if (!emitted.contains(&remark))
616 emit(remark);
617}
