35static Value getPointer(Location loc, Value value,
36 ConversionPatternRewriter &rewriter) {
37 Type ptrTy = LLVM::LLVMPointerType::get(rewriter.getContext());
38 if (!value)
39 return LLVM::ZeroOp::create(rewriter, loc, ptrTy);
40 if (value.getType() == ptrTy)
41 return value;
42
43 if (isa<PointerLikeType>(value.getType()))
44 return castPointerLikeTypeIfNeeded(rewriter, loc, value, ptrTy);
45
46 Value one = LLVM::ConstantOp::create(rewriter, loc, rewriter.getI32Type(), 1);
47 Value storage =
48 LLVM::AllocaOp::create(rewriter, loc, ptrTy, value.getType(), one);
49 LLVM::StoreOp::create(rewriter, loc, value, storage);
50 return storage;
51}

53/// Returns the address of a slot holding \p value, so that the runtime reads
54/// the value from memory rather than receiving it directly.
55static Value getIndirectPointer(Location loc, Value value,
56 ConversionPatternRewriter &rewriter) {
57 Type ptrTy = LLVM::LLVMPointerType::get(rewriter.getContext());
58 Value one = LLVM::ConstantOp::create(rewriter, loc, rewriter.getI32Type(), 1);
59 Value storage =
60 LLVM::AllocaOp::create(rewriter, loc, ptrTy, value.getType(), one);
61 LLVM::StoreOp::create(rewriter, loc, value, storage);
62 return storage;
63}

65Value mlir::createACCDataArray(Location loc, Type elementType,
66 ArrayRef<Value> values, RewriterBase &rewriter) {
67 Type ptrTy = LLVM::LLVMPointerType::get(rewriter.getContext());
68 Type i32Ty = rewriter.getI32Type();
69 Value count = LLVM::ConstantOp::create(rewriter, loc, i32Ty, values.size());
70 Value array =
71 LLVM::AllocaOp::create(rewriter, loc, ptrTy, elementType, count);
72 for (auto [index, value] : llvm::enumerate(values)) {
73 Value indexValue = LLVM::ConstantOp::create(rewriter, loc, i32Ty, index);
74 Value element = LLVM::GEPOp::create(rewriter, loc, ptrTy, elementType,
75 array, ValueRange{indexValue});
76 LLVM::StoreOp::create(rewriter, loc, value, element);
77 }
78 return array;
79}

81void mlir::createACCMappedObjectArrays(Location loc,
82 ArrayRef<ACCMappedObjectArgs> objects,
83 ACCDataRuntimeArgs &runtimeArgs,
84 RewriterBase &rewriter) {
85 Type i32Ty = rewriter.getI32Type();
86 Type i64Ty = rewriter.getI64Type();
87 Type ptrTy = LLVM::LLVMPointerType::get(rewriter.getContext());
88 auto createArray = [&](Type elementType, Value ACCMappedObjectArgs::*field) {
89 SmallVector<Value> values =
90 llvm::map_to_vector(objects, [&](const ACCMappedObjectArgs &object) {
91 return object.*field;
92 });
93 return createACCDataArray(loc, elementType, values, rewriter);
94 };
95 // The data-call emitter states the count before the object values.
96 if (!runtimeArgs.argNum)
97 runtimeArgs.argNum =
98 LLVM::ConstantOp::create(rewriter, loc, i32Ty, objects.size());
99 runtimeArgs.argBasePtrs = createArray(ptrTy, &ACCMappedObjectArgs::base);
100 runtimeArgs.argPtrs = createArray(ptrTy, &ACCMappedObjectArgs::pointer);
101 runtimeArgs.argSizes = createArray(i64Ty, &ACCMappedObjectArgs::size);
102 runtimeArgs.argTypes = createArray(i64Ty, &ACCMappedObjectArgs::type);
103 runtimeArgs.argNames = createArray(ptrTy, &ACCMappedObjectArgs::name);
104 if (!runtimeArgs.argMappers)
105 runtimeArgs.argMappers = LLVM::ZeroOp::create(rewriter, loc, ptrTy);
106 runtimeArgs.argDescs = createArray(ptrTy, &ACCMappedObjectArgs::descriptor);
107}

109/// Privatized storage only ever exists on the device, so there is no host
110/// address to give. A device address names storage the runtime must not look
111/// up, and an object mapped by value has no address at all, so both are handed
112/// over in a slot the runtime reads them from. A pointer that no clause maps
113/// but the device can already access is a device address as well.
114Value mlir::createACCMappedObjectPointer(Location loc, Value operand,
115 Value converted, MapFlags mapFlags,
116 ConversionPatternRewriter &rewriter) {
117 Type ptrTy = LLVM::LLVMPointerType::get(rewriter.getContext());
118 if (isa<PrivateType>(operand.getType()))
119 return LLVM::ZeroOp::create(rewriter, loc, ptrTy);
120 if (converted && converted.getType() == ptrTy &&
121 (bitEnumContainsAny(mapFlags, MapFlags::devptr) ||
122 (!isDataEntryOp(operand.getDefiningOp()) &&
123 isDeviceAccessibleValue(operand))))
124 return getIndirectPointer(loc, converted, rewriter);
125 return getPointer(loc, converted, rewriter);
126}

128Value mlir::createACCMappedObjectBase(Location loc, Operation *mapOp,
129 MapFlags mapFlags,
130 ConversionPatternRewriter &rewriter) {
131 Value base;
132 if (mapOp) {
133 if (Value attach = acc::getVarPtrPtr(mapOp))
134 base = remapOrSelf(attach, rewriter);
135 else if (bitEnumContainsAny(mapFlags, MapFlags::ptr_and_obj))
136 base = remapOrSelf(acc::getDesc(mapOp), rewriter);
137 }
138 return getPointer(loc, base, rewriter);
139}

141Value mlir::createACCMappedObjectName(Location loc, Value operand,
142 OpenACCSupport &accSupport,
143 Region &globalSymbolRegion,
144 SymbolTable *symbolTable,
145 ConversionPatternRewriter &rewriter) {
146 // The runtime resolves an object that lives on the device against the
147 // symbols of the binary by this name, so it is the name the object is
148 // emitted under rather than the one the source spells.
149 VariableNameConfig nameConfig;
150 nameConfig.preferDemangledName = false;
151 std::string name = accSupport.getVariableName(operand, nameConfig);
152 if (name.empty())
153 return LLVM::ZeroOp::create(
154 rewriter, loc, LLVM::LLVMPointerType::get(rewriter.getContext()));
155 return getOrCreateGlobalString(loc, rewriter,
156 getInternalGlobalName("var_name", name), name,
157 globalSymbolRegion, symbolTable);
158}

160static std::optional<MapFlags> computePackedMapFlags(Operation *mapOp) {
161 if (auto mapInfo = dyn_cast<MapInfoOp>(mapOp))
162 return mapInfo.getMapFlags();
163 if (isa<ACC_DATA_ENTRY_OPS>(mapOp))
164 return computeDataClauseMapFlags(mapOp, hasAttachPoint(mapOp));
165 return std::nullopt;

168FailureOr<Value>
169mlir::createACCMappedObjectSize(Location loc, Value operand, Value converted,
170 MapFlags mapFlags, OpenACCSupport &accSupport,
171 ConversionPatternRewriter &rewriter) {
172 Operation *scope = rewriter.getInsertionBlock()->getParentOp();
173 auto module = dyn_cast<ModuleOp>(scope);
174 if (!module)
175 module = scope->getParentOfType<ModuleOp>();
176 // An object mapped by value is passed as the argument itself, so its size is
177 // the size of what is passed and not of the object it was read from. The two
178 // differ when the argument holds one field of a decomposed aggregate.
179 if (bitEnumContainsAny(mapFlags, MapFlags::literal) && converted) {
180 auto sizeAndAlignment =
181 acc::getTypeSizeAndAlignment(converted.getType(), module, &accSupport);
182 if (!sizeAndAlignment) {
183 std::string message;
184 llvm::raw_string_ostream os(message);
185 os << "unhandled type size computation for type: " << converted.getType();
186 (void)accSupport.emitNYI(loc, message);
187 return failure();
188 }
189 return constantI64(loc, sizeAndAlignment->first.getFixedValue(), rewriter);
190 }
191 Operation *mapOp = operand.getDefiningOp();
192 if (!isDataEntryOp(mapOp))
193 return constantI64(loc, 0, rewriter);
194 if (Value size = acc::getMapSize(mapOp))
195 return castToI64(loc, remapOrSelf(size, rewriter), rewriter);
196
197 Value var = acc::getVar(mapOp);
198 if (!var)
199 return constantI64(loc, 0, rewriter);
200 Type varType = acc::getVarType(mapOp);
201 if (!varType)
202 varType = var.getType();
203 // A memref is handed over with a descriptor stating its extents and element
204 // size, which is what states the size of such a mapping.
205 if (isa<MemRefType>(var.getType()) || isa<MemRefType>(varType))
206 return constantI64(loc, 0, rewriter);
207
208 // A data clause operation that was not turned into an `acc.map_info` has no
209 // size of its own, so it is sized the way that operation would have been.
210 // A size that is not statically known is left to the runtime, as an unsized
211 // mapping is.
212 int64_t size =
213 acc::computeMapInfoSizeBytes(var, varType, acc::getDataDescKind(mapOp),
214 acc::getBounds(mapOp), &accSupport);
215 return constantI64(loc, std::max<int64_t>(size, 0), rewriter);
216}

218static Value getElementSize(Operation *mapOp, OpenACCSupport &accSupport,
219 ConversionPatternRewriter &rewriter) {
220 Location loc = mapOp->getLoc();
221 ModuleOp module = mapOp->getParentOfType<ModuleOp>();
222 if (std::optional<int64_t> size = acc::getMapElementSize(mapOp))
223 return constantI64(loc, *size, rewriter);
224
225 Type type = acc::getVarType(mapOp);
226 if (auto shaped = dyn_cast<ShapedType>(type))
227 type = shaped.getElementType();
228 if (!type)
229 return constantI64(loc, 0, rewriter);
230 if (auto sizeAndAlignment =
231 acc::getTypeSizeAndAlignment(type, module, &accSupport))
232 return constantI64(loc, sizeAndAlignment->first.getFixedValue(), rewriter);
233 return constantI64(loc, 0, rewriter);
234}

243Value mlir::createACCDataDescriptor(Location loc, Value baseDescriptor,
244 Type baseDescriptorType, uint32_t version,
245 ValueRange bounds, Value elementSize,
246 ConversionPatternRewriter &rewriter) {
247 MLIRContext *context = rewriter.getContext();
248 Type i8Ty = rewriter.getI8Type();
249 Type i32Ty = rewriter.getI32Type();
250 Type i64Ty = rewriter.getI64Type();
251 Type ptrTy = LLVM::LLVMPointerType::get(context);
252
253 Value versionValue = LLVM::ConstantOp::create(
254 rewriter, loc, i32Ty,
255 version | static_cast<uint32_t>(
256 getDataDescriptorKind(DataDescriptor::AccDataDescOpenACC)));
257 baseDescriptor = LLVM::InsertValueOp::create(
258 rewriter, loc, baseDescriptorType, baseDescriptor, versionValue,
259 ArrayRef<int64_t>{0});
260
261 Type descriptorType = getDataDescriptorType(
262 context, DataDescriptor::AccDataDescOpenACC, baseDescriptorType);
263 Value descriptor = LLVM::ZeroOp::create(rewriter, loc, descriptorType);
264 Value rank = LLVM::ConstantOp::create(rewriter, loc, i8Ty, bounds.size());
265
266 SmallVector<Value> lowerBounds;
267 SmallVector<Value> upperBounds;
268 SmallVector<Value> extents;
269 SmallVector<Value> strides;
270 SmallVector<Value> starts;
271 for (Value boundValue : bounds) {
272 auto bound = cast<DataBoundsOp>(boundValue.getDefiningOp());
273 lowerBounds.push_back(
274 getBoundValue(bound.getLowerbound(), bound.getLoc(), rewriter));
275 upperBounds.push_back(
276 getBoundValue(bound.getUpperbound(), bound.getLoc(), rewriter));
277 Value sourceExtent =
278 bound.getSourceExtent() ? bound.getSourceExtent() : bound.getExtent();
279 extents.push_back(getBoundValue(sourceExtent, bound.getLoc(), rewriter));
280 Value stride = getBoundValue(bound.getStride(), bound.getLoc(), rewriter);
281 if (!bound.getStrideInBytes())
282 stride =
283 LLVM::MulOp::create(rewriter, bound.getLoc(), stride, elementSize);
284 strides.push_back(stride);
285 starts.push_back(
286 getBoundValue(bound.getStartIdx(), bound.getLoc(), rewriter));
287 }
288
289 Value lowerBoundsArray =
290 createACCDataArray(loc, i64Ty, lowerBounds, rewriter);
291 Value upperBoundsArray =
292 createACCDataArray(loc, i64Ty, upperBounds, rewriter);
293 Value extentsArray = createACCDataArray(loc, i64Ty, extents, rewriter);
294 Value stridesArray = createACCDataArray(loc, i64Ty, strides, rewriter);
295 Value startsArray = createACCDataArray(loc, i64Ty, starts, rewriter);
296
297 using Field = AccDataDescOpenACCField;
298 auto insert = [&](Value value, Field field) {
299 descriptor = LLVM::InsertValueOp::create(
300 rewriter, loc, descriptorType, descriptor, value,
301 ArrayRef<int64_t>{getDataDescriptorFieldIndex(field)});
302 };
303 insert(baseDescriptor, Field::Base);
304 insert(rank, Field::Rank);
305 insert(elementSize, Field::ElementSize);
306 insert(lowerBoundsArray, Field::LowerBounds);
307 insert(upperBoundsArray, Field::UpperBounds);
308 insert(extentsArray, Field::Extents);
309 insert(stridesArray, Field::StridesInBytes);
310 insert(startsArray, Field::StartIndices);
311
312 Value one = LLVM::ConstantOp::create(rewriter, loc, i32Ty, 1);
313 Value storage =
314 LLVM::AllocaOp::create(rewriter, loc, ptrTy, descriptorType, one);
315 LLVM::StoreOp::create(rewriter, loc, descriptor, storage);
316 return storage;
317}

319Value mlir::createACCMemRefDescriptorWrapperArg(
320 Location loc, MemRefType memrefType, Value convertedMemref,
321 ValueRange bounds, Value elementSize, ConversionPatternRewriter &rewriter) {
322 MLIRContext *context = rewriter.getContext();
323 Type i8Ty = rewriter.getI8Type();
324 Type i32Ty = rewriter.getI32Type();
325 Type ptrTy = LLVM::LLVMPointerType::get(context);
326 using Field = AccDataDescMemRefField;
327 Type descriptorType =
328 getDataDescriptorType(context, DataDescriptor::AccDataDescMemRef);
329 auto version = static_cast<uint32_t>(
330 getDataDescriptorKind(DataDescriptor::AccDataDescMemRef));
331 Value descriptor = LLVM::ZeroOp::create(rewriter, loc, descriptorType);
332 Value rank =
333 LLVM::ConstantOp::create(rewriter, loc, i8Ty, memrefType.getRank());
334 auto insert = [&](Value value, Field field) {
335 descriptor = LLVM::InsertValueOp::create(
336 rewriter, loc, descriptorType, descriptor, value,
337 ArrayRef<int64_t>{getDataDescriptorFieldIndex(field)});
338 };
339 insert(rank, Field::Rank);
340 insert(elementSize, Field::ElementSize);
341 insert(getPointer(loc, convertedMemref, rewriter), Field::MemRefDescriptor);
342 if (!bounds.empty())
343 return createACCDataDescriptor(loc, descriptor, descriptorType, version,
344 bounds, elementSize, rewriter);
345
346 insert(LLVM::ConstantOp::create(rewriter, loc, i32Ty, version),
347 Field::Version);
348 Value one = LLVM::ConstantOp::create(rewriter, loc, i32Ty, 1);
349 Value storage =
350 LLVM::AllocaOp::create(rewriter, loc, ptrTy, descriptorType, one);
351 LLVM::StoreOp::create(rewriter, loc, descriptor, storage);
352 return storage;
353}

355Value mlir::createACCArgumentDescriptor(Operation *mapOp,
356 Value convertedOperand,
357 OpenACCSupport &accSupport,
358 ConversionPatternRewriter &rewriter) {
359 Location loc = mapOp->getLoc();
360 MLIRContext *context = rewriter.getContext();
361 Type i32Ty = rewriter.getI32Type();
362 Type ptrTy = LLVM::LLVMPointerType::get(context);
363 SmallVector<Value> bounds = acc::getBounds(mapOp);
364 DataDescKind descKind = acc::getDataDescKind(mapOp);
365 // Only a descriptor stating bounds reads the element size, so it is
366 // materialized where it is used, leaving no constant behind for a mapping
367 // that needs no descriptor at all.
368 auto elementSize = [&] {
369 return getElementSize(mapOp, accSupport, rewriter);
370 };
371
372 Value var = acc::getVar(mapOp);
373 if (var && isa<MemRefType>(var.getType())) {
374 auto memrefType = cast<MemRefType>(var.getType());
375 return createACCMemRefDescriptorWrapperArg(
376 loc, memrefType, convertedOperand, bounds, elementSize(), rewriter);
377 }
378
379 if (bitEnumContainsAny(descKind, DataDescKind::cfi)) {
380 using Field = AccDataDescCFIField;
381 Type descriptorType =
382 getDataDescriptorType(context, DataDescriptor::AccDataDescCFI);
383 auto version = static_cast<uint32_t>(
384 getDataDescriptorKind(DataDescriptor::AccDataDescCFI));
385 Value descriptor = LLVM::ZeroOp::create(rewriter, loc, descriptorType);
386 Value descriptorStorage = remapOrSelf(acc::getDesc(mapOp), rewriter);
387 if (!descriptorStorage)
388 descriptorStorage = convertedOperand;
389 descriptor = LLVM::InsertValueOp::create(
390 rewriter, loc, descriptorType, descriptor,
391 getPointer(loc, descriptorStorage, rewriter),
392 ArrayRef<int64_t>{getDataDescriptorFieldIndex(Field::CFIDescriptor)});
393 if (!bounds.empty())
394 return createACCDataDescriptor(loc, descriptor, descriptorType, version,
395 bounds, elementSize(), rewriter);
396
397 Value versionValue =
398 LLVM::ConstantOp::create(rewriter, loc, i32Ty, version);
399 descriptor = LLVM::InsertValueOp::create(
400 rewriter, loc, descriptorType, descriptor, versionValue,
401 ArrayRef<int64_t>{getDataDescriptorFieldIndex(Field::Version)});
402 Value one = LLVM::ConstantOp::create(rewriter, loc, i32Ty, 1);
403 Value storage =
404 LLVM::AllocaOp::create(rewriter, loc, ptrTy, descriptorType, one);
405 LLVM::StoreOp::create(rewriter, loc, descriptor, storage);
406 return storage;
407 }
408
409 if (!bounds.empty()) {
410 Type descriptorType =
411 getDataDescriptorType(context, DataDescriptor::AccDataDescGeneric);
412 auto version = static_cast<uint32_t>(
413 getDataDescriptorKind(DataDescriptor::AccDataDescGeneric));
414 Value descriptor = LLVM::ZeroOp::create(rewriter, loc, descriptorType);
415 return createACCDataDescriptor(loc, descriptor, descriptorType, version,
416 bounds, elementSize(), rewriter);
417 }
418 return LLVM::ZeroOp::create(rewriter, loc, ptrTy);
419}

426LogicalResult mlir::emitACCDataRuntimeArgs(
427 Location loc, ValueRange mappingOperands, ValueRange convertedOperands,
428 ConversionPatternRewriter &rewriter, Region &globalSymbolRegion,
429 acc::OpenACCSupport &accSupport, const acc::ACCRuntimeCallConfig &config,
430 ACCDataRuntimeArgs &runtimeArgs, const ACCDataCallOptions &options,
431 SymbolTable *symbolTable) {
432 if (mappingOperands.size() != convertedOperands.size())
433 return failure();
434
435 runtimeArgs.ident =
436 createIdent(loc, getParentFunctionName(mappingOperands), rewriter,
437 globalSymbolRegion, config, symbolTable);
438 runtimeArgs.flags = constantI64(loc, options.flags, rewriter);
439 runtimeArgs.deviceType = constantI64(
440 loc, config.getDeviceTypeRuntimeValue(DeviceType::None), rewriter);
441 runtimeArgs.argMappers = LLVM::ZeroOp::create(
442 rewriter, loc, LLVM::LLVMPointerType::get(rewriter.getContext()));
443
444 // An aggregate passed by value is decomposed into one argument per field by
445 // the target ABI, so the runtime is told about one object per field as well.
446 // Each of them keeps the map entry of the aggregate it came from. The device
447 // address of an aggregate is returned in the slot of its first field.
448 struct MappedObject {
449 Value operand;
450 Value converted;
451 bool firstField;
452 };
453 SmallVector<MappedObject> mappedObjects;
454 for (auto [operand, converted] :
455 llvm::zip_equal(mappingOperands, convertedOperands)) {
456 auto structType = dyn_cast<LLVM::LLVMStructType>(converted.getType());
457 if (structType && !structType.isIdentified() &&
458 structType.getBody().size() > 1) {
459 for (unsigned field = 0, fields = structType.getBody().size();
460 field != fields; ++field)
461 mappedObjects.push_back(
462 {operand,
463 LLVM::ExtractValueOp::create(rewriter, loc, converted, field),
464 field == 0});
465 continue;
466 }
467 mappedObjects.push_back({operand, converted, true});
468 }
469
470 // The count precedes the values of the objects, which is where a decomposed
471 // aggregate states how many fields it became.
472 runtimeArgs.argNum = LLVM::ConstantOp::create(
473 rewriter, loc, rewriter.getI32Type(), mappedObjects.size());
474
475 constexpr unsigned noSlot = ~0u;
476 runtimeArgs.returnAddressSlots.assign(options.returnAddressOperands.size(),
477 noSlot);
478 SmallVector<ACCMappedObjectArgs> objects;
479 for (auto [slot, object] : llvm::enumerate(mappedObjects)) {
480 auto [operand, converted, firstField] = object;
481 Operation *mapOp = operand.getDefiningOp();
482 if (!isDataEntryOp(mapOp))
483 return failure();
484 std::optional<MapFlags> mapFlags = computePackedMapFlags(mapOp);
485 if (!mapFlags)
486 return failure();
487 *mapFlags = config.postProcessMapFlags(mapOp, *mapFlags);
488 LLVM_DEBUG(llvm::dbgs() << "mapping " << *mapOp << "\n as "
489 << config.formatMapFlags(*mapFlags) << "\n");
490
491 Location mapLoc = mapOp->getLoc();
492 ACCMappedObjectArgs args;
493 args.pointer = createACCMappedObjectPointer(mapLoc, operand, converted,
494 *mapFlags, rewriter);
495 // An object that only lives on the device has no host address to state as
496 // its base.
497 args.base =
498 bitEnumContainsAny(*mapFlags, MapFlags::device_resident)
499 ? getPointer(mapLoc, Value(), rewriter)
500 : createACCMappedObjectBase(mapLoc, mapOp, *mapFlags, rewriter);
501
502 // A local object that lives on the device is torn down when the region
503 // that declared it ends, which the runtime only does for a mapping that
504 // states it is deleted.
505 if (options.callKind == ACCDataCallKind::DataExit &&
506 bitEnumContainsAny(*mapFlags, MapFlags::device_resident)) {
507 Value var = acc::getVar(mapOp);
508 if (var &&
509 !isa_and_nonnull<AddressOfGlobalOpInterface>(var.getDefiningOp()))
510 *mapFlags = *mapFlags | MapFlags::delete_;
511 }
512
513 if (firstField) {
514 for (auto [index, returned] :
515 llvm::enumerate(options.returnAddressOperands)) {
516 if (returned != operand)
517 continue;
518 *mapFlags = *mapFlags | MapFlags::return_param;
519 runtimeArgs.returnAddressSlots[index] = slot;
520 }
521 }
522 int64_t mapType = config.getMapFlagsRuntimeValue(*mapFlags);
523
524 FailureOr<Value> size = createACCMappedObjectSize(
525 mapLoc, operand, converted, *mapFlags, accSupport, rewriter);
526 if (failed(size))
527 return failure();
528 args.size = *size;
529 args.type = constantI64(mapLoc, mapType, rewriter);
530 args.name = createACCMappedObjectName(
531 mapLoc, operand, accSupport, globalSymbolRegion, symbolTable, rewriter);
532 args.descriptor =
533 createACCArgumentDescriptor(mapOp, converted, accSupport, rewriter);
534 objects.push_back(args);
535 }
536 if (llvm::is_contained(runtimeArgs.returnAddressSlots, noSlot))
537 return failure();
538
539 createACCMappedObjectArrays(loc, objects, runtimeArgs, rewriter);
540 return success();
541}

543FailureOr<SmallVector<Value>> mlir::emitACCDataRuntimeCall(
544 Location loc, RuntimeFunction fn, ValueRange mappingOperands,
545 ValueRange convertedOperands, ConversionPatternRewriter &rewriter,
546 OpenACCSupport &accSupport, Region &globalSymbolRegion,
547 SymbolTable &symbolTable, const ACCRuntimeCallConfig &config,
548 ArrayRef<Value> trailingArguments, const ACCDataCallOptions &options) {
549 SmallVector<Value> returnedAddresses;
550 if (mappingOperands.empty())
551 return returnedAddresses;
552
553 ACCDataRuntimeArgs runtimeArgs;
554 if (failed(emitACCDataRuntimeArgs(
555 loc, mappingOperands, convertedOperands, rewriter, globalSymbolRegion,
556 accSupport, config, runtimeArgs, options, &symbolTable)))
557 return rewriter.notifyMatchFailure(
558 mappingOperands.front().getDefiningOp(),
559 "unsupported OpenACC data-clause operand");
560
561 SmallVector<Value> arguments = runtimeArgs.getCallArgs();
562 arguments.append(trailingArguments.begin(), trailingArguments.end());
563 if (failed(createRuntimeCall(loc, rewriter, globalSymbolRegion, symbolTable,
564 fn, config, arguments)))
565 return failure();
566
567 Type i32Ty = rewriter.getI32Type();
568 Type ptrTy = LLVM::LLVMPointerType::get(rewriter.getContext());
569 for (unsigned slot : runtimeArgs.returnAddressSlots) {
570 Value index = LLVM::ConstantOp::create(rewriter, loc, i32Ty, slot);
571 Value element =
572 LLVM::GEPOp::create(rewriter, loc, ptrTy, ptrTy,
573 runtimeArgs.argBasePtrs, ValueRange{index});
574 returnedAddresses.push_back(
575 LLVM::LoadOp::create(rewriter, loc, ptrTy, element));
576 }
577 return returnedAddresses;
