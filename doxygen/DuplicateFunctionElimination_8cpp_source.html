20// Define a notion of function equivalence that allows for reuse. Ignore the
21// symbol name for this purpose.
22struct DuplicateFuncOpEquivalenceInfo
23 : public llvm::DenseMapInfo<func::FuncOp> {
24
25 static unsigned getHashValue(const func::FuncOp cFunc) {
26 if (!cFunc) {
27 return DenseMapInfo<func::FuncOp>::getHashValue(cFunc);
28 }
29
30 // Aggregate attributes, ignoring the symbol name.
31 llvm::hash_code hash = {};
32 func::FuncOp func = const_cast<func::FuncOp &>(cFunc);
33 StringAttr symNameAttrName = func.getSymNameAttrName();
34 for (NamedAttribute namedAttr :
35 cFunc->getDiscardableAttrDictionary().getValue()) {
36 StringAttr attrName = namedAttr.getName();
37 if (attrName == symNameAttrName)
38 continue;
39 hash = llvm::hash_combine(hash, namedAttr);
40 }
41
42 // Also hash the func body.
43 func.getBody().walk([&](Operation *op) {
44 hash = llvm::hash_combine(
45 hash, OperationEquivalence::computeHash(
46 op, /*hashOperands=*/OperationEquivalence::ignoreHashValue,
47 /*hashResults=*/OperationEquivalence::ignoreHashValue,
48 OperationEquivalence::IgnoreLocations));
49 });
50
51 return hash;
52 }
53
54 static bool isEqual(func::FuncOp lhs, func::FuncOp rhs) {
55 if (lhs == rhs)
56 return true;
57 if (lhs.isDeclaration() || rhs.isDeclaration())
58 return false;
59
60 // Check discardable attributes equivalence
61 if (lhs->getDiscardableAttrDictionary() !=
62 rhs->getDiscardableAttrDictionary())
63 return false;
64
65 // Check properties equivalence, ignoring the symbol name.
66 // Make a copy, so that we can erase the symbol name and perform the
67 // comparison.
68 auto pLhs = lhs.getProperties();
69 auto pRhs = rhs.getProperties();
70 pLhs.sym_name = nullptr;
71 pRhs.sym_name = nullptr;
72 if (pLhs != pRhs)
73 return false;
74
75 // Compare inner workings.
76 return OperationEquivalence::isRegionEquivalentTo(
77 &lhs.getBody(), &rhs.getBody(), OperationEquivalence::IgnoreLocations);
78 }
79};

81struct DuplicateFunctionEliminationPass
82 : public func::impl::DuplicateFunctionEliminationPassBase<
83 DuplicateFunctionEliminationPass> {
84
85 using DuplicateFunctionEliminationPassBase<
86 DuplicateFunctionEliminationPass>::DuplicateFunctionEliminationPassBase;
87
88 void runOnOperation() override {
89 auto module = getOperation();
90
91 // Find unique representant per equivalent func ops.
92 DenseSet<func::FuncOp, DuplicateFuncOpEquivalenceInfo> uniqueFuncOps;
93 DenseMap<StringAttr, func::FuncOp> getRepresentant;
94 DenseSet<func::FuncOp> toBeErased;
95 module.walk([&](func::FuncOp f) {
96 auto [repr, inserted] = uniqueFuncOps.insert(f);
97 getRepresentant[f.getSymNameAttr()] = *repr;
98 if (!inserted) {
99 toBeErased.insert(f);
100 }
101 });
102
103 // Update all symbol uses to reference unique func op
104 // representants and erase redundant func ops.
105 for (auto it : toBeErased) {
106 StringAttr oldSymbol = it.getSymNameAttr();
107 StringAttr newSymbol = getRepresentant[oldSymbol].getSymNameAttr();
108 if (failed(
109 SymbolTable::replaceAllSymbolUses(oldSymbol, newSymbol, module)))
110 return signalPassFailure();
111 it.erase();
112 }
113 }
114};
