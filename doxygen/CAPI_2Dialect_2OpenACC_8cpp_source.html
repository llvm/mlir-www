19// Keep the C API's complete registration behavior. Frontends that only need
20// OpenACC IR can link MLIROpenACCDialect directly instead.
21void registerOpenACC(DialectRegistry &registry) {
22 registry.insert<OpenACCDialect, gpu::GPUDialect>();
23 acc::registerAllExtensions(registry);
24 registry.addExtension(+[](MLIRContext *context, OpenACCDialect *) {
25 context->getOrLoadDialect<gpu::GPUDialect>();
26 });
27}
28
29void insertOpenACC(MlirDialectRegistry registry) {
30 registerOpenACC(*unwrap(registry));
31}

33MlirDialect loadOpenACC(MlirContext context) {
34 DialectRegistry registry;
35 registerOpenACC(registry);
36 unwrap(context)->appendDialectRegistry(registry);
37 return wrap(unwrap(context)->getOrLoadDialect<OpenACCDialect>());
38}

40MlirStringRef getNamespace() {
